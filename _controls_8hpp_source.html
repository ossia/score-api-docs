31template <typename Field>
32struct NormalizerFromMapper
33{
34 static inline constexpr auto mapper = avnd::get_mapper<Field>();
35 static float map(float val) noexcept { return mapper.map(val); }
36 static float unmap(float val) noexcept { return mapper.unmap(val); }
37
38 template <typename T>
39 static float to01(const T& slider, float val) noexcept
40 {
41 const auto min = slider.getMin();
42 const auto max = slider.getMax();
43
44 // [a; b] -> [0; 1] in linear space
45 const auto in_01 = (val - min) / (max - min);
46
47 // [0; 1] -> [0; 1] in e.g. log space
48 return map(in_01);
49 }
50
51 template <typename T>
52 static float from01(const T& slider, float in_01) noexcept
53 {
54 const auto min = slider.getMin();
55 const auto max = slider.getMax();
56
57 // [0; 1] in e.g. log space -> [0; 1] in linear space
58 const auto unmapped = unmap(in_01);
59
60 // [0; 1] in linear space -> [a; b]
61 return min + unmapped * (max - min);
62 }
63};
