435 void prepare(const value_loop& loop, uint16_t key, bool own) noexcept
436 {
437 own_mute = own;
438 count = 0;
439 const int64_t len = loop.length;
440 if(len <= 0)
441 return;
442 for(int i = 0; i < loop.event_count; i++)
443 {
444 const auto& e = loop.events[i];
445 if(e.key != key || e.impulse || !loop.heard(e, own_mute))
446 continue;
447 const int64_t period = loop.period_for(e.pass);
448 for(int64_t q = e.pos % period; q < len && count < max_refs; q += period)
449 refs[count++] = {q, e.serial, i};
450 }
451 std::sort(refs.begin(), refs.begin() + count, [](const ref& a, const ref& b) {
452 if(a.serial != b.serial)
453 return a.serial < b.serial;
454 if(a.pos != b.pos)
455 return a.pos < b.pos;
456 return a.event < b.event;
457 });
458 }

463 int value_at(
464 const value_loop& loop, uint16_t key, int64_t x, int first = -1,
465 int last = -1) const noexcept
466 {
467 bool any = false;
468 uint32_t top = 0;
469 for(int s = 0; s < loop.stretch_count; s++)
470 {
471 const auto& st = loop.stretches[s];
472 if(st.key != key || !in(st.pass, first, last) || !loop.heard(st, own_mute)
473 || !value_loop::holds(st, x, loop.period_for(st.pass)))
474 continue;
475 if(!any || st.serial > top)
476 top = st.serial;
477 any = true;
478 }
479 if(!any)
480 return -1;
481 auto lo = std::lower_bound(
482 refs.begin(), refs.begin() + count, top,
483 [](const ref& r, uint32_t s) { return r.serial < s; });
484 auto hi = std::upper_bound(
485 lo, refs.begin() + count, top, [](uint32_t s, const ref& r) { return s < r.serial; });
486 if(lo == hi)
487 return -1;
488 auto it = std::upper_bound(lo, hi, x, [](int64_t x, const ref& r) { return x < r.pos; });
489 return it == lo ? std::prev(hi)->event : std::prev(it)->event;
490 }

533 void update(
534 const value_loop& loop, uint16_t key, held_scratch& scratch, int first = -1,
535 int last = -1, bool own_mute = true, bool force = false) noexcept
536 {
537 if(!force && edits == loop.edits && length == loop.length)
538 return;
539 edits = loop.edits;
540 length = loop.length;
541 count = 0;
542 const int64_t len = loop.length;
543 if(len <= 0)
544 return;
545 scratch.prepare(loop, key, own_mute);
546
547 // Where the held value may change: on a value, and where a stretch
548 // starts or ends.
549 auto& cuts = scratch.cuts;
550 int c = 0;
551 for(int i = 0; i < scratch.count && c < held_scratch::max_refs; i++)
552 if(held_scratch::in(loop.events[scratch.refs[i].event].pass, first, last))
553 cuts[c++] = scratch.refs[i].pos;
554 for(int s = 0; s < loop.stretch_count; s++)
555 {
556 const auto& st = loop.stretches[s];
557 if(st.key != key || !held_scratch::in(st.pass, first, last)
558 || !loop.heard(st, own_mute))
559 continue;
560 const int64_t period = loop.period_for(st.pass);
561 if(st.span >= period)
562 continue;
563 for(int64_t base = 0; base < len && c + 2 <= held_scratch::max_refs; base += period)
564 for(const int64_t q : {base + st.from % period, base + (st.from + st.span) % period})
565 if(q < len)
566 cuts[c++] = q;
567 }
568 if(c == 0)
569 return;
570 if(c < held_scratch::max_refs)
571 cuts[c++] = 0;
572 std::sort(cuts.begin(), cuts.begin() + c);
573 c = int(std::unique(cuts.begin(), cuts.begin() + c) - cuts.begin());
574
575 for(int i = 0; i < c; i++)
576 {
577 const int e = scratch.value_at(loop, key, cuts[i], first, last);
578 if(count > 0)
579 {
580 const int prev = points[count - 1].event;
581 if(prev == e
582 || (prev >= 0 && e >= 0 && loop.events[prev].value.same(loop.events[e].value)))
583 continue;
584 }
585 points[count++] = {cuts[i], e};
586 }
587 }
