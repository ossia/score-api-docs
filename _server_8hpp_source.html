57// What arrived since the last take(), coalesced: only the newest scene, and
58// only the newest record of each kind for each node, bone or material,
59// survive. The receiving node may tick far slower than the sender sends (or
60// not at all while nothing is rendered), and a stale intermediate value is of
61// no use to it. Structural messages are kept in order, except those older
62// than the newest scene, which contains them.
63struct inbox
64{

72 latest<visibility_record> visibility;
73 latest<light_record> lights;
74 latest<camera_record> cameras;
75 latest<material_record> materials;
76
77 // The sender's hello, as received (JSON); empty until one arrives.
78 std::string peer_hello;

80 // Bumped on every connection and disconnection, so the node can notice a
81 // DCC that came back without polling the socket state.
