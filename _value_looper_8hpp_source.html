794inline void Node::handle_input(
795 int64_t from, int64_t to, double pos0, bool taking, double rate, bool backwards)
796{
797 const Passthrough pt = inputs.passthrough;
798 const bool echo = taking ? pt != Passthrough::None : pt == Passthrough::Full;
799 const auto hold = int64_t(input_hold_seconds * m_rate);
800
801 for(; m_in_index < m_in_count; m_in_index++)
802 {
803 const int64_t ts = input_ts(m_in_index);
804 if(ts >= to)
805 break;
806 send_due(ts);
807 const auto& v = inputs.values.value->get_data()[std::size_t(m_in_order[m_in_index])].value;
808 const uint16_t key = 0;
809 auto& k = m_keys[key];
810 const bool impulse = v.get_type() == ossia::val_type::IMPULSE;
811
812 {
813 auto& h = m_history[m_history_count++ % history_size];
814 if(m_history_count > history_size && !h.value.impulse())
815 m_history_floor[h.key] = h.value;
816 h.frame = m_now + ts;
817 h.key = key;
818 h.value.set(v);
819 }
820
821 // Where the value falls in the loop. A recording's loop is as long as
822 // what was recorded so far, so its positions are not wrapped.
823 // Backwards, pos0 counts from the loop's end, and a value falls on the
824 // edge of the position the playhead has just gone over.
825 int64_t pos = -1;
826 int64_t written = 0;
827 if(pos0 >= 0)
828 {
829 const double into = (std::max(ts, from) - from) * rate;
830 pos = int64_t(pos0 + into);
831 written = int64_t(pass_progress() + into);
832 if(backwards && loop.length > 0)
833 {
834 pos = (loop.length - pos) % loop.length;
835 if(pos < 0)
836 pos += loop.length;
837 }
838 else if(transport.actualMode != LoopMode::Record && loop.length > 0)
839 pos %= loop.length;
840 else
841 written = pos;
842 }
843
844 // A value is new unless it is the one the key held: only a new value
845 // is recorded.
846 const bool fresh = impulse || k.in.empty() || !k.in.same(v);
847 if(!impulse)
848 k.in.set(v);
849 if(taking && pos >= 0 && fresh)
850 record_value(pos, written, key, v, backwards);
851
852 if(echo)
853 {
854 emit(ts, v);
855 if(!impulse)
856 {
857 k.out.set(v);
858 k.owned_until = m_now + ts + hold;
859 }
860 }
861 }
862 send_due(to - 1);
863}

865inline void Node::record_value(
866 int64_t pos, int64_t written, uint16_t key, const ossia::value& v, bool backwards)
867{
868 material_arriving();
869 if(v.get_type() == ossia::val_type::IMPULSE)
870 {
871 loop.add_value(pos, written, key, v);
872 return;
873 }
874 auto& k = m_keys[key];
875 if(!loop.touched(key))
876 loop.touch_key(key, pos, written, backwards);
877 // Backwards, a value holds for the positions under the one it came at:
878 // the loop keeps, at that position, the value from before, which is what
879 // it goes back to going forwards; and where the playhead stops going
880 // backwards, the last value that came.
881 if(!backwards)
882 loop.add_value(pos, written, key, v);
883 else if(!k.recorded.empty())
884 loop.add_value(pos, written, key, k.recorded.get());
885 k.recorded.set(v);
886}

917inline void Node::looper_trim_recording(int64_t frames)
918{
919 const int64_t end = std::max<int64_t>(0, loop.length - frames);
920 // What came after the grid point belongs to the next pass: it is kept
921 // aside, from the grid point on, for an overdub that starts there.
922 const int64_t cut = loop.length - end;
923 for(int i = 0; i < m_carried_count; i++)
924 m_carried[i].pos += cut;
925 m_carried_span += cut;
926 for(int i = 0; i < loop.event_count; i++)
927 {
928 const auto& e = loop.events[i];
929 if(e.pos < end || e.impulse)
930 continue;
931 if(m_carried_count < int(m_carried.size()))
932 {
933 auto& c = m_carried[m_carried_count++];
934 c.pos = e.pos - end;
935 c.value = e.value;
936 c.key = e.key;
937 }
938 }
939 loop.remove_events([end](const Looper::value_event& e) { return e.pos >= end; });
940 loop.length = end;
941 loop.pass_written = end;
942 loop.edits++;
943}

949inline void Node::begin_pass_carried(int64_t k)
950{
951 const int64_t len = loop.length;
952 const bool from_point = m_carried_span > 0 && m_carried_span < len && k == m_carried_span;
953 begin_pass(from_point ? 0 : k);
954 for(int i = 0; i < m_carried_count; i++)
955 {

962 forget_carried();

965inline int64_t Node::looper_run(int64_t start, int64_t length)
966{
967 if(!loop.empty() && transport.playbackPos % loop.length != m_expected_pos)
968 m_phase = double(transport.playbackPos % loop.length);
969 if(m_reverse != m_was_reverse)
970 {
971 if(m_was_reverse && loop.taking && !loop.empty())
972 close_backwards(int64_t(std::ceil(m_phase - 1e-9)));
973 m_was_reverse = m_reverse;
974 }
975
976 const LoopMode mode = transport.actualMode;
977 switch(mode)
978 {
979 case LoopMode::Stop:
980 handle_input(start, start + length, -1, false);
981 break;
982
983 case LoopMode::Play:
984 advance(start, length, false);
985 break;
986
987 case LoopMode::Record: {
988 if(!transport.is_taking())
989 {
990 begin_recording();
991 m_switches.soloed.forget();
992 transport.playbackPos = 0;
993 m_export_pending = false;
994 }
995 transport.taking(LoopMode::Record);
996 const int64_t pos0 = loop.length;
997 handle_input(start, start + length, double(pos0), true);
998 loop.length += length;
999 loop.pass_written = loop.length;
1000 loop.pass_progress = double(loop.length);
1001 loop.update_open_stretches();
1002 transport.playbackPos = loop.length;
1003 m_phase = 0.;
1004 m_expected_pos = 0;
1005 return length;
1006 }
1007
1008 case LoopMode::Overdub: {
1009 if(loop.empty())
1010 {
1011 handle_input(start, start + length, -1, false);
1012 break;
1013 }
1014 if(!transport.is_taking())
1015 {
1016 const int64_t k = int64_t(m_phase) % loop.length;
1017 begin_pass_carried(k);
1018 m_cycle_pending = false;
1019 // An extending pass grows the loop by the loop it started over.
1020 m_extend_unit = loop.length;
1021 m_extended = false;
1022 }
1023 transport.taking(LoopMode::Overdub);
1024 advance(start, length, true);
1025 break;
1026 }
1027 }
1028
1029 if(!loop.empty())
1030 {
1031 transport.playbackPos = int64_t(m_phase) % loop.length;
1032 m_expected_pos = transport.playbackPos;
1033 }
1034 return length;
1035}

1059inline void Node::advance(int64_t start, int64_t length, bool overdub)
1060{
1061 const double r = rate();
1062 const bool back = m_reverse;
1063 // Backwards, the playhead reaches the loop's start, not its end.
1064 const bool extend = overdub && inputs.past_end.value == Looper::PastEnd::Extend && !back;
1065 const bool per_cycle = overdub && inputs.passes.value == Looper::Passes::PerCycle;
1066 // Undoing an extension can leave the playhead past the end of the loop.
1067 const double ph = std::fmod(std::max(0., m_phase), double(loop.length));
1068 m_phase = ph;
1069 double y = back ? double(loop.length) - ph : ph;
1070 if(y >= double(loop.length))
1071 y -= double(loop.length);
1072 double out = double(start);
1073 const double out_end = double(start + length);
1074
1075 chase(start);
1076 while(out < out_end - 1e-9)
1077 {
1078 const double L = double(loop.length);
1079 const double b = std::min(L, y + (out_end - out) * r);
1080 const double seg_out = (b - y) / r;
1081 if(overdub && std::exchange(m_cycle_pending, false))
1082 next_cycle_pass(back ? 0 : int64_t(y));
1083
1084 // Gone round again, the pass writes over what it recorded a round
1085 // before.
1086 if(overdub && !back && !pass_pending())
1087 loop.write_over(int64_t(std::ceil(y)), int64_t(std::ceil(b)), loop.pass_progress);
1088 if(back)
1089 play_range_reverse(y, b, out, r);
1090 else
1091 play_range(y, b, out, r);
1092 // The input of the frames this part of the loop covers.
1093 const int64_t from = int64_t(std::ceil(out - 1e-9));
1094 const int64_t to = b >= L ? int64_t(std::ceil(out + seg_out - 1e-9))
1095 : int64_t(std::ceil(out_end - 1e-9));
1096 handle_input(from, to, y + (from - out) * r, overdub, r, back);
1097 send_due(std::numeric_limits<int64_t>::max());
1098 if(overdub)
1099 {
1100 pass_advanced(b - y, back);
1101 if(!pass_pending())
1102 loop.update_open_stretches();
1103 }
1104
1105 out += seg_out;
1106 y = b;
1107 if(y >= L - 1e-9)
1108 {
1109 if(extend
1110 && loop.extend(m_extend_unit, int64_t(max_loop_seconds * m_rate), !pass_pending()))
1111 {
1112 // A cycle is the loop's length when the overdub started: the loop
1113 // growing by one is a cycle gone round as well.
1114 m_extended = true;
1115 loop.edits++;
1116 if(per_cycle)
1117 m_cycle_pending = true;
1118 }
1119 else
1120 {
1121 y -= L;
1122 if(y < 0.)
1123 y = 0.;
1124 if(per_cycle)
1125 {
1126 // The pass is over at the loop's start.
1127 if(back)
1128 close_backwards(0);
1129 m_cycle_pending = true;
1130 }
1131 }
1132 }
1133 }
1134 // Input on the last frame, when the span ended on the loop's end.
1135 if(!overdub)
1136 handle_input(start + length, start + length, -1, false);
1137 if(back)
1138 {
1139 y = double(loop.length) - y;
1140 if(y >= double(loop.length))
1141 y -= double(loop.length);
1142 }
1143 m_phase = y;
1144}

1150inline void Node::play_range(double a, double b, double out0, double r)
1151{
1152 m_due_count = 0;
1153 m_due_index = 0;
1154 update_timeline();
1155 const int64_t len = loop.length;
1156 auto frame = [&](int64_t q) { return int64_t(out0 + (q - a) / r); };
1157 const int64_t lo = int64_t(std::ceil(a));
1158 const int64_t hi = std::min(len, int64_t(std::ceil(b)));
1159 if(lo >= hi)
1160 return;
1161
1162 for(int i = m_timeline.at(lo); i >= 0 && i < m_timeline.count; i++)
1163 {
1164 const auto& p = m_timeline.points[i];
1165 if(p.pos < lo)
1166 continue;
1167 if(p.pos >= hi)
1168 break;
1169 add_due(frame(p.pos), i, false);
1170 }
1171 for(int i = 0; i < loop.event_count; i++)
1172 {
1173 const auto& e = loop.events[i];
1174 if(!e.impulse || !loop.heard(e))
1175 continue;
1176 const int64_t period = loop.period_for(e.pass);
1177 const int64_t first = e.pos % period;
1178 for(int64_t q = first + std::max<int64_t>(0, (lo - first + period - 1) / period) * period;
1179 q < hi; q += period)
1180 add_due(frame(q), i, true);
1181 }
1182 std::sort(m_due.begin(), m_due.begin() + m_due_count, [](const due& x, const due& y) {
1183 return x.ts != y.ts ? x.ts < y.ts : x.order < y.order;
1184 });
1185}

1191inline void Node::play_range_reverse(double a, double b, double out0, double r)
1192{
1193 m_due_count = 0;
1194 m_due_index = 0;
1195 update_timeline();
1196 const int64_t len = loop.length;
1197 auto frame = [&](int64_t q) { return int64_t(out0 + (q - a) / r); };
1198 const int64_t lo = int64_t(std::ceil(a));
1199 const int64_t hi = std::min(len, int64_t(std::ceil(b)));
1200 if(lo >= hi)
1201 return;
1202 auto mirrored = [](int64_t pos, int64_t period) {
1203 int64_t m = (-pos) % period;
1204 if(m < 0)
1205 m += period;
1206 return m;
1207 };
1208
1209 for(int i = 0; i < m_timeline.count; i++)
1210 {
1211 const int64_t q = mirrored(m_timeline.points[i].pos, len);
1212 if(q >= lo && q < hi)
1213 add_due(frame(q), m_timeline.previous(i), false);
1214 }
1215 for(int i = 0; i < loop.event_count; i++)
1216 {
1217 const auto& e = loop.events[i];
1218 if(!e.impulse || !loop.heard(e))
1219 continue;
1220 int64_t period = loop.period_for(e.pass);
1221 if(len % period != 0)
1222 period = len;
1223 const int64_t first = mirrored(e.pos % period, period);
1224 for(int64_t q = first + std::max<int64_t>(0, (lo - first + period - 1) / period) * period;
1225 q < hi; q += period)
1226 add_due(frame(q), i, true);
1227 }
1228 std::sort(m_due.begin(), m_due.begin() + m_due_count, [](const due& x, const due& y) {
1229 return x.ts != y.ts ? x.ts < y.ts : x.order < y.order;
1230 });
1231}

1235inline void Node::capture_into(int64_t end, int64_t n)
1236{
1237 const int64_t from = end - n;
1238 loop.begin_recording();
1239 loop.length = n;
1240 forget_recorded();
1241
1242 const int64_t count = std::min<int64_t>(m_history_count, history_size);
1243 // The value held before the window.
1244 for(int key = 0; key < max_keys; key++)
1245 {
1246 const Looper::value_slot* before = m_history_floor[key].empty() ? nullptr : &m_history_floor[key];
1247 for(int64_t i = m_history_count - count; i < m_history_count; i++)
1248 {
1249 const auto& h = m_history[i % history_size];
1250 if(h.key == key && h.frame < from && !h.value.impulse())
1251 before = &h.value;
1252 }
1253 if(before)
1254 record_value(0, 0, uint16_t(key), before->get(), false);
1255 }
1256 for(int64_t i = m_history_count - count; i < m_history_count; i++)
1257 {
1258 const auto& h = m_history[i % history_size];
1259 if(h.frame < from || h.frame >= end)
1260 continue;
1261 const int64_t pos = h.frame - from;
1262 const auto& last = m_keys[h.key].recorded;
1263 if(!h.value.impulse() && !last.empty() && last.same(h.value))
1264 continue;
1265 record_value(pos, pos, h.key, h.value.get(), false);
1266 }
1267 loop.pass_written = n;
1268 loop.pass_progress = double(n);
1269 loop.end_pass();
1270}

1283 if(h.frame >= w.end)
1284 break;
1285 const bool impulse = h.value.impulse();
1286 const bool fresh = impulse || !held[h.key] || !held[h.key]->same(h.value);
1287 if(!impulse)
1288 held[h.key] = &h.value;
1289 if(fresh && w.contains(h.frame))
1290 if(!f(h))

1412inline void Node::fill_message(Looper::view_message& v)
1413{
1414 fill_view(v);
1415 v.values = true;
1416 v.lanes = std::min(loop.count, Looper::view_message::max_lanes);
1417 v.max_seconds = 0.f;
1418 v.notes = 0;
1419
1420 constexpr int bins = Looper::view_message::bins;
1421 const int lanes = v.lanes;
1422 const int merged = loop.count - lanes;
1423 const int64_t len = std::max<int64_t>(1, loop.length);
1424 v.peaks.fill(-1.f);
1425 v.value_flags.fill(0);
1426 if(loop.empty())
1427 return;
1428 update_timeline();
1429
1430 // What each lane holds, sampled at the start of each bin: its passes
1431 // alone, muted or not, and whether that is what is heard there.
1432 m_scratch.prepare(loop, 0, false);
1433 for(int l = 0; l < lanes; l++)
1434 {
1435 const int first = l == 0 ? 0 : l + merged;
1436 const int last = l + merged;
1437 float* y = v.peaks.data() + l * bins;
1438 uint8_t* f = v.value_flags.data() + l * bins;
1439 double lo = 0., hi = 0.;
1440 bool any = false;
1441 for(int b = 0; b < bins; b++)
1442 {
1443 const int64_t x = std::min<int64_t>(len - 1, int64_t(double(b) * double(len) / bins));
1444 const int e = m_scratch.value_at(loop, 0, x, first, last);
1445 if(e < 0)
1446 continue;
1447 const auto& val = loop.events[e].value.get();
1448 const int at = m_timeline.at(x);
1449 if(at >= 0 && m_timeline.points[at].event == e)
1450 f[b] |= Looper::view_message::flag_heard;
1451 if(Looper::value_export::numbers_of(val) > 0)
1452 {
1453 const double n = Looper::value_export::number(val, 0);
1454 y[b] = float(n);
1455 f[b] |= Looper::view_message::flag_number;
1456 lo = any ? std::min(lo, n) : n;
1457 hi = any ? std::max(hi, n) : n;
1458 any = true;
1459 }
1460 else
1461 {
1462 f[b] |= Looper::view_message::flag_other;
1463 }
1464 }
1465 // Scaled to the values seen in the lane.
1466 for(int b = 0; b < bins; b++)
1467 if(f[b] & Looper::view_message::flag_number)
1468 y[b] = hi > lo ? float((y[b] - lo) / (hi - lo)) : 0.5f;
1469 }
1470
1471 // The impulses, and the values that are not numbers, as marks where they
1472 // are.
1473 for(int i = 0; i < loop.event_count; i++)
1474 {
1475 const auto& e = loop.events[i];
1476 if(e.pass >= loop.count)
1477 continue;
1478 const bool number = !e.impulse && Looper::value_export::numbers_of(e.value.get()) > 0;
1479 if(number)
1480 continue;
1481 const int l = std::clamp(e.pass - merged, 0, lanes - 1);
1482 const int64_t period = loop.period_for(e.pass);
1483 for(int64_t q = e.pos % period; q < len; q += period)
1484 {
1485 const int b = std::clamp(int(q * bins / len), 0, bins - 1);
1486 auto& f = v.value_flags[l * bins + b];
1487 f |= e.impulse ? Looper::view_message::flag_impulse : Looper::view_message::flag_mark;
1488 if(e.impulse && loop.heard(e))
1489 f |= Looper::view_message::flag_heard_mark;
1490 }
1491 }
1492}
