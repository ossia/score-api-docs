95inline QLinearGradient
96zoneGradient(double y_lo, double y_hi, double lo, double hi, double warn, double over, double alpha = 1.)
97{
98 QLinearGradient g{QPointF{0., y_lo}, QPointF{0., y_hi}};
99 const double span = std::max(hi - lo, 1e-9);
100 auto stop = [&](double v, QColor c) {
101 c.setAlphaF(alpha);
102 g.setColorAt(std::clamp((v - lo) / span, 0., 1.), c);
103 };
104 stop(lo, zoneColor(lo, warn, over));
105 // Each zone edge inside the range: the colour just under it, then over it
106 const double eps = 1e-6 * span;
107 for(double edge : {warn, over})
108 if(edge - eps > lo && edge < hi)
109 {
110 stop(edge - eps, zoneColor(edge - eps, warn, over));
111 stop(edge, zoneColor(edge, warn, over));
112 }
113 stop(hi, zoneColor(hi, warn, over));
114 return g;
115}

132inline const std::array<QRgb, 256>& colorMap(ColorMap m)
133{
134 auto build = [](std::initializer_list<std::array<int, 3>> anchors) {
135 std::array<QRgb, 256> lut{};
136 const std::vector<std::array<int, 3>> a(anchors);
137 const int n = int(a.size()) - 1;
138 for(int i = 0; i < 256; i++)
139 {
140 const double t = i / 255. * n;
141 const int k = std::min(int(t), n - 1);
142 const double f = t - k;
143 auto mix = [&](int c) { return int(std::lround(a[k][c] * (1 - f) + a[k + 1][c] * f)); };
144 lut[i] = qRgb(mix(0), mix(1), mix(2));
145 }
146 return lut;
147 };

221inline std::vector<Tick>
222linearTicks(double lo, double hi, double pixels, double min_spacing = 18., QString unit = {})
223{
224 std::vector<Tick> ticks;
225 if(!(hi > lo) || pixels <= 0)
226 return ticks;
227 const double span = hi - lo;
228 const double raw = span * min_spacing / pixels;
229 const double mag = std::pow(10., std::floor(std::log10(raw)));
230 double step = mag;
231 for(double m : {1., 2., 2.5, 5., 10.})
232 {
233 step = m * mag;
234 if(step >= raw)
235 break;
236 }
237 const double first = std::ceil(lo / step) * step;
238 for(double v = first; v <= hi + step * 1e-6; v += step)
239 {
240 const double r = std::abs(v) < step * 1e-6 ? 0. : v;
241 ticks.push_back({r, QString::number(r, 'g', 4) + unit, true});
242 }
243 return ticks;
244}

247inline std::vector<Tick> frequencyTicks(double lo, double hi, double pixels)
248{
249 std::vector<Tick> ticks;
250 if(lo <= 0 || hi <= lo)
251 return ticks;
252 const double decades = std::log10(hi / lo);
253 const bool dense = pixels / std::max(decades, 1e-3) > 90.;
254 for(double dec = std::pow(10., std::floor(std::log10(lo))); dec <= hi; dec *= 10.)
255 {
256 for(double m : {1., 2., 5.})
257 {
258 const double f = dec * m;
259 if(f < lo || f > hi)
260 continue;
261 const bool major = m == 1.;
262 if(!major && !dense && m == 5.)
263 continue;
264 ticks.push_back({f, formatHz(f), major});
265 }
266 }
267 return ticks;
268}

312 return;

318 void line(double x0, double y0, double x1, double y1, float energy = 1.f)
319 {
320 const double dx = x1 - x0, dy = y1 - y0;
321 const int n = std::max(1, int(std::ceil(std::max(std::abs(dx), std::abs(dy)))));
322 const float e = energy / n;
323 for(int i = 0; i <= n; i++)
324 hit(x0 + dx * i / n, y0 + dy * i / n, e);
325 }

327 void buildGamma(float gamma)
328 {
329 if(gamma == lut_gamma)
330 return;
331 lut_gamma = gamma;
332 for(int i = 0; i <= levels; i++)
333 gamma_lut[i] = uint8_t(std::lround(std::pow(float(i) / levels, gamma) * 255.f));
334 }

337 template <typename F>
338 void map(float full, float gamma, F&& f)
339 {
340 buildGamma(gamma);
341 const float k = levels / std::max(full, 1e-9f);
342 for(int y = 0; y < height; y++)
343 {
344 auto* line = reinterpret_cast<QRgb*>(image.scanLine(y));
345 const float* a = acc.data() + std::size_t(y) * width;
346 for(int x = 0; x < width; x++)
347 line[x] = f(gamma_lut[std::size_t(std::min(a[x] * k, float(levels)))]);
348 }
349 }

352 void render(QColor color, float full, float gamma = 0.5f)
353 {
354 std::array<QRgb, 256> shade;
355 for(int i = 0; i < 256; i++)
356 shade[i] = qPremultiply(qRgba(color.red(), color.green(), color.blue(), i));
357 map(full, gamma, [&](uint8_t v) { return shade[v]; });
358 }

388void drawXAxisLabels(QPainter& p, const QRectF& plot, const std::vector<Tick>& ticks, XOf x_of)
389{
390 constexpr double gap = 6.;
391 const QFontMetricsF fm{smallFont()};
392 std::vector<std::pair<double, double>> taken;
393 auto place = [&](const Tick& t) {
394 const double w = fm.horizontalAdvance(t.label);
395 if(w > plot.width())
396 return;
397 const double x0 = std::clamp(x_of(t.value) - w / 2., plot.left(), plot.right() - w);
398 for(auto [a, b] : taken)
399 if(x0 < b + gap && x0 + w > a - gap)
400 return;
401 taken.emplace_back(x0, x0 + w);
402 drawLabel(p, QRectF(x0, plot.bottom() + 1, w + 1, 11), Qt::AlignLeft | Qt::AlignTop, t.label);
403 };
404 for(const auto& t : ticks)
405 if(t.major)
406 place(t);
407 for(const auto& t : ticks)
408 if(!t.major)
409 place(t);
410}
