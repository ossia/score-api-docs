48struct OffscreenFrame
49{
50 explicit OffscreenFrame(QRhi& rhi) noexcept
51 {
52 QRhiCommandBuffer* cb{};
53 m_beginResult = simulateDeviceLost.load(std::memory_order_relaxed)
54 ? QRhi::FrameOpDeviceLost
55 : rhi.beginOffscreenFrame(&cb);
56 if(m_beginResult == QRhi::FrameOpSuccess)
57 {
58 if(cb)
59 {
60 m_rhi = &rhi;
61 m_commands = cb;
62 }
63 else
64 {
65 thread_local int nested = 0;
66 if((nested++ % 600) == 0)
67 qWarning() << "score.gfx: beginOffscreenFrame entered while a frame is "
68 "already recording; skipping this render (occurrence"
69 << nested << ")";
70 }
71 }
72 }
73
74 ~OffscreenFrame() noexcept { end(); }
75
76 OffscreenFrame(const OffscreenFrame&) = delete;
77 OffscreenFrame(OffscreenFrame&&) = delete;
78 OffscreenFrame& operator=(const OffscreenFrame&) = delete;
79 OffscreenFrame& operator=(OffscreenFrame&&) = delete;
80
81 explicit operator bool() const noexcept { return m_commands; }
82 QRhiCommandBuffer& commands() const noexcept { return *m_commands; }
83
84 QRhi::FrameOpResult beginResult() const noexcept { return m_beginResult; }
85 QRhi::FrameOpResult endResult() const noexcept { return m_endResult; }
86 bool deviceLost() const noexcept
87 {
88 return m_beginResult == QRhi::FrameOpDeviceLost
89 || m_endResult == QRhi::FrameOpDeviceLost;
90 }
91
92 QRhi::FrameOpResult end() noexcept
93 {
94 if(auto* rhi = m_rhi)
95 {
96 m_rhi = nullptr;
97 m_commands = nullptr;
98 m_endResult = rhi->endOffscreenFrame();
99 }
100 return m_endResult;
101 }
102
107 static inline std::atomic_bool simulateDeviceLost{false};
108
109private:
110 QRhi* m_rhi{};
111 QRhiCommandBuffer* m_commands{};
112 QRhi::FrameOpResult m_beginResult{QRhi::FrameOpSuccess};
113 QRhi::FrameOpResult m_endResult{QRhi::FrameOpSuccess};
114};

290struct TextureRenderTarget
291{
292 // The first five members must keep this order: out-of-tree addons
293 // aggregate-initialize {texture, colorRenderBuffer, depthRenderBuffer,
294 // renderPass, renderTarget}.
295 QRhiTexture* texture{}; // Primary color attachment (location 0)
296 QRhiRenderBuffer* colorRenderBuffer{};
297 QRhiRenderBuffer* depthRenderBuffer{};
298 QRhiRenderPassDescriptor* renderPass{};
299 QRhiRenderTarget* renderTarget{};
300
301 std::vector<QRhiTexture*> additionalColorTextures; // MRT: locations 1..N
302 std::vector<QRhiRenderBuffer*> additionalColorRenderBuffers; // MRT: MSAA attachments for locations 1..N
303 QRhiTexture* depthTexture{}; // Sampleable depth (alternative to depthRenderBuffer)
304 QRhiTexture* msDepthTexture{}; // MSAA depth attachment when depthTexture is the resolve target
305
306 // A 1×1 color texture allocated when the backend requires a color attachment
307 // but the user only wants depth-only rendering. Owned by this RT.
308 QRhiTexture* dummyColorTexture{};
309
310 // Number of array layers on `texture` (1 = non-layered, >1 = texture array).
311 int arrayLayers{1};
312
313 // Multiview view count (0/1 = disabled).
314 int multiViewCount{0};
315
316 // Which layer of `texture`/`additionalColorTextures` this RT renders to.
317 // -1 = not applicable (non-layered, or MultiView handles it automatically).
318 int renderLayer{-1};
319
320 operator bool() const noexcept { return texture != nullptr || dummyColorTexture != nullptr || depthTexture != nullptr; }
321
322 int colorAttachmentCount() const noexcept
323 {
324 if(texture)
325 return 1 + (int)additionalColorTextures.size();
326 if(dummyColorTexture)
327 return 1;
328 return 0;
329 }
330
331 // Returns the actual MSAA sample count of this render target, or -1 if it
332 // cannot be determined from the stored fields. Callers must treat -1 as
333 // "unknown — fall back to the renderlist's global sample count".
334 //
335 // Lookup priority:
336 // 1. colorRenderBuffer (owned MSAA attachment — always authoritative).
337 // 2. texture (single-sample resolve target OR non-MSAA render target).
338 // 3. depthTexture (depth-only RTs).
339 // 4. msDepthTexture (MSAA depth attachment when depth resolve is used).
340 // 5. renderTarget — BUT only when this RT genuinely owns its attachments
341 // (colorRenderBuffer/texture/depthTexture set). A "bare" RT that only
342 // carries renderTarget + renderPass (e.g. a swap-chain wrapper
343 // returned by QRhiSwapChain::currentFrameRenderTarget()) is NOT
344 // queried because swap-chain render-target objects lazily write
345 // their sampleCount only when createOrResize() runs — any read before
346 // that returns the default 1, which would silently mismatch a
347 // multi-sample renderPassDescriptor and produce
348 // VUID-VkGraphicsPipelineCreateInfo-multisampledRenderToSingleSampled-06853.
349 // 6. Otherwise return -1 so the caller uses RenderList::samples(), which
350 // IS authoritative for externally-managed swap-chain RTs (it drove
351 // the swap-chain sample count in the first place).
352 int sampleCount() const noexcept
353 {
354 if(colorRenderBuffer)
355 return colorRenderBuffer->sampleCount();
356 if(texture)
357 return texture->sampleCount();
358 if(msDepthTexture)
359 return msDepthTexture->sampleCount();
360 if(depthTexture)
361 return depthTexture->sampleCount();
362 // renderTarget alone without any owned attachment = swap-chain wrapper.
363 // Its sampleCount is unreliable pre-createOrResize; fall through.
364 return -1;
365 }
366
367 void release()
368 {
369 if(texture || dummyColorTexture || depthTexture)
370 {
371 // Use deleteLater() for all GPU resources: Qt RHI commands are async
372 // and resources may still be referenced by in-flight frames until
373 // endFrame() completes. deleteLater() defers actual destruction to
374 // the next beginFrame().
375 if(texture)
376 texture->deleteLater();
377 texture = nullptr;
378
379 if(dummyColorTexture)
380 dummyColorTexture->deleteLater();
381 dummyColorTexture = nullptr;
382
383 for(auto* t : additionalColorTextures)
384 t->deleteLater();
385 additionalColorTextures.clear();
386
387 if(colorRenderBuffer)
388 colorRenderBuffer->deleteLater();
389 colorRenderBuffer = nullptr;
390
391 for(auto* rb : additionalColorRenderBuffers)
392 rb->deleteLater();
393 additionalColorRenderBuffers.clear();
394
395 if(depthRenderBuffer)
396 depthRenderBuffer->deleteLater();
397 depthRenderBuffer = nullptr;
398
399 if(depthTexture)
400 depthTexture->deleteLater();
401 depthTexture = nullptr;
402
403 if(msDepthTexture)
404 msDepthTexture->deleteLater();
405 msDepthTexture = nullptr;
406
407 if(renderPass)
408 renderPass->deleteLater();
409 renderPass = nullptr;
410
411 if(renderTarget)
412 renderTarget->deleteLater();
413 renderTarget = nullptr;
414 }
415 }
