46struct OffscreenFrame
47{
48 explicit OffscreenFrame(QRhi& rhi) noexcept
49 {
50 QRhiCommandBuffer* cb{};
51 if(rhi.beginOffscreenFrame(&cb) == QRhi::FrameOpSuccess)
52 {
53 if(cb)
54 {
55 m_rhi = &rhi;
56 m_commands = cb;
57 }
58 else
59 {
60 thread_local int nested = 0;
61 if((nested++ % 600) == 0)
62 qWarning() << "score.gfx: beginOffscreenFrame entered while a frame is "
63 "already recording; skipping this render (occurrence"
64 << nested << ")";
65 }
66 }
67 }
68
69 ~OffscreenFrame() noexcept { end(); }
70
71 OffscreenFrame(const OffscreenFrame&) = delete;
72 OffscreenFrame(OffscreenFrame&&) = delete;
73 OffscreenFrame& operator=(const OffscreenFrame&) = delete;
74 OffscreenFrame& operator=(OffscreenFrame&&) = delete;
75
76 explicit operator bool() const noexcept { return m_commands; }
77 QRhiCommandBuffer& commands() const noexcept { return *m_commands; }
78
79 void end() noexcept
80 {
81 if(auto* rhi = m_rhi)
82 {
83 m_rhi = nullptr;
84 m_commands = nullptr;
85 rhi->endOffscreenFrame();
86 }
87 }
88
89private:
90 QRhi* m_rhi{};
91 QRhiCommandBuffer* m_commands{};
92};

197struct Edge
198{
199 Edge(Port* source, Port* sink, Process::CableType t)
200 : source{source}
201 , sink{sink}
202 , type{t}
203 {
204 source->edges.push_back(this);
205 sink->edges.push_back(this);

207
208 ~Edge()
209 {
210 if(auto it = std::find(source->edges.begin(), source->edges.end(), this);
211 it != source->edges.end())
212 source->edges.erase(it);
213 if(auto it = std::find(sink->edges.begin(), sink->edges.end(), this);
214 it != sink->edges.end())
215 sink->edges.erase(it);
216 }

246struct TextureRenderTarget
247{
248 // The first five members must keep this order: out-of-tree addons
249 // aggregate-initialize {texture, colorRenderBuffer, depthRenderBuffer,
250 // renderPass, renderTarget}.
251 QRhiTexture* texture{}; // Primary color attachment (location 0)
252 QRhiRenderBuffer* colorRenderBuffer{};
253 QRhiRenderBuffer* depthRenderBuffer{};
254 QRhiRenderPassDescriptor* renderPass{};
255 QRhiRenderTarget* renderTarget{};
256
257 std::vector<QRhiTexture*> additionalColorTextures; // MRT: locations 1..N
258 std::vector<QRhiRenderBuffer*> additionalColorRenderBuffers; // MRT: MSAA attachments for locations 1..N
259 QRhiTexture* depthTexture{}; // Sampleable depth (alternative to depthRenderBuffer)
260 QRhiTexture* msDepthTexture{}; // MSAA depth attachment when depthTexture is the resolve target
261
262 // A 1×1 color texture allocated when the backend requires a color attachment
263 // but the user only wants depth-only rendering. Owned by this RT.
264 QRhiTexture* dummyColorTexture{};
265
266 // Number of array layers on `texture` (1 = non-layered, >1 = texture array).
267 int arrayLayers{1};
268
269 // Multiview view count (0/1 = disabled).
270 int multiViewCount{0};
271
272 // Which layer of `texture`/`additionalColorTextures` this RT renders to.
273 // -1 = not applicable (non-layered, or MultiView handles it automatically).
274 int renderLayer{-1};
275
276 operator bool() const noexcept { return texture != nullptr || dummyColorTexture != nullptr || depthTexture != nullptr; }
277
278 int colorAttachmentCount() const noexcept
279 {
280 if(texture)
281 return 1 + (int)additionalColorTextures.size();
282 if(dummyColorTexture)
283 return 1;
284 return 0;
285 }
286
287 // Returns the actual MSAA sample count of this render target, or -1 if it
288 // cannot be determined from the stored fields. Callers must treat -1 as
289 // "unknown — fall back to the renderlist's global sample count".
290 //
291 // Lookup priority:
292 // 1. colorRenderBuffer (owned MSAA attachment — always authoritative).
293 // 2. texture (single-sample resolve target OR non-MSAA render target).
294 // 3. depthTexture (depth-only RTs).
295 // 4. msDepthTexture (MSAA depth attachment when depth resolve is used).
296 // 5. renderTarget — BUT only when this RT genuinely owns its attachments
297 // (colorRenderBuffer/texture/depthTexture set). A "bare" RT that only
298 // carries renderTarget + renderPass (e.g. a swap-chain wrapper
299 // returned by QRhiSwapChain::currentFrameRenderTarget()) is NOT
300 // queried because swap-chain render-target objects lazily write
301 // their sampleCount only when createOrResize() runs — any read before
302 // that returns the default 1, which would silently mismatch a
303 // multi-sample renderPassDescriptor and produce
304 // VUID-VkGraphicsPipelineCreateInfo-multisampledRenderToSingleSampled-06853.
305 // 6. Otherwise return -1 so the caller uses RenderList::samples(), which
306 // IS authoritative for externally-managed swap-chain RTs (it drove
307 // the swap-chain sample count in the first place).
308 int sampleCount() const noexcept
309 {
310 if(colorRenderBuffer)
311 return colorRenderBuffer->sampleCount();
312 if(texture)
313 return texture->sampleCount();
314 if(msDepthTexture)
315 return msDepthTexture->sampleCount();
316 if(depthTexture)
317 return depthTexture->sampleCount();
318 // renderTarget alone without any owned attachment = swap-chain wrapper.
319 // Its sampleCount is unreliable pre-createOrResize; fall through.
320 return -1;
321 }
322
323 void release()
324 {
325 if(texture || dummyColorTexture || depthTexture)
326 {
327 // Use deleteLater() for all GPU resources: Qt RHI commands are async
328 // and resources may still be referenced by in-flight frames until
329 // endFrame() completes. deleteLater() defers actual destruction to
330 // the next beginFrame().
331 if(texture)
332 texture->deleteLater();
333 texture = nullptr;
334
335 if(dummyColorTexture)
336 dummyColorTexture->deleteLater();
337 dummyColorTexture = nullptr;
338
339 for(auto* t : additionalColorTextures)
340 t->deleteLater();
341 additionalColorTextures.clear();
342
343 if(colorRenderBuffer)
344 colorRenderBuffer->deleteLater();
345 colorRenderBuffer = nullptr;
346
347 for(auto* rb : additionalColorRenderBuffers)
348 rb->deleteLater();
349 additionalColorRenderBuffers.clear();
350
351 if(depthRenderBuffer)
352 depthRenderBuffer->deleteLater();
353 depthRenderBuffer = nullptr;
354
355 if(depthTexture)
356 depthTexture->deleteLater();
357 depthTexture = nullptr;
358
359 if(msDepthTexture)
360 msDepthTexture->deleteLater();
361 msDepthTexture = nullptr;
362
363 if(renderPass)
364 renderPass->deleteLater();
365 renderPass = nullptr;
366
367 if(renderTarget)
368 renderTarget->deleteLater();
369 renderTarget = nullptr;
370 }
371 }
