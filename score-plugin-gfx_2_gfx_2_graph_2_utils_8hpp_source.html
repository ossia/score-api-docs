53struct OffscreenFrame
54{
55 explicit OffscreenFrame(QRhi& rhi) noexcept
56 {
57 QRhiCommandBuffer* cb{};
58 m_beginResult = simulateDeviceLost.load(std::memory_order_relaxed)
59 ? QRhi::FrameOpDeviceLost
60 : rhi.beginOffscreenFrame(&cb);
61 if(m_beginResult == QRhi::FrameOpSuccess)
62 {
63 if(cb)
64 {
65 m_rhi = &rhi;
66 m_commands = cb;
67 }
68 else
69 {
70 thread_local int nested = 0;
71 if((nested++ % 600) == 0)
72 qWarning() << "score.gfx: beginOffscreenFrame entered while a frame is "
73 "already recording; skipping this render (occurrence"
74 << nested << ")";
75 }
76 }
77 }
78
79 ~OffscreenFrame() noexcept { end(); }
80
81 OffscreenFrame(const OffscreenFrame&) = delete;
82 OffscreenFrame(OffscreenFrame&&) = delete;
83 OffscreenFrame& operator=(const OffscreenFrame&) = delete;
84 OffscreenFrame& operator=(OffscreenFrame&&) = delete;
85
86 explicit operator bool() const noexcept { return m_commands; }
87 QRhiCommandBuffer& commands() const noexcept { return *m_commands; }
88
89 QRhi::FrameOpResult beginResult() const noexcept { return m_beginResult; }
90 QRhi::FrameOpResult endResult() const noexcept { return m_endResult; }
91 bool deviceLost() const noexcept
92 {
93 return m_beginResult == QRhi::FrameOpDeviceLost
94 || m_endResult == QRhi::FrameOpDeviceLost;
95 }
96
97 QRhi::FrameOpResult end() noexcept
98 {
99 if(auto* rhi = m_rhi)
100 {
101 m_rhi = nullptr;
102 m_commands = nullptr;
103 m_endResult = rhi->endOffscreenFrame();
104 }
105 return m_endResult;
106 }
107
112 static inline std::atomic_bool simulateDeviceLost{false};
113
114private:
115 QRhi* m_rhi{};
116 QRhiCommandBuffer* m_commands{};
117 QRhi::FrameOpResult m_beginResult{QRhi::FrameOpSuccess};
118 QRhi::FrameOpResult m_endResult{QRhi::FrameOpSuccess};
119};

236{
240 Edge(Port* source, Port* sink, Process::CableType t);
241
242 ~Edge()
243 {
244 generation.fetch_add(1, std::memory_order_relaxed);
245 if(auto it = std::find(source->edges.begin(), source->edges.end(), this);
246 it != source->edges.end())
247 source->edges.erase(it);
248 if(auto it = std::find(sink->edges.begin(), sink->edges.end(), this);
249 it != sink->edges.end())
250 sink->edges.erase(it);
251 }

306struct TextureRenderTarget
307{
308 // The first five members must keep this order: out-of-tree addons
309 // aggregate-initialize {texture, colorRenderBuffer, depthRenderBuffer,
310 // renderPass, renderTarget}.
311 QRhiTexture* texture{}; // Primary color attachment (location 0)
312 QRhiRenderBuffer* colorRenderBuffer{};
313 QRhiRenderBuffer* depthRenderBuffer{};
314 QRhiRenderPassDescriptor* renderPass{};
315 QRhiRenderTarget* renderTarget{};
316
317 std::vector<QRhiTexture*> additionalColorTextures; // MRT: locations 1..N
318 std::vector<QRhiRenderBuffer*> additionalColorRenderBuffers; // MRT: MSAA attachments for locations 1..N
319 QRhiTexture* depthTexture{}; // Sampleable depth (alternative to depthRenderBuffer)
320 QRhiTexture* msDepthTexture{}; // MSAA depth attachment when depthTexture is the resolve target
321
322 // A 1×1 color texture allocated when the backend requires a color attachment
323 // but the user only wants depth-only rendering. Owned by this RT.
324 QRhiTexture* dummyColorTexture{};
325
326 // Number of array layers on `texture` (1 = non-layered, >1 = texture array).
327 int arrayLayers{1};
328
329 // Multiview view count (0/1 = disabled).
330 int multiViewCount{0};
331
332 // Which layer of `texture`/`additionalColorTextures` this RT renders to.
333 // -1 = not applicable (non-layered, or MultiView handles it automatically).
334 int renderLayer{-1};
335
336 operator bool() const noexcept { return texture != nullptr || dummyColorTexture != nullptr || depthTexture != nullptr; }
337
338 int colorAttachmentCount() const noexcept
339 {
340 if(texture)
341 return 1 + (int)additionalColorTextures.size();
342 if(dummyColorTexture)
343 return 1;
344 return 0;
345 }
346
347 // Whether the depth attachment has a stencil aspect: a depth-stencil
348 // renderbuffer, or a D24S8 / D32FS8 depth texture. The D32F targets the
349 // engine creates for 3D rendering have none.
350 bool hasStencil() const noexcept
351 {
352 if(depthRenderBuffer)
353 return depthRenderBuffer->type() == QRhiRenderBuffer::DepthStencil;
354 for(const QRhiTexture* t : {depthTexture, msDepthTexture})
355 {
356 if(!t)
357 continue;
358 switch(t->format())
359 {
360 case QRhiTexture::D24S8:
361#if QT_VERSION >= QT_VERSION_CHECK(6, 9, 0)
362 case QRhiTexture::D32FS8:
363#endif
364 return true;
365 default:
366 return false;
367 }
368 }
369 return false;
370 }
371
372 // Returns the actual MSAA sample count of this render target, or -1 if it
373 // cannot be determined from the stored fields. Callers must treat -1 as
374 // "unknown — fall back to the renderlist's global sample count".
375 //
376 // Lookup priority:
377 // 1. colorRenderBuffer (owned MSAA attachment — always authoritative).
378 // 2. texture (single-sample resolve target OR non-MSAA render target).
379 // 3. depthTexture (depth-only RTs).
380 // 4. msDepthTexture (MSAA depth attachment when depth resolve is used).
381 // 5. renderTarget — BUT only when this RT genuinely owns its attachments
382 // (colorRenderBuffer/texture/depthTexture set). A "bare" RT that only
383 // carries renderTarget + renderPass (e.g. a swap-chain wrapper
384 // returned by QRhiSwapChain::currentFrameRenderTarget()) is NOT
385 // queried because swap-chain render-target objects lazily write
386 // their sampleCount only when createOrResize() runs — any read before
387 // that returns the default 1, which would silently mismatch a
388 // multi-sample renderPassDescriptor and produce
389 // VUID-VkGraphicsPipelineCreateInfo-multisampledRenderToSingleSampled-06853.
390 // 6. Otherwise return -1 so the caller uses RenderList::samples(), which
391 // IS authoritative for externally-managed swap-chain RTs (it drove
392 // the swap-chain sample count in the first place).
393 int sampleCount() const noexcept
394 {
395 if(colorRenderBuffer)
396 return colorRenderBuffer->sampleCount();
397 if(texture)
398 return texture->sampleCount();
399 if(msDepthTexture)
400 return msDepthTexture->sampleCount();
401 if(depthTexture)
402 return depthTexture->sampleCount();
403 // renderTarget alone without any owned attachment = swap-chain wrapper.
404 // Its sampleCount is unreliable pre-createOrResize; fall through.
405 return -1;
406 }
407
408 void release()
409 {
410 if(texture || dummyColorTexture || depthTexture)
411 {
412 // Use deleteLater() for all GPU resources: Qt RHI commands are async
413 // and resources may still be referenced by in-flight frames until
414 // endFrame() completes. deleteLater() defers actual destruction to
415 // the next beginFrame().
416 if(texture)
417 texture->deleteLater();
418 texture = nullptr;
419
420 if(dummyColorTexture)
421 dummyColorTexture->deleteLater();
422 dummyColorTexture = nullptr;
423
424 for(auto* t : additionalColorTextures)
425 t->deleteLater();
426 additionalColorTextures.clear();
427
428 if(colorRenderBuffer)
429 colorRenderBuffer->deleteLater();
430 colorRenderBuffer = nullptr;
431
432 for(auto* rb : additionalColorRenderBuffers)
433 rb->deleteLater();
434 additionalColorRenderBuffers.clear();
435
436 if(depthRenderBuffer)
437 depthRenderBuffer->deleteLater();
438 depthRenderBuffer = nullptr;
439
440 if(depthTexture)
441 depthTexture->deleteLater();
442 depthTexture = nullptr;
443
444 if(msDepthTexture)
445 msDepthTexture->deleteLater();
446 msDepthTexture = nullptr;
447
448 if(renderPass)
449 renderPass->deleteLater();
450 renderPass = nullptr;
451
452 if(renderTarget)
453 renderTarget->deleteLater();
454 renderTarget = nullptr;
455 }
456 }
