48struct OffscreenFrame
49{
50 explicit OffscreenFrame(QRhi& rhi) noexcept
51 {
52 QRhiCommandBuffer* cb{};
53 m_beginResult = simulateDeviceLost.load(std::memory_order_relaxed)
54 ? QRhi::FrameOpDeviceLost
55 : rhi.beginOffscreenFrame(&cb);
56 if(m_beginResult == QRhi::FrameOpSuccess)
57 {
58 if(cb)
59 {
60 m_rhi = &rhi;
61 m_commands = cb;
62 }
63 else
64 {
65 thread_local int nested = 0;
66 if((nested++ % 600) == 0)
67 qWarning() << "score.gfx: beginOffscreenFrame entered while a frame is "
68 "already recording; skipping this render (occurrence"
69 << nested << ")";
70 }
71 }
72 }
73
74 ~OffscreenFrame() noexcept { end(); }
75
76 OffscreenFrame(const OffscreenFrame&) = delete;
77 OffscreenFrame(OffscreenFrame&&) = delete;
78 OffscreenFrame& operator=(const OffscreenFrame&) = delete;
79 OffscreenFrame& operator=(OffscreenFrame&&) = delete;
80
81 explicit operator bool() const noexcept { return m_commands; }
82 QRhiCommandBuffer& commands() const noexcept { return *m_commands; }
83
84 QRhi::FrameOpResult beginResult() const noexcept { return m_beginResult; }
85 QRhi::FrameOpResult endResult() const noexcept { return m_endResult; }
86 bool deviceLost() const noexcept
87 {
88 return m_beginResult == QRhi::FrameOpDeviceLost
89 || m_endResult == QRhi::FrameOpDeviceLost;
90 }
91
92 QRhi::FrameOpResult end() noexcept
93 {
94 if(auto* rhi = m_rhi)
95 {
96 m_rhi = nullptr;
97 m_commands = nullptr;
98 m_endResult = rhi->endOffscreenFrame();
99 }
100 return m_endResult;
101 }
102
107 static inline std::atomic_bool simulateDeviceLost{false};
108
109private:
110 QRhi* m_rhi{};
111 QRhiCommandBuffer* m_commands{};
112 QRhi::FrameOpResult m_beginResult{QRhi::FrameOpSuccess};
113 QRhi::FrameOpResult m_endResult{QRhi::FrameOpSuccess};
114};

278struct TextureRenderTarget
279{
280 // The first five members must keep this order: out-of-tree addons
281 // aggregate-initialize {texture, colorRenderBuffer, depthRenderBuffer,
282 // renderPass, renderTarget}.
283 QRhiTexture* texture{}; // Primary color attachment (location 0)
284 QRhiRenderBuffer* colorRenderBuffer{};
285 QRhiRenderBuffer* depthRenderBuffer{};
286 QRhiRenderPassDescriptor* renderPass{};
287 QRhiRenderTarget* renderTarget{};
288
289 std::vector<QRhiTexture*> additionalColorTextures; // MRT: locations 1..N
290 std::vector<QRhiRenderBuffer*> additionalColorRenderBuffers; // MRT: MSAA attachments for locations 1..N
291 QRhiTexture* depthTexture{}; // Sampleable depth (alternative to depthRenderBuffer)
292 QRhiTexture* msDepthTexture{}; // MSAA depth attachment when depthTexture is the resolve target
293
294 // A 1×1 color texture allocated when the backend requires a color attachment
295 // but the user only wants depth-only rendering. Owned by this RT.
296 QRhiTexture* dummyColorTexture{};
297
298 // Number of array layers on `texture` (1 = non-layered, >1 = texture array).
299 int arrayLayers{1};
300
301 // Multiview view count (0/1 = disabled).
302 int multiViewCount{0};
303
304 // Which layer of `texture`/`additionalColorTextures` this RT renders to.
305 // -1 = not applicable (non-layered, or MultiView handles it automatically).
306 int renderLayer{-1};
307
308 operator bool() const noexcept { return texture != nullptr || dummyColorTexture != nullptr || depthTexture != nullptr; }
309
310 int colorAttachmentCount() const noexcept
311 {
312 if(texture)
313 return 1 + (int)additionalColorTextures.size();
314 if(dummyColorTexture)
315 return 1;
316 return 0;
317 }
318
319 // Returns the actual MSAA sample count of this render target, or -1 if it
320 // cannot be determined from the stored fields. Callers must treat -1 as
321 // "unknown — fall back to the renderlist's global sample count".
322 //
323 // Lookup priority:
324 // 1. colorRenderBuffer (owned MSAA attachment — always authoritative).
325 // 2. texture (single-sample resolve target OR non-MSAA render target).
326 // 3. depthTexture (depth-only RTs).
327 // 4. msDepthTexture (MSAA depth attachment when depth resolve is used).
328 // 5. renderTarget — BUT only when this RT genuinely owns its attachments
329 // (colorRenderBuffer/texture/depthTexture set). A "bare" RT that only
330 // carries renderTarget + renderPass (e.g. a swap-chain wrapper
331 // returned by QRhiSwapChain::currentFrameRenderTarget()) is NOT
332 // queried because swap-chain render-target objects lazily write
333 // their sampleCount only when createOrResize() runs — any read before
334 // that returns the default 1, which would silently mismatch a
335 // multi-sample renderPassDescriptor and produce
336 // VUID-VkGraphicsPipelineCreateInfo-multisampledRenderToSingleSampled-06853.
337 // 6. Otherwise return -1 so the caller uses RenderList::samples(), which
338 // IS authoritative for externally-managed swap-chain RTs (it drove
339 // the swap-chain sample count in the first place).
340 int sampleCount() const noexcept
341 {
342 if(colorRenderBuffer)
343 return colorRenderBuffer->sampleCount();
344 if(texture)
345 return texture->sampleCount();
346 if(msDepthTexture)
347 return msDepthTexture->sampleCount();
348 if(depthTexture)
349 return depthTexture->sampleCount();
350 // renderTarget alone without any owned attachment = swap-chain wrapper.
351 // Its sampleCount is unreliable pre-createOrResize; fall through.
352 return -1;
353 }
354
355 void release()
356 {
357 if(texture || dummyColorTexture || depthTexture)
358 {
359 // Use deleteLater() for all GPU resources: Qt RHI commands are async
360 // and resources may still be referenced by in-flight frames until
361 // endFrame() completes. deleteLater() defers actual destruction to
362 // the next beginFrame().
363 if(texture)
364 texture->deleteLater();
365 texture = nullptr;
366
367 if(dummyColorTexture)
368 dummyColorTexture->deleteLater();
369 dummyColorTexture = nullptr;
370
371 for(auto* t : additionalColorTextures)
372 t->deleteLater();
373 additionalColorTextures.clear();
374
375 if(colorRenderBuffer)
376 colorRenderBuffer->deleteLater();
377 colorRenderBuffer = nullptr;
378
379 for(auto* rb : additionalColorRenderBuffers)
380 rb->deleteLater();
381 additionalColorRenderBuffers.clear();
382
383 if(depthRenderBuffer)
384 depthRenderBuffer->deleteLater();
385 depthRenderBuffer = nullptr;
386
387 if(depthTexture)
388 depthTexture->deleteLater();
389 depthTexture = nullptr;
390
391 if(msDepthTexture)
392 msDepthTexture->deleteLater();
393 msDepthTexture = nullptr;
394
395 if(renderPass)
396 renderPass->deleteLater();
397 renderPass = nullptr;
398
399 if(renderTarget)
400 renderTarget->deleteLater();
401 renderTarget = nullptr;
402 }
403 }
