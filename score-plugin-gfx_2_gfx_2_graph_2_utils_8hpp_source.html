114 {

119 it != sink->edges.end())
120 sink->edges.erase(it);
121 }

151struct TextureRenderTarget
152{
153 // The first five members must keep this order: out-of-tree addons
154 // aggregate-initialize {texture, colorRenderBuffer, depthRenderBuffer,
155 // renderPass, renderTarget}.
156 QRhiTexture* texture{}; // Primary color attachment (location 0)
157 QRhiRenderBuffer* colorRenderBuffer{};
158 QRhiRenderBuffer* depthRenderBuffer{};
159 QRhiRenderPassDescriptor* renderPass{};
160 QRhiRenderTarget* renderTarget{};
161
162 std::vector<QRhiTexture*> additionalColorTextures; // MRT: locations 1..N
163 std::vector<QRhiRenderBuffer*> additionalColorRenderBuffers; // MRT: MSAA attachments for locations 1..N
164 QRhiTexture* depthTexture{}; // Sampleable depth (alternative to depthRenderBuffer)
165 QRhiTexture* msDepthTexture{}; // MSAA depth attachment when depthTexture is the resolve target
166
167 // A 1×1 color texture allocated when the backend requires a color attachment
168 // but the user only wants depth-only rendering. Owned by this RT.
169 QRhiTexture* dummyColorTexture{};
170
171 // Number of array layers on `texture` (1 = non-layered, >1 = texture array).
172 int arrayLayers{1};
173
174 // Multiview view count (0/1 = disabled).
175 int multiViewCount{0};
176
177 // Which layer of `texture`/`additionalColorTextures` this RT renders to.
178 // -1 = not applicable (non-layered, or MultiView handles it automatically).
179 int renderLayer{-1};
180
181 operator bool() const noexcept { return texture != nullptr || dummyColorTexture != nullptr || depthTexture != nullptr; }
182
183 int colorAttachmentCount() const noexcept
184 {
185 if(texture)
186 return 1 + (int)additionalColorTextures.size();
187 if(dummyColorTexture)
188 return 1;
189 return 0;
190 }
191
192 // Returns the actual MSAA sample count of this render target, or -1 if it
193 // cannot be determined from the stored fields. Callers must treat -1 as
194 // "unknown — fall back to the renderlist's global sample count".
195 //
196 // Lookup priority:
197 // 1. colorRenderBuffer (owned MSAA attachment — always authoritative).
198 // 2. texture (single-sample resolve target OR non-MSAA render target).
199 // 3. depthTexture (depth-only RTs).
200 // 4. msDepthTexture (MSAA depth attachment when depth resolve is used).
201 // 5. renderTarget — BUT only when this RT genuinely owns its attachments
202 // (colorRenderBuffer/texture/depthTexture set). A "bare" RT that only
203 // carries renderTarget + renderPass (e.g. a swap-chain wrapper
204 // returned by QRhiSwapChain::currentFrameRenderTarget()) is NOT
205 // queried because swap-chain render-target objects lazily write
206 // their sampleCount only when createOrResize() runs — any read before
207 // that returns the default 1, which would silently mismatch a
208 // multi-sample renderPassDescriptor and produce
209 // VUID-VkGraphicsPipelineCreateInfo-multisampledRenderToSingleSampled-06853.
210 // 6. Otherwise return -1 so the caller uses RenderList::samples(), which
211 // IS authoritative for externally-managed swap-chain RTs (it drove
212 // the swap-chain sample count in the first place).
213 int sampleCount() const noexcept
214 {
215 if(colorRenderBuffer)
216 return colorRenderBuffer->sampleCount();
217 if(texture)
218 return texture->sampleCount();
219 if(msDepthTexture)
220 return msDepthTexture->sampleCount();
221 if(depthTexture)
222 return depthTexture->sampleCount();
223 // renderTarget alone without any owned attachment = swap-chain wrapper.
224 // Its sampleCount is unreliable pre-createOrResize; fall through.
225 return -1;
226 }
227
228 void release()
229 {
230 if(texture || dummyColorTexture || depthTexture)
231 {
232 // Use deleteLater() for all GPU resources: Qt RHI commands are async
233 // and resources may still be referenced by in-flight frames until
234 // endFrame() completes. deleteLater() defers actual destruction to
235 // the next beginFrame().
236 if(texture)
237 texture->deleteLater();
238 texture = nullptr;
239
240 if(dummyColorTexture)
241 dummyColorTexture->deleteLater();
242 dummyColorTexture = nullptr;
243
244 for(auto* t : additionalColorTextures)
245 t->deleteLater();
246 additionalColorTextures.clear();
247
248 if(colorRenderBuffer)
249 colorRenderBuffer->deleteLater();
250 colorRenderBuffer = nullptr;
251
252 for(auto* rb : additionalColorRenderBuffers)
253 rb->deleteLater();
254 additionalColorRenderBuffers.clear();
255
256 if(depthRenderBuffer)
257 depthRenderBuffer->deleteLater();
258 depthRenderBuffer = nullptr;
259
260 if(depthTexture)
261 depthTexture->deleteLater();
262 depthTexture = nullptr;
263
264 if(msDepthTexture)
265 msDepthTexture->deleteLater();
266 msDepthTexture = nullptr;
267
268 if(renderPass)
269 renderPass->deleteLater();
270 renderPass = nullptr;
271
272 if(renderTarget)
273 renderTarget->deleteLater();
274 renderTarget = nullptr;
275 }
276 }
277};
