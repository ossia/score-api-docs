22inline bool videoDecoderNeedsRebuild(
23 bool hasDecoder, const Video::ImageFormat& built, const Video::ImageFormat& src,
24 AVPixelFormat fmt, int w, int h) noexcept
25{
26 if(!hasDecoder)
27 return true;
28
29 // A fielded source hands over half-height frames while describing the whole
30 // picture, so the two heights are compared in the same units -- get this
31 // wrong and every frame looks like a size change.
32 const int pictureH = (src.interlacing == Video::Interlacing::Fields) ? h * 2 : h;
33
34 return fmt != built.pixel_format || w != built.width || pictureH != built.height
35 || src.output_format != built.output_format || src.tonemap != built.tonemap
36 // The colour description comes from the decoder, not the frame, and a
37 // live input may revise it while running -- a sender that starts
38 // declaring HDR. The matrix is baked into the shader, so that
39 // rebuilds.
40 || src.color_space != built.color_space || src.color_range != built.color_range
41 || src.color_trc != built.color_trc
42 || src.color_primaries != built.color_primaries
43 // Interlacing changes the texture geometry; the deinterlace MODE does
44 // not -- it is a uniform, so switching weave to bob must not rebuild.
45 || src.interlacing != built.interlacing;
46}
