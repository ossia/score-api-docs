189 static std::vector<std::string> captureGroups(const std::string& pattern)
190 {
191 std::vector<std::string> groups;
192 RE2::Options opts;
193 opts.set_log_errors(false);
194 RE2 re{pattern, opts};
195 if(re.ok())
196 {
197 const auto& names = re.CapturingGroupNames();
198 for(int i = 1; i <= re.NumberOfCapturingGroups(); i++)
199 {
200 auto it = names.find(i);
201 groups.push_back(it != names.end() ? it->second : std::string{});
202 }
203 return groups;
204 }
205
206 bool in_class = false;
207 for(std::size_t i = 0; i < pattern.size(); i++)
208 {
209 const char c = pattern[i];
210 if(c == '\\')
211 {
212 i++;
213 continue;
214 }
215 if(in_class)
216 {
217 in_class = c != ']';
218 continue;
219 }
220 if(c == '[')
221 {
222 in_class = true;
223 continue;
224 }
225 if(c != '(')
226 continue;
227 if(i + 1 < pattern.size() && pattern[i + 1] == '?')
228 {
229 // (?P<name>...) and (?<name>...) capture; (?:...), (?i)... do not.
230 std::size_t n = i + 2;
231 if(n < pattern.size() && pattern[n] == 'P')
232 n++;
233 if(n < pattern.size() && pattern[n] == '<')
234 {
235 const auto close = pattern.find('>', n + 1);
236 groups.push_back(
237 pattern.substr(n + 1, close == std::string::npos ? std::string::npos
238 : close - n - 1));
239 }
240 continue;
241 }
242 groups.emplace_back();
243 }
244 return groups;
245 }

249 static halp::string_list_value captureRows(const std::string& pattern)
250 {
251 const auto groups = captureGroups(pattern);
252 const int count = std::min<int>(groups.size(), max_capture_outlets);
253 halp::string_list_value rows;
254 rows.reserve(count);
255 std::vector<int> used;
256 for(int i = 1; i <= count; i++)
257 {
258 int key{};
259 const std::string& name = groups[i - 1];
260 if(!name.empty())
261 {
262 // FNV-1a of the name: the group keeps its outlet wherever it moves.
263 uint32_t h = 2166136261u;
264 for(unsigned char c : name)
265 h = (h ^ c) * 16777619u;
266 key = 20000 + int(h % 1'000'000u);
267 while(std::find(used.begin(), used.end(), key) != used.end())
268 key++;
269 }
270 else
271 {
272 key = 10000 + i;
273 }
274 used.push_back(key);
275 rows.emplace_back(key, name);
276 }
277 return rows;
278 }

281 static std::string toRE2Rewrite(std::string_view s)
282 {
283 std::string out;
284 out.reserve(s.size());
285 for(std::size_t i = 0; i < s.size(); i++)
286 {
287 const char c = s[i];
288 if((c == '$' || c == '%') && i + 1 < s.size())
289 {
290 const char n = s[i + 1];
291 if(n >= '0' && n <= '9')
292 {
293 out += '\\';
294 out += n;
295 i++;
296 continue;
297 }
298 if(n == c)
299 {
300 out += c;
301 i++;
302 continue;
303 }
304 }
305 out += c;
306 }
307 return out;
308 }

329 static Compiled build(const Spec& spec)
330 {
331 Compiled c;
332 RE2::Options opts;
333 opts.set_log_errors(false);
334 opts.set_case_sensitive(!spec.ignore_case);
335 auto re = std::make_unique<RE2>(
336 spec.multiline ? "(?m)" + spec.pattern : spec.pattern, opts);
337 if(!re->ok())
338 {
339 c.error = re->error();
340 return c;
341 }
342
343 std::string rw = toRE2Rewrite(spec.replacement);
344 if(spec.mode == Replace)
345 {
346 if(std::string why; !re->CheckRewriteString(rw, &why))
347 {
348 c.error = std::move(why);
349 return c;
350 }
351 }
352 c.submatches.assign(1 + re->NumberOfCapturingGroups(), {});
353 c.re = std::move(re);
354 c.rewrite = std::move(rw);
355 return c;

420 static void appendText(std::string& out, const ossia::value& v)
421 {
422 switch(v.get_type())
423 {
424 case ossia::val_type::STRING:
425 out += *v.target<std::string>();
426 break;
427 case ossia::val_type::INT:
428 out += std::to_string(*v.target<int>());
429 break;

432 break;

434 case ossia::val_type::BOOL:
435 out += *v.target<bool>() ? "true" : "false";

439 for(auto& e : *v.target<std::vector<ossia::value>>())
440 {
441 if(!first)
442 out += ' ';
443 first = false;
444 appendText(out, e);
445 }
446 break;
447 }
448 case ossia::val_type::VEC2F:
449 case ossia::val_type::VEC3F:
450 case ossia::val_type::VEC4F:
451 appendText(out, ossia::convert<std::vector<ossia::value>>(v));
452 break;
453 default:
454 break;
455 }

505 void process(const ossia::value& input, int64_t ts)
506 {
507 // The text buffers are reused from one input to the next
508 if(input.target<ossia::impulse>())
509 {
510 text = last_text;
511 }
512 else
513 {
514 text.clear();
515 appendText(text, input);
516 last_text = text;
517 }
518
519 const re2::StringPiece piece{text};
520 const int n = submatches.size();
521 bool found = false;
522
523 switch(inputs.mode.value)
524 {
525 case Search:
526 case Match: {
527 const auto anchor = inputs.mode.value == Match ? RE2::ANCHOR_BOTH : RE2::UNANCHORED;
528 found = re->Match(piece, 0, piece.size(), anchor, submatches.data(), n);
529 if(found)
530 {
531 write(outputs.match.value, std::string{submatches[0]}, ts);
532 write(outputs.groups.value, sendGroups(ts), ts);
533 }
534 break;
535 }
536 case SearchAll:
537 case Split: {
538 std::vector<ossia::value> matches, groups, pieces;
539 std::size_t pos = 0, last = 0;
540 while(pos <= piece.size()
541 && re->Match(piece, pos, piece.size(), RE2::UNANCHORED, submatches.data(), n))
542 {
543 found = true;
544 const auto& m = submatches[0];
545 const std::size_t begin = m.data() - piece.data();
546 const std::size_t end = begin + m.size();
547 if(inputs.mode.value == Split)
548 {
549 pieces.emplace_back(std::string{text.substr(last, begin - last)});
550 }
551 else
552 {
553 matches.emplace_back(std::string{m});
554 groups.emplace_back(sendGroups(ts));
555 }
556 last = end;
557 // An empty match moves on by one character, or it would match again
558 pos = end > begin ? end : end + 1;
559 }
560 if(!found)
561 break;
562 if(inputs.mode.value == Split)
563 {
564 pieces.emplace_back(std::string{text.substr(std::min(last, text.size()))});
565 write(outputs.match.value, std::move(pieces), ts);
566 }
567 else
568 {
569 write(outputs.match.value, std::move(matches), ts);
570 write(outputs.groups.value, std::move(groups), ts);
571 }
572 break;
573 }
574 case Replace: {
575 std::string result = text;
576 found = inputs.global.value ? RE2::GlobalReplace(&result, *re, rewrite) > 0
577 : RE2::Replace(&result, *re, rewrite);
578 if(found)
579 write(outputs.match.value, std::move(result), ts);
580 break;
581 }
582 }
583
584 write(outputs.matched.value, found, ts);
585 if(!found)
586 write(outputs.unmatched.value, input, ts);
587 }
