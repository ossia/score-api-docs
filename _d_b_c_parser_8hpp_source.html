199inline uint64_t extractRawBits(
200 const uint8_t* data, int dataSize, int startBit, int length,
201 ByteOrder order) noexcept
202{
203 if(length <= 0 || length > 64)
204 return 0;
205
206 const int frameBits = dataSize * 8;
207 uint64_t out = 0;
208
209 // Walk the value from its MSB down to its LSB, so that each step is a plain
210 // shift-in. `k` is the significance of the bit being read.
211 for(int k = length - 1; k >= 0; --k)
212 {
213 int pos;
214 if(order == ByteOrder::LittleEndian)
215 {
216 // The start bit is the LSB and significance grows with the position.
217 pos = startBit + k;
218 }
219 else
220 {
221 // The start bit is the MSB; significance grows *backwards* along the
222 // flipped axis, on which the signal's bits are contiguous.
223 pos = flipBitPos(flipBitPos(startBit) + (length - 1 - k));
224 }
225
226 out <<= 1;
227 if(pos >= 0 && pos < frameBits)
228 out |= uint64_t((data[pos >> 3] >> (pos & 7)) & 1u);
229 }
230
231 return out;
232}

235constexpr int64_t signExtend(uint64_t raw, int length) noexcept
236{
237 if(length <= 0 || length >= 64)
238 return int64_t(raw);
239
240 const uint64_t signBit = uint64_t(1) << (length - 1);
241 if(raw & signBit)
242 return int64_t(raw | ~((uint64_t(1) << length) - 1));
243 return int64_t(raw);
244}

247inline double decodeSignal(const Signal& sig, const uint8_t* data, int dataSize) noexcept
248{
249 const uint64_t raw
250 = extractRawBits(data, dataSize, sig.startBit, sig.length, sig.byteOrder);
251
252 double physical{};
253 switch(sig.valueType)
254 {
255 case ValueType::Float32: {
256 const auto bits = uint32_t(raw);
257 float f{};
258 std::memcpy(&f, &bits, sizeof(f));
259 physical = double(f);
260 break;
261 }
262 case ValueType::Double64: {
263 double d{};
264 std::memcpy(&d, &raw, sizeof(d));
265 physical = d;
266 break;
267 }
268 case ValueType::Signed:
269 physical = double(signExtend(raw, sig.length));
270 break;
271 case ValueType::Unsigned:
272 physical = double(raw);
273 break;
274 }
275
276 return physical * sig.factor + sig.offset;
277}
