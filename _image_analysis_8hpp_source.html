103inline Mat3 rgbToXyz(const Primaries& p) noexcept
104{
105 auto xyz = [](double x, double y) { return std::array<double, 3>{x / y, 1., (1. - x - y) / y}; };
106 const auto r = xyz(p.rx, p.ry), g = xyz(p.gx, p.gy), b = xyz(p.bx, p.by), w = xyz(p.wx, p.wy);
107 const Mat3 m{{{r[0], g[0], b[0]}, {r[1], g[1], b[1]}, {r[2], g[2], b[2]}}};
108 // Scale each primary so that they sum to the white point
109 const double det = m[0][0] * (m[1][1] * m[2][2] - m[1][2] * m[2][1])
110 - m[0][1] * (m[1][0] * m[2][2] - m[1][2] * m[2][0])
111 + m[0][2] * (m[1][0] * m[2][1] - m[1][1] * m[2][0]);
112 Mat3 inv{};
113 inv[0][0] = (m[1][1] * m[2][2] - m[1][2] * m[2][1]) / det;
114 inv[0][1] = (m[0][2] * m[2][1] - m[0][1] * m[2][2]) / det;
115 inv[0][2] = (m[0][1] * m[1][2] - m[0][2] * m[1][1]) / det;
116 inv[1][0] = (m[1][2] * m[2][0] - m[1][0] * m[2][2]) / det;
117 inv[1][1] = (m[0][0] * m[2][2] - m[0][2] * m[2][0]) / det;
118 inv[1][2] = (m[0][2] * m[1][0] - m[0][0] * m[1][2]) / det;
119 inv[2][0] = (m[1][0] * m[2][1] - m[1][1] * m[2][0]) / det;
120 inv[2][1] = (m[0][1] * m[2][0] - m[0][0] * m[2][1]) / det;
121 inv[2][2] = (m[0][0] * m[1][1] - m[0][1] * m[1][0]) / det;
122 std::array<double, 3> s{};
123 for(int i = 0; i < 3; i++)
124 s[i] = inv[i][0] * w[0] + inv[i][1] * w[1] + inv[i][2] * w[2];
125 Mat3 out{};
126 for(int i = 0; i < 3; i++)
127 for(int j = 0; j < 3; j++)
128 out[i][j] = m[i][j] * s[j];
129 return out;
130}
