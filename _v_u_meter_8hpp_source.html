81inline double toScale(Mode m, double lin, double reference) noexcept
82{
83 const double db = ossia::detail::LinearGainToDecibels(std::max(lin, 1e-10));
84 switch(m)
85 {
86 case Mode::VU:
87 case Mode::PpmNordic:
88 case Mode::PpmEbu:
89 return db - reference;
90 case Mode::PpmDin:
91 // Alignment level reads -9 on a DIN meter
92 return db - reference - 9.;
93 case Mode::PpmBbc:
94 // Marks 1 to 7, 4 dB apart, 4 at alignment level
95 return 4. + (db - reference) / 4.;
96 default:
97 return db + kOffset(m);
98 }
99}

282 void set_ballistics() noexcept
283 {

285 for(auto& c : channels)

296 void operator()(int frames)
297 {
298 const int nch = std::min(inputs.audio.channels, max_channels);
299 if(nch == 0 || frames <= 0)
300 return;
301
302 if(inputs.mode.value != current)
303 {
304 current = inputs.mode.value;
305 reset_followers();
306 }
307 else if(inputs.hold.value != current_hold || inputs.fall.value != current_fall)
308 {
309 set_ballistics();
310 }
311 if(inputs.reset || monitor->reset_clips.exchange(false, std::memory_order_relaxed))
312 clipped = 0;
313
314 const double dc_coef = 1. - std::exp(-frames / (1. * rate));
315
316 for(int c = 0; c < nch; c++)
317 {
318 auto& ch = channels[c];
319 const double* in = inputs.audio[c];
320 ch.peak.process(in, frames);
321 double block_peak = 0;
322 for(int i = 0; i < frames; i++)
323 block_peak = std::max(block_peak, std::abs(in[i]));
324
325 float level{}, secondary{};
326 double clip_level = block_peak;
327 switch(current)
328 {
329 case Mode::PeakRms:
330 case Mode::Surround:
331 ch.rms.process(in, frames);
332 level = ch.peak.level;
333 secondary = ch.rms.level();
334 ch.hold.feed(block_peak, frames);
335 break;
336 case Mode::TruePeak:
337 ch.tp.process(in, frames);
338 ch.tp_level = std::max(ch.tp.block_peak, ch.tp_level * std::pow(tp_fall, frames));
339 level = ch.tp_level;
340 secondary = ch.peak.level;
341 clip_level = ch.tp.block_peak;
342 ch.hold.feed(ch.tp.block_peak, frames);
343 break;
344 case Mode::VU:
345 ch.vu.process(in, frames);
346 level = ch.vu.level();
347 ch.hold.feed(level, frames);
348 break;
349 case Mode::PpmDin:
350 case Mode::PpmNordic:
351 case Mode::PpmBbc:
352 case Mode::PpmEbu:
353 ch.ppm.process(in, frames);
354 level = ch.ppm.level();
355 ch.hold.feed(level, frames);
356 break;
357 case Mode::K20:
358 case Mode::K14:
359 case Mode::K12:
360 ch.rms.process(in, frames);
361 level = ch.rms.level();
362 secondary = ch.peak.level;
363 ch.hold.feed(block_peak, frames);
364 break;
365 case Mode::Bits: {
366 level = ch.peak.level;
367 double sum = 0;
368 uint32_t bits = 0;
369 for(int i = 0; i < frames; i++)
370 {
371 sum += in[i];
372 const auto q = (int64_t)std::llround(std::clamp(in[i], -1., 1.) * 8388607.);
373 bits |= uint32_t(q < 0 ? -q : q);
374 }
375 ch.bits = bits;
376 ch.dc += dc_coef * (sum / frames - ch.dc);
377 ch.hold.feed(block_peak, frames);
378 break;
379 }
380 }
381 if(clip_level >= 1.)
382 clipped |= (uint64_t(1) << c);
383
384 frame.level[c] = level;
385 frame.secondary[c] = secondary;
386 frame.hold[c] = ch.hold.value;
387 frame.bits[c] = ch.bits;
388 frame.dc[c] = ch.dc;
389 }
390
391 if(current == Mode::Surround)
392 {
393 const auto r = Ui::Monitor::ring(inputs.layout.value, nch);
394 for(int i = 0; i < r.size; i++)
395 {
396 const int a = r.channel[i], b = r.channel[(i + 1) % r.size];
397 auto& corr = channels[a].ring_corr;
398 corr.process(inputs.audio[a], inputs.audio[b], frames);
399 frame.ring_correlation[i] = corr.value();
400 }
401 }
402
403 frame.channels = nch;
404 frame.mode = current;
405 frame.clipped = clipped;
406 if(monitor->watched())
407 monitor->frames.produce(frame);
408
409 outputs.clip.value = clipped != 0;
410
411 // The levels are for patching: a few lists a second are plenty, and
412 // each one is a new message
413 since_output += frames;
414 if(since_output >= rate / 30.)
415 {
416 since_output = 0;
417 auto& out = outputs.levels.value.emplace(nch);
418 for(int c = 0; c < nch; c++)
419 out[c] = float(toScale(current, frame.level[c], inputs.reference.value));
420 }
421 }
