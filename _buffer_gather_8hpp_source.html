107constexpr GatherPlan planGather(
108 const Layout& l, int64_t origin, int64_t buffer_bytes, const Sampling& s, bool reduce,
109 int64_t max_bind, int max_groups_per_dim) noexcept
110{
111 GatherPlan p;
112 if(s.count <= 0 || origin < 0)
113 return p;
114 constexpr int64_t u32 = std::numeric_limits<uint32_t>::max();
115 const int64_t e = elementBytes(l);
116 const int64_t stride = strideOf(l);
117 const int64_t start = origin + s.first * stride;
118 const int64_t sample_stride = s.count > 1 ? s.step * stride : 0;
119 int64_t end = start + (s.count - 1) * sample_stride + e;
120 const int64_t reduce_count = reduce ? s.window : 0;
121 if(reduce_count > gather_max_reduce)
122 return p;
123 if(reduce)
124 end = std::max(end, start + (reduce_count - 1) * stride + e);
125 const int64_t readable = gatherReadable(buffer_bytes);
126 if(end > readable || buffer_bytes > u32 || sample_stride > u32)
127 return p;
128
129 p.bind_offset = start / gather_bind_alignment * gather_bind_alignment;
130 p.bind_size = std::min(readable, (end + 3) / 4 * 4) - p.bind_offset;
131 if(p.bind_size > max_bind)
132 return p;
133
134 const int64_t out_words = (s.count * e + 3) / 4;
135 auto& g = p.params;
136 g.base = uint32_t(start - p.bind_offset);
137 g.sample_stride = uint32_t(sample_stride);
138 g.element_bytes = uint32_t(e);
139 g.out_words = uint32_t(out_words);
140 g.src_words = uint32_t(p.bind_size / 4);
141 g.reduce_count = uint32_t(reduce_count);
142 g.reduce_base = g.base;
143 g.reduce_stride = uint32_t(stride);
144 g.type = uint32_t(l.type);
145 g.components = uint32_t(components(l));
146 g.component_bytes = uint32_t(componentBytes(l.type));
147
148 p.staging_bytes = gather_header_bytes + out_words * 4;
149 p.readback_offset = reduce ? 0 : gather_header_bytes;
150 p.readback_size = p.staging_bytes - p.readback_offset;
151
152 // Wider than a dimension allows: the rows of groups go on y
153 const int64_t invocations = std::max(out_words, reduce_count);
154 const int64_t groups = std::clamp<int64_t>(
155 (invocations + gather_local_size - 1) / gather_local_size, 1, gather_max_groups);
156 const int64_t max_dim = std::max(1, max_groups_per_dim);
157 p.groups_x = int(std::min(groups, max_dim));
158 p.groups_y = int((groups + p.groups_x - 1) / p.groups_x);
159 p.valid = true;
160 return p;
161}

165constexpr uint32_t gatherBytesAt(std::span<const uint32_t> words, uint32_t b, uint32_t n) noexcept
166{
167 auto word = [&](uint32_t w) { return w < words.size() ? words[w] : 0u; };
168 const uint32_t w = b >> 2;
169 const uint32_t shift = (b & 3u) * 8u;
170 uint32_t v = word(w) >> shift;
171 if(shift + n * 8u > 32u)
172 v |= word(w + 1u) << (32u - shift);
173 return n == 4u ? v : (v & ((1u << (n * 8u)) - 1u));
174}

177constexpr uint32_t
178gatherWord(const GatherParams& g, std::span<const uint32_t> words, uint32_t w) noexcept
179{
180 const uint32_t b = w * 4u;
181 if(g.element_bytes % 4u == 0u)
182 return gatherBytesAt(
183 words, g.base + (b / g.element_bytes) * g.sample_stride + b % g.element_bytes, 4u);
184 uint32_t v = 0;
185 for(uint32_t i = 0; i < 4u; i++)
186 {
187 const uint32_t at = b + i;
188 v |= gatherBytesAt(
189 words, g.base + (at / g.element_bytes) * g.sample_stride + at % g.element_bytes,
190 1u)
191 << (i * 8u);
192 }
193 return v;
194}

216inline GatherRange readGatherHeader(std::span<const uint32_t> header, int components) noexcept
217{
218 GatherRange r;
219 r.lo.fill(std::numeric_limits<float>::max());
220 r.hi.fill(std::numeric_limits<float>::lowest());
221 if(header.size() < std::size_t(gather_header_words))
222 return r;
223 for(int c = 0; c < std::clamp(components, 1, 4); c++)
224 {
225 // The values the header is reset to: nothing finite was found
226 if(header[c] == 0xffffffffu && header[4 + c] == 0u)
227 continue;
228 r.lo[c] = unorderedFloat(header[c]);
229 r.hi[c] = unorderedFloat(header[4 + c]);
230 }
231 r.non_finite = header[8];
232 return r;
233}
