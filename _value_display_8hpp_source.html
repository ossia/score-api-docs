60 struct
61 {
62 // Reading the port directly rather than through a control input: a control
63 // input is fed with get_data().back(), so only the last value of a tick
64 // would ever be seen, and a burst - a note-off immediately followed by a
65 // note-on, several values written at the same sample - would show up as a
66 // single entry.
67 raw_port<"in", ossia::value_port> port;
68
69 halp::spinbox_i32<"Log", halp::range{1, max_log, 1}> log;

71 // Spread nested lists and maps over indented lines instead of a single one.
72 halp::toggle<"Pretty"> pretty;

77 // [sequence number of the first entry, values...]
78 struct : halp::val_port<"values", std::optional<ossia::value>>
79 {
80 enum widget
81 {
82 control

85 } outputs;
86
87 // The engine -> UI queue for control outputs is drained keeping only the last
88 // entry, so a push has to carry everything the layer may still need. That is
89 // never more than the log length, which bounds the ring exactly.
90 boost::container::devector<ossia::value> m_ring;
91 int m_next_seq = 0;
92
93 void prepare(halp::setup) noexcept
94 {
95 m_ring.clear();
96 m_next_seq = 0;
97 }
98
99 void operator()()
100 {
101 outputs.values.value.reset();
102
103 if(!inputs.port.value)
104 return;
105
106 const auto& data = inputs.port.value->get_data();
107 if(data.empty())
108 return;
109
110 // Never more than what the layer displays: with a log of 1 this sends the
111 // last value and nothing else.
112 const int keep = std::clamp(inputs.log.value, 1, max_log);
113
114 for(const ossia::timed_value& tv : data)
115 {
116 m_ring.push_back(tv.value);
117 m_next_seq++;
118 if(std::ssize(m_ring) > keep)
119 m_ring.pop_front();
120 }
121 while(std::ssize(m_ring) > keep)
122 m_ring.pop_front();
123
124 std::vector<ossia::value> payload;
125 payload.reserve(1 + m_ring.size());
126 payload.push_back(int(m_next_seq - std::ssize(m_ring)));
127 for(const auto& v : m_ring)
128 payload.push_back(v);
129
130 outputs.values.value = ossia::value{std::move(payload)};
131 }

219 void on_values(const ossia::value& v)
220 {
221 auto* list = v.target<std::vector<ossia::value>>();
222 if(!list || list->size() < 2)
223 return;
224
225 const int base = ossia::convert<int>((*list)[0]);
226 const int n = std::ssize(*list) - 1;
227
228 // The batch is entirely older than what we hold: the process restarted.
229 if(base + n < m_next_seq)
230 {
231 values.clear();
232 m_next_seq = base;
233 }
234
235 for(int i = 0; i < n; i++)
236 {
237 const int seq = base + i;
238 if(seq < m_next_seq)
239 continue;
240 values.push_front((*list)[i + 1]);
241 m_next_seq = seq + 1;
242 }
243
244 while(std::ssize(values) > logging())
245 values.pop_back();
246
247 rebuildText();
248 }
