46struct EdgeSpec
47{
48 port_index first{};
49 port_index second{};
50 Process::CableType type{};
51
52 // type takes part in identity: the exec thread republishes the full edge
53 // set every tick and endTick dedups against prev_edges, so a comparison that
54 // ignored the cable type would make an Immediate/Delayed flip invisible. As
55 // it stands a type flip diffs as remove(old) + add(new) in the incremental
56 // path.
57 bool operator==(const EdgeSpec& other) const noexcept
58 {
59 return first == other.first && second == other.second && type == other.type;
60 }
61 bool operator!=(const EdgeSpec& other) const noexcept { return !(*this == other); }
62 bool operator<(const EdgeSpec& other) const noexcept
63 {
64 if(first != other.first)
65 return first < other.first;
66 if(second != other.second)
67 return second < other.second;
68 return type < other.type;
69 }
70};
