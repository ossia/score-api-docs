138struct capture_window
139{
140 int64_t end{};
141 int64_t n{};
142 int64_t length{};
143 double y_end{};
144 double rate{1.};
145
146 static int64_t wrap(int64_t k, int64_t len) noexcept
147 {
148 k %= len;
149 return k < 0 ? k + len : k;
150 }
152 int64_t positions() const noexcept
153 {
154 return std::clamp<int64_t>(std::llround(double(n) * rate), 0, length);
155 }
157 int64_t start() const noexcept
158 {
159 return wrap(int64_t(std::floor(y_end)) - positions(), length);
160 }
162 int64_t position(int64_t t) const noexcept
163 {
164 double y = std::fmod(y_end - double(end - t) * rate, double(length));
165 if(y < 0.)
166 y += double(length);
167 return std::min(int64_t(y), length - 1);
168 }
170 int64_t progress(int64_t t) const noexcept
171 {
172 return std::clamp<int64_t>(int64_t(double(t - (end - n)) * rate), 0, positions());
173 }
174 bool contains(int64_t t) const noexcept { return t >= end - n && t < end; }
175};

465template <typename Node>
466void looper_node<Node>::run_tick(const tick& tk)
467{
468 auto& in = self().inputs;
469 auto& st = stack();
470 const auto state = self().ossia_state;
471 m_rate = state.sampleRate();
472 m_tempo = tk.tempo;
473 m_signature = tk.signature;
474 m_speed = Looper::clamp_speed(in.speed.value);
475 m_reverse = in.reverse.value;
476 self().tick_started(tk);
477
478 apply_buttons();
479 follow_merges();
480 if(in.capture.value)
481 capture(tk);
482 self().prepare_loop();
483
484 const bool empty = st.empty() || self().busy();
485 Looper::read_loop(in.mode, transport, m_loop, empty);
486 m_switches.read(in, transport, self(), empty);
487 transport.settle(empty);
488
489 m_mode = transport.wanted;
490 const Looper::transport_settings settings{
491 .mode = m_mode,
492 .quantif = in.quantif.value,
493 .after_record = in.after_record,
494 .postaction_bars = in.postaction_bars,
495 .start_from = in.start_from,
496 .late_seconds = in.late.value / 1000.,
497 .stop_at = in.stop_at.value};
498 transport(tk, state, settings, self());
499 // Only a pass that starts with the take closing gets what it cut off.
500 self().forget_carried();
501 Looper::show_loop(in.mode, transport, m_loop);
502 follow_merges();
503 m_switches.follow(in, st.count);
504
505 self().tick_finished(tk);
506 follow_merges();
507 run_export();
508
509 self().outputs.phase.value
510 = st.length > 0 ? float(transport.playbackPos % st.length) / float(st.length) : 0.f;
511}

513template <typename Node>
514void looper_node<Node>::apply_buttons()
515{
516 auto& in = self().inputs;
517 if(in.clear.value)
518 clear_loop();
519
520 // The m and x of a lane are layer actions as any: on the grid.
521 for(int i = 0, n = std::exchange(m_erase_count, 0); i < n; i++)
522 {
523 const auto& r = m_erase_requests[std::size_t(i)];
524 m_switches.add(
525 {.kind = r.stopped >= 0 ? Looper::layer_action::stop
526 : r.mute ? Looper::layer_action::mute
527 : Looper::layer_action::erase,
528 .pass = r.pass,
529 .layers = r.layers,
530 .id = r.id,
531 .muted = r.muted,
532 .stopped = r.stopped},
533 transport);
534 }
535
536 if(in.undo.value || in.redo.value)
537 m_switches.soloed.forget();
538 // An m, an x or a layer action still waiting for the grid was done last:
539 // undo takes it back first. While a pass is taken in, undo drops it, unless
540 // what was done last is an action, done since it started.
541 if(in.undo.value && !m_switches.withdraw_last(undo_mark()))
542 {
543 if(pass_open() && !stack().undoes_action())
544 drop_taken_pass();
545 else
546 {
547 passes_changing();
548 undo_pass();
549 }
550 }
551
552 if(in.redo.value && m_switches.may_press_again(undo_mark()))
553 m_switches.press_again(transport);
554 else if(in.redo.value && may_redo())
555 {
556 passes_changing();
557 const auto step = stack().redo();
558 if(step.done)
559 {
560 if(step.removed >= 0)
561 m_switches.removed(in, step.removed);
562 merge_extra_pass();
563 }
564 }
565}

746template <typename Node>
747void looper_node<Node>::close_pass(bool cycle)
748{
749 // Nothing came in: there is no pass to close.
750 if(std::exchange(m_pass_pending, false))
751 {
752 if(!cycle)
753 m_extended = false;
754 return;
755 }
756 auto* top = stack().top();
757 if(!top)
758 return;
759 // A loop grown by the cycle that ended can still be cut back by a late
760 // stop.
761 if(!cycle)
762 m_extended = false;
763 m_version++;
764 m_export_pending = true;

766 return;

768 self().pass_closed();

783template <typename Node>
784void looper_node<Node>::looper_close_take(
785 const ossia::token_request& tk, LoopMode mode, bool ended_by_postaction)
786{
787 auto& st = stack();
788 if(st.count == 0)
789 return;
790
791 if(mode == LoopMode::Record)
792 {
793 m_loop_tempo = transport.loop_tempo(tk, st.length);
794 const double bar_frames = m_rate * 4. * (double(tk.signature.upper) / tk.signature.lower)
795 * (60. / tk.tempo);
796 const auto bars = int64_t(std::floor(transport.postaction_bars * bar_frames));
797
798 // The bar count says when the post-action takes over, so it is the length
799 // of a recording the post-action ended: what was recorded past the bars,
800 // e.g. the tempo having dropped while it went on, is cut off as a late
801 // stop cuts it, for the overdub that follows. A recording stopped by hand
802 // is as long as it was played for.
803 const bool to_bars = ended_by_postaction && transport.quantized()
804 && transport.postaction_bars > 0 && bars > 0;
805 if(to_bars && bars < st.length)
806 self().looper_trim_recording(st.length - bars);
807 self().close_recording(to_bars ? bars : 0);
808 m_export_date = transport.recordStartDate.impl / double(ossia::flicks_per_second<int64_t>);
809
810 // Position 0 of the loop on the timeline: the loop plays on from
811 // playbackPos at the point the take closes.
812 const auto at = double(transport.playbackPos % std::max<int64_t>(1, st.length));
813 m_origin = transport.timeline(tk);
814 m_origin.frames -= at;
815 m_origin.quarters -= at / m_rate * m_loop_tempo / 60.;
816 // The transport counts how far the loop went on since it closed, e.g.
817 // from the grid point a late stop cut it back to: backwards, that is read
818 // from its end, and a pass that goes on from the recording is as far
819 // into the loop.
820 if(m_reverse && st.length > 0)
821 self().set_head(at, st.length);
822 }
823 else
824 {
825 // An overdub that got nothing leaves no take.
826 if(m_pass_pending)
827 {
828 self().close_overdub();
829 m_extended = false;
830 return;
831 }
832 self().close_overdub();
833 }
834 m_version++;
835 m_extended = false;
836 m_export_pending = true;
837}

841template <typename Node>
842int64_t looper_node<Node>::looper_trim_extension(int64_t max_frames)
843{
844 auto& st = stack();
845 if(!m_extended || m_extend_unit <= 0 || st.count == 0)
846 return -1;
847 const int64_t end = st.length - m_extend_unit;
848 const int64_t into_last = transport.playbackPos - end;
849 if(into_last < 0 || into_last > max_frames || end < m_extend_unit)
850 return -1;
851 if(m_pass_pending)
852 {
853 // The pass after the cut got nothing: it was never a layer.
854 m_pass_pending = false;
855 self().cut_pass_dropped();
856 st.length = end;
857 if(auto* t = st.top())
858 t->loop_after = end;
859 }
860 else if(st.count > 1 && self().cut_after(end))
861 {
862 // A pass per cycle cut the overdub on the old end: the pass after the cut
863 // has nothing left, and goes as undo takes a pass out. The one before it
864 // ends on the old end, closed already.
865 self().pass_dropped();
866 drop_top_pass();
867 self().cut_pass_dropped();
868 st.length = end;
869 if(auto* t = st.top())
870 t->loop_after = end;
871 }
872 else
873 {
874 self().trim_top_pass(end);
875 st.length = end;
876 }
877 m_version++;
878 m_extended = false;
879 self().head_moved(into_last);
880 return into_last;
881}

901template <typename Node>
902Looper::loop_period
903looper_node<Node>::looper_period(const ossia::token_request& tk) const noexcept
904{
905 const LoopMode m = transport.actualMode;
906 const auto& st = stack();
907 if(st.empty() || self().busy() || st.length <= 0 || m == LoopMode::Record)
908 return {};
909 const double rate = self().rate_for(m);
910 const int64_t round = Looper::loop_round(st.length, rate);
911 if(m == LoopMode::Stop)
912 return {
913 round, int64_t(std::floor(rounds_since_made(tk) * double(round) + 1e-6)), false};

935template <typename Node>
936void looper_node<Node>::capture(const tick& tk)
937{
938 if(transport.is_taking() || self().busy())
939 return;
940 auto& in = self().inputs;
941 const double bar = m_rate * 4. * (double(tk.signature.upper) / tk.signature.lower)
942 * (60. / std::max(1., tk.tempo));
943 const int64_t now = self().capture_clock();
944 int64_t end = now;
945 if(transport.quantized() && transport.lastPointFrame)
946 {
947 const int64_t late = transport.frame - *transport.lastPointFrame;
948 if(late >= 0 && late <= int64_t(in.late.value * m_rate / 1000.))
949 end -= late;
950 }
951 if(!stack().empty())
952 {
953 capture_layer(tk, now, end);
954 return;
955 }
956 const int64_t n = std::min<int64_t>(
957 {int64_t(in.capture_bars.value * bar), end, self().capture_limit()});
958 if(n <= 0)
959 return;
960
961 passes_changing();
962 m_switches.soloed.forget();
963 self().capture_into(end, n);
964
965 transport.play_captured((now - end) % n);
966 m_loop_tempo = tk.tempo;
967 m_origin = Looper::timeline_start(tk, self().ossia_state);
968 m_origin.frames -= double(transport.playbackPos);
969 m_origin.quarters -= double(transport.playbackPos) / m_rate * tk.tempo / 60.;
970 const double model_per_frame = self().ossia_state.samplesToModel();
971 m_export_date = (tk.prev_date.impl - (now - end + n) * model_per_frame)
972 / double(ossia::flicks_per_second<int64_t>);
973 m_export_pending = true;
974 self().captured();
975}

983template <typename Node>
984void looper_node<Node>::capture_layer(const tick& tk, int64_t now, int64_t end)
985{
986 auto& st = stack();
987 const int64_t len = st.length;
988 const LoopMode m = transport.actualMode;
989 const bool stopped = m != LoopMode::Play;
990 const double rate = self().rate_for(m);
991 if(len <= 0 || !(rate > 0.))
992 return;
993 double y_now{};
994 if(stopped)
995 {
996 const double rounds = rounds_since_made(tk);
997 y_now = std::floor((rounds - std::floor(rounds)) * double(len) + 1e-6);
998 }
999 else
1000 y_now = self().head_y(len);
1001
1002 const auto round = int64_t(std::ceil(double(len) / rate - 1e-9));
1003 double y_end = std::fmod(y_now - double(now - end) * rate, double(len));
1004 if(y_end < 0.)
1005 y_end += double(len);
1006 Looper::capture_window w{
1007 .end = end,
1008 .n = std::min<int64_t>({round, end, self().capture_limit()}),
1009 .length = len,
1010 .y_end = y_end,
1011 .rate = rate};
1012 if(w.n <= 0 || w.positions() <= 0 || !self().capture_layer_ready(w))
1013 return;
1014
1015 passes_changing();
1016 const auto pushes = st.pushes;
1017 begin_pass_now(w.start());
1018 if(st.pushes == pushes)
1019 return;
1020 self().capture_layer_into(w);
1021 close_pass();
1022
1023 if(stopped)
1024 {
1025 self().set_head(y_now, len);
1026 transport.play_captured(transport.playbackPos);
1027 self().captured();
1028 }

1091 // Nothing to take: the last take is what there is.
1092 m_view.answered = m_take_request;
1093 m_view_countdown = 0;
1094 }
1095 // Passes asked for alone, of the loop the view showed: one that changed
1096 // since has nothing to give.
1097 auto& part = m_part_request;
1098 const bool part_asked = part.take > m_view.part_answered;
1099 if(part_asked
1100 && (st.empty() || part.layers != st.count || part.pass < 0 || part.last < part.pass
1101 || part.last >= st.count))
1102 {
1103 m_view.part_answered = part.take;
1104 part.take = 0;
1105 m_view_countdown = 0;
1106 }
1107 else if(
1108 part_asked && !m_writing && self().writer && self().export_ready())
1109 {
1110 // The files of the passes as they are now, the one taken in as far as it
1111 // went, under a take number of their own.
1112 m_job_part = true;
1113 m_job_request = part.take;
1114 part.take = 0;

1117 m_job_stretch = self().take_stretch();
1118 m_export_tempo = recording ? m_tempo : m_loop_tempo;
1119 m_export_signature = m_signature;
1120 m_export_mix = false;
1121 m_export_first = part.pass;
1122 m_export_layers = part.last + 1;
1123 m_export_number = 0;
1124 m_export_slot = claim_take_slot();
1125 self().start_export(st.count, open, recording);
1126 }
1127 else if(
1128 (m_export_pending || asked) && !m_writing && self().writer && !st.empty()
1129 && self().export_ready())
1130 {
1131 m_export_pending = false;
1132 m_job_part = false;
1133 m_job_request = m_take_request;
1134 m_job_version = m_version;
1135 const bool open = asked && pass_open();

1137 if(recording)

1139 m_job_date = m_export_date;
1140 m_job_stretch = self().take_stretch();
1141 m_export_tempo = recording ? m_tempo : m_loop_tempo;
1142 m_export_signature = m_signature;

1154 }
