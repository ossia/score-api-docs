125struct capture_window
126{
127 int64_t end{};
128 int64_t n{};
129 int64_t length{};
130 double y_end{};
131 double rate{1.};
132 bool backwards{};
133
134 static int64_t wrap(int64_t k, int64_t len) noexcept
135 {
136 k %= len;
137 return k < 0 ? k + len : k;
138 }
140 int64_t positions() const noexcept
141 {
142 return std::clamp<int64_t>(std::llround(double(n) * rate), 0, length);
143 }
145 int64_t start() const noexcept
146 {
147 const auto y = int64_t(std::floor(y_end));
148 return backwards ? wrap(length - y, length) : wrap(y - positions(), length);
149 }
151 int64_t position(int64_t t) const noexcept
152 {
153 double y = std::fmod(y_end - double(end - t) * rate, double(length));
154 if(y < 0.)
155 y += double(length);
156 const int64_t k = std::min(int64_t(y), length - 1);
157 return backwards ? wrap(length - k, length) : k;
158 }
160 int64_t progress(int64_t t) const noexcept
161 {
162 return std::clamp<int64_t>(int64_t(double(t - (end - n)) * rate), 0, positions());
163 }
164 bool contains(int64_t t) const noexcept { return t >= end - n && t < end; }
165};

440template <typename Node>
441void looper_node<Node>::run_tick(const tick& tk)
442{
443 auto& in = self().inputs;
444 auto& st = stack();
445 const auto state = self().ossia_state;
446 m_rate = state.sampleRate();
447 m_tempo = tk.tempo;
448 m_signature = tk.signature;
449 m_speed = Looper::clamp_speed(in.speed.value);
450 m_reverse = in.reverse.value;
451 self().tick_started(tk);
452
453 apply_buttons();
454 follow_merges();
455 if(in.capture.value)
456 capture(tk);
457 self().prepare_loop();
458
459 const bool empty = st.empty() || self().busy();
460 Looper::read_loop(in.mode, transport, m_loop, empty);
461 m_switches.read(in, transport, self(), empty);
462 transport.settle(empty);
463
464 m_mode = transport.wanted;
465 const Looper::transport_settings settings{
466 .mode = m_mode,
467 .quantif = in.quantif.value,
468 .after_record = in.after_record,
469 .postaction_bars = in.postaction_bars,
470 .start_from = in.start_from,
471 .late_seconds = in.late.value / 1000.,
472 .stop_at = in.stop_at.value};
473 transport(tk, state, settings, self());
474 // Only a pass that starts with the take closing gets what it cut off.
475 self().forget_carried();
476 Looper::show_loop(in.mode, transport, m_loop);
477 follow_merges();
478 m_switches.follow(in, st.count);
479
480 self().tick_finished(tk);
481 follow_merges();
482 run_export();
483
484 self().outputs.phase.value
485 = st.length > 0 ? float(transport.playbackPos % st.length) / float(st.length) : 0.f;
486}

488template <typename Node>
489void looper_node<Node>::apply_buttons()
490{
491 auto& in = self().inputs;
492 if(in.clear.value)
493 clear_loop();
494
495 // The m and x of a lane are layer actions as any: on the grid.
496 if(std::exchange(m_erase_requested, false))
497 m_switches.add(
498 {m_erase_request.mute ? Looper::layer_action::mute : Looper::layer_action::erase,
499 m_erase_request.pass, m_erase_request.layers},
500 transport);
501
502 if(in.undo.value || in.redo.value)
503 m_switches.soloed.forget();
504 if(in.undo.value)
505 {
506 if(pass_open())
507 drop_taken_pass();
508 else
509 {
510 passes_changing();
511 undo_pass();
512 }
513 }
514
515 if(in.redo.value && !pass_open())
516 {
517 passes_changing();
518 const int out = stack().leaves_on_redo();
519 if(stack().redo())
520 {
521 if(out >= 0)
522 m_switches.removed(in, out);
523 merge_extra_pass();
524 }
525 }

615{
616 auto& st = stack();

663template <typename Node>
664void looper_node<Node>::close_pass(bool cycle)
665{
666 // Nothing came in: there is no pass to close.
667 if(std::exchange(m_pass_pending, false))
668 {
669 if(!cycle)
670 m_extended = false;
671 return;
672 }
673 auto* top = stack().top();
674 if(!top)
675 return;
676 // A loop grown by the cycle that ended can still be cut back by a late
677 // stop.
678 if(!cycle)
679 m_extended = false;
680 m_version++;
681 m_export_pending = true;
682 if(!self().pass_closing())
683 return;
684 top->loop_after = stack().length;
685 self().pass_closed();
686 merge_extra_pass();

700template <typename Node>
701void looper_node<Node>::looper_close_take(
702 const ossia::token_request& tk, LoopMode mode, bool ended_by_postaction)
703{
704 auto& st = stack();
705 if(st.count == 0)
706 return;
707
708 if(mode == LoopMode::Record)
709 {
710 m_loop_tempo = transport.loop_tempo(tk, st.length);
711 const double bar_frames = m_rate * 4. * (double(tk.signature.upper) / tk.signature.lower)
712 * (60. / tk.tempo);
713 const auto bars = int64_t(std::floor(transport.postaction_bars * bar_frames));
714
715 // The bar count says when the post-action takes over, so it is the length
716 // of a recording the post-action ended: what was recorded past the bars,
717 // e.g. the tempo having dropped while it went on, is cut off as a late
718 // stop cuts it, for the overdub that follows. A recording stopped by hand
719 // is as long as it was played for.
720 const bool to_bars = ended_by_postaction && transport.quantized()
721 && transport.postaction_bars > 0 && bars > 0;
722 if(to_bars && bars < st.length)
723 self().looper_trim_recording(st.length - bars);
724 self().close_recording(to_bars ? bars : 0);
725 m_export_date = transport.recordStartDate.impl / double(ossia::flicks_per_second<int64_t>);
726
727 // Position 0 of the loop on the timeline: the loop plays on from
728 // playbackPos at the point the take closes.
729 const auto at = double(transport.playbackPos % std::max<int64_t>(1, st.length));
730 m_origin = transport.timeline(tk);
731 m_origin.frames -= at;
732 m_origin.quarters -= at / m_rate * m_loop_tempo / 60.;
733 }
734 else
735 {
736 // An overdub that got nothing leaves no take.
737 if(m_pass_pending)
738 {
739 self().close_overdub();
740 m_extended = false;
741 return;
742 }
743 self().close_overdub();
744 }
745 m_version++;
746 m_extended = false;
747 m_export_pending = true;
748}

752template <typename Node>
753int64_t looper_node<Node>::looper_trim_extension(int64_t max_frames)
754{
755 auto& st = stack();
756 if(!m_extended || m_extend_unit <= 0 || st.count == 0)
757 return -1;
758 const int64_t end = st.length - m_extend_unit;
759 const int64_t into_last = transport.playbackPos - end;
760 if(into_last < 0 || into_last > max_frames || end < m_extend_unit)
761 return -1;
762 if(m_pass_pending)
763 {
764 // The pass after the cut got nothing: it was never a layer.
765 m_pass_pending = false;
766 self().cut_pass_dropped();
767 st.length = end;
768 if(auto* t = st.top())
769 t->loop_after = end;
770 }
771 else if(st.count > 1 && self().cut_after(end))
772 {
773 // A pass per cycle cut the overdub on the old end: the pass after the cut
774 // has nothing left, and goes as undo takes a pass out. The one before it
775 // ends on the old end, closed already.
776 self().pass_dropped();
777 undo_pass();
778 st.drop_redo();
779 self().cut_pass_dropped();
780 st.length = end;
781 if(auto* t = st.top())
782 t->loop_after = end;
783 }
784 else
785 {
786 self().trim_top_pass(end);
787 st.length = end;
788 }
789 m_version++;
790 m_extended = false;
791 self().head_moved(into_last);
792 return into_last;
793}

813template <typename Node>
814Looper::loop_period
815looper_node<Node>::looper_period(const ossia::token_request& tk) const noexcept
816{
817 const LoopMode m = transport.actualMode;
818 const auto& st = stack();
819 if(st.empty() || self().busy() || st.length <= 0 || m == LoopMode::Record)
820 return {};
821 const double rate = self().rate_for(m);
822 const int64_t round = Looper::loop_round(st.length, rate);
823 if(m == LoopMode::Stop)
824 return {
825 round, int64_t(std::floor(rounds_since_made(tk) * double(round) + 1e-6)), false};
826 const int64_t to_end = Looper::frames_to_loop_end(self().head_y(st.length), st.length, rate);
827 return {round, std::clamp<int64_t>(round - to_end, 0, round), true};
828}

847template <typename Node>
848void looper_node<Node>::capture(const tick& tk)
849{
850 if(transport.is_taking() || self().busy())
851 return;
852 auto& in = self().inputs;
853 const double bar = m_rate * 4. * (double(tk.signature.upper) / tk.signature.lower)
854 * (60. / std::max(1., tk.tempo));
855 const int64_t now = self().capture_clock();
856 int64_t end = now;
857 if(transport.quantized() && transport.lastPointFrame)
858 {
859 const int64_t late = transport.frame - *transport.lastPointFrame;
860 if(late >= 0 && late <= int64_t(in.late.value * m_rate / 1000.))
861 end -= late;
862 }
863 if(!stack().empty())
864 {
865 capture_layer(tk, now, end);
866 return;
867 }
868 const int64_t n = std::min<int64_t>(
869 {int64_t(in.capture_bars.value * bar), end, self().capture_limit()});
870 if(n <= 0)
871 return;
872
873 passes_changing();
874 m_switches.soloed.forget();
875 self().capture_into(end, n);
876
877 transport.play_captured((now - end) % n);
878 m_loop_tempo = tk.tempo;
879 m_origin = Looper::timeline_start(tk, self().ossia_state);
880 m_origin.frames -= double(transport.playbackPos);
881 m_origin.quarters -= double(transport.playbackPos) / m_rate * tk.tempo / 60.;
882 const double model_per_frame = self().ossia_state.samplesToModel();
883 m_export_date = (tk.prev_date.impl - (now - end + n) * model_per_frame)
884 / double(ossia::flicks_per_second<int64_t>);
885 m_export_pending = true;
886 self().captured();
887}

895template <typename Node>
896void looper_node<Node>::capture_layer(const tick& tk, int64_t now, int64_t end)
897{
898 auto& st = stack();
899 const int64_t len = st.length;
900 const LoopMode m = transport.actualMode;
901 const bool stopped = m != LoopMode::Play;
902 const double rate = self().rate_for(m);
903 if(len <= 0 || !(rate > 0.))
904 return;
905 double y_now{};
906 if(stopped)
907 {
908 const double rounds = rounds_since_made(tk);
909 y_now = std::floor((rounds - std::floor(rounds)) * double(len) + 1e-6);
910 }
911 else
912 y_now = self().head_y(len);
913
914 const auto round = int64_t(std::ceil(double(len) / rate - 1e-9));
915 double y_end = std::fmod(y_now - double(now - end) * rate, double(len));
916 if(y_end < 0.)
917 y_end += double(len);
918 Looper::capture_window w{
919 .end = end,
920 .n = std::min<int64_t>({round, end, self().capture_limit()}),
921 .length = len,
922 .y_end = y_end,
923 .rate = rate,
924 .backwards = m_reverse};
925 if(w.n <= 0 || w.positions() <= 0 || !self().capture_layer_ready(w))
926 return;
927
928 passes_changing();
929 const auto pushes = st.pushes;
930 begin_pass_now(w.start());
931 if(st.pushes == pushes)
932 return;
933 self().capture_layer_into(w);
934 close_pass();
935
936 if(stopped)
937 {
938 self().set_head(y_now, len);
939 transport.play_captured(transport.playbackPos);
940 self().captured();
941 }

997template <typename Node>
998void looper_node<Node>::run_export()
999{
1000 auto& st = stack();
1001 const bool asked = m_take_request > m_view.answered;
1002 if(asked && st.empty())
1003 {
1004 // Nothing to take: the last take is what there is.
1005 m_view.answered = m_take_request;
1006 m_view_countdown = 0;
1007 }
1008 // Passes asked for alone, of the loop the view showed: one that changed
1009 // since has nothing to give.
1010 auto& part = m_part_request;
1011 const bool part_asked = part.take > m_view.part_answered;
1012 if(part_asked
1013 && (st.empty() || part.layers != st.count || part.pass < 0 || part.last < part.pass
1014 || part.last >= st.count))
1015 {
1016 m_view.part_answered = part.take;
1017 part.take = 0;
1018 m_view_countdown = 0;
1019 }
1020 else if(
1021 part_asked && !m_writing && self().writer && self().export_ready())
1022 {
1023 // The files of the passes as they are now, the one taken in as far as it
1024 // went, under a take number of their own.
1025 m_job_part = true;
1026 m_job_request = part.take;
1027 part.take = 0;
1028 const bool open = pass_open();
1029 const bool recording = open && transport.actualMode == LoopMode::Record;
1030 m_job_stretch = self().take_stretch();
1031 m_export_tempo = recording ? m_tempo : m_loop_tempo;
1032 m_export_signature = m_signature;
1033 m_export_mix = false;
1034 m_export_first = part.pass;
1035 m_export_layers = part.last + 1;
1036 m_export_number = 0;
1037 m_export_slot = claim_take_slot();
1038 self().start_export(st.count, open, recording);
1039 }
1040 else if(
1041 (m_export_pending || asked) && !m_writing && self().writer && !st.empty()
1042 && self().export_ready())
1043 {
1044 m_export_pending = false;
1045 m_job_part = false;
1046 m_job_request = m_take_request;
1047 m_job_version = m_version;
1048 const bool open = asked && pass_open();
1049 const bool recording = open && transport.actualMode == LoopMode::Record;
1050 if(recording)
1051 m_export_date = transport.recordStartDate.impl / double(ossia::flicks_per_second<int64_t>);
1052 m_job_date = m_export_date;
1053 m_job_stretch = self().take_stretch();
1054 m_export_tempo = recording ? m_tempo : m_loop_tempo;
1055 m_export_signature = m_signature;
1056 const int passes
1057 = open || !pass_open() ? st.count : std::max(1, st.count - 1);
1058 // A take the view asks for has every file, which Export only chooses
1059 // from; one a pass closing makes has the ones Export says.
1060 const auto files = self().inputs.exported.value;
1061 m_export_mix = asked || files != Looper::TakeFiles::Layers;
1062 m_export_layers = asked || files != Looper::TakeFiles::Mix ? passes : 0;
1063 m_export_first = m_export_mix ? -1 : 0;
1064 m_export_number = 0;
1065 m_export_slot = claim_take_slot();
1066 self().start_export(passes, open, recording);
1067 }
1068 self().continue_export();

1072void looper_node<Node>::take_written(int slot, int number, bool ok, bool alone)
1073{
1074 auto& writer = self().writer;
1075 if(alone)
1076 {
1077 self().stop_take_written();
1078 if(!ok || !writer)
1079 return;
1080 // The views were told of it from the worker: a looper that plays on
1081 // after its stop shows it in its own messages as well.
1082 const auto& v = self().stop_notice().view;
1083 m_takes_written++;

1085 v.take_base.data(), v.take_ext.data(), v.take_mix, v.take_layers, v.take_stretch,
1086 v.take_placeable);
1087 m_view.take = m_takes_written;
1088 m_view.take_version = v.take_version;
1089 m_view.take_date = v.take_date;
1090 m_view_countdown = 0;
1091 return;
1092 }
1093 m_writing = false;
1094 if(self().take_cancelled() || !ok || !writer)
1095 {
1096 // What was being written changed, or the file could not be: the take
1097 // stops there.
1098 self().take_released();
1099 auto& answered = m_job_part ? m_view.part_answered : m_view.answered;
1100 answered = std::max(answered, m_job_request);
1101 m_view_countdown = 0;
1102 return;
1103 }
1104
1105 m_export_number = number;
1106 if(m_export_item + 1 < m_export_layers)
1107 {
1108 self().export_item(m_export_item + 1);
1109 return;
1110 }
1111
1112 self().take_released();
1113 if(m_job_part)
1114 {
1115 m_view.set_part(
1116 writer->take_base(slot), m_export_first, m_export_layers - 1, m_job_request,
1117 m_job_stretch, placeable_take());
1118 m_view.part_answered = std::max(m_view.part_answered, m_job_request);
1119 m_view_countdown = 0;
1120 return;
1121 }
1122 m_takes_written++;
1123 m_view.take_version = m_job_version;
1124 m_view.set_take(
1125 writer->take_base(slot), Node::take_extension, m_export_mix, m_export_layers,
1126 m_job_stretch, placeable_take());
1127 m_view.take = m_takes_written;
1128 m_view.take_date = m_job_date;
1129 m_view.answered = std::max(m_view.answered, m_job_request);
1130 // Force a view update with the new take.
1131 m_view_countdown = 0;
1132}
