228 using tick = halp::tick_musical;
229 void operator()(halp::tick_musical t)
230 {
231 const double t0 = m_frames / m_rate;
232 const double t1 = (m_frames + t.frames) / m_rate;
233 m_frames += t.frames;
234
235 const bool watched = monitor->watched();
236 bool got_message = false;
237 for(const auto& m : inputs.midi)
238 {
239 if(m.bytes.empty())
240 continue;
241
242 const double ts = t0 + double(m.timestamp) / m_rate;
243 const auto n = m.bytes.size();
244 const uint8_t status = m.bytes[0];
245 const uint8_t d1 = n > 1 ? m.bytes[1] : 0;
246 const uint8_t d2 = n > 2 ? m.bytes[2] : 0;
247 const auto flags = uint8_t(track(status, d1, d2, float(ts)));
248 if(watched)
249 monitor->events.push({m_seq, ts, status, d1, d2, flags, uint32_t(n)});
250 m_seq++;
251 got_message = true;
252 }
253
254 if(watched && (got_message || (t1 - m_last_push) >= heartbeat_seconds))
255 {
256 auto& h = m_held_state;
257 h = {};
258 for(int chan = 0; chan < 16; chan++)
259 {
260 for(int pitch = 0; pitch < 128; pitch++)
261 {
262 const auto& note = m_held_notes[chan][pitch];
263 if(!note.active)
264 continue;
265 h.bits[chan][pitch / 64] |= uint64_t(1) << (pitch % 64);
266 h.held++;
267 h.stuck += (t1 - note.on) > stuck_after_seconds;
268 }
269 }
270 h.now = t1;
271 h.window = inputs.window.value;
272 monitor->held.produce(h);
273 m_last_push = t1;
274 }
275 }

279 int track(uint8_t status, uint8_t d1, uint8_t d2, float t) noexcept
280 {
281 using T = libremidi::message_type;
282 const int chan = status & 0x0F;
283 switch(Ui::Monitor::midiMessageType(status, d1, d2))
284 {
285 case T::NOTE_ON: {
286 auto& note = m_held_notes[chan][d1];
287 const bool was_held = note.active;
288 note = {true, t};
289 return was_held ? flag_retrigger : flag_none;
290 }
291 case T::NOTE_OFF: {
292 auto& note = m_held_notes[chan][d1];
293 if(!note.active)
294 return flag_orphan;
295 note = {};
296 return flag_none;
297 }
298 default:
299 return flag_none;
300 }

549 void track(const Event& e)
550 {
551 using T = libremidi::message_type;
552 const auto type = Ui::Monitor::midiMessageType(e.status, e.d1, e.d2);
553 if(!Ui::Monitor::isChannelMessage(type))
554 return;
555 const int chan = e.status & 0x0F;
556 m_last_activity[chan] = e.t;
557 m_last_status[chan] = e.status;
558
559 auto lane = [&](int key, float v) {
560 auto& l = m_lanes[chan * 256 + key];
561 Ui::Monitor::pushRing(l.points, std::pair{e.t, v});
562 };
563 auto& pressure = m_note_pressure[chan];
564 // A point on the path of a sounding note, from its current expression
565 auto point = [&](int pitch) {
566 if(m_open[chan][pitch].active)
567 Ui::Monitor::pushRing(
568 m_paths[chan * 128 + pitch],
569 ExpressionPoint{
570 e.t, float(pitch + m_bend[chan] * mpe_bend_range), pressure[pitch],
571 m_timbre[chan]});
572 };
573 // What applies to the whole channel shapes every note sounding on it
574 auto every_note = [&] {
575 for(int pitch = 0; pitch < 128; pitch++)
576 point(pitch);
577 };
578
579 switch(type)
580 {
581 case T::CONTROL_CHANGE:
582 lane(e.d1, e.d2 / 127.f);
583 if(e.d1 == 74)
584 {
585 m_timbre[chan] = e.d2 / 127.f;
586 every_note();
587 }
588 break;
589 case T::PITCH_BEND: {
590 const float bend = (int(e.d1) + (int(e.d2) << 7) - 8192) / 8192.f;
591 lane(128, bend * 0.5f + 0.5f);
592 m_bend[chan] = bend;
593 every_note();
594 break;
595 }
596 case T::AFTERTOUCH:
597 lane(129, e.d1 / 127.f);
598 pressure.fill(e.d1 / 127.f);
599 every_note();
600 break;
601 case T::POLY_PRESSURE:
602 pressure[e.d1] = e.d2 / 127.f;
603 point(e.d1);
604 break;
605 case T::NOTE_ON:
606 m_paths[chan * 128 + e.d1].clear();
607 pressure[e.d1] = e.d2 / 127.f;
608 point(e.d1);
609 break;
610 default:
611 break;
612 }
613 }

618 // The flags are computed by the node against the full stream. This only
619 // turns them into geometry: an unmatched message here means our window
620 // simply does not reach far enough back, which is not an anomaly.
621 void apply(const Event& e)
622 {
623 using T = libremidi::message_type;
624 const int chan = e.status & 0x0F;
625 switch(Ui::Monitor::midiMessageType(e.status, e.d1, e.d2))
626 {
627 case T::NOTE_ON: {
628 auto& open = m_open[chan][e.d1];
629 if(open.active)
630 {
631 Ui::Monitor::pushRing(
632 m_notes, Note{open.on, e.t, uint8_t(chan), e.d1, open.vel,
633 bool(e.flags & flag_retrigger)});
634 }
635 open = {true, e.t, e.d2};
636 break;
637 }
638 case T::NOTE_OFF: {
639 if(e.flags & flag_orphan)
640 {
641 Ui::Monitor::pushRing(m_orphans, OrphanOff{e.t, uint8_t(chan), e.d1});
642 break;
643 }
644
645 auto& open = m_open[chan][e.d1];
646 if(open.active)
647 {
648 Ui::Monitor::pushRing(
649 m_notes, Note{open.on, e.t, uint8_t(chan), e.d1, open.vel, false});
650 open = {};
651 }
652 break;
653 }
654 default:
655 break;
656 }
657 }

659 void trim()
660 {
661 // Keep a bit more than the largest possible window so that changing the
662 // window slider does not blank the past.
663 const double horizon = m_now - 61.;
664 while(!m_notes.empty() && m_notes.front().off < horizon)
665 m_notes.pop_front();
666 while(!m_orphans.empty() && m_orphans.front().t < horizon)
667 m_orphans.pop_front();
668 for(auto& [k, l] : m_lanes)
669 while(l.points.size() > 1 && l.points[1].first < horizon)
670 l.points.pop_front();
671 for(auto it = m_paths.begin(); it != m_paths.end();)
672 {
673 while(!it->second.empty() && it->second.front().t < horizon)
674 it->second.pop_front();
675 it = it->second.empty() ? m_paths.erase(it) : std::next(it);
676 }
677 }

735 void paintRoll(QPainter* p, bool expression) const
736 {
737 const auto rect = boundingRect();
738 const double W = rect.width();
739 const double H = rect.height();
740 if(W <= left_gutter + 8. || H <= header_height + system_lane_height + 8.)
741 return;
742
743 const double roll_x = left_gutter;
744 const double roll_w = W - left_gutter - 2.;
745 const double roll_y = header_height;
746 const double roll_h = H - header_height - system_lane_height;
747
748 const double win = window();
749 const double t_min = m_now - win;
750
751 const auto to_x = TimeAxis(m_now, win, roll_x, roll_w);
752
753 // Vertical pitch range: fit what is on screen, with an octave minimum
754 int lo = 127, hi = 0;
755 const auto extend = [&](int pitch) {
756 lo = std::min(lo, pitch);
757 hi = std::max(hi, pitch);
758 };
759 for(const auto& n : m_notes)
760 if(n.off >= t_min)
761 extend(n.pitch);
762 for(const auto& chan : m_open)
763 for(int pitch = 0; pitch < 128; pitch++)
764 if(chan[pitch].active)
765 extend(pitch);
766
767 if(lo > hi)
768 {
769 lo = default_low_pitch;
770 hi = default_high_pitch;
771 }
772 lo = std::max(0, lo - 1);
773 hi = std::min(127, hi + 1);
774 if(hi - lo < 12)
775 {
776 const int mid = (lo + hi) / 2;
777 lo = std::max(0, mid - 6);
778 hi = std::min(127, lo + 12);
779 }
780
781 const int rows = hi - lo + 1;
782 const double row_h = roll_h / rows;
783 const auto to_y = [&](int pitch) { return roll_y + (hi - pitch) * row_h; };
784
785 p->save();
786 p->setRenderHint(QPainter::Antialiasing, false);
787
788 draw_grid(p, roll_x, roll_w, row_h, lo, hi, to_y);
789 if(expression)
790 draw_expression(p, row_h, t_min, to_x, to_y);
791 else
792 draw_notes(p, row_h, t_min, to_x, to_y);
793 draw_orphans(p, row_h, t_min, to_x, to_y);
794 draw_system_lane(p, roll_x, H - system_lane_height, roll_w, t_min, to_x);
795 draw_header(p, W);
796
797 p->restore();
798 }

958 void draw_expression(QPainter* p, double row_h, double t_min, auto to_x, auto to_y) const
959 {
960 p->setRenderHint(QPainter::Antialiasing, true);
961 constexpr int steps = 32;
962 const auto quantize = [](float v) { return int(std::clamp(v, 0.f, 1.f) * (steps - 1) + 0.5f); };
963 const auto pen_for = [](int timbre, int pressure) {
964 const double t = double(timbre) / (steps - 1), w = double(pressure) / (steps - 1);
965 return QPen(
966 QColor::fromHsvF(0.6 - 0.6 * t, 0.8, 0.95), 1. + 6. * w, Qt::SolidLine,
967 Qt::RoundCap);
968 };
969 const int filter = channelFilter();
970 auto& run = m_polyline;
971 for(const auto& [key, path] : m_paths)
972 {
973 const int chan = key / 128;
974 if(path.empty() || (filter >= 0 && chan != filter))
975 continue;
976 // From the last point before the window, which still shapes its left edge
977 auto it = std::lower_bound(
978 path.begin(), path.end(), t_min,
979 [](const ExpressionPoint& pt, double t) { return pt.t < t; });
980 if(it != path.begin())
981 --it;
982
983 run.resize(0);
984 int run_key = -1;
985 for(; it != path.end(); ++it)
986 {
987 const auto& pt = *it;
988 // Fractional pitch: the row of the note plus its bend
989 const double y = to_y(int(std::floor(pt.pitch)))
990 - (pt.pitch - std::floor(pt.pitch)) * row_h + row_h / 2;
991 const QPointF here{to_x(pt.t), y};
992 // A segment takes the expression of the point it leads to
993 const int k = quantize(pt.timbre) * steps + quantize(pt.pressure);
994 if(k != run_key && run.size() > 1)
995 {
996 p->setPen(pen_for(run_key / steps, run_key % steps));
997 p->drawPolyline(run);
998 const QPointF last = run.back();
999 run.resize(0);
1000 run << last;
1001 }
1002 run_key = k;
1003 run << here;
1004 }
1005 // Still sounding: run to now
1006 if(!run.isEmpty() && m_open[chan][key % 128].active)
1007 run << QPointF{to_x(m_now), run.back().y()};
1008 if(run.size() > 1)
1009 {
1010 p->setPen(pen_for(run_key / steps, run_key % steps));
1011 p->drawPolyline(run);
1012 }
1013 }
1014 p->setRenderHint(QPainter::Antialiasing, false);
1015 }
