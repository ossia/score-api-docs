158 // Bottom two rungs of the capture ladder (RDMA -> DVP -> host-import ->
159 // uploadTexture). SCORE_GFX_CAPTURE_STRATEGY pins one:
160 // "uploadtexture" skips the import; "hostimport" requires it and fails
161 // init when unavailable, so the ladder reports honestly instead of
162 // silently landing a rung lower than asked for.
163 const auto wantRung = qEnvironmentVariable("SCORE_GFX_CAPTURE_STRATEGY").toLower();
164 const bool forceUpload = wantRung == "uploadtexture"
165 || qEnvironmentVariableIsSet("SCORE_GFX_NO_VK_HOST_IMPORT");
166 const bool requireImport = wantRung == "hostimport";
167 const std::size_t vkAlign
168 = forceUpload ? 0u : VkHostImportUpload::requiredAlignment(*cfg.rhi);
169 const std::size_t d3dAlign
170 = forceUpload ? 0u : D3D12HostImportUpload::requiredAlignment(*cfg.rhi);
171 const std::size_t importAlign = vkAlign ? vkAlign : d3dAlign;
172 const std::size_t rowPitch
173 = cfg.height > 0 ? cfg.frameByteSize / std::size_t(cfg.height) : 0;
174 // Both host-import paths need the row pitch.
175 m_rowPitch = rowPitch;
176 // D3D12 additionally constrains the row pitch; refusing here keeps the
177 // ladder honest rather than importing and then failing every copy.
178 const bool d3dUsable = d3dAlign != 0 && D3D12HostImportUpload::pitchUsable(rowPitch);
179 if(requireImport && importAlign == 0)
180 {
181 qWarning() << "CpuStagedCapture: SCORE_GFX_CAPTURE_STRATEGY=hostimport but "
182 "no host-import path is available on this backend";
183 return false;
184 }
185 // Align to the import requirement when it is available so the same slots
186 // can serve both paths; otherwise plain max_align is enough.
187 const std::size_t slotAlign = importAlign ? importAlign : alignof(std::max_align_t);
188 m_slotsImportable = importAlign != 0;
189
190 for(std::size_t i = 0; i < kSlotCount; ++i)
191 {
192 // importableAlloc when a GPU is going to import these pages: D3D12 needs
193 // a 64 KB-granularity VirtualAlloc address and refuses _aligned_malloc.
194 m_slots[i] = static_cast<std::uint8_t*>(
195 importAlign ? importableAlloc(cfg.frameByteSize)
196 : alignedSlotAlloc(cfg.frameByteSize, slotAlign));
197 if(!m_slots[i])
198 {
199 release();
200 return false;
201 }
202 std::memset(m_slots[i], 0, cfg.frameByteSize);
203 if constexpr(Policy::has_dma_lock)
204 {
205 // Page-lock (paged, not RDMA) so the card's DMA into the host buffer
206 // doesn't re-pin pages every frame.
207 if(m_policy.dmaLock(m_slots[i], cfg.frameByteSize))
208 m_dmaLocked[i] = true;
209 }
210 }
211
212 if(importAlign)
213 {
214 std::vector<void*> ptrs;
215 ptrs.reserve(kSlotCount);
216 for(auto* p : m_slots)
217 ptrs.push_back(p);
218 const bool ok
219 = vkAlign ? m_hostImport.init(*cfg.rhi, ptrs, cfg.frameByteSize)
220 : (d3dUsable
221 && m_d3dImport.init(
222 *cfg.rhi, ptrs, cfg.frameByteSize, rowPitch));
223 if(ok)
224 {
225 qDebug() << "DMA capture:" << (vkAlign ? "Vulkan" : "D3D12")
226 << "host-import upload engaged (zero-copy)";
227 }
228 else
229 {
230 m_hostImport.release();
231 m_d3dImport.release();
232 if(requireImport)
233 {
234 qWarning() << "CpuStagedCapture: host-import rung was pinned but the "
235 "import failed";
236 return false;
237 }
238 }
239 }
240 return true;
241 }

243 void release() override
244 {
245 m_hostImport.release();
246 m_d3dImport.release();
247 for(std::size_t i = 0; i < kSlotCount; ++i)
248 {
249 // NB: no DMA unlock here — the vendor capture session's close() does the
250 // unlock-all, and by teardown the card object may already be gone.
251 m_dmaLocked[i] = false;
252 if(m_slotsImportable)

257 }

290 void acquireForRender(QRhiResourceUpdateBatch& res, QRhiCommandBuffer* cb) override
291 {
292 const int slotIdx = m_publisher.consume();
293 if(slotIdx < 0 || static_cast<std::size_t>(slotIdx) >= kSlotCount)
294 return;
295 const void* src = m_slots[static_cast<std::size_t>(slotIdx)];
296
297 // Vulkan zero-copy: the slot pages are already imported, so all that is
298 // left is the device-side DMA. Needs the command buffer, hence the overload.
299 if(cb && (m_hostImport.valid() || m_d3dImport.valid()))
300 {
301 const auto doCopy = [&](QRhiTexture& t, std::size_t s, int w, int h,
302 std::size_t off) {
303 return m_hostImport.valid()
304 ? m_hostImport.copyToTexture(*cb, t, s, w, h, off, m_rowPitch)
305 : m_d3dImport.copyToTexture(*cb, t, s, w, h, off);
306 };
307 // Planar frames are one contiguous slot holding N planes, so the
308 // zero-copy path walks them exactly as the portable path below does:
309 // uploading only plane 0 would leave chroma uninitialised.
310 if(cfg.planes.size() > 1)
311 {
312 std::size_t offset = 0;
313 bool all = true;
314 for(auto* tex : cfg.planes)
315 {
316 if(!tex)
317 {
318 all = false;
319 break;
320 }
321 const auto psz = tex->pixelSize();
322 const std::size_t bytes
323 = std::size_t(psz.width()) * texelBytes(tex->format()) * psz.height();
324 if(offset + bytes > cfg.frameByteSize
325 || !doCopy(
326 *tex, static_cast<std::size_t>(slotIdx), psz.width(),
327 psz.height(), offset))
328 {
329 all = false;
330 break;
331 }
332 offset += bytes;
333 }
334 if(all)
335 return;
336 // Fall through to the portable path rather than ship a half-uploaded
337 // frame.
338 }
339 else
340 {
341 const auto pxsz = cfg.outputTexture->pixelSize();
342 if(doCopy(
343 *cfg.outputTexture, static_cast<std::size_t>(slotIdx),
344 pxsz.width(), pxsz.height(), 0))
345 return;
346 }
347 }
348
349#if QT_CONFIG(opengl)
350 if constexpr(Policy::has_gl_fast_path)
351 {
352 if(m_glCtx)
353 {
354 // Raw-GL fast path: one glTexSubImage2D, no staging copy.
355 uploadClientToGLTexture(
356 *m_glCtx, *cfg.outputTexture, src, m_policy.bgra());
357 return;
358 }
359 }
360#endif
361
362 // Portable path: backend-neutral QRhi upload.
363 QRhiTextureSubresourceUploadDescription sub(src, cfg.frameByteSize);
364 if constexpr(Policy::has_dma_lock)
365 {
366 // AJA: the slot is tightly packed to the texture's byte layout
367 // (validateCaptureTextureBytes enforces frameByteSize == texW*4*texH),
368 // so the stride is texWidth*4.
369 const auto pxsz = cfg.outputTexture->pixelSize();
370 sub.setDataStride(static_cast<quint32>(pxsz.width()) * 4u);
371 }
372 else
373 {
374 // CPU-only vendors: source stride = the card raster's true pitch (v210
375 // rows are 48-pixel-group padded, wider than texWidth*4 when w % 48 != 0,
376 // e.g. 720p, and a texture-width stride would shear the frame).
377 sub.setDataStride(
378 cfg.height > 0 ? cfg.frameByteSize / static_cast<quint32>(cfg.height)
379 : 0);
380 }
381 // Planar formats arrive as one contiguous buffer holding N planes back to
382 // back. The decoder's own texture list defines the layout: each plane's
383 // byte size is its texture geometry times its texel size, and offsets
384 // accumulate in order. Deriving it this way keeps the strategy free of
385 // per-format tables -- adding a planar decoder needs no change here.
386 if(cfg.planes.size() > 1)
387 {
388 std::size_t offset = 0;
389 for(std::size_t i = 0; i < cfg.planes.size(); ++i)
390 {
391 auto* tex = cfg.planes[i];
392 if(!tex)
393 continue;
394 const auto psz = tex->pixelSize();
395 const auto texel = texelBytes(tex->format());
396 const auto stride = static_cast<quint32>(psz.width()) * texel;
397 const std::size_t bytes = std::size_t(stride) * psz.height();
398 if(offset + bytes > cfg.frameByteSize)
399 {
400 qWarning() << "CpuStagedCapture: plane" << i << "overruns the frame ("
401 << offset << "+" << bytes << ">" << cfg.frameByteSize
402 << ") - skipping the rest";
403 break;
404 }
405 QRhiTextureSubresourceUploadDescription psub(
406 static_cast<const char*>(src) + offset, bytes);
407 psub.setDataStride(stride);
408 res.uploadTexture(
409 tex, QRhiTextureUploadDescription{QRhiTextureUploadEntry{0, 0, psub}});
410 offset += bytes;
411 }
412 return;
413 }
414
415 res.uploadTexture(
416 cfg.outputTexture,
417 QRhiTextureUploadDescription{QRhiTextureUploadEntry{0, 0, sub}});
418 }
