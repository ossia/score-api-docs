67 // GPU-decided draw count / dispatch size — Qt 6.13-era QRhi API
68 // (QRhi::DrawIndirectCount, QRhi::DispatchIndirect). Populated through
69 // RhiIndirectCompat.hpp's member DETECTION, not QT_VERSION: the ossia SDK
70 // pins the Qt 6.12 branch with the 6.13 indirect changes cherry-picked,
71 // so version macros lie on the reference builds (see that header).
72 //
73 // Fallback ladder these caps select, from most to least capable — every
74 // rung must paint the SAME pixels (GfxIndirectFallbackLadder.cpp holds
75 // each rung to that):
76 // drawIndirectCount → drawIndexedIndirectCount / drawIndirectCount
77 // (count read by the GPU from a count buffer)
78 // drawIndirect+multi → one drawIndexedIndirect of the full command
79 // capacity. With a count buffer, the raw raster
80 // first clamps a copy of the commands to the count
81 // on the GPU (IndirectCountClamp: dead slots
82 // zeroed, a zero command draws nothing) and draws
83 // that; without compute, producers must leave the
84 // slots past their count zeroed themselves
85 // drawIndirect only → per-command loop of drawCount=1 indirect draws,
86 // over the same clamped copy
87 // none → CPU loop over cpuDrawCommands, filled by the
88 // producer or by the readback fallback (which also
89 // reads the count buffer and clamps).
90 //
91 // Kill switches (Caps::populate): SCORE_GFX_NO_GPU_INDIRECT_COUNT,
92 // SCORE_GFX_NO_GPU_INDIRECT_MULTI, SCORE_GFX_NO_GPU_DISPATCH_INDIRECT,
93 // and SCORE_GFX_NO_GPU_INDIRECT, which forces the CPU rung outright.
94 // Each fallback stays exercisable on capable hardware.
95 bool drawIndirectCount{false};
96 bool dispatchIndirect{false};
97
98 // Always queryable.
99 bool multiview{false};
100 bool resolveDepthStencil{false};
101 bool tessellation{false};
102 bool geometryShader{false};
103
104 // Extended capability set, driving shader feature gating and observability.
105 //
106 // baseInstance: indirect draws can use firstInstance as the draw ID through
107 // gl_BaseInstance (ARB_shader_draw_parameters), which MDI's per-draw lookup
108 // reads.
109 // instanceIndexIncludesBaseInstance: whether gl_InstanceIndex already contains
110 // the firstInstance offset. The shader prepass injects
111 // SCORE_INSTANCE_INDEX_INCLUDES_BASE_INSTANCE from this so presets work on
112 // both paths.
113 // variableRateShading: per-tile shading-rate maps
114 // (VK_EXT_fragment_shading_rate, D3D12 VRS).
115 // timestamps: whether lastCompletedGpuTime() returns meaningful values.
116 // pipelineCacheDataLoadSave: pipeline binary cache round-trip, used by
117 // tryLoadPipelineCache / tryStorePipelineCache.
118 // textureViewFormat: R32UI <-> R32F aliasing, needed by the visibility buffer
119 // preset.
120 // depthClamp: reverse-Z shadow passes avoiding near-plane clipping.
121 // triangleFan: triangleFanSupported(), false on D3D and Metal; fans are
122 // drawn as triangle lists there (lowerTriangleFan, the VSA index list).

237inline QRhiBuffer::UsageFlags
238compatibleBufferUsage(QRhi& rhi, QRhiBuffer::UsageFlags usage) noexcept
239{
240 if(!usage.testFlag(QRhiBuffer::StorageBuffer))
241 return usage;
242 // Storage-only: there is no other role to fall back to, keep it as asked.
243 if(int(usage) == int(QRhiBuffer::StorageBuffer))
244 return usage;
245 // QRhi::Compute is exactly the GL backend's own SSBO line: caps.compute is
246 // set for GL >= 4.3 / GLES >= 3.1 (qrhigles2.cpp), the same versions that
247 // introduce GL_SHADER_STORAGE_BUFFER. Ask the capability, do not name a
248 // backend: it stays right if a backend gains or loses the ability.
249 if(rhi.isFeatureSupported(QRhi::Compute))
250 return usage;
251 return usage & ~QRhiBuffer::UsageFlags(QRhiBuffer::StorageBuffer);
252}

370inline bool viewIndexNeedsPassIndexFallback(
371 GraphicsApi api, const QShaderVersion& version, int multiViewCount) noexcept
372{
373 // SCORE_GFX_DISABLE_MULTIVIEW means "pretend this backend has no multiview".
374 // It also makes this path -- otherwise reachable only on a D3D target --
375 // testable everywhere.
376 if(qEnvironmentVariableIsSet("SCORE_GFX_DISABLE_MULTIVIEW"))
377 return true;
378
379 if(api != GraphicsApi::D3D11 && api != GraphicsApi::D3D12)
380 return false;
381
382 // D3D11 is pinned to SM 5.0 for good, so it can never have SV_ViewID.
383 if(api == GraphicsApi::D3D11)
384 return true;
385
386 // D3D12 without dxcompiler.dll drops to SM 5.0, which has no SV_ViewID
387 // either. The ossia SDK ships that runtime, so shipping builds DO reach
388 // 6.1 -- but a source build without it still lands here.
389 if(version.version() < 61)
390 return true;
391
392 // D3D12 ViewInstancing is capped at D3D12_MAX_VIEW_INSTANCE_COUNT == 4.
393 // Above that, CreatePipelineState rejects the PSO outright:
394 //
395 // Failed to create graphics pipeline state: COM error 0x80070057:
396 // The parameter is incorrect.
397 // Warning! MRT Pipeline not created
398 //
399 // and with no pipeline nothing draws, so every cube face reads back
400 // (0,0,0,255) -- which looks like a rasterizer or copy fault and is neither.
401 // Qt encodes the same limit in its QVarLengthArray<D3D12_VIEW_INSTANCE_LOCATION, 4>.
402 //
403 // A cubemap is inherently 6 views, so CUBEMAP+MULTIVIEW can never use D3D12
404 // ViewInstancing -- it is not a Qt bug and not a shim bug, it is the API's
405 // limit.
406 //
407 // Keep the fast path where it is legal: 2- and 4-view shaders still get real
408 // ViewInstancing. Only what D3D12 cannot express falls back to N passes.
409 if(multiViewCount > 4)
410 return true;
411
412 return false;
413}
