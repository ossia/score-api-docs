13// Wire format of the glTF live link: a DCC (the Blender add-on,
14// https://github.com/ossia/score-blender-livelink) connects over TCP and
15// streams frames.
16// That repository's PROTOCOL.md is the specification for writing a sender.
17//
18// Every integer and float is little-endian. A frame is
19//
20// u32 length byte count of what follows (type + body)
21// u8 type
22// u8 body[length - 1]
23//
24// A `str` below is a u16 byte count followed by that many UTF-8 bytes.
25//
26// hello (1) UTF-8 JSON, see "Negotiation" below
27// scene (2) u64 seq, then a complete binary glTF (.glb)
28//
29// Every other type is u64 seq, u32 count, then count records:
30//
31// transforms (3) str node, f32 translation[3], f32 rotation[4] (x, y, z, w),
32// f32 scale[3]: the node's local TRS
33// joints (4) str armature node, u32 bone count, then per bone a
34// transforms record named after the bone: bone names are
35// only unique within their armature
36// weights (5) str node, u16 n, f32 weights[n]: the morph target weights
37// of the node's mesh
38// visibility (6) str node, u8 visible
39// lights (7) str node, u8 type (0 directional, 1 point, 2 spot),
40// f32 color[3], f32 intensity, f32 range (0: infinite),
41// f32 inner cone, f32 outer cone (radians)
42// cameras (8) str node, u8 projection (0 perspective, 1 orthographic),
43// f32 yfov (radians), f32 aspect (0: unset), f32 znear,
44// f32 zfar, f32 xmag, f32 ymag
45// materials (9) str material, u32 fields, f32 base color[4],
46// f32 metallic, f32 roughness, f32 emissive[3],
47// f32 emissive strength; `fields` says which are meant
48// remove (10) str node: removes the node, and its subtree, from the
49// scene
50// environment (12) str world, f32 ambient[3], f32 strength: the light
51// the world sends from every direction (glTF has none)
52// active camera (13) str node: the camera node to render from, of the
53// several a scene may hold (glTF has no notion of it)
54//
55// and, laid out like a scene frame:
56//
57// patch (11) u64 seq, then a binary glTF of a few objects, whose root
58// nodes are added to the scene's roots
59//
60// Patches and removals change the scene without sending all of it: an object
61// added, deleted, or re-exported alone (a removal and a patch of the same
62// node). They apply in seq order, after the scene frame they follow.
63//
64// All values are what the glTF of the last scene frame would hold for them:
65// node names are glTF node names, units are glTF units, axes are glTF axes.
66// A record overrides that value in the last scene, so a scene is sent only
67// when geometry, materials' structure or the hierarchy change.
68//
69// `seq` is a single counter shared by every frame type, increasing on the
70// sender. Comparing seqs is how the receiver drops overrides that a later
71// scene already contains.
72//
73// Negotiation
74// -----------
75// The sender opens with a hello, {"protocol": 1, "app": ..., "features":
76// [...]}, and the receiver answers with its own: "protocol" and "accepts",
77// the names of the frame types it handles (accepted_frames below). A sender
78// uses only frame types the receiver accepts. One that gets no answer takes
79// the receiver for one that does not negotiate, and sends it scene frames
80// only. glTF is the baseline every sender can fall back to: an exporter only
81// has to produce a .glb to drive the receiver; every other frame type is an
82// optimisation over re-sending it.
83//
84// Native frames
85// -------------
86// What a glTF export would carry, without the export: the sender's own
87// evaluated meshes and its images' original files, and a JSON scene that
88// refers to them by key. They apply in seq order with patches and removals.
89//
90// mesh (14) u64 seq, str key, u16 primitive count, then per primitive
91// u32 vertex count v, u32 index count i (0: not indexed),
92// u8 attributes (1 normal, 2 texcoord0, 4 texcoord1,
93// 8 color0, 16 joints0 + weights0), u16 morph target count
94// t, f32 position[3v], then the attributes present in that
95// order (normal[3v], texcoord[2v], texcoord1[2v],
96// color[4v], u16 joints[4v] then f32 weights[4v]), then
97// u32 indices[i], then per morph target u8 (1 position, 2
98// normal deltas) and those f32 deltas[3v]. glTF axes and
99// UV convention; one primitive per material slot. Joints
100// index the "joints" of the skin of the node using it.
101// image (15) u64 seq, str key, str MIME type, then the encoded image
102// file (PNG, JPEG: whatever the receiver can decode)
103// native_scene (16) u64 seq, u8 reset, then UTF-8 JSON:
104// {"nodes": [{"name", "parent" (index in "nodes", -1 for
105// none: names repeat, bones), "t": [3], "r": [4] (xyzw),
106// "s": [3], "visible", "mesh": key, "materials": [name
107// per primitive], "light": [type, r, g, b, intensity,
108// range, inner, outer], "camera": [projection, yfov,
109// aspect, znear, zfar, xmag, ymag], "weights": [morph
110// target weights], "skin": {"armature": node index, "joints":
111// [bone node names, under the armature node],
112// "inverse_bind": [16 per joint, column-major]}}, ...]
113// (parents before children),
114// "materials": {key: {"name", "base_color": [4], "metallic",
115// "roughness", "emissive": [3], "emissive_strength",
116// "alpha": "opaque" | "mask" | "blend", "alpha_cutoff",
117// "double_sided", "unlit", and the KHR_materials_*
118// factors: "ior", "transmission", "specular",
119// "specular_color": [3], "clearcoat",
120// "clearcoat_roughness", "sheen_color": [3],
121// "sheen_roughness", "anisotropy", "anisotropy_rotation",
122// "iridescence", "iridescence_ior",
123// "iridescence_thickness": [min, max];
124// "textures": {slot: {"image": key, "uv": set,
125// "wrap": "repeat" | "clamp" | "mirror",
126// "filter": "linear" | "nearest", "scale", "strength",
127// "transform": {"offset": [2], "rotation", "scale": [2]}
128// (KHR_texture_transform)}}}}}
129// ("name" is the material's, for streamed values; nodes
130// name materials by key), with slots as glTF's: base_color,
131// transmission (R), specular (A), specular_color,
132// clearcoat (R), clearcoat_roughness (G), clearcoat_normal,
133// sheen_color, sheen_roughness (A), metallic_roughness (G roughness,
134// B metallic), normal, occlusion (R), emissive. With reset,
135// the nodes replace the whole scene, as a scene frame;
136// without, they are added to it, as a patch. Meshes and
137// images are kept by key until a reset no longer uses them.

183struct reader
184{
185 std::string_view in;
186
187 template <typename T>
188 bool operator()(T& out) noexcept
189 {
190 if(in.size() < sizeof(T))
191 return false;
192 std::memcpy(&out, in.data(), sizeof(T));
193 in.remove_prefix(sizeof(T));
194 return true;
195 }
196
197 template <typename T, std::size_t N>
198 bool operator()(T (&out)[N]) noexcept
199 {
200 for(auto& v : out)
201 if(!(*this)(v))
202 return false;
203 return true;
204 }
205
206 bool operator()(std::string& out)
207 {
208 uint16_t len{};
209 if(!(*this)(len) || in.size() < len)
210 return false;
211 out.assign(in.data(), len);
212 in.remove_prefix(len);
213 return true;
214 }
215};

477// Body of a scene or patch frame (without the length and type bytes).
478inline std::optional<scene_message> parse_scene(std::string_view body)
479{
480 scene_message m;
481 detail::reader r{body};
482 if(!r(m.seq))
483 return std::nullopt;
484 // A .glb starts with the "glTF" magic; anything else cannot be a binary
485 // glTF and would only fail later, on a worker, with a vaguer message.
486 if(r.in.size() < 12 || r.in.substr(0, 4) != "glTF")
487 return std::nullopt;
488 m.glb.assign(r.in.data(), r.in.size());
489 return m;
490}

494template <typename T>
495bool read_array(reader& r, std::size_t count, std::shared_ptr<std::vector<T>>& out)
496{
497 // Bounded by what the body holds before anything is allocated.
498 if(count > r.in.size() / sizeof(T))
499 return false;
500 out = std::make_shared<std::vector<T>>(count);
501 std::memcpy(out->data(), r.in.data(), count * sizeof(T));
502 r.in.remove_prefix(count * sizeof(T));
503 return true;
504}

509 mesh_normal = 1,
510 mesh_texcoord0 = 2,
511 mesh_texcoord1 = 4,
512 mesh_color0 = 8,
513 mesh_skin = 16,

516inline std::optional<mesh_message> parse_mesh(std::string_view body)
517{
518 mesh_message m;
519 detail::reader r{body};
520 uint16_t count{};
521 if(!r(m.seq) || !r(m.key) || !r(count))
522 return std::nullopt;
523 m.primitives.resize(count);
524 for(auto& p : m.primitives)
525 {
526 uint32_t indices{};
527 uint8_t attrs{};
528 uint16_t targets{};
529 if(!r(p.vertex_count) || !r(indices) || !r(attrs) || !r(targets))
530 return std::nullopt;
531 const std::size_t v = p.vertex_count;
532 if(!detail::read_array(r, 3 * v, p.positions))
533 return std::nullopt;
534 if((attrs & mesh_normal) && !detail::read_array(r, 3 * v, p.normals))
535 return std::nullopt;
536 if((attrs & mesh_texcoord0) && !detail::read_array(r, 2 * v, p.texcoords))
537 return std::nullopt;
538 if((attrs & mesh_texcoord1) && !detail::read_array(r, 2 * v, p.texcoords1))
539 return std::nullopt;
540 if((attrs & mesh_color0) && !detail::read_array(r, 4 * v, p.colors))
541 return std::nullopt;
542 if((attrs & mesh_skin)
543 && (!detail::read_array(r, 4 * v, p.joints) || !detail::read_array(r, 4 * v, p.weights)))
544 return std::nullopt;
545 if(indices > 0)
546 {
547 if(!detail::read_array(r, indices, p.indices))
548 return std::nullopt;
549 for(uint32_t i : *p.indices)
550 if(i >= p.vertex_count)
551 return std::nullopt;
552 }
553 else if(v % 3 != 0)
554 {
555 return std::nullopt; // not indexed: a triangle list
556 }
557 p.targets.resize(targets);
558 for(auto& t : p.targets)
559 {
560 uint8_t which{};
561 if(!r(which))
562 return std::nullopt;
563 if((which & 1) && !detail::read_array(r, 3 * v, t.positions))
564 return std::nullopt;
565 if((which & 2) && !detail::read_array(r, 3 * v, t.normals))
566 return std::nullopt;
567 }
568 }
569 if(!r.in.empty())
570 return std::nullopt;
571 return m;
572}

597// Body of any records frame. All-or-nothing: a truncated or inconsistent
598// record rejects the frame rather than applying part of it.
599template <typename Record>
600std::optional<records_message<Record>> parse_records(std::string_view body)
601{
602 records_message<Record> m;
603 detail::reader r{body};
604 uint32_t count{};
605 if(!r(m.seq) || !r(count))
606 return std::nullopt;
607
608 // Bounds count before the resize below can be asked for an absurd size.
609 if(count > r.in.size() / Record::min_size)
610 return std::nullopt;
611
612 m.records.resize(count);
613 for(auto& rec : m.records)
614 if(!rec.read(r))
615 return std::nullopt;
616 if(!r.in.empty())
617 return std::nullopt;
618 return m;
619}

646inline std::string encode_mesh(
647 uint64_t seq, const std::string& key, const std::vector<native_primitive>& prims)
648{
649 std::string body;
650 detail::writer w{body};
651 w(seq), w(key), w(uint16_t(prims.size()));
652 auto put = [&](const auto& v) {
653 if(v)
654 body.append((const char*)v->data(), v->size() * sizeof((*v)[0]));
655 };
656 for(const auto& p : prims)
657 {
658 const uint8_t attrs = (p.normals ? mesh_normal : 0) | (p.texcoords ? mesh_texcoord0 : 0)
659 | (p.texcoords1 ? mesh_texcoord1 : 0) | (p.colors ? mesh_color0 : 0)
660 | (p.joints && p.weights ? mesh_skin : 0);
661 w(p.vertex_count), w(uint32_t(p.indices ? p.indices->size() : 0)), w(attrs),
662 w(uint16_t(p.targets.size()));
663 put(p.positions), put(p.normals), put(p.texcoords), put(p.texcoords1), put(p.colors);
664 if(p.joints && p.weights)
665 put(p.joints), put(p.weights);
666 put(p.indices);
667 for(const auto& t : p.targets)
668 {
669 w(uint8_t((t.positions ? 1 : 0) | (t.normals ? 2 : 0)));
670 put(t.positions), put(t.normals);
671 }
672 }
673 return encode_frame(frame_type::mesh, body);

712// Incremental frame splitter for a byte stream: feed() whatever the socket
713// delivered, get called back once per complete frame. Once it has failed (a
714// zero or oversized length), the stream cannot be resynchronised and the
715// connection should be dropped.
716class frame_decoder
717{
718public:
719 template <typename F>
720 bool feed(std::string_view bytes, F&& on_frame)
721 {
722 if(m_failed)
723 return false;
724 m_buffer.append(bytes);
725
726 std::size_t pos = 0;
727 while(m_buffer.size() - pos >= 4)
728 {
729 uint32_t len{};
730 std::memcpy(&len, m_buffer.data() + pos, 4);
731 if(len == 0 || len > max_frame_size)
732 {
733 m_failed = true;
734 m_buffer.clear();
735 return false;
736 }
737 if(m_buffer.size() - pos - 4 < len)
738 break;
739
740 const auto type = frame_type(uint8_t(m_buffer[pos + 4]));
741 on_frame(type, std::string_view{m_buffer.data() + pos + 5, len - 1});
742 pos += 4 + len;
743 }
744 m_buffer.erase(0, pos);
745 return true;
746 }
747
748 bool failed() const noexcept { return m_failed; }
749 std::size_t buffered() const noexcept { return m_buffer.size(); }
750
751private:
752 std::string m_buffer;
753 bool m_failed{false};
754};
