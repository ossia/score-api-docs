990inline int64_t Node::run_mode(int64_t start, int64_t length)
991{
992 layers.now = m_clock;
993 // A pass cut on the last frame of an overdub that ended has nothing going
994 // on from it.
995 if(m_tail_open >= 0 && transport.actualMode != LoopMode::Overdub)
996 end_open_tail();
997 layers.update_seams();
998 // Stretching starts again from where the loop is, the next time it plays.
999 if(transport.actualMode != LoopMode::Play)
1000 m_stretch_ready = false;
1001 if(transport.playbackPos != m_head_pos)
1002 m_frac = 0.;
1003
1004 // Record passthrough is what its name says: the input while something is
1005 // being taken in, and nothing else. Only Full is heard the rest of the time.
1006 const Passthrough pt = inputs.passthrough;
1007 const bool echo_while_taking_in = pt != Passthrough::None;
1008 const bool echo_otherwise = pt == Passthrough::Full;
1009 switch(transport.actualMode)
1010 {
1011 case LoopMode::Play:
1012 if(layers.empty())
1013 echo_otherwise ? through(start, length) : silence(start, length);
1014 else
1015 {
1016 if(m_stretch_mode != ossia::audio_stretch_mode::None)
1017 play_stretched(start, length);
1018 else if(m_speed != 1. || m_reverse || m_frac != 0.)
1019 play_tape(start, length);
1020 else
1021 play(start, length);
1022 if(echo_otherwise)
1023 mix_input(start, length);
1024 }
1025 return length;
1026 case LoopMode::Stop:
1027 echo_otherwise ? through(start, length) : silence(start, length);
1028 break;
1029 case LoopMode::Record:
1030 return record(start, length, echo_while_taking_in);
1031 case LoopMode::Overdub:
1032 overdub(start, length, echo_while_taking_in);
1033 break;
1034 }
1035 return length;
1036}

1040inline void Node::close_recording(int64_t bars)
1041{
1042 auto* top = layers.top();
1043 if(!top || layers.length <= 0)
1044 return;
1045
1046 // Overdubbing from the frame the recording ends on carries on what was
1047 // played: the end of the loop goes on into the first pass over it. The
1048 // edges are those of what was recorded, wherever the latency puts them.
1049 m_seamless = transport.actualMode == LoopMode::Overdub;
1050 fade_edges(top->slot, !m_seamless);
1051 top->start = 0;
1052 top->written = layers.length;
1053 if(const int64_t latency = latency_frames(); latency > 0)
1054 layers.shift_earlier(*top, latency);
1055 m_tail_open = -1;
1056 if(m_seamless)
1057 {
1058 layers.open_seam(*top, true, Looper::audio_seam_end::pending);
1059 m_tail_open = top->slot;
1060 }
1061
1062 if(bars > layers.length)
1063 {
1064 const int64_t end = std::min(bars, layers.capacity);
1065 for(int c = 0; c < layers.channels; c++)
1066 std::fill(layers.data(top->slot, c) + layers.length, layers.data(top->slot, c) + end, 0.f);
1067 layers.length = end;
1068 }
1069 top->written = layers.length;
1070}

1242inline void Node::capture_layer_into(const Looper::capture_window& w)
1243{
1244 if(layers.reserved_overwritten)
1245 cancel_export();
1246 auto* top = layers.top();
1247 if(!top)
1248 return;
1249 const int64_t len = layers.length;
1250 const int64_t positions = w.positions();
1251 const auto y_last = int64_t(std::floor(w.y_end));
1252 const int64_t first = w.end - w.n;
1253 const int fade = Looper::audio_layers::fade_frames;
1254 const auto input = [&](int c, double t) {
1255 if(c >= m_ring_channels)
1256 return 0.f;
1257 const float* ring = m_ring.data() + std::size_t(c) * std::size_t(m_ring_capacity);
1258 const auto at = [&](int64_t f) {
1259 return ring[std::clamp<int64_t>(f, first, w.end - 1) % m_ring_capacity];
1260 };
1261 const auto f = int64_t(std::floor(t));
1262 const float a = float(t - double(f));
1263 return at(f) + a * (at(f + 1) - at(f));
1264 };
1265 for(int64_t i = 0; i < positions; i++)
1266 {
1267 const int64_t y = y_last - positions + i;
1268 const double t = double(w.end) - (w.y_end - double(y)) / w.rate;
1269 const int64_t yw = Looper::capture_window::wrap(y, len);
1270 const int64_t k = w.backwards ? len - 1 - yw : yw;
1271 const float ramp = i < fade ? float(i) / float(fade) : 1.f;
1272 for(int c = 0; c < layers.channels; c++)
1273 {
1274 const float v = input(c, t) * ramp;
1275 layers.data(top->slot, c)[k] = v;
1276 layers.note_peak(top->slot, k, v);
1277 }
1278 }
1279 top->start = w.start();
1280 top->written = positions;
1281 top->backwards = w.backwards;
1282 if(y_last - positions < 0)
1283 layers.decay(std::clamp(inputs.feedback.value, 0.f, 1.f));
1284}

1289inline void Node::load_file()
1290{
1291 const auto& f = inputs.file.soundfile;
1292 if(f.data != m_file_data || f.frames != m_file_frames)
1293 {
1294 m_file_data = f.data;
1295 m_file_frames = f.frames;
1296 m_loading = false;
1297
1298 if(!inputs.file || transport.is_taking())
1299 return;
1300
1301 const double ratio = f.rate > 0 ? m_rate / f.rate : 1.;
1302 m_file_target = std::min<int64_t>(int64_t(f.frames * ratio), layers.capacity);
1303 if(m_file_target <= 0)
1304 return;
1305 cancel_export();
1306 layers.begin_recording();
1307 m_tail_open = -1;
1308 m_switches.soloed.forget();
1309 // Nothing is heard until the whole file is in.
1310 layers.count = 0;
1311 m_file_pos = 0;
1312 m_loading = true;
1313 m_full = false;
1314 }
1315
1316 if(!m_loading)
1317 return;
1318
1319 const double step = f.rate > 0 ? double(f.rate) / m_rate : 1.;
1320 const int slot = layers.layers[0].slot;
1321 const int64_t end = std::min(m_file_target, m_file_pos + 65536);
1322 for(int c = 0; c < layers.channels; c++)
1323 {
1324 float* dst = layers.data(slot, c);
1325 // A mono file goes to every channel.
1326 const float* src = f.channels > 0 ? f.data[std::min(c, f.channels - 1)] : nullptr;
1327 for(int64_t k = m_file_pos; k < end; k++)
1328 {
1329 const double x = k * step;
1330 const int64_t i = int64_t(x);
1331 const float a = i < f.frames ? src[i] : 0.f;
1332 const float b = i + 1 < f.frames ? src[i + 1] : a;
1333 const float v = a + float(x - i) * (b - a);
1334 dst[k] = v;
1335 layers.note_peak(slot, k, v);
1336 }
1337 }
1338 m_file_pos = end;
1339 if(m_file_pos < m_file_target)
1340 return;
1341
1342 m_loading = false;
1343 m_loop_tempo = m_tempo;
1344 layers.length = m_file_target;
1345 layers.layers[0].written = m_file_target;
1346 layers.count = 1;
1347 layers.total = 1;
1348 fade_edges(slot);
1349 transport.play_captured(0);
1350 m_origin = m_tick_start;
1351}

1355inline void Node::play_stretched(int64_t first_pos, int64_t samples)
1356{
1357 const int64_t len = layers.length;
1358 const int chans = layers.channels;
1359 if(std::ssize(m_stretch_out) < chans || std::ssize(m_stretch_span) < chans
1360 || samples > std::ssize(m_stretch_out[0]))
1361 {
1362 play(first_pos, samples);
1363 return;
1364 }
1365 if(!m_stretch_ready || transport.playbackPos % len != m_expected_pos
1366 || m_stretch_speed != m_speed || m_stretch_reverse != m_reverse)
1367 {
1368 // The stretchers read the tape at m_speed: their frame u is position
1369 // u * m_speed of it, which goes round every len / m_speed frames.
1370 m_stretch_speed = m_speed;
1371 m_stretch_reverse = m_reverse;
1372 const auto u = int64_t(head_y(len) / m_speed);
1373 m_stretcher.transport(u);
1374 m_heard = double(u);
1375 m_cycle_ratio = m_loop_tempo / std::max(1., m_tempo);
1376 m_stretch_ready = true;
1377 }
1378
1379 for(int c = 0; c < chans; c++)
1380 m_stretch_span[c] = std::span<double>(m_stretch_out[c].data(), std::size_t(samples));
1381
1382 ossia::token_request tk;
1383 tk.prev_date = ossia::time_value{0};
1384 tk.date = ossia::time_value{samples};
1385 m_wrapped = false;
1386 // Frames of output per frame of the loop, for this run.
1387 const double ratio = m_stretcher.stretch() ? m_cycle_ratio : 1.;
1388 m_stretcher.run(
1389 *this, tk, ossia_state, m_cycle_ratio, 1., std::size_t(chans), len, samples,
1390 samples, 0, m_stretch_span);
1391 if(m_wrapped)
1392 m_cycle_ratio = m_loop_tempo / std::max(1., m_tempo);
1393
1394 auto& p2 = outputs.audio;
1395 const int loop_chans = std::min(chans, p2.channels);
1396 for(int c = 0; c < loop_chans; c++)
1397 for(int64_t j = 0; j < samples; j++)
1398 p2.samples[c][first_pos + j] = m_stretch_out[c][j];
1399 for(int c = loop_chans; c < p2.channels; c++)
1400 for(int64_t j = 0; j < samples; j++)
1401 p2.samples[c][first_pos + j] = 0.;
1402
1403 // The stretcher reads ahead of what it outputs, by what it keeps buffered
1404 // and its own latency: the position is the one of what was heard, which an
1405 // overdub that carries on from here writes at.
1406 m_heard = std::fmod(
1407 m_heard + double(samples) / std::max(1e-9, ratio), double(len) / m_stretch_speed);
1408 if(m_stretch_speed == 1. && !m_stretch_reverse)
1409 {
1410 transport.playbackPos = int64_t(m_heard) % len;
1411 m_frac = 0.;
1412 m_head_pos = transport.playbackPos;
1413 }
1414 else
1415 {
1416 set_head_y(std::fmod(m_heard * m_stretch_speed, double(len)), len);
1417 }
1418 m_expected_pos = transport.playbackPos;
1419}

1424{

1426 double y = m_reverse ? double(len) - h : h;
1427 if(y >= double(len))
1428 y -= double(len);
1429 if(y < 0.)
1430 y += double(len);

1434inline void Node::set_head_y(double y, int64_t len) noexcept
1435{
1436 double h = m_reverse ? double(len) - y : y;
1437 h = std::fmod(h, double(len));
1438 if(h < 0.)
1439 h += double(len);
1440 const double whole = std::floor(h);
1441 transport.playbackPos = int64_t(whole) % len;
1442 m_frac = h - whole;
1443 // A position that rounds to the loop's end is its start.
1444 if(m_frac >= 1. - 1e-9)
1445 {
1446 transport.playbackPos = (transport.playbackPos + 1) % len;
1447 m_frac = 0.;
1448 }
1449 m_head_pos = transport.playbackPos;

1455inline float Node::read_span(int c, double y, double s, int64_t len) const noexcept
1456{
1457 if(s > 1.)
1458 {
1459 const int n = int(std::lround(s));
1460 const auto k0 = int64_t(y);
1461 float sum = 0.f;
1462 for(int i = 0; i < n; i++)
1463 sum += tape(c, (k0 + i) % len, len);
1464 return sum / float(n);
1465 }
1466 const double centre = y + 0.5 * s - 0.5;
1467 const double whole = std::floor(centre);
1468 const double f = centre - whole;
1469 int64_t k = int64_t(whole) % len;
1470 if(k < 0)
1471 k += len;
1472 const float a = tape(c, k, len);
1473 if(f < 1e-9)
1474 return a;
1475 return a + float(f) * (tape(c, (k + 1) % len, len) - a);
1476}

1479inline void Node::play_tape(int64_t first_pos, int64_t samples)
1480{
1481 auto& p2 = outputs.audio;
1482 const int64_t last = first_pos + samples;
1483 const int out_chans = p2.channels;
1484 const int loop_chans = std::min(layers.channels, out_chans);
1485 const int64_t len = layers.length;
1486 const double s = m_speed;
1487 double y = head_y(len);
1488 // Faster than x1, whole positions are gone over.
1489 if(s > 1.)
1490 y = std::floor(y);
1491
1492 for(int64_t j = first_pos; j < last; j++)
1493 {
1494 layers.now = m_clock + (j - first_pos);
1495 for(int i = 0; i < loop_chans; i++)
1496 p2.samples[i][j] = read_span(i, y, s, len);
1497 y += s;
1498 while(y >= double(len))
1499 y -= double(len);
1500 }
1501 for(int i = loop_chans; i < out_chans; i++)
1502 for(int64_t j = first_pos; j < last; j++)
1503 p2.samples[i][j] = 0.;
1504
1505 set_head_y(y, len);
1506}

1509{

1511 const int64_t last = first_pos + samples;
1512 const int out_chans = p2.channels;
1513 const int loop_chans = std::min(layers.channels, out_chans);
1514 const int64_t len = layers.length;
1515 const int64_t start = transport.playbackPos % len;
1516
1517 for(int i = 0; i < loop_chans; i++)
1518 {
1519 auto& out = p2.samples[i];
1520 // The loop is free to be shorter than the tick, and to end anywhere
1521 // inside it.
1522 int64_t k = start;
1523 for(int64_t j = first_pos; j < last; j++)
1524 {
1525 layers.now = m_clock + (j - first_pos);
1526 out[j] = layers.mix(i, k);
1527 if(++k == len)
1528 k = 0;
1529 }
1530 }
1531
1532 // The bus can carry more channels than the loop was recorded with.
1533 for(int i = loop_chans; i < out_chans; i++)
1534 {
1535 auto& out = p2.samples[i];
1536 for(int64_t j = first_pos; j < last; j++)
1537 out[j] = 0.;
1538 }
1539
1540 // The loop goes on with no output to play it to, e.g. nothing patched in.
1541 transport.playbackPos = (start + samples) % len;
1542}

1594inline int64_t Node::record(int64_t first_pos, int64_t samples, bool echo)
1595{
1596 auto& p1 = inputs.audio;
1597 if(!transport.is_taking())
1598 {
1599 // A new recording: the passes go, and an export of them that was still
1600 // being flattened would read the new one.
1601 cancel_export();
1602 layers.begin_recording();
1603 m_tail_open = -1;
1604 m_switches.soloed.forget();
1605 m_seamless = false;
1606 transport.playbackPos = 0;
1607 m_export_pending = false;
1608 m_full = false;
1609 m_loading = false;
1610 }
1611 transport.taking(LoopMode::Record);
1612
1613 auto& base = layers.layers[0];
1614 const int64_t pos = layers.length;
1615 const int64_t room = std::max<int64_t>(0, layers.capacity - pos);
1616 const int64_t n = std::min(samples, room);
1617 const int chans = std::min(p1.channels, layers.channels);
1618
1619 for(int c = 0; c < layers.channels; c++)
1620 {
1621 float* dst = layers.data(base.slot, c) + pos;
1622 if(c < chans)
1623 {
1624 auto& in = p1.samples[c];
1625 for(int64_t j = 0; j < n; j++)
1626 {
1627 const float v = float(in[first_pos + j]);
1628 dst[j] = v;
1629 layers.note_peak(base.slot, pos + j, v);
1630 }
1631 }
1632 else
1633 {
1634 std::fill_n(dst, n, 0.f);
1635 }
1636 }
1637 layers.length += n;
1638 base.written = layers.length;
1639 transport.playbackPos = layers.length;
1640
1641 echo ? through(first_pos, n) : silence(first_pos, n);
1642
1643 if(n < samples)
1644 {
1645 // The buffer is full: the loop is what fitted, and the post-action takes
1646 // over for the rest of the span.
1647 m_full = true;
1648 transport.end_recording(inputs.after_record);
1649 transport.playbackPos = 0;
1650 }
1651 return n;
1652}

1657inline void Node::write_tape(int64_t y, const float* v, float feedback)
1658{
1659 if(std::exchange(m_cycle_pending, false))
1660 {
1661 // The pass of the cycle that ended is closed as any pass is, but for its
1662 // end: the next one goes on from it.
1663 m_cutting = true;
1664 close_pass(true);
1665 m_cutting = false;
1666 layers.reserved_overwritten = false;
1667 begin_pass(written_at(m_reverse ? layers.length - 1 - y : y, layers.length));
1668 m_pass_in_place = true;
1669 if(layers.reserved_overwritten)
1670 cancel_export();
1671 // It is not faded in where it meets the one before it.
1672 m_pass_frames = continue_seam() ? Looper::audio_layers::fade_frames : 0;
1673 }
1674 auto* t = layers.top();
1675 if(!t)
1676 return;
1677 auto& top = *t;
1678 const int64_t len = layers.length;
1679 const int64_t k = written_at(m_reverse ? len - 1 - y : y, len);
1680 const bool first_time = !layers.valid(top, k);
1681 const int fade = Looper::audio_layers::fade_frames;
1682 const float ramp
1683 = m_pass_frames < fade ? float(m_pass_frames) / fade : 1.f;
1684 m_pass_frames++;
1685 const float g = layers.heard_gain(top);
1686 for(int c = 0; c < layers.channels; c++)
1687 {
1688 float& d = layers.data(top.slot, c)[k];
1689 const float before = first_time ? 0.f : d;
1690 // Later laps of the pass fade what it laid down on the earlier ones.
1691 d = before * feedback + v[c] * ramp;
1692 layers.note_peak(top.slot, k, d);
1693 m_fresh_delta[c] = top.muted ? 0.f : g * (d - before);
1694 m_last_v[c * last_frames + m_last_i] = v[c] * ramp;
1695 }
1696 m_fresh_k = k;
1697 m_last_k[m_last_i] = k;
1698 m_last_i = (m_last_i + 1) % last_frames;
1699 m_last_n = std::min(m_last_n + 1, last_frames);
1700 m_lapped = m_lapped || !first_time;
1701 layers.cover(top, k);
1702 top.backwards = m_reverse;
1703}

1709inline bool Node::cycle_end(int64_t& len, bool extend, float feedback)
1710{
1711 // A cycle is the loop's length when the overdub started: the loop growing
1712 // by one is a cycle gone round as well.
1713 const bool per_cycle = inputs.passes.value == Looper::Passes::PerCycle;
1714 if(extend && layers.extend(m_extend_unit))
1715 {
1716 len = layers.length;
1717 m_extended = true;
1718 if(per_cycle)
1719 m_cycle_pending = true;
1720 return false;
1721 }
1722 // The gains glide while the playhead goes over the frames the seams weigh.
1723 layers.decay(
1724 feedback,
1725 int64_t(std::ceil(double(Looper::audio_layers::fade_frames) / std::max(1e-3, m_speed))));
1726 if(per_cycle)
1727 m_cycle_pending = true;
1728 return true;
1729}

1735inline void Node::begin_pass_carried(int64_t len)
1736{
1737 const int64_t span = m_carried_span;
1738 m_carried_span = 0;
1739 // A pass that carries on a recording starts at full level: it is not faded
1740 // in, as if it had already written its fade.
1741 const bool seamless = std::exchange(m_seamless, false);
1742 const int fade = seamless ? 0 : Looper::audio_layers::fade_frames;
1743 m_pass_frames = seamless ? Looper::audio_layers::fade_frames : 0;
1744 const int64_t k = transport.playbackPos % len;
1745 const bool carried = span > 0 && span < len && k == span && layers.count == 1
1746 && m_speed == 1. && !m_reverse
1747 && len + span <= layers.capacity;
1748 m_fresh_k = -1;
1749 begin_pass(written_at(carried ? 0 : k, len));
1750 m_pass_in_place = true;
1751 if(seamless)
1752 continue_seam();
1753 else
1754 end_open_tail();
1755 auto* top = layers.top();
1756 if(!carried || !top || layers.count != 2)
1757 return;
1758 const auto& base = layers.layers[0];
1759 for(int64_t i = 0; i < span; i++)
1760 {
1761 const float ramp = i < fade ? float(i) / fade : 1.f;
1762 const int64_t at = written_at(i, len);
1763 for(int c = 0; c < layers.channels; c++)
1764 {
1765 const float v = layers.data(base.slot, c)[len + i] * ramp;
1766 layers.data(top->slot, c)[at] = v;
1767 layers.note_peak(top->slot, at, v);
1768 }
1769 layers.cover(*top, at);
1770 }
1771 m_pass_frames = std::max<int64_t>(m_pass_frames, span);
1772}

1774inline void Node::overdub(int64_t first_pos, int64_t samples, bool echo)
1775{
1776 auto& p1 = inputs.audio;
1777 auto& p2 = outputs.audio;
1778 const int64_t last = first_pos + samples;
1779
1780 // Nothing to layer onto: an overdub of an empty loop is an echo.
1781 if(layers.empty())
1782 {
1783 echo ? through(first_pos, samples) : silence(first_pos, samples);
1784 return;
1785 }
1786
1787 const int in_chans = std::min(p1.channels, layers.channels);
1788 const int out_chans = p2.channels;
1789 auto input = [&](int c, int64_t j) {
1790 return c < in_chans ? float(p1.samples[c][j]) : 0.f;
1791 };
1792
1793 int64_t len = layers.length;
1794 if(!transport.is_taking())
1795 {
1796 layers.reserved_overwritten = false;
1797 begin_pass_carried(len);
1798 // No buffer was left but one a take being written still reads.
1799 if(layers.reserved_overwritten)
1800 cancel_export();
1801 // An extending pass grows the loop by the loop it started over.
1802 m_extend_unit = len;
1803 m_extended = false;
1804 m_cycle_pending = false;
1805 m_acc_n = 0;
1806 for(int c = 0; c < layers.channels; c++)
1807 {
1808 m_acc[c] = 0.f;
1809 m_prev_in[c] = input(c, first_pos);
1810 }
1811 }
1812 transport.taking(LoopMode::Overdub);
1813 // Backwards, the playhead reaches the loop's start, not its end.
1814 const bool extend = inputs.past_end.value == Looper::PastEnd::Extend && !m_reverse;
1815
1816 if(!layers.top())
1817 return;
1818
1819 const float feedback = std::clamp(inputs.feedback.value, 0.f, 1.f);
1820 const double s = m_speed;
1821 double y = head_y(len);
1822 if(s >= 1.)
1823 y = std::floor(y);
1824 const int steps = s >= 1. ? int(std::lround(s)) : 1;
1825 std::array<float, Looper::audio_layers::max_channels> v{};
1826
1827 for(int64_t j = first_pos; j < last; j++)
1828 {
1829 layers.now = m_clock + (j - first_pos);
1830 m_hide_fresh = true;
1831 for(int c = 0; c < layers.channels; c++)
1832 {
1833 const float in = input(c, j);
1834 const float before = read_span(c, y, s, len);
1835 if(c < out_chans)
1836 p2.samples[c][j] = echo ? before + in : before;
1837 }
1838 m_hide_fresh = false;
1839 for(int c = layers.channels; c < out_chans; c++)
1840 p2.samples[c][j] = 0.;
1841
1842 if(s >= 1.)
1843 {
1844 // Each frame goes over `steps` positions: the input between the last
1845 // frame and this one, interpolated, the last of them this frame's.
1846 auto w = int64_t(y);
1847 for(int i = 0; i < steps; i++)
1848 {
1849 const float t = float(i + 1) / float(steps);
1850 for(int c = 0; c < layers.channels; c++)
1851 v[c] = steps == 1 ? input(c, j) : m_prev_in[c] + t * (input(c, j) - m_prev_in[c]);
1852 write_tape(w, v.data(), feedback);
1853 if(++w == len && cycle_end(len, extend, feedback))
1854 w = 0;
1855 }
1856 y = double(w);
1857 }
1858 else
1859 {
1860 // A position takes several frames: what came in over them, averaged,
1861 // once the playhead leaves it.
1862 for(int c = 0; c < layers.channels; c++)
1863 m_acc[c] += input(c, j);
1864 m_acc_n++;
1865 const auto k = int64_t(y);
1866 y += s;
1867 if(y >= double(k + 1) - 1e-9)
1868 {
1869 for(int c = 0; c < layers.channels; c++)
1870 {
1871 v[c] = m_acc[c] / float(m_acc_n);
1872 m_acc[c] = 0.f;
1873 }
1874 m_acc_n = 0;
1875 write_tape(k, v.data(), feedback);
1876 if(k + 1 == len && cycle_end(len, extend, feedback))
1877 y -= double(len);
1878 }
1879 }
1880 for(int c = 0; c < layers.channels; c++)
1881 m_prev_in[c] = input(c, j);
1882 }
1883 set_head_y(y, len);
1884}

1911inline void Node::continue_export()
1912{
1913 if(!m_flattening)
1914 return;
1915
1916 if(m_copy_left > 0)
1917 {
1918 const int64_t n = std::min(
1919 m_copy_left,
1920 std::max<int64_t>(65536, work_per_tick / std::max(1, m_export_channels)));
1921 for(int c = 0; c < m_export_channels; c++)
1922 {
1923 const float* src = layers.data(m_copy_from, c);
1924 float* dst = layers.data(m_copy_to, c);
1925 int64_t q = m_copy_pos;
1926 for(int64_t i = 0; i < n; i++)
1927 {
1928 dst[q] = src[q];
1929 q += m_copy_dir;
1930 if(q >= m_copy_period)
1931 q = 0;
1932 else if(q < 0)
1933 q = m_copy_period - 1;
1934 }
1935 }
1936 m_copy_pos = (m_copy_pos + m_copy_dir * n) % m_copy_period;
1937 if(m_copy_pos < 0)
1938 m_copy_pos += m_copy_period;
1939 m_copy_left -= n;
1940 if(m_copy_left > 0)
1941 return;
1942 }
1943
1944 // A bounded amount of work per tick: a minute-long loop of many passes and
1945 // channels does not make one tick a minute of work.
1946 const int64_t per_frame
1947 = int64_t(m_export_channels) * (m_export_item < 0 ? std::max(1, m_export_count) : 1);
1948 const int64_t chunk = std::max<int64_t>(1024, work_per_tick / std::max<int64_t>(1, per_frame));
1949 const int64_t end = std::min(m_export_length, m_flatten_pos + chunk);
1950 flatten(
1951 m_export_snapshot.data(), m_export_count, m_export_item, m_export_length,
1952 m_flatten_pos, end);
1953 m_flatten_pos = end;
1954
1955 if(m_flatten_pos < m_export_length)
1956 return;
1957
1958 m_flattening = false;
1959 m_writing = true;
1960
1961 Looper::take_job job;
1962 job.writer = writer;
1963 job.audio = export_audio;
1964 job.slot = m_export_slot;
1965 job.number = m_export_number;
1966 job.layer = m_export_item;
1967 job.channels = m_export_channels;
1968 job.frames = m_export_length;
1969 job.rate = int(m_rate);
1970 job.tempo = m_export_tempo;

2097inline void Node::scan_peaks(bool whole)
2098{
2099 using layers_t = Looper::audio_layers;
2100 constexpr int bins = Looper::view_message::bins;
2101 const int count = layers.count;
2102 const int64_t len = layers.length;
2103 const int lanes = std::min(count, Looper::view_message::max_lanes);
2104 if(count != m_scan_count || len != m_scan_length || layers.changes != m_scan_changes)
2105 {
2106 m_scan_count = count;
2107 m_scan_length = len;
2108 m_scan_changes = layers.changes;
2109 m_scan_layer = 0;
2110 m_scan_pos = 0;
2111 std::fill(m_scan_peaks.begin(), m_scan_peaks.end(), 0.f);
2112 }
2113
2114 const int64_t micro = (len + layers_t::peak_frames - 1) / layers_t::peak_frames;
2115 int64_t budget = whole ? std::numeric_limits<int64_t>::max() / 2 : work_per_tick / 8;
2116 while(m_scan_layer < count && budget > 0)
2117 {
2118 const int i = m_scan_layer;
2119 const int lane = std::max(0, i - (count - lanes));
2120 const auto& l = layers.layers[i];
2121 const float* p = layers.peaks_of(l.slot);
2122 float* dst = m_scan_peaks.data() + lane * bins;
2123 // A pass shorter than the loop plays again over the rest of it.
2124 const int64_t own = std::max<int64_t>(
2125 1, (layers.period_of(l) + layers_t::peak_frames - 1) / layers_t::peak_frames);
2126 const int64_t end = std::min(micro, m_scan_pos + budget);
2127 // A muted pass of a lane of its own is drawn as it would be heard, the
2128 // view dims it; in the lane of the oldest passes, it is left out.
2129 const float gain = count > lanes && i <= count - lanes ? l.gain : l.unmuted_gain;
2130 for(int64_t m = m_scan_pos; m < end; m++)
2131 {
2132 const int b = int(m * bins / micro);
2133 dst[b] = std::max(dst[b], p[m % own] * gain);
2134 }
2135 budget -= end - m_scan_pos;
2136 m_scan_pos = end;
2137 if(m_scan_pos >= micro)
2138 {
2139 m_scan_layer++;
2140 m_scan_pos = 0;
2141 }
2142 }
2143
2144 if(m_scan_layer < count)
2145 return;
2146 m_view.lanes = lanes;
2147 m_view.peaks = m_scan_peaks;
2148 // The next message goes through the passes again: gains change as an
2149 // overdub goes round, and what it records with them.
2150 m_scan_layer = 0;
2151 m_scan_pos = 0;
2152 std::fill(m_scan_peaks.begin(), m_scan_peaks.end(), 0.f);
2153}
