136// PQ EOTF (PQ -> linear, inverse of PQ OETF, for decoding PQ-encoded input)
137#define SCORE_GFX_PQ_EOTF R"_(
138vec3 pqEotf(vec3 pq) {
139 const float m1 = 0.1593017578125;
140 const float m2 = 78.84375;
141 const float c1 = 0.8359375;
142 const float c2 = 18.8515625;
143 const float c3 = 18.6875;
144 vec3 Nm2 = pow(pq, vec3(1.0 / m2));
145 return pow(max(Nm2 - c1, 0.0) / (c2 - c3 * Nm2), vec3(1.0 / m1));
146}
147)_"
148
149// HLG EOTF (HLG -> linear, inverse of HLG OETF)
150#define SCORE_GFX_HLG_EOTF R"_(
151vec3 hlgEotf(vec3 hlg) {
152 const float a = 0.17883277;
153 const float b = 0.28466892;
154 const float c = 0.55991073;
155 vec3 lo = hlg * hlg / 3.0;
156 vec3 hi = (exp((hlg - c) / a) + b) / 12.0;
157 return mix(lo, hi, step(vec3(0.5), hlg));
158}
159)_"
160
161// sRGB -> linear (inverse sRGB OETF, used before gamut conversion for HDR output)
162#define SCORE_GFX_INVERSE_SRGB_OETF R"_(
163vec3 inverseSrgbOetf(vec3 c) {
164 vec3 lo = c / 12.92;
165 vec3 hi = pow((c + 0.055) / 1.055, vec3(2.4));
166 return mix(lo, hi, step(vec3(0.04045), c));
167}
168)_"
169
170// Linear -> PQ (SMPTE ST 2084 OETF, for HDR10 output)
171// Input: linear light in [0, 1] normalized to SDR peak (1.0 = sdrPeakNits)
172// Output: PQ-encoded [0, 1]
173#define SCORE_GFX_PQ_OETF R"_(
174vec3 pqOetf(vec3 linearRgb) {
175 const float m1 = 0.1593017578125;
176 const float m2 = 78.84375;
177 const float c1 = 0.8359375;
178 const float c2 = 18.8515625;
179 const float c3 = 18.6875;
180 vec3 Ym1 = pow(linearRgb, vec3(m1));
181 return pow((c1 + c2 * Ym1) / (1.0 + c3 * Ym1), vec3(m2));
182}
183)_"
184
185// Linear -> HLG (ARIB STD-B67 OETF, for HLG output)
186// Input: linear light in [0, 1] (scene-referred)
187// Output: HLG-encoded [0, 1]
188#define SCORE_GFX_HLG_OETF R"_(
189vec3 hlgOetf(vec3 linearRgb) {
190 const float a = 0.17883277;
191 const float b = 0.28466892; // 1.0 - 4.0 * a
192 const float c = 0.55991073; // 0.5 - a * ln(4.0 * a)
193 vec3 lo = sqrt(3.0 * linearRgb);
194 vec3 hi = a * log(12.0 * linearRgb - b) + c;
195 return mix(lo, hi, step(vec3(1.0 / 12.0), linearRgb));
196}
197)_"

199// Linear -> BT.709 gamma (approximate 0.45 power)
200#define SCORE_GFX_BT709_OETF R"_(
201vec3 bt709Oetf(vec3 linearRgb) {
202 vec3 lo = 4.5 * linearRgb;
203 vec3 hi = 1.099 * pow(linearRgb, vec3(0.45)) - 0.099;
204 return mix(lo, hi, step(vec3(0.018), linearRgb));
205}
206)_"

237// Builds an SDR output shader with the given matrix macro.
238// Handles all input transfer functions for SDR YUV output.
239static inline QString sdrOutShader(
240 const char* matrix, AVColorTransferCharacteristic input_trc)
241{
242 QString shader;
243 shader.reserve(2048);
244
245 if(input_trc == AVCOL_TRC_LINEAR)
246 {
247 // Linear -> sRGB gamma -> YUV matrix
248 shader += SCORE_GFX_SRGB_OETF_OUT;
249 shader += matrix;
250 shader += "vec3 convert_from_rgb(vec3 rgb) { return (encode_matrix * vec4(srgbOetfOut(rgb), 1.0)).xyz; }\n";
251 }
252 else if(input_trc == AVCOL_TRC_SMPTE2084)
253 {
254 // PQ -> linear -> sRGB gamma -> YUV matrix
255 shader += SCORE_GFX_PQ_EOTF;
256 shader += SCORE_GFX_SRGB_OETF_OUT;
257 shader += matrix;
258 shader += "vec3 convert_from_rgb(vec3 rgb) { return (encode_matrix * vec4(srgbOetfOut(pqEotf(rgb)), 1.0)).xyz; }\n";
259 }
260 else if(input_trc == AVCOL_TRC_ARIB_STD_B67)
261 {
262 // HLG -> linear -> sRGB gamma -> YUV matrix
263 shader += SCORE_GFX_HLG_EOTF;
264 shader += SCORE_GFX_SRGB_OETF_OUT;
265 shader += matrix;
266 shader += "vec3 convert_from_rgb(vec3 rgb) { return (encode_matrix * vec4(srgbOetfOut(hlgEotf(rgb)), 1.0)).xyz; }\n";
267 }
268 else
269 {
270 // sRGB / passthrough / unspecified -> direct matrix multiply
271 shader += matrix;
272 shader += "vec3 convert_from_rgb(vec3 rgb) { return (encode_matrix * vec4(rgb, 1.0)).xyz; }\n";
273 }
274 return shader;
275}

513static inline QString colorMatrixOut(
514 AVColorSpace color_space,
515 AVColorTransferCharacteristic color_trc,
516 AVColorRange color_range,
517 AVColorPrimaries color_primaries,
518 AVColorTransferCharacteristic input_trc = DefaultInputTrc)
519{
520 switch(color_space)
521 {
522 case AVCOL_SPC_RGB:
523 return "vec3 convert_from_rgb(vec3 rgb) { return rgb; }\n";
524
525 case AVCOL_SPC_BT709:
526 return bt709OutShader(color_range, input_trc);
527
528 case AVCOL_SPC_BT470BG:
529 case AVCOL_SPC_SMPTE170M:
530 return bt601OutShader(color_range, input_trc);
531
532 case AVCOL_SPC_SMPTE240M:
533 {
534 const bool full = (color_range == AVCOL_RANGE_JPEG);
535 if(full)
536 {
537 QString shader;
538 shader += "const mat4 encode_matrix = " SCORE_GFX_RGB_TO_SMPTE240M_FULL_MATRIX ";\n";
539 shader += "vec3 convert_from_rgb(vec3 rgb) { return (encode_matrix * vec4(rgb, 1.0)).xyz; }\n";
540 return shader;
541 }
542 return bt709OutShader(color_range, input_trc);
543 }
544
545 case AVCOL_SPC_BT2020_NCL:
546 case AVCOL_SPC_BT2020_CL:
547 if(color_trc == AVCOL_TRC_SMPTE2084)
548 return bt2020PqOutShader(color_range, input_trc);
549 else if(color_trc == AVCOL_TRC_ARIB_STD_B67)
550 return bt2020HlgOutShader(color_range, input_trc);
551 else
552 return bt2020SdrOutShader(color_range, input_trc);
553
554 default:
555 break;
556 }
557
558 // Fallback based on primaries
559 if(color_primaries == AVCOL_PRI_BT2020)
560 {
561 if(color_trc == AVCOL_TRC_SMPTE2084)
562 return bt2020PqOutShader(color_range, input_trc);
563 else if(color_trc == AVCOL_TRC_ARIB_STD_B67)
564 return bt2020HlgOutShader(color_range, input_trc);
565 else
566 return bt2020SdrOutShader(color_range, input_trc);
567 }
568
569 // Ultimate fallback: BT.709
570 return bt709OutShader(color_range, input_trc);
571}
