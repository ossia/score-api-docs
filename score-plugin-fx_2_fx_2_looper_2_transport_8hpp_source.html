394 void press(TransportAction a, bool empty, AfterRecord after_record, AfterOverdub after) noexcept
395 {
396 const LoopMode now = m_queued > 0 ? m_queue[m_queued - 1].mode : wanted;
397 // A recording waiting makes the loop that the next change acts on.
398 bool none = empty && now != LoopMode::Record;
399 for(int i = 0; i < m_queued; i++)
400 none = none && m_queue[i].mode != LoopMode::Record;
401 none = none && wanted != LoopMode::Record;
402
403 switch(a)
404 {
405 case TransportAction::RecPlayDub:
406 if(now == LoopMode::Record)
407 push({end_of_recording(after_record), false});
408 else if(none)
409 push({LoopMode::Record, false});
410 else if(now == LoopMode::Play)
411 push({LoopMode::Overdub, false});
412 else if(now == LoopMode::Overdub)
413 push({after == AfterOverdub::Play ? LoopMode::Play : LoopMode::Overdub,
414 after == AfterOverdub::NewLayer});
415 else
416 push({LoopMode::Play, false});
417 break;
418 case TransportAction::PlayStop:
419 if(now != LoopMode::Stop)
420 push({LoopMode::Stop, false});
421 else if(!none)
422 push({LoopMode::Play, false});
423 break;
424 case TransportAction::NewLayer:
425 if(now == LoopMode::Overdub && !empty)
426 push({LoopMode::Overdub, true});
427 break;
428 case TransportAction::Retrigger:
429 if(!empty || wanted == LoopMode::Record)
430 m_restart = true;
431 break;
432 }
433 }

506 template <typename Engine>
507 void operator()(
508 const ossia::token_request& tk, ossia::exec_state_facade st,
509 const transport_settings& s, Engine& e)
510 {
511 using namespace ossia;
512
513 m_facade = st;
514 m_tick = tk;
515 const auto timings = st.timings(tk);
516 this->tickStartSample = timings.start_sample;
517 this->postaction_bars = s.postaction_bars;
518 this->sampleRate = st.sampleRate();
519 this_buffer_quantif_time = std::nullopt;
520
521 const LoopMode m = s.mode;
522 const bool on_grid = !no_grid(s.quantif) && tk.prev_date != 0_tv;
523 const loop_period period = e.looper_period(tk);
524 count_rounds(period);
525
526 // The timeline jumped, e.g. the score was moved to another date: a loop
527 // in phase with it goes there as well.
528 if(s.start_from == StartFrom::InPhase && m_last_date && tk.prev_date != *m_last_date
529 && actualMode == LoopMode::Play && m_taking == LoopMode::Stop)
530 m_rephase = true;
531 m_last_date = tk.date;
532
533 if(on_grid)
534 {
535 this->quantif = s.quantif;
536 // A point on the tick's very first sample counts. Bar lines land there
537 // whenever the bar divides evenly into the buffer -- which at 120bpm and
538 // 48kHz is every buffer of 256 frames or fewer -- and refusing them left
539 // the mode waiting for a point that would never come.
540 if(const auto pt = grid_point(tk, s, period, timings.length))
541 {
542 this_buffer_quantif_time = date_at(tk, *pt, timings.length);
543 m_point_this_tick = frame + *pt;
544 }
545 }
546 else
547 {
548 this->quantif = {};
549 }
550
551 if(m != quantizedPlayMode || m_restart || m_retake || m_point_requested)
552 {
553 const bool stopping = m == LoopMode::Stop && quantizedPlayMode != LoopMode::Stop
554 && !m_restart && !m_retake && !m_point_requested;
555 const int64_t to_end = stopping && s.stop_at == StopAt::LoopEnd
556 ? e.looper_frames_to_end()
557 : int64_t(-1);
558 if(late_extension_stop(s, e))
559 {
560 take_up(tk, s, e, m_late_frames);
561 preAction(tk, s, e);
562 }
563 else if(to_end >= 0)
564 {
565 if(to_end == 0)
566 {
567 take_up(tk, s, e);
568 preAction(tk, s, e);
569 }
570 else if(to_end < timings.length)
571 {
572 const auto at = tk.prev_date
573 + ossia::time_value{int64_t(
574 std::round(double(to_end) * st.samplesToModel()))};
575 auto sub_tk = tk;
576 sub_tk.set_end_time(at);
577 preAction(sub_tk, s, e);
578 sub_tk = tk;
579 sub_tk.set_start_time(at);
580 take_up(sub_tk, s, e);
581 preAction(sub_tk, s, e);
582 }
583 else
584 {
585 preAction(tk, s, e);
586 }
587 }
588 else if(on_grid && !(stopping && s.stop_at == StopAt::Now))
589 {
590 if(late_stop(s, e))
591 {
592 m_closing_late = true;
593 take_up(tk, s, e, m_late_frames);
594 m_closing_late = false;
595 preAction(tk, s, e);
596 }
597 else if(auto& time = this_buffer_quantif_time)
598 {
599 // tempo = 60 -> 1 quarter = 1 second
600 // tempo = 100 -> 1 quarter = 1 * 60/100 = 0.6 second
601 // tempo = 100 -> 1 bar = 2 second
602 // tempo = 120 -> 1 bar = 2 second
603 if(*time > tk.prev_date)
604 {
605 // Finish what we were doing until the quantization date. The span
606 // is half-open, so the two halves tile the tick exactly; ending a
607 // flick early instead rounds down to a sample nobody writes.
608 {
609 auto sub_tk = tk;
610 sub_tk.set_end_time(*time);
611 preAction(sub_tk, s, e);
612 }
613
614 // We can switch to the new mode, for the remaining of the tick
615 {
616 auto sub_tk = tk;
617 sub_tk.set_start_time(*time);
618 take_up(sub_tk, s, e);
619 preAction(sub_tk, s, e);
620 }
621 }
622 else
623 {
624 // We can switch to the new mode
625 take_up(tk, s, e);
626 preAction(tk, s, e);
627 }
628 }
629 else
630 {
631 // We cannot switch yet
632 preAction(tk, s, e);
633 }
634 }
635 else
636 {
637 // No quantization, we can switch to the new mode
638 take_up(tk, s, e);
639 preAction(tk, s, e);
640 }
641 }
642 else
643 {
644 // No change
645 preAction(tk, s, e);
646 }
647
648 if(m_point_this_tick)
649 {
650 lastPointFrame = m_point_this_tick;
651 m_point_this_tick.reset();
652 }
653 frame += timings.length;
654
655 // What was asked for is taken up: the change queued after it waits for
656 // the next grid point.
657 if(m_queued > 0 && wanted == quantizedPlayMode && !m_retake)
658 {
659 const step next = m_queue[0];
660 std::move(m_queue.begin() + 1, m_queue.begin() + m_queued, m_queue.begin());
661 m_queued--;
662 take_step(next);
663 }
664 }

699 void count_rounds(const loop_period& p) noexcept
700 {
701 if(p.length <= 0 || !p.going_round)
702 {
703 m_rounds = 0;
704 m_last_round_position = -1;
705 return;
706 }
707 // Back past half the loop is a restart: the end reached, or Retrigger.
708 if(m_last_round_position >= 0 && p.position + p.length / 2 < m_last_round_position)
709 m_rounds++;
710 m_last_round_position = p.position;

717 std::optional<int> grid_point(
718 const ossia::token_request& tk, const transport_settings& s,
719 const loop_period& period, int64_t frames) const noexcept
720 {
721 halp::tick_musical t{};
722 t.frames = int(frames);
723 t.tempo = tk.tempo;
724 t.signature.num = tk.signature.upper;
725 t.signature.denom = tk.signature.lower;
726 t.position_in_frames = frame;
727 t.start_position_in_quarters = tk.musical_start_position;
728 t.end_position_in_quarters = tk.musical_end_position;
729 t.last_signature_change = tk.musical_start_last_signature;
730 t.bar_at_start = tk.musical_start_last_bar;
731 t.bar_at_end = tk.musical_end_last_bar;
732
733 halp::grid_value g = s.quantif;
734 halp::grid_period p{};
735 if(period.length > 0)
736 {
737 p.length = period.length;
738 p.position = period.going_round ? m_rounds * period.length + period.position
739 : period.position;
740 }
741 else if(g.unit == halp::grid_unit::period)
742 {
743 g = halp::grid_value::bars(g.value * std::max(1, s.postaction_bars));
744 }
745 return halp::next_grid_point(g, t, sampleRate, p);
746 }

783 template <typename Engine>
784 bool late_stop(const transport_settings& s, Engine& e)
785 {
786 if(actualMode != LoopMode::Record || m_taking != LoopMode::Record)
787 return false;
788 if(s.mode == LoopMode::Record || s.late_seconds <= 0.)
789 return false;
790 // A point on this tick's first sample is not late: switch on it.
791 if(this_buffer_quantif_time && m_point_this_tick && *m_point_this_tick == frame)
792 return false;
793 if(!lastPointFrame)
794 return false;
795
796 const int64_t late = frame - *lastPointFrame;
797 if(late <= 0 || late > int64_t(s.late_seconds * sampleRate))
798 return false;
799
800 e.looper_trim_recording(late);
801 m_late_frames = late;
802 return true;
803 }

811 template <typename Engine>
812 bool late_extension_stop(const transport_settings& s, Engine& e)
813 {
814 if(actualMode != LoopMode::Overdub || m_taking != LoopMode::Overdub)
815 return false;
816 // Only the end of the pass is late: a restart carries it on.
817 if(s.mode == quantizedPlayMode && !m_retake)
818 return false;
819 const int64_t into = e.looper_trim_extension(
820 std::max<int64_t>(0, int64_t(s.late_seconds * sampleRate)));
821 if(into < 0)
822 return false;
823 m_late_frames = into;
824 return true;
825 }

859 template <typename Engine>
860 void take_up(
861 const ossia::token_request& tk, const transport_settings& s, Engine& e,
862 std::optional<int64_t> position = {})
863 {
864 if(std::exchange(m_point_requested, false))
865 e.looper_point();
866 const bool change = s.mode != quantizedPlayMode;
867 const bool retake = std::exchange(m_retake, false);
868 const bool restart = std::exchange(m_restart, false);
869 if(change)
870 enterMode(s.mode, s.start_from);
871 // A recording that starts again drops what it had: nothing to close.
872 const bool rerecord = restart && !change && actualMode == LoopMode::Record;
873 if(position)
874 {
875 playbackPos = *position;
876 }
877 else if(restart)
878 {
879 playbackPos = 0;
880 m_rephase = false;
881 if(rerecord)
882 m_taking = LoopMode::Stop;
883 }
884 // A restart alone carries the pass on from the loop's start.
885 if(change || retake || rerecord)
886 changeAction(tk, e);
887 }

903 void begin_recording_window(const ossia::token_request& tk) noexcept
904 {
905 // Zero bars means no automatic end: the end bar would otherwise be the
906 // start bar and preAction() would leave Record on its first tick.
907 if(postaction_bars > 0)
908 {
909 recordStartBar = tk.musical_start_position;
910 recordEndBar = tk.musical_start_position
911 + (4. * double(tk.signature.upper) / tk.signature.lower)
912 * postaction_bars;
913 }
914 else
915 {
916 recordStartBar = -1.;
917 recordEndBar = -1.;
918 }
919 endedByPostaction = false;
920 recordStartDate = tk.prev_date;
921 recordStartPosition = tk.musical_start_position;
922 }

939 template <typename Engine>
940 void preAction(ossia::token_request tk, const transport_settings& s, Engine& e)
941 {
942 using namespace ossia;
943 if(recordStartBar == -1.)
944 {
945 action(tk, e);
946 }
947 else
948 {
949 // If we are in the token request which steps into the bar in which
950 // we change because we are e.g. 4 bars after the recording started
951 auto switch_to_main_mode = [&] {
952 actualMode = end_of_recording(s.after_record);
953 recording_handed_over();
954
955 recordStartBar = -1.;
956 endedByPostaction = true;
957 if(s.start_from == StartFrom::LoopStart)
958 playbackPos = 0;
959 };
960
961 // Change of bar at the first sample
962 if(tk.musical_start_last_bar >= recordEndBar)
963 {
964 switch_to_main_mode();
965 close(tk, e);
966 action(tk, e);
967 }
968
969 // Change of bar in the middle
970 else if(tk.musical_end_last_bar >= recordEndBar)
971 {
972 const auto quant_date = tk.get_quantification_date(1.0);
973 if(quant_date && *quant_date > tk.prev_date && *quant_date < tk.date)
974 {
975 const ossia::time_value t = *quant_date;
976
977 // Finish what we were doing until the quantization date
978 {
979 auto sub_tk = tk;
980 sub_tk.set_end_time(t);
981 action(sub_tk, e);
982 }
983
984 // We can switch to the new mode. The take closes on the bar line,
985 // which is where the loop starts again.
986 auto sub_tk = tk;
987 sub_tk.set_start_time(t);
988 switch_to_main_mode();
989 close(sub_tk, e);
990
991 // Remaining of the tick
992 action(sub_tk, e);
993 }
994 else if(tk.musical_end_position - tk.musical_end_last_bar < 1e-9)
995 {
996 // The bar line is this tick's own end, so the whole of it still
997 // belongs to what came before -- and a point there is not reported,
998 // the tick's end being exclusive. The next tick starts on the line
999 // and takes the switch; ending here instead drops this tick's
1000 // recording, which comes back as a buffer of silence in the loop.
1001 action(tk, e);
1002 }
1003 else
1004 {
1005 // The bar line is not locatable inside the tick: switch on its first
1006 // sample rather than leaving the buffer unwritten.
1007 switch_to_main_mode();
1008 close(tk, e);
1009 action(tk, e);
1010 }
1011 }
1012
1013 // No change of bar yet, we continue
1014 else
1015 {
1016 action(tk, e);
1017 }
1018 }
1019 }

1021 template <typename Engine>
1022 void action(const ossia::token_request& tk, Engine& e)
1023 {
1024 const auto [start, length] = m_facade.timings(tk);
1025 const int64_t first = start - tickStartSample;
1026 if(std::exchange(m_rephase, false))
1027 e.looper_rephase(tk);
1028 if(std::exchange(m_rerecord, false) && actualMode == LoopMode::Record)
1029 begin_recording_window(tk);
1030 const int64_t done = e.looper_run(first, length);
1031 // The engine ended the recording itself (its buffer is full): the take is
1032 // closed, then the rest of the span is After record's.
1033 if(m_forced_close)
1034 {
1035 m_forced_close = false;
1036 // The loop starts again where the buffer filled up.
1037 m_close_offset = done;
1038 close(tk, e);
1039 m_close_offset = 0;
1040 if(done < length)
1041 e.looper_run(first + done, length - done);
1042 }
1043 }

1045 ossia::exec_state_facade m_facade{};
1046 int64_t m_close_offset{};
1047};
