110constexpr Sampling sampleElements(
111 int64_t available, int64_t first, int64_t max_elements, int64_t element_bytes,
112 int64_t cap_bytes, bool overview) noexcept
113{
114 Sampling s;
115 s.first = std::clamp<int64_t>(first, 0, std::max<int64_t>(available, 0));
116 s.window = std::max<int64_t>(available, 0) - s.first;
117 if(s.window <= 0)
118 return s;
119 const int64_t fit = std::max<int64_t>(1, cap_bytes / std::max<int64_t>(element_bytes, 1));
120 const int64_t asked = std::max<int64_t>(1, max_elements);
121 const int64_t limit = std::min(asked, fit);
122 s.capped = fit < asked && s.window > fit;
123 if(overview && s.window > limit)
124 {
125 s.step = (s.window + limit - 1) / limit;
126 s.count = (s.window + s.step - 1) / s.step;
127 }
128 else
129 s.count = std::min(s.window, limit);
130 return s;
131}

154inline float halfToFloat(uint16_t h) noexcept
155{
156 const uint32_t sign = uint32_t(h & 0x8000u) << 16;
157 const uint32_t exponent = (h >> 10) & 0x1fu;
158 const uint32_t mantissa = h & 0x3ffu;
159 if(exponent == 0)
160 {
161 const float f = std::ldexp(float(mantissa), -24);
162 return sign ? -f : f;
163 }
164 if(exponent == 31)
165 return std::bit_cast<float>(sign | 0x7f800000u | (mantissa << 13));
166 return std::bit_cast<float>(sign | ((exponent + 112u) << 23) | (mantissa << 13));
167}

170inline float readComponent(ElementType t, const unsigned char* p) noexcept
171{
172 switch(t)

196 case ElementType::UInt8:

208decode(const Layout& l, std::span<const unsigned char> bytes, std::span<float> out) noexcept
209{
210 const int n = components(l);
211 const int cb = componentBytes(l.type);
212 const int64_t stride = strideOf(l);
213 const int64_t count = std::min(

215 const unsigned char* src = bytes.data() + std::max<int64_t>(l.offset, 0);
216 float* dst = out.data();
217 for(int64_t i = 0; i < count; i++, src += stride)
218 for(int c = 0; c < n; c++)

220 return count;
221}
