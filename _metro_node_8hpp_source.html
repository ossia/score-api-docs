65inline void fade_tail(click_sound& s, int rate) noexcept
66{
67 const int64_t n = std::min<int64_t>(s.frames(), std::max(1, rate / 500));
68 for(auto* ch : {&s.left, &s.right})
69 {
70 if(ch->empty())
71 continue;
72 const int64_t end = ch->size();
73 for(int64_t i = 0; i < n; i++)
74 (*ch)[end - 1 - i] *= float(i) / float(n);
75 }
76}

81inline click_sound make_builtin_sound(sound_kind kind, bool accent, int rate)
82{
83 constexpr double two_pi = 2. * std::numbers::pi;
84 click_sound s;
85 if(rate <= 0)
86 return s;
87
89 struct partial
90 {
91 double freq;
92 double level;
93 };
94 const auto render = [&](double seconds, double decay, double attack, double level,
95 std::initializer_list<partial> partials) {
96 const int64_t n = std::max<int64_t>(1, std::llround(seconds * rate));
97 s.left.resize(n);
98 for(int64_t i = 0; i < n; i++)
99 {
100 const double t = double(i) / rate;
101 double v = 0.;
102 for(const auto& p : partials)
103 // A partial above the Nyquist frequency would fold back.
104 if(p.freq < 0.45 * rate)
105 v += p.level * std::sin(two_pi * p.freq * t);
106 const double env
107 = std::exp(-t / decay) * (attack > 0. ? std::min(1., t / attack) : 1.);
108 s.left[i] = float(level * env * v);
109 }
110 };
111
112 switch(kind)
113 {
114 case Woodblock:
115 if(accent)
116 render(0.12, 0.016, 0.0003, 0.6, {{1180., 1.}, {3210., 0.35}, {5290., 0.12}});
117 else
118 render(0.12, 0.016, 0.0003, 0.5, {{820., 1.}, {2230., 0.35}, {3680., 0.12}});
119 break;
120 case Beep:
121 if(accent)
122 render(0.07, 0.03, 0.002, 0.7, {{1760., 1.}});
123 else
124 render(0.07, 0.03, 0.002, 0.6, {{880., 1.}});
125 break;
126 case Cowbell:
127 // The two detuned tones of the analog drum machines' cowbell, with the
128 // first harmonics a square wave would give them.
129 if(accent)
130 render(
131 0.3, 0.07, 0.0005, 0.3,
132 {{800., 1.}, {1200., 1.}, {2400., 0.33}, {3600., 0.33}, {4000., 0.2}});
133 else
134 render(
135 0.25, 0.05, 0.0005, 0.26,
136 {{540., 1.}, {800., 1.}, {1620., 0.33}, {2400., 0.33}, {2700., 0.2}});
137 break;
138 case Click:
139 default:
140 if(accent)
141 render(0.03, 0.003, 0., 0.9, {{4000., 1.}});
142 else
143 render(0.03, 0.003, 0., 0.8, {{2500., 1.}});
144 break;
145 }
146 fade_tail(s, rate);
147 return s;
148}

187template <typename F>
188void for_each_click(
189 const ossia::token_request& tk, int unit, int subdivision, F&& f) noexcept
190{
191 const double ms = tk.musical_start_position;
192 const double me = tk.musical_end_position;
193 const bool rewinding = tk.backward();
194 if(tk.paused() || ms == me || (me < ms) != rewinding)
195 return;
196
197 const auto& sig = tk.signature;
198 const double beat
199 = (unit == SignatureBeat && sig.upper > 0 && sig.lower > 0) ? 4. / sig.lower : 1.;
200 subdivision = std::clamp(subdivision, 1, 16);
201 const double step = beat / subdivision;
202
203 constexpr double eps = 1e-9;
204 const double lo = rewinding ? me : ms;
205 const double hi = rewinding ? ms : me;
206
207 tk.for_each_bar_segment(lo, hi, [&](double bar_line, double next_bar) {
208 // The first point of this bar the tick can hold.
209 int64_t k = std::max<int64_t>(0, int64_t(std::floor((lo - bar_line) / step)) - 1);
210 // A tick never holds more clicks than this, whatever it is handed.
211 for(int guard = 0; guard < 4096; guard++, k++)
212 {
213 const double p = bar_line + k * step;
214 if(p >= next_bar - eps || p > hi)
215 break;
216 if(rewinding ? (p <= lo) : (p < lo || p >= hi))
217 continue;
218 f(p,
219 k == 0 ? click_kind::Accent
220 : (k % subdivision == 0 ? click_kind::Beat : click_kind::Subdivision));
221 }
222 });
223}

343 void run(const ossia::token_request& tk, ossia::exec_state_facade st) noexcept override
344 {
345 // Before any early return: the inlets are cleared once the node has run.
346 read_controls();
347
348 const auto [tick_start, length] = st.timings(tk);
349 if(length <= 0)
350 return;
351
352 // The clicks of this tick, in the order of their samples.
353 struct event
354 {
355 int64_t offset;
356 click_kind kind;
357 };
358 std::array<event, max_events> events;
359 int n_events = 0;
360 const double ratio = st.modelToSamples();
361 for_each_click(tk, beat, subdivision, [&](double position, click_kind kind) {
362 if(n_events >= max_events)
363 return;
364 const int64_t offset
365 = std::clamp<int64_t>(tk.physical_position(position, ratio), 0, length - 1);
366 // Rewinding, the bars are still walked in musical order, which is the
367 // reverse of the samples'.
368 int at = n_events++;
369 while(at > 0 && events[at - 1].offset > offset)
370 {
371 events[at] = events[at - 1];
372 at--;
373 }
374 events[at] = {offset, kind};
375 });
376
377 auto& port = *audio_out.target<ossia::audio_port>();
378 port.set_channels(2);
379 auto& ap = port.get();
380 const std::size_t needed = std::size_t(tick_start + length);
381 for(auto& channel : ap)
382 if(channel.size() < needed)
383 channel.resize(std::max<std::size_t>(needed, st.bufferSize()));
384 double* const out[2] = {ap[0].data() + tick_start, ap[1].data() + tick_start};
385
386 // The volume moves over the whole tick rather than stepping at its start.
387 const float target = std::isfinite(volume) ? std::clamp(volume, 0.f, 4.f) : 0.f;
388 const float from = m_last_volume < 0.f ? target : m_last_volume;
389 const float volume_step = (target - from) / float(length);
390 m_last_volume = target;
391
392 const sound_set* set = current_sounds();
393 auto& bangs = *bang_out.target<ossia::value_port>();
394
395 int64_t cursor = 0;
396 for(int i = 0; i < n_events; i++)
397 {
398 const auto& ev = events[i];
399 render(out, cursor, ev.offset, from, volume_step);
400 cursor = ev.offset;
401
402 bangs.write_value(ossia::impulse{}, tick_start + ev.offset);
403 if(set)
404 start_voice(*set, ev.kind);
405 }
406 render(out, cursor, length, from, volume_step);
407 }

448 void start_voice(const sound_set& set, click_kind kind) noexcept
449 {
450 const click_sound* snd = (kind == click_kind::Accent && accent) ? &set.accent : &set.beat;
451 // One file is enough for both.
452 if(snd->empty())
453 snd = snd == &set.accent ? &set.beat : &set.accent;
454
455 voice* free_voice{};
456 voice* oldest{&m_voices[0]};
457 for(auto& v : m_voices)
458 {
459 if(!v.snd)
460 {
461 if(!free_voice)
462 free_voice = &v;
463 continue;
464 }
465 if(v.fade_left < 0)
466 {
467 // Shorter when the click ends before: the fade still starts at 1.
468 v.fade_total
469 = int32_t(std::max<int64_t>(1, std::min<int64_t>(m_fade, v.snd->frames() - v.pos)));
470 v.fade_left = v.fade_total;
471 }
472 if(v.pos > oldest->pos)
473 oldest = &v;
474 }
475 if(snd->empty())
476 return;
477
478 // All of them fading: the one closest to silence makes room.
479 auto& v = free_voice ? *free_voice : *oldest;
480 v.snd = snd;
481 v.pos = 0;
482 v.gain = kind == click_kind::Subdivision ? subdivision_level : 1.f;
483 v.fade_left = -1;
484 }

487 void render(
488 double* const out[2], int64_t begin, int64_t end, float volume_from,
489 float volume_step) noexcept
490 {
491 if(end <= begin)
492 return;
493 for(auto& v : m_voices)
494 {
495 if(!v.snd)
496 continue;
497 const float* const left = v.snd->left.data();
498 const float* const right
499 = v.snd->right.size() == v.snd->left.size() ? v.snd->right.data() : left;
500 int64_t count = std::min(end - begin, v.snd->frames() - v.pos);
501 bool finished = count >= v.snd->frames() - v.pos;
502 if(v.fade_left >= 0 && v.fade_left <= count)
503 {
504 count = v.fade_left;
505 finished = true;
506 }
507
508 const float fade_step = 1.f / float(v.fade_total);
509 float fade = v.fade_left >= 0 ? float(v.fade_left) * fade_step : 1.f;
510 for(int64_t i = 0; i < count; i++)
511 {
512 const float g = v.gain * fade * (volume_from + volume_step * float(begin + i));
513 out[0][begin + i] += double(left[v.pos + i] * g);
514 out[1][begin + i] += double(right[v.pos + i] * g);
515 if(v.fade_left >= 0)
516 fade -= fade_step;
517 }
518
519 v.pos += count;
520 if(v.fade_left >= 0)
521 v.fade_left -= int32_t(count);
522 if(finished)
523 v.snd = nullptr;
524 }
525 }
