129// 1-in/1-out plug-ins get N replicated voices for multichannel host buses
130inline voice_strategy
131choose_voice_strategy(const LV2Data& data, std::size_t replicate_count = 8) noexcept
132{
133 const auto ains = data.audio_in_ports.size();
134 const auto aouts = data.audio_out_ports.size();
135 if(ains == 1 && aouts == 1 && replicate_count > 1)
136 return {voice_routing::per_channel, replicate_count};
137 return {voice_routing::single, 1};
138}

572 void receive_ui_messages() noexcept
573 {
574 if(!ui)
575 return;
576 while(ui->to_plugin.try_read(
577 [this](const message_ring::header& h) -> std::optional<uint8_t*> {
578 auto it = std::find_if(m_uiInputs.begin(), m_uiInputs.end(), [&](const auto& in) {
579 return uint32_t(in.port) == h.index;
580 });
581 if(it == m_uiInputs.end() || h.size < sizeof(LV2_Atom))
582 return nullptr;
583 const uint32_t size = UiChannel::event_size(h.size);
584 if(size > it->room)
585 return nullptr;
586 if(it->used + size > it->room)
587 return std::nullopt;
588 uint8_t* ev = it->events.get() + it->used;
589 std::memset(ev, 0, sizeof(int64_t));
590 it->used += size;
591 return ev + sizeof(int64_t);
592 }))
593 {
594 }
595 }

606 at += UiChannel::event_size(sizeof(LV2_Atom) + atom->size);
607 }

677 void forgeTime(
678 uint8_t* buffer, const ossia::token_request& tk, ossia::exec_state_facade st,
679 float speed)
680 {
681 LV2::HostContext& host = data.host;
682 auto& forge = m_forge;
683 lv2_atom_forge_set_buffer(&forge, buffer, 256);
684 LV2_Atom_Forge_Frame frame;
685 lv2_atom_forge_object(&forge, &frame, 0, host.time_Position_id);
686
687 // The playhead at the first sample of this run
688 lv2_atom_forge_key(&forge, host.time_frame_id);
689 lv2_atom_forge_long(&forge, this->m_transport_frames);
690
691 lv2_atom_forge_key(&forge, host.time_framesPerSecond_id);
692 lv2_atom_forge_long(&forge, st.sampleRate());
693
694 lv2_atom_forge_key(&forge, host.time_speed_id);
695 lv2_atom_forge_float(&forge, speed);
696
697 lv2_atom_forge_key(&forge, host.time_bar_id);
698 lv2_atom_forge_long(&forge, tk.musical_start_last_bar / 4.);
699
700 lv2_atom_forge_key(&forge, host.time_beat_id);
701 lv2_atom_forge_double(&forge, tk.musical_start_position);
702
703 auto barBeat = float(tk.musical_start_position - tk.musical_start_last_bar);
704 lv2_atom_forge_key(&forge, host.time_barBeat_id);
705 lv2_atom_forge_float(&forge, barBeat);
706
707 lv2_atom_forge_key(&forge, host.time_beatUnit_id);
708 lv2_atom_forge_int(&forge, 4);
709
710 lv2_atom_forge_key(&forge, host.time_beatsPerBar_id);
711 lv2_atom_forge_float(
712 &forge, 4 * double(tk.signature.upper) / double(tk.signature.lower));
713
714 lv2_atom_forge_key(&forge, host.time_beatsPerMinute_id);
715 lv2_atom_forge_float(&forge, tk.tempo);
716
717 lv2_atom_forge_pop(&forge, &frame);
718 }

720 // Vec slices by voice (clamped); scalars broadcast. Mirrors faust_mono_fx::set_control.
721 static std::optional<float>
722 voice_control_value(const ossia::value& val, std::size_t voice_idx) noexcept
723 {
724 return ossia::apply_nonnull(
725 [voice_idx](const auto& v) -> std::optional<float> {
726 using T = std::decay_t<decltype(v)>;
727 if constexpr(std::is_same_v<T, std::vector<ossia::value>>)
728 {
729 if(v.empty())
730 return std::nullopt;
731 const auto& elem = (voice_idx < v.size()) ? v[voice_idx] : v.back();
732 return ossia::convert<float>(elem);
733 }
734 else if constexpr(std::is_same_v<T, ossia::vec2f>)
735 return v[std::min<std::size_t>(voice_idx, 1)];
736 else if constexpr(std::is_same_v<T, ossia::vec3f>)
737 return v[std::min<std::size_t>(voice_idx, 2)];
738 else if constexpr(std::is_same_v<T, ossia::vec4f>)
739 return v[std::min<std::size_t>(voice_idx, 3)];
740 else if constexpr(
741 std::is_same_v<T, float> || std::is_same_v<T, int>
742 || std::is_same_v<T, bool>)
743 return float(v);
744 else
745 return std::nullopt;
746 },
747 val.v);
748 }

750 void preProcessVoice(voice& v, std::size_t voice_idx)
751 {
752 const std::size_t audio_in_size = data.audio_in_ports.size();
753 const std::size_t cv_size = data.cv_ports.size();
754 const std::size_t midi_in_size = data.midi_in_ports.size();
755 const std::size_t atom_in_size = data.atom_in_ports.size();
756 const std::size_t control_in_size = data.control_in_ports.size();
757
758 int first_midi_idx = (audio_in_size > 0 ? 1 : 0) + cv_size;
759 for(std::size_t i = 0; i < v.midi_atom_ins.size(); i++)
760 {
761 auto& ossia_port
762 = this->m_inlets[i + first_midi_idx]->template cast<ossia::midi_port>();
763 auto& lv2_port = v.midi_atom_ins[i];
764 Iterator it{lv2_port.buf};
765
766 // The position first: the sequence is in time order, and the events of
767 // this block are at this position.
768 if(ossia::any_of(v.atom_timePosition_shared, [&](const auto& t) {
769 return t.buffer == &lv2_port;
770 }))
771 writeTime(it);
772
773 write_ui_messages(it, i);
774 if(i == 0)
775 write_notes_off(it);
776
777 // Port timestamps are relative to the buffer, the plug-in's to the tick:
778 // as the VST3 host does, what falls outside it goes to its nearest end.
779 auto& conv = v.midi_2to1[i];
780 for(const libremidi::ump& msg : ossia_port.messages)
781 {
782 const int64_t frame = std::clamp<int64_t>(
783 msg.timestamp - m_span.start, 0,
784 m_span.samples > 0 ? m_span.samples - 1 : 0);
785 conv.convert(
786 msg.data, msg.size(), frame,
787 [&](const unsigned char* midi1, std::size_t bytes, int64_t) {
788 if(bytes > 0)
789 it.write(frame, 0, data.host.midi_event_id, bytes, midi1);
790 return stdx::error{};
791 });
792 }
793 }
794
795 writeAtomIns(v);
796 for(auto& [_, buf] : v.atom_timePosition_owned)
797 {
798 Iterator it{buf->buf};
799 writeTime(it);
800 }
801
802 const auto control_start
803 = (audio_in_size > 0 ? 1 : 0) + midi_in_size + atom_in_size + cv_size;
804 for(std::size_t i = 0; i < control_in_size; i++)
805 {
806 auto& in = m_inlets[control_start + i]
807 ->template cast<ossia::value_port>()
808 .get_data();
809 if(in.empty())
810 continue;
811 if(auto new_val = voice_control_value(in.back().value, voice_idx))
812 v.fInControls[i] = *new_val * fInScale[i];
813 }
814
815 (void)atom_in_size;
816 }

828 template <typename Channel>
829 static void readChannel(const Channel& ch, float* dst, const buffer_span& s) noexcept
830 {
831 const int64_t n = std::clamp<int64_t>(int64_t(ch.size()) - s.start, 0, s.samples);
832 for(int64_t j = 0; j < n; j++)
833 dst[j] = float(ch[s.start + j]);
834 std::fill(dst + n, dst + s.samples, 0.f);
835 }

839 {
840 if(int64_t(ch.size()) < s.size)
841 ch.resize(s.size);
842 std::copy_n(src, s.samples, ch.begin() + s.start);
843 }

1035 void run(const ossia::token_request& tk, ossia::exec_state_facade st) noexcept override
1036 {
1037 // Before the paused check: the tick requested by stop() is empty.
1038 if(mustStop)
1039 {
1040 stop_voices();
1041 mustStop = false;
1042 m_sendStopped = true;
1043 return;
1044 }
1045 if(tk.paused())
1046 return;
1047
1048 if(!data.time_Position_ports.empty())
1049 updateTime(tk, st);
1050
1051 const auto [tick_start, samples] = st.timings(tk);
1052 const bool samples_valid
1053 = samples > 0 && std::size_t(samples) <= max_block_size && tick_start >= 0;
1054 m_span = {
1055 std::max<int64_t>(tick_start, 0), samples_valid ? samples : 0,
1056 std::max<int64_t>(st.bufferSize(), tick_start + samples)};
1057
1058 on_start();
1059 receive_ui_messages();
1060 for(std::size_t c = 0; c < voices.size(); ++c)
1061 preProcessVoice(*voices[c], c);
1062
1063 if(samples_valid)
1064 {
1065 for(std::size_t c = 0; c < voices.size(); ++c)
1066 feedCVInputs(*voices[c], c);
1067 switch(routing)
1068 {
1069 case voice_routing::single:
1070 runSingle(samples);
1071 break;
1072 case voice_routing::per_channel:
1073 runPerChannel(samples);
1074 break;
1075 }
1076 }
1077
1078 surfaceVoiceZeroOutputs(tk.physical_start(st.modelToSamples()));
1079 send_ui_messages();
1080 on_finished();
1081
1082 for(auto& v : voices)
1083 postProcessVoice(*v);
1084 clear_ui_messages();
1085 m_sendStopped = false;
1086 }

1128 void runPerChannel(int samples) noexcept
1129 {

1131 return;

1134 = m_inlets.empty() ? nullptr : m_inlets[0]->template target<ossia::audio_port>();
1135 const auto in_channels = audio_in_inlet ? audio_in_inlet->channels() : 0;
1136

1138 const auto requested
1139 = (audio_in_inlet ? std::max<std::size_t>(in_channels, 1) : voices.size());

1143

1145 out.set_channels(active);
1146

1148 {
1149 auto& v = *voices[c];
1150
1151 if(audio_in_inlet)
1152 {

1154 if(in_channels > c)
1155 readChannel(audio_in_inlet->channel(c), in_scratch.data(), m_span);
1156 else

1158 lilv_instance_connect_port(
1159 v.instance, data.audio_in_ports[0], in_scratch.data());
1160 }
1161
1162 auto& out_scratch = v.audio_out_scratch[0];
1163 lilv_instance_connect_port(
1164 v.instance, data.audio_out_ports[0], out_scratch.data());
1165
1166 lilv_instance_run(v.instance, samples);
1167 v.ran = true;
1168
1169 writeChannel(out.channel(c), out_scratch.data(), m_span);
1170 if(c == 0)
1171 writeCVOutputs(v);
1172 }
1173 }
