118struct GraphicsStorageResources
119{
120 std::vector<GraphicsSSBO> ssbos;
121 std::vector<GraphicsStorageImage> images;
122 std::vector<GraphicsUBO> ubos;
123
124 // Quick aliases: first SSBO with BUFFER_USAGE="indirect_draw*". Populated
125 // by collectGraphicsStorageResources. Updated by callers when the underlying
126 // SSBO's buffer pointer changes (e.g. when an upstream CSF rebuilds it).
127 QRhiBuffer* indirectDrawBuffer{};
128 bool indirectDrawIndexed{false};
129 int indirectDrawSsboIndex{-1};
130
131 // Next free binding index after all graphics-visible storage resources
132 // (SSBOs + images + UBOs) have been assigned by
133 // collectGraphicsStorageResources. This is exactly the value libisf's
134 // isf_emit_graphics_storage() returns (isf.cpp:3406-3449) and the binding
135 // at which the codegen places the multiview UBO (isf.cpp:3773-3783). Callers
136 // that append a multiview UBO MUST use this rather than re-deriving a max
137 // over ssbos/images alone: that omits uniform_input UBOs and collides the
138 // multiview binding with the last UBO's slot. -1 until the first
139 // collectGraphicsStorageResources() call.
140 int nextBinding{-1};
141
142 // Sentinel zero-buffer bound when an SSBO/UBO upstream port disconnects
143 // mid-session. QRhi (especially Vulkan) requires every SRB binding to
144 // point at a valid resource — without a sentinel, a disconnect leaves
145 // the binding pointing at a dangling QRhiBuffer* (the prior upstream's
146 // buffer, which was deleteLater'd when the upstream node was destroyed).
147 // Lazily allocated on first disconnect, sized to the largest binding
148 // observed (sentinelSize). Single buffer reused for both SSBO and UBO
149 // disconnects since the descriptor type is set on the SRB binding side,
150 // not the buffer side; QRhi accepts a buffer with both StorageBuffer and
151 // UniformBuffer usage flags. owned=true; freed in release().
152 QRhiBuffer* sentinelBuffer{};
153 QRhiBuffer* sentinelUniformBuffer{};
154 uint32_t sentinelSize{0};
155
158 SCORE_PLUGIN_GFX_EXPORT void release(RenderList& renderer);
159};
