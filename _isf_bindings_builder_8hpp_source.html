142struct GraphicsStorageResources
143{
144 std::vector<GraphicsSSBO> ssbos;
145 std::vector<GraphicsStorageImage> images;
146 std::vector<GraphicsUBO> ubos;
147
148 // Quick aliases: first SSBO with BUFFER_USAGE="indirect_draw*". Populated
149 // by collectGraphicsStorageResources. Updated by callers when the underlying
150 // SSBO's buffer pointer changes (e.g. when an upstream CSF rebuilds it).
151 QRhiBuffer* indirectDrawBuffer{};
152 bool indirectDrawIndexed{false};
153 int indirectDrawSsboIndex{-1};
154
155 // Next free binding index after all graphics-visible storage resources
156 // (SSBOs + images + UBOs) have been assigned by
157 // collectGraphicsStorageResources. This is exactly the value libisf's
158 // isf_emit_graphics_storage() returns (isf.cpp:3406-3449) and the binding
159 // at which the codegen places the multiview UBO (isf.cpp:3773-3783). Callers
160 // that append a multiview UBO MUST use this rather than re-deriving a max
161 // over ssbos/images alone: that omits uniform_input UBOs and collides the
162 // multiview binding with the last UBO's slot. -1 until the first
163 // collectGraphicsStorageResources() call.
164 int nextBinding{-1};
165
166 // Sentinel zero-buffer bound when an SSBO/UBO upstream port disconnects
167 // mid-session. QRhi (especially Vulkan) requires every SRB binding to
168 // point at a valid resource — without a sentinel, a disconnect leaves
169 // the binding pointing at a dangling QRhiBuffer* (the prior upstream's
170 // buffer, which was deleteLater'd when the upstream node was destroyed).
171 // Lazily allocated on first disconnect, sized to the largest binding
172 // observed (sentinelSize). Single buffer reused for both SSBO and UBO
173 // disconnects since the descriptor type is set on the SRB binding side,
174 // not the buffer side; QRhi accepts a buffer with both StorageBuffer and
175 // UniformBuffer usage flags. owned=true; freed in release().
176 QRhiBuffer* sentinelBuffer{};
177 QRhiBuffer* sentinelUniformBuffer{};
178 uint32_t sentinelSize{0};
179
182 SCORE_PLUGIN_GFX_EXPORT void release(RenderList& renderer);
183};
