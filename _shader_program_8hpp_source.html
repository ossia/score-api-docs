196// Cache key. `originDir` is the *canonical directory* the shader was
197// loaded from (derived by the cache from the caller-supplied origin
198// path). Keying on both means two models loading the same source text
199// from different directories don't collide — include resolution against
200// each shader's own sibling dir stays correct.

208 {

210 }
211};
