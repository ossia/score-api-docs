228vec3 evaluateSH(uint base, vec3 dir) {
229 // Degree 0
230 vec3 result = SH_C0 * vec3(
231 rawData[base + 6],
232 rawData[base + 7],
233 rawData[base + 8]
234 );
235
236 if (shDegree < 1) {
237 return result + 0.5;
238 }
239
240 // Degree 1
241 float x = dir.x, y = dir.y, z = dir.z;
242
243 // f_rest layout: [0..14] = R channel rest, [15..29] = G, [30..44] = B
244 // But the INRIA convention interleaves: [0..2] = degree1 for R,G,B etc.
245 // Actually the standard layout is:
246 // f_rest[0..14]: coeffs 1..15 for channel 0 (R)
247 // f_rest[15..29]: coeffs 1..15 for channel 1 (G)
248 // f_rest[30..44]: coeffs 1..15 for channel 2 (B)
249
250 uint r = base + 9; // f_rest_0 start
251 // Degree 1: 3 coefficients per channel, interleaved as RGB triplets
252 // Coeff indices in f_rest: R=[0,1,2], G=[15,16,17], B=[30,31,32]
253 result += SH_C1 * (
254 - y * vec3(rawData[r+0], rawData[r+15], rawData[r+30])
255 + z * vec3(rawData[r+1], rawData[r+16], rawData[r+31])
256 - x * vec3(rawData[r+2], rawData[r+17], rawData[r+32])
257 );
258
259 if (shDegree < 2) {
260 return result + 0.5;
261 }
262
263 // Degree 2: 5 coefficients per channel
264 // R=[3..7], G=[18..22], B=[33..37]
265 float xx = x*x, yy = y*y, zz = z*z, xy = x*y, yz = y*z, xz = x*z;
266
267 result += SH_C2[0] * xy * vec3(rawData[r+3], rawData[r+18], rawData[r+33]);
268 result += SH_C2[1] * yz * vec3(rawData[r+4], rawData[r+19], rawData[r+34]);
269 result += SH_C2[2] * (2.*zz - xx - yy)
270 * vec3(rawData[r+5], rawData[r+20], rawData[r+35]);
271 result += SH_C2[3] * xz * vec3(rawData[r+6], rawData[r+21], rawData[r+36]);
272 result += SH_C2[4] * (xx - yy)* vec3(rawData[r+7], rawData[r+22], rawData[r+37]);
273
274 if (shDegree < 3) {
275 return result + 0.5;
276 }
277
278 // Degree 3: 7 coefficients per channel
279 // R=[8..14], G=[23..29], B=[38..44]
280 result += SH_C3[0] * y*(3.*xx - yy)
281 * vec3(rawData[r+8], rawData[r+23], rawData[r+38]);
282 result += SH_C3[1] * xy*z * vec3(rawData[r+9], rawData[r+24], rawData[r+39]);
283 result += SH_C3[2] * y*(4.*zz - xx - yy)
284 * vec3(rawData[r+10], rawData[r+25], rawData[r+40]);
285 result += SH_C3[3] * z*(2.*zz - 3.*xx - 3.*yy)
286 * vec3(rawData[r+11], rawData[r+26], rawData[r+41]);
287 result += SH_C3[4] * x*(4.*zz - xx - yy)
288 * vec3(rawData[r+12], rawData[r+27], rawData[r+42]);
289 result += SH_C3[5] * z*(xx - yy)
290 * vec3(rawData[r+13], rawData[r+28], rawData[r+43]);
291 result += SH_C3[6] * x*(xx - 3.*yy)
292 * vec3(rawData[r+14], rawData[r+29], rawData[r+44]);
293
294 return result + 0.5;
295}

382void main() {

386 // Transform to view space

390 // Front-to-back sort key: top 16 bits = depth, bottom 16 bits = splat index.
391 // The depth gives correct rendering order; the index provides stable
392 // tie-breaking for splats at similar depths (same buffer order every frame).
393 // This eliminates the "wave" artifact from coherent sort-order swaps.
394 // Combined with "under" blending for correct front-to-back compositing.
395 const uint keyMax = 0xFFFFFFFFu;

398 // Behind camera: draw last, but keep stable index-based sub-order
399 key = (0xFFFFu << 16u) | (idx & 0xFFFFu);
400 } else {
401 float t = log2(depth / nearPlane) / log2(farPlane / nearPlane);
402 t = clamp(t, 0.0, 1.0);
403 uint depthKey = uint(t * 65535.0);
404 key = (depthKey << 16u) | (idx & 0xFFFFu);
405 }
406
407 keys[idx] = key;
408 indices[idx] = idx;
409}

416static constexpr auto histogram_shader = R"_(#version 450
417layout(local_size_x = 256) in;
418
419layout(std430, binding = 0) readonly buffer KeyBuffer {
420 uint keys[];
421};
422
423layout(std430, binding = 1) buffer HistogramBuffer {

425};
426
427layout(std140, binding = 2) uniform Params {
428 uint splatCount;
429 uint bitOffset; // Which 8 bits to sort (0, 8, 16, 24)

436void main() {

443 barrier();

495void main() {
496 uint localId = gl_LocalInvocationID.x;
497 uint globalId = gl_GlobalInvocationID.x;
498 uint workgroupId = gl_WorkGroupID.x;
499
500 // Load global prefix sum for this workgroup's digit
501 localOffset[localId] = histogram[workgroupId * 256 + localId];
502
503 // Load this thread's element
504 uint key = 0u;
505 uint idx = 0u;
506 uint digit = 256u; // invalid sentinel (> any real digit)
507 bool valid = globalId < splatCount;
508 if (valid) {
509 key = keysIn[globalId];
510 idx = indicesIn[globalId];
511 digit = (key >> bitOffset) & 0xFFu;
512 }
513 localDigits[localId] = digit;
514 barrier();
515

517 // Stable rank: count threads with LOWER ID that share the same digit.
518 // This is deterministic (no atomicAdd race), so the sort is stable
519 // and identical across frames — eliminates flickering.
520 uint rank = 0u;
521 for (uint i = 0u; i < localId; i++) {
522 if (localDigits[i] == digit)
523 rank++;
524 }
525
526 uint globalPos = localOffset[digit] + rank;
527 if (globalPos < splatCount) {
528 keysOut[globalPos] = key;
529 indicesOut[globalPos] = idx;
530 }

549static constexpr auto prefix_sum_shader = R"_(#version 450
550layout(local_size_x = 256) in;
551
552layout(std430, binding = 0) buffer HistogramBuffer {
553 uint histogram[]; // Layout: histogram[workgroup * 256 + digit]
554};
555
556layout(std140, binding = 1) uniform Params {
557 uint numWorkgroups;
558 uint _pad0;
559 uint _pad1;
560 uint _pad2;
561};
562
563shared uint digitTotal[256];
564shared uint digitPrefix[256];
565
566void main() {
567 uint digit = gl_LocalInvocationID.x; // 0-255, one thread per digit
568
569 // Step 1: Sum all workgroup counts for this digit
570 uint total = 0;
571 for (uint wg = 0; wg < numWorkgroups; wg++) {
572 total += histogram[wg * 256 + digit];
573 }
574 digitTotal[digit] = total;
575 barrier();
576
577 // Step 2: Thread 0 computes exclusive prefix sum across all digits
578 // This determines the global starting offset for each digit bucket
579 if (digit == 0) {
580 digitPrefix[0] = 0;
581 for (uint d = 1; d < 256; d++) {
582 digitPrefix[d] = digitPrefix[d-1] + digitTotal[d-1];
583 }
584 }
585 barrier();
586
587 // Step 3: Convert per-workgroup counts to global offsets
588 // For each workgroup: offset = digitPrefix[digit] + sum of same-digit counts in earlier workgroups
589 uint running = digitPrefix[digit];
590 for (uint wg = 0; wg < numWorkgroups; wg++) {
591 uint idx = wg * 256 + digit;
592 uint val = histogram[idx];
593 histogram[idx] = running;
594 running += val;
595 }
596}
597)_";

655void main() {
656 // Get splat index (sorted or unsorted)
657 uint splatIdx = useSorting != 0 ? sortedIndices[gl_InstanceIndex] : gl_InstanceIndex;
658 RenderSplat splat = splats[splatIdx];
659 vec2 quadPos = positions[gl_VertexIndex];
660
661 // Early opacity cull: skip splats that are nearly invisible
662 if (splat.color.a < 1.0 / 255.0) {
663 gl_Position = vec4(0.0, 0.0, 2.0, 1.0);
664 return;
665 }
666
667 // View space position
668 vec4 viewPos = view * vec4(splat.position.xyz, 1.0);
669
670 // Focal lengths in pixels
671 float focal = projection[0][0] * viewport.x * 0.5;
672 float focal_y = projection[1][1] * viewport.y * 0.5;
673 float tanFovX = 0.5 * viewport.x / focal;
674 float tanFovY = 0.5 * viewport.y / focal_y;
675
676 // Frustum culling: project to clip space and check NDC bounds
677 // (matches INRIA reference: cull behind camera + outside 1.3x viewport)
678 vec4 clipPos = projection * viewPos;
679 if (clipPos.w <= 0.2) {
680 gl_Position = vec4(0.0, 0.0, 2.0, 1.0);
681 return;
682 }
683 vec3 ndc = clipPos.xyz / clipPos.w;
684 if (abs(ndc.x) > 1.3 || abs(ndc.y) > 1.3) {
685 gl_Position = vec4(0.0, 0.0, 2.0, 1.0);
686 return;
687 }
688
689 // Clamp view-space position to prevent numerical issues at screen edges
690 // (matches INRIA CUDA reference: 1.3x FOV tangent)
691 float limX = 1.3 * tanFovX;
692 float limY = 1.3 * tanFovY;
693 float txtz = viewPos.x / viewPos.z;
694 float tytz = viewPos.y / viewPos.z;
695 viewPos.x = clamp(txtz, -limX, limX) * viewPos.z;
696 viewPos.y = clamp(tytz, -limY, limY) * viewPos.z;
697
698 // Build 3D covariance from scale and rotation (already preprocessed)
699 // INRIA convention: Sigma = R * S * S^T * R^T = R * S² * R^T
700 // The principal axes are the COLUMNS of R.
701 vec3 scale = splat.scale.xyz;
702 mat3 R = quatToMat(splat.rotation);
703 mat3 S = mat3(scale.x, 0, 0, 0, scale.y, 0, 0, 0, scale.z);
704 mat3 M = R * S;
705 mat3 Sigma = M * transpose(M);
706
707 // 2D covariance via EWA projection
708 mat3 W = mat3(view);
709 float z2 = viewPos.z * viewPos.z;
710
711 // Jacobian of projection (column-major: mat3(col0, col1, col2))
712 mat3 J = mat3(
713 focal / viewPos.z, 0.0, 0.0, // col 0
714 0.0, focal_y / viewPos.z, 0.0, // col 1
715 -focal * viewPos.x / z2, -focal_y * viewPos.y / z2, 0.0 // col 2
716 );
717
718 mat3 T = J * W;
719 mat3 cov = T * Sigma * transpose(T);
720
721 float cov_xx = cov[0][0], cov_xy = cov[0][1], cov_yy = cov[1][1];
722
723 // Mip-Splatting 2D filter (Yu et al. 2024): approximate the pixel box filter
724 // as a Gaussian and convolve with the projected 2D covariance.
725 // Opacity is compensated to preserve each splat's total contribution:
726 // alpha' = alpha * sqrt(det(Sigma) / det(Sigma + kernel_size * I))
727 float kernel_size = 0.3;
728 float det_0 = max(1e-6, cov_xx * cov_yy - cov_xy * cov_xy);
729 cov_xx += kernel_size;
730 cov_yy += kernel_size;
731 float det_1 = max(1e-6, cov_xx * cov_yy - cov_xy * cov_xy);
732 float mipCoef = sqrt(det_0 / det_1);
733
734 float det = cov_xx * cov_yy - cov_xy * cov_xy;
735 float mid = 0.5 * (cov_xx + cov_yy);
736 float disc = max(0.0, mid * mid - det);
737 float lambda1 = mid + sqrt(disc);
738 float lambda2 = mid - sqrt(disc);
739
740 // Eigenvectors of 2D covariance for ellipse-aligned quad
741 vec2 eigVec1;
742 if (abs(cov_xy) > 1e-6) {
743 eigVec1 = normalize(vec2(cov_xy, lambda1 - cov_xx));
744 } else {
745 eigVec1 = (cov_xx >= cov_yy) ? vec2(1.0, 0.0) : vec2(0.0, 1.0);
746 }
747 vec2 eigVec2 = vec2(-eigVec1.y, eigVec1.x);
748
749 float maxExtent = 2048.0;
750 float r1 = min(ceil(3.0 * sqrt(max(lambda1, 0.0))), maxExtent);
751 float r2 = min(ceil(3.0 * sqrt(max(lambda2, 0.0))), maxExtent);
752
753 // Cull degenerate or invisible splats
754 if (det < 1e-3 || max(r1, r2) < 0.1) {
755 gl_Position = vec4(0.0, 0.0, 2.0, 1.0);
756 return;
757 }
758
759 // Inverse covariance (conic) for fragment Gaussian evaluation.
760 // The cross-term sign must match the screen-space convention of gl_FragCoord:
761 // Vulkan/Metal/D3D (clipSpaceCorr[1][1] < 0): both screen axes flip
762 // relative to J-space, preserving the cross-product sign.
763 // OpenGL (clipSpaceCorr[1][1] > 0): only X flips, requiring correction.
764 float inv_det = 1.0 / det;
765 float crossSign = sign(clipSpaceCorr[1][1]);
766 f_conic = vec3(cov_yy * inv_det, crossSign * cov_xy * inv_det, cov_xx * inv_det);
767
768 // Oriented quad: major axis along eigVec1, minor along eigVec2
769 vec2 pixelOffset = quadPos.x * r1 * eigVec1 + quadPos.y * r2 * eigVec2;
770 vec2 center = ndc.xy;
771 vec2 ndcOffset = pixelOffset * 2.0 / viewport;
772
773 gl_Position = clipSpaceCorr * vec4(center + ndcOffset, ndc.z, 1.0);
774
775 // Score's texture compositing pipeline flips Y when sampling for Vulkan/HLSL/Metal.
776 // To match this convention (same as ISF shaders), we undo clipSpaceCorr's Y-flip here
777 // so the compositing re-flip produces a correctly oriented final image.
778 gl_Position.y = -gl_Position.y;
779
780 // Screen-space center in pixels (matches gl_FragCoord coordinate system)
781 vec4 centerClip = clipSpaceCorr * vec4(ndc.xy, ndc.z, 1.0);
782 centerClip.y = -centerClip.y;
783 f_center = (centerClip.xy / centerClip.w * 0.5 + 0.5) * viewport;
784
785 // Fade out excessively large projected splats.
786 float alpha = splat.color.a * mipCoef;
787 float maxR = max(r1, r2);
788 float fadeRadius = 512.0;
789 if (maxR > fadeRadius) {
790 float fade = fadeRadius / maxR;
791 alpha *= fade;
792 if (alpha < 1.0 / 255.0) {
793 gl_Position = vec4(0.0, 0.0, 2.0, 1.0);
794 return;
795 }
796 }
797 f_color = vec4(splat.color.rgb, alpha);
798}
799)_";

801static constexpr auto fragment_shader = R"_(#version 450
802
803layout(location = 0) in vec2 f_center; // screen-space splat center (pixels)
804layout(location = 1) in vec4 f_color;
805layout(location = 2) in vec3 f_conic;
806
807layout(location = 0) out vec4 fragColor;

809void main() {
810 // Pixel offset from splat center, computed per-fragment for precision.
811 // Unlike interpolated UVs, this is exact regardless of quad orientation.
812 vec2 d = gl_FragCoord.xy - f_center;
813
814 float power = -0.5 * (f_conic.x * d.x * d.x +
815 2.0 * f_conic.y * d.x * d.y +
816 f_conic.z * d.y * d.y);
817
818 if (power > 0.0) discard;
819
820 float gaussian = exp(power);

823

825}
