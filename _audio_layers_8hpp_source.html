349 static float
350 repeat_weight(const audio_layer& l, int64_t k, int64_t p, int64_t len) noexcept
351 {
352 float w = 1.f;
353 for(const bool tail : {false, true})
354 {
355 const auto& e = tail ? l.tail : l.head;
356 if(e.at < 0 || e.n <= 0 || e.rho <= 0.f || (tail && e.meet < 0))
357 continue;
358 int64_t d = ((k % p - e.at) * e.dir) % p;
359 if(d < 0)
360 d += p;
361 if(d >= e.n)
362 continue;
363 int64_t seam = (k - e.dir * d) % len;
364 if(seam < 0)
365 seam += len;
366 int64_t met = tail ? (e.meet + e.dir) % len : e.at;
367 if(met < 0)
368 met += len;
369 if(seam != met)
370 w *= seam_weight(tail, 0.f, d, e.n) / seam_weight(tail, e.rho, d, e.n);
371 }
372 return w;
373 }

378 void cover(audio_layer& l, int64_t k) noexcept
379 {
380 const int64_t p = period_of(l);
381 if(p <= 0 || l.written >= p)
382 return;
383 if(l.written == 0)
384 {
385 l.start = k;
386 l.written = 1;
387 return;
388 }
389 int64_t d = (k - l.start) % p;
390 if(d < 0)
391 d += p;
392 if(d < l.written)
393 return;
394 const int64_t after = d - l.written;
395 const int64_t before = p - 1 - d;
396 if(after <= before)
397 {
398 silence(l, l.start + l.written, after);
399 l.written = d + 1;
400 }
401 else
402 {
403 silence(l, k + 1, before);
404 l.start = k;
405 l.written += before + 1;
406 }
407 l.written = std::min(l.written, p);
408 }

551 void open_seam(audio_layer& l, bool tail, int partner) noexcept
552 {
553 const int64_t p = period_of(l);
554 const int64_t written = std::min(l.written, p);
555 auto& e = tail ? l.tail : l.head;
556 e = {};
557 if(p <= 0 || written <= 0 || !allocated() || kept.empty())
558 return;
559 const int64_t first = l.start % p;
560 const int64_t last = (l.start + written - 1) % p;
561 // Backwards, the part written grows towards the loop's start: its first
562 // frame is its last position.
563 const bool at_last = tail != l.backwards;
564 e.at = at_last ? last : first;
565 e.dir = at_last ? -1 : 1;
566 e.n = int(std::min<int64_t>(fade_frames, written));
567 e.partner = partner;
568 e.rho = 1.f;
569 for(int c = 0; c < channels; c++)
570 {
571 const float* d = data(l.slot, c);
572 float* orig = kept_of(l.slot, tail, false, c);
573 float* part = kept_of(l.slot, tail, true, c);
574 for(int i = 0; i < e.n; i++)
575 orig[i] = part[i] = d[seam_index(l, e, i)];
576 }
577 }

580 void weigh_seam(audio_layer& l, bool tail, float rho) noexcept
581 {
582 auto& e = tail ? l.tail : l.head;
583 e.rho = rho;
584 for(int c = 0; c < channels; c++)
585 {
586 float* d = data(l.slot, c);
587 const float* orig = kept_of(l.slot, tail, false, c);
588 const float* part = kept_of(l.slot, tail, true, c);
589 for(int i = 0; i < e.n; i++)
590 d[seam_index(l, e, i)] = orig[i] - part[i] * (1.f - seam_weight(tail, rho, i, e.n));
591 }
592 }

606 void update_seams() noexcept
607 {
608 for(int i = 0; i < count; i++)
609 {
610 auto& l = layers[i];
611 if(l.muted)
612 continue;
613 for(const bool tail : {false, true})
614 {
615 auto& e = tail ? l.tail : l.head;
616 if(e.at < 0 || e.partner == audio_seam_end::pending)
617 continue;
618 const int j = e.partner >= 0 ? heard_in(e.partner) : -1;
619 if(j == i)
620 continue;
621 float rho = 0.f;
622 if(j >= 0)
623 {
624 const float mine = settled_gain(l);
625 const float theirs = settled_gain(layers[j]);
626 rho = mine > 0.f ? std::min(1.f, theirs / mine) : 1.f;
627 if(tail && layers[j].head.at >= 0)
628 e.meet = layers[j].head.at;
629 }
630 if(std::abs(rho - e.rho) > 1e-6f)
631 weigh_seam(l, tail, rho);
632 }
633 }

643 {

645 forget_aside();

647 set_aside(replace);
648 if(!oldest && count > slots)
649 {

651 // be taken out than when its merge ended before this pass: the two
652 // buffers kept for that are free for this one.

690 void start_merge(int at = 0, bool undoable = false) noexcept
691 {
692 if(at < 0 || at + 1 >= count || merging.active)
693 return;
694 const int target = take_slot();
695 if(target < 0)
696 return;
697 const auto& a = layers[at];
698 const auto& b = layers[at + 1];
699 // A muted pass merged with one that is not is left out of the merged one,
700 // and takes nothing from the pass under it: merging is where it is lost.
701 // Two muted passes make a muted one that has both.
702 const bool lost_a = a.muted && !b.muted, lost_b = b.muted && !a.muted;
703 merging = merge_state{
704 .active = true,
705 .at = at,
706 .target = target,
707 .a_slot = a.slot,
708 .b_slot = b.slot,
709 .a_gain = lost_a ? 0.f : lost_b ? 1.f : b.decay,
710 .b_gain = lost_b ? 0.f : 1.f,
711 .muted = a.muted && b.muted,
712 .undoable = undoable,
713 .a = a,
714 .b = b,
715 .lost_a = lost_a,
716 .lost_b = lost_b,
717 .a_len = std::max<int64_t>(1, period_of(a)),
718 .b_len = std::max<int64_t>(1, period_of(b)),
719 .merged = std::max(period_of(a), period_of(b)),
720 .pos = 0};
721 }

726 void merge_step(int64_t frames) noexcept
727 {
728 auto& m = merging;
729 if(!m.active)
730 return;
731 if(m.at + 1 >= count || layers[m.at].slot != m.a_slot
732 || layers[m.at + 1].slot != m.b_slot)
733 {
734 m.active = false;
735 return;
736 }
737
738 // The merged pass is as long as the longer of the two: the shorter one
739 // repeats into it.
740 const int64_t end = std::min(m.merged, m.pos + std::max<int64_t>(1, frames));
741 for(int c = 0; c < channels; c++)
742 {
743 float* dst = data(m.target, c);
744 const float* da = data(m.a_slot, c);
745 const float* db = data(m.b_slot, c);
746 for(int64_t k = m.pos; k < end; k++)
747 dst[k] = merged_at(da, db, k);
748 }
749 {
750 float* pt = peaks_of(m.target);
751 const float* pa = peaks_of(m.a_slot);
752 const float* pb = peaks_of(m.b_slot);
753 const int64_t a_bins = std::max<int64_t>(1, (m.a_len + peak_frames - 1) / peak_frames);
754 const int64_t b_bins = std::max<int64_t>(1, (m.b_len + peak_frames - 1) / peak_frames);
755 for(int64_t i = m.pos / peak_frames; i < (end + peak_frames - 1) / peak_frames; i++)
756 pt[i] = m.a_gain * pa[i % a_bins] + m.b_gain * pb[i % b_bins];
757 }
758 m.pos = end;
759 if(m.pos < m.merged)
760 return;
761
762 m.active = false;
763 begin_merge();
764 // What the two took from the passes under them, the merged one takes.
765 audio_layer a = merged_pass(layers[m.at], layers[m.at + 1]);
766 merge_seams(a);
767 a.from = layers[m.at + 1].from;
768 a.slot = m.target;
769 a.decay = (m.lost_a ? 1.f : m.a.decay) * (m.lost_b ? 1.f : m.b.decay);
770 a.start = 0;
771 a.period = m.merged == length ? 0 : m.merged;
772 a.written = m.merged;
773 a.muted = m.muted;
774 a.backwards = false;
775 merged(m.at, a, m.undoable, m.a, m.b);
776 }

784 float merged_at(const float* da, const float* db, int64_t k) const noexcept
785 {
786 const auto& m = merging;
787 const auto& a = layers[m.at];
788 const auto& b = layers[m.at + 1];
789 float va = 0.f, vb = 0.f;
790 if(valid_in(m.a, k, m.a_len))
791 {
792 va = da[k % m.a_len];
793 if(m.a_len < m.merged)
794 va *= repeat_weight(a, k, m.a_len, length);
795 }
796 if(valid_in(m.b, k, m.b_len))
797 {
798 vb = db[k % m.b_len];
799 if(m.b_len < m.merged)
800 vb *= repeat_weight(b, k, m.b_len, length);
801 }
802 return m.a_gain * va + m.b_gain * vb;
803 }

808 void merge_seams(audio_layer& into) noexcept
809 {
810 auto& m = merging;
811 const auto& a = layers[m.at];
812 const auto& b = layers[m.at + 1];
813 // The ends of the two are merged again from their buffers as they are
814 // now: a seam may have been weighed while the merge went on.
815 for(int c = 0; c < channels; c++)
816 {
817 float* dst = data(m.target, c);
818 const float* da = data(m.a_slot, c);
819 const float* db = data(m.b_slot, c);
820 for(const auto* l : {&a, &b})
821 {
822 const int64_t p = std::max<int64_t>(1, period_of(*l));
823 for(const auto* e : {&l->head, &l->tail})
824 if(e->at >= 0)
825 for(int i = 0; i < e->n; i++)
826 for(int64_t k = seam_index(*l, *e, i); k < m.merged; k += p)
827 dst[k] = merged_at(da, db, k);
828 }
829 }

844 const float* dst = data(m.target, c);
845 const float* part = kept_of(l.slot, tail, true, c);
846 float* to_orig = kept_of(m.target, tail, false, c);
847 float* to_part = kept_of(m.target, tail, true, c);
848 for(int i = 0; i < e.n; i++)
849 {
850 to_part[i] = gain * part[i];
851 to_orig[i] = dst[seam_index(l, e, i)]
852 + to_part[i] * (1.f - seam_weight(tail, e.rho, i, e.n));
853 }
854 }
855 };
856 // One end a side: another one there is kept in the merged pass as it is
857 // heard now.
858 if(outer(a, false, b, m.a_len, m.lost_a))
859 take(a, false, m.a_gain);

861 take(b, false, m.b_gain);
862 if(outer(b, true, a, m.b_len, m.lost_b))
863 take(b, true, m.b_gain);
864 else if(outer(a, true, b, m.a_len, m.lost_a))
865 take(a, true, m.a_gain);
866
867 merged_into[m.a_slot] = m.lost_a ? -1 : m.target;
868 merged_into[m.b_slot] = m.lost_b ? -1 : m.target;
869 }

919 void shift_earlier(audio_layer& l, int64_t frames) noexcept
920 {
921 const int64_t len = period_of(l);
922 if(len <= 0)
923 return;
924 // A negative count moves it later: the input of a pass the playhead went
925 // over backwards belongs further on in the loop.
926 frames %= len;
927 if(frames < 0)
928 frames += len;
929 if(frames <= 0)
930 return;
931 for(int c = 0; c < channels; c++)
932 {
933 float* d = data(l.slot, c);
934 std::rotate(d, d + frames, d + len);
935 }
936 float* p = peaks_of(l.slot);
937 const int64_t bins = (len + peak_frames - 1) / peak_frames;
938 const int64_t bin_shift = frames / peak_frames;
939 if(bin_shift > 0 && bins > 0)
940 std::rotate(p, p + (bin_shift % bins), p + bins);
941 l.start -= frames;
942 if(l.start < 0)
943 l.start += len;
944 for(auto* e : {&l.head, &l.tail})
945 if(e->at >= 0)
946 e->at = ((e->at - frames) % len + len) % len;
947 }

953void mix_passes(
954 Data&& data, const audio_layer* passes, int count, int c, int64_t len, int64_t from,
955 int64_t to, float* dst)
956{
957 std::fill(dst + from, dst + to, 0.f);

959 {

962 const int64_t p = audio_layers::period_in(l, len);
963 for(int64_t k = from; k < to; k++)

965 dst[k] += l.gain * src[audio_layers::index_in(l, k, len)]
966 * (p < len ? audio_layers::repeat_weight(l, k, p, len) : 1.f);
967 }
