110 bool triggered(bool received, bool changed) const noexcept
111 {
112 const bool full = buffer.full();
113 switch(inputs.mode)
114 {
115 case OutputMode::EveryTick:
116 return true;
117 case OutputMode::OnInput:
118 return received;
119 case OutputMode::OnChange:
120 return changed;
121 case OutputMode::WhenFull:
122 return full;
123 case OutputMode::OnInputWhenFull:
124 return received && full;
125 case OutputMode::OnChangeWhenFull:
126 return changed && full;
127 case OutputMode::OnBang:
128 case OutputMode::OnBangPopOldest:
129 return false;
130 }
131 return false;
132 }

134 void send()
135 {
136 if(buffer.empty())
137 {
138 if(inputs.data == OutputData::WholeBuffer)
139 outputs.output.value = std::vector<ossia::value>{};
140 else
141 outputs.output.value = ossia::value{};
142 return;
143 }
144
145 switch(inputs.data)
146 {
147 case OutputData::Oldest:
148 outputs.output.value = buffer.front();
149 if(inputs.pop || inputs.mode == OutputMode::OnBangPopOldest)
150 buffer.pop_front();
151 break;
152 case OutputData::WholeBuffer:
153 outputs.output.value = std::vector<ossia::value>(buffer.begin(), buffer.end());
154 if(inputs.pop)
155 buffer.clear();
156 else if(inputs.mode == OutputMode::OnBangPopOldest)
157 buffer.pop_front();
158 break;
159 case OutputData::Newest:
160 outputs.output.value = buffer.back();
161 if(inputs.pop)
162 buffer.pop_back();
163 else if(inputs.mode == OutputMode::OnBangPopOldest)
164 buffer.pop_front();
165 break;
166 }
167 }
168
169 void operator()()
170 {
171 const bool bang = std::exchange(banged, false);
172 outputs.output.value.reset();
173
174 const bool received = bool(inputs.input.value);
175 bool changed = false;
176 if(received)
177 {
178 if(!inputs.lock && !inputs.clear && buffer.capacity() > 0)
179 {
180 buffer.push_back(std::move(*inputs.input.value));
181 changed = true;
182 }
183 inputs.input.value.reset();
184 }
185
186 if(inputs.clear && !buffer.empty())
187 {
188 buffer.clear();
189 changed = true;
190 }
191
192 if(bang || triggered(received, changed))
193 send();
194 }
