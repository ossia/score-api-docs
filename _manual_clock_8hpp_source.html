46 StepGate(
47 ossia::audio_engine::fun_type t, int64_t bs, double rate, int ins, int outs)
48 : tick{std::move(t)}
49 , bufferSize{std::max<int64_t>(bs, 1)}
50 , sampleRate{rate > 0. ? rate : 44100.}

56 for(auto& c : outputs)

65 void run(const ossia::audio_tick_state& t)
66 {
67 for(int chan = 0; chan < t.n_out; chan++)
68 std::fill_n(t.outputs[chan], t.frames, 0.f);
69
70 using clk = std::chrono::steady_clock;
71 constexpr auto linger = std::chrono::milliseconds(5);
72 auto deadline = clk::now() + linger;
73 for(;;)
74 {
75 const int64_t target = requested.load(std::memory_order_acquire);
76 int64_t cur = done.load(std::memory_order_relaxed);
77 if(cur < target)
78 {
79 while(cur < target)
80 {
81 const int64_t n = std::min(target - cur, bufferSize);
82 ossia::audio_tick_state st{
83 inputPointers.data(),
84 outputPointers.data(),
85 int32_t(inputPointers.size()),
86 int32_t(outputPointers.size()),
87 uint64_t(n),
88 double(cur) / sampleRate};
89 tick(st);
90 cur += n;
91 }
92 done.store(cur, std::memory_order_release);
93 deadline = clk::now() + linger;
94 continue;
95 }
96 if(clk::now() >= deadline)
97 break;
98 std::this_thread::yield();
99 }
100 }
101
102 bool waitFor(int64_t target, const ossia::audio_engine& engine)
103 {
104 int64_t prev = requested.load(std::memory_order_relaxed);
105 if(target > prev)
106 requested.store(target, std::memory_order_release);
107
108 using clk = std::chrono::steady_clock;
109 const auto t0 = clk::now();
110 int spins = 0;
111 while(done.load(std::memory_order_acquire) < target)
112 {
113 if(!engine.running())
114 return false;
115 if(clk::now() - t0 > std::chrono::seconds(60))
116 return false;
117 if(spins++ < 1000)
118 std::this_thread::yield();
119 else
120 std::this_thread::sleep_for(std::chrono::microseconds(50));
121 }
122 return true;
123 }
