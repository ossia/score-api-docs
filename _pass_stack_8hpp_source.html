421 step undo() noexcept
422 {
423 if(done == 0)
424 {
425 // Past the actions kept, what redo would do again of them is not
426 // where it was any more.
427 if(count > 0)
428 recorded = 0;
429 return undo_top();
430 }
431 step s;
432 const uint32_t g = history[done - 1].group;
433 while(done > 0 && history[done - 1].group == g)
434 {
435 auto& a = history[--done];
436 switch(a.kind)
437 {
438 case action::pass: {
439 const step t = undo_top();
440 s.done |= t.done;
441 if(t.inserted >= 0)
442 s.inserted = t.inserted;
443 break;
444 }
445 case action::erase:
446 if(a.kept >= 0 && unkeep(a.at, a.kept))
447 {
448 a.kept = -1;
449 s.done = true;
450 s.inserted = a.at;
451 }
452 break;
453 case action::mute:
454 s.done |= apply_mute(a.at, !a.muted);
455 break;
456 }
457 }
458 return s;
459 }
460
462 step redo() noexcept
463 {
464 if(recorded <= done)
465 return redo_top();
466 step s;
467 const uint32_t g = history[done].group;
468 while(done < recorded && history[done].group == g)
469 {
470 switch(history[done].kind)
471 {
472 case action::pass: {
473 const step t = redo_top();
474 s.done |= t.done;
475 if(t.removed >= 0)
476 s.removed = t.removed;
477 break;
478 }
479 case action::erase: {
480 const int k = history[done].at;
481 if(k >= count || count < 2 || !self().may_remove(k))
482 break;
483 // Room for it may cost the oldest actions.
484 const int j = take_kept();
485 auto& a = history[done];
486 drop_unmerge();
487 forget_aside();
488 keep(k, j);
489 a.kept = int8_t(j);
490 s.done = true;
491 s.removed = k;
492 break;
493 }
494 case action::mute:
495 s.done |= apply_mute(history[done].at, history[done].muted);
496 break;
497 }
498 done++;
499 }
500 return s;
501 }

516 done = recorded = out;
517 }

520 return s;
521 }

524 step undo_top() noexcept
525 {
526 if(count == 0)
527 return {};
528 self().undoing();
529 // The pass merged into the newest one comes out of it.
530 if(unmerge.valid && unmerge.at == count - 1)
531 {
532 forget_aside();
533 const int at = unmerge.at;
534 unmerge.valid = false;
535 self().unmerged(at);
536 layers[at] = unmerge.a;
537 layers[at + 1] = unmerge.b;
538 layers[at + 1].loop_after = length;
539 length = unmerge.b.loop_before;
540 total = count + 1;
541 self().passes_changed();
542 return {.done = true};
543 }
544 drop_unmerge();
545 const int back = returns_on_undo();
546 auto& l = layers[count - 1];
547 if(count > 1)
548 {
549 l.loop_after = length;
550 length = l.loop_before;
551 }
552 count--;
553 if(back >= 0)
554 put_back();
555 self().passes_changed();
556 return {.done = true, .inserted = back};
557 }

560 step redo_top() noexcept
561 {
562 if(count >= total)
563 return {};
564 self().undoing();
565 drop_unmerge();
566 // The pass that replaced a layer takes its place again.
567 const int k = leaves_on_redo();
568 if(k >= 0)
569 {
570 const int by = aside.by;
571 move_aside(k);
572 aside.by = by;
573 }
574 if(count > 0)
575 length = layers[count].loop_after;
576 count++;
577 self().passes_changed();
578 return {.done = true, .removed = k};
579 }

678 void drop_oldest(int n) noexcept
679 {
680 n = std::clamp(n, 0, recorded);
681 for(int i = 0; i < n; i++)
682 if(history[i].kind == action::erase && history[i].kept >= 0)
683 {
684 const int j = history[i].kept;
685 kept_used &= ~(uint32_t(1) << j);
686 self().kept_forgotten(j);

696 int take_kept() noexcept
697 {
698 for(;;)
699 {
700 for(int j = 0; j < max_kept; j++)
701 if(!(kept_used & (uint32_t(1) << j)))
702 return j;
703 if(!forget_oldest_kept())
704 {
705 // Kept by actions redo would do again: those go.
706 recorded = done;
707 for(int j = 0; j < max_kept; j++)
708 self().kept_forgotten(j);
709 kept_used = 0;
710 }
711 }
712 }

717 self().removing(k);
718 kept_passes[j] = layers[k];

729 return false;

734 layers[i] = layers[i - 1];

741 }

799template <int MaxPasses, int MaxKept>
800struct tagged_material
801{
802 static constexpr int aside_pass = MaxPasses;
803 static constexpr int lost_pass = MaxPasses + 1;
804 static constexpr int kept_pass = MaxPasses + 2;
805 static_assert(kept_pass + MaxKept <= 64);
806 static constexpr uint64_t bit(int k) noexcept { return uint64_t(1) << k; }
808 static constexpr uint64_t passes_mask = bit(aside_pass) - 1;
809
810 static constexpr uint64_t without(uint64_t m, int k) noexcept
811 {
812 return (without_pass(m & passes_mask, k) & passes_mask) | (m & ~passes_mask);
813 }
814 static constexpr uint64_t with_room(uint64_t m, int k) noexcept
815 {
816 return (with_room_at(m & passes_mask, k) & passes_mask) | (m & ~passes_mask);
817 }
818
821 template <typename T>
822 static void set_aside(T& e, int k, int to = aside_pass) noexcept
823 {
824 const bool by_k = e.replaced_by & bit(k);
825 e.replaced_by = without(e.replaced_by, k);
826 if(by_k)
827 e.replaced_by |= bit(to);
828 if(e.pass == k)
829 e.pass = uint8_t(to);
830 else if(e.pass > k && e.pass < aside_pass)
831 e.pass--;
832 }
833
835 template <typename T>
836 static void put_back(T& e, int k, int from = aside_pass) noexcept
837 {
838 const bool by_it = e.replaced_by & bit(from);
839 e.replaced_by = with_room(e.replaced_by & ~bit(from), k);
840 if(by_it)
841 e.replaced_by |= bit(k);
842 if(e.pass == from)
843 e.pass = uint8_t(k);
844 else if(e.pass >= k && e.pass < aside_pass)
845 e.pass++;
846 }
847
852 template <typename T>
853 static void merge(T& e, int a, bool lost_a, bool lost_b) noexcept
854 {
855 const int b = a + 1;
856 const bool had_a = e.replaced_by & bit(a);
857 const bool had_b = e.replaced_by & bit(b);
858 e.merged = uint8_t(
859 (e.pass == b ? from_upper : 0) | (had_a ? replaced_by_lower : 0)
860 | (had_b ? replaced_by_upper : 0));
861 uint64_t r = e.replaced_by & ~(bit(a) | bit(b));
862 if((had_a && !lost_a) || (had_b && !lost_b))
863 r |= bit(a);
864 e.replaced_by = without(r, b);
865 if((e.pass == a && lost_a) || (e.pass == b && lost_b))
866 e.pass = uint8_t(lost_pass);
867 else if(e.pass >= b && e.pass < aside_pass)
868 e.pass--;
869 }
870
872 template <typename T>
873 static void unmerge(T& e, int a) noexcept
874 {
875 const int b = a + 1;
876 uint64_t r = with_room(e.replaced_by, b) & ~bit(a);
877 if(e.merged & replaced_by_lower)
878 r |= bit(a);
879 if(e.merged & replaced_by_upper)
880 r |= bit(b);
881 e.replaced_by = r;
882 if(e.pass == lost_pass || (e.merged & from_upper))
883 e.pass = uint8_t((e.merged & from_upper) ? b : a);
884 e.merged = 0;
885 }
886};
