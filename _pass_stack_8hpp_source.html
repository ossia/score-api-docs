404 step undo() noexcept
405 {
406 if(done == 0)
407 {
408 // Past the actions kept, what redo would do again of them is not
409 // where it was any more.
410 if(count > 0)
411 recorded = 0;
412 return undo_top();
413 }
414 step s;
415 const uint32_t g = history[done - 1].group;
416 while(done > 0 && history[done - 1].group == g)
417 {
418 auto& a = history[--done];
419 switch(a.kind)
420 {
421 case action::pass: {
422 const step t = undo_top();
423 s.done |= t.done;
424 if(t.inserted >= 0)
425 s.inserted = t.inserted;
426 break;
427 }
428 case action::erase:
429 if(a.kept >= 0 && unkeep(a.at, a.kept))
430 {
431 a.kept = -1;
432 s.done = true;
433 s.inserted = a.at;
434 }
435 break;
436 case action::mute:
437 s.done |= apply_mute(a.at, !a.muted);
438 break;
439 }
440 }
441 return s;
442 }
443
445 step redo() noexcept
446 {
447 if(recorded <= done)
448 return redo_top();
449 step s;
450 const uint32_t g = history[done].group;
451 while(done < recorded && history[done].group == g)
452 {
453 switch(history[done].kind)
454 {
455 case action::pass: {
456 const step t = redo_top();
457 s.done |= t.done;
458 if(t.removed >= 0)
459 s.removed = t.removed;
460 break;
461 }
462 case action::erase: {
463 const int k = history[done].at;
464 if(k >= count || count < 2 || !self().may_remove(k))
465 break;
466 // Room for it may cost the oldest actions.
467 const int j = take_kept();
468 auto& a = history[done];
469 drop_unmerge();
470 forget_aside();
471 keep(k, j);
472 a.kept = int8_t(j);
473 s.done = true;
474 s.removed = k;
475 break;
476 }
477 case action::mute:
478 s.done |= apply_mute(history[done].at, history[done].muted);
479 break;
480 }
481 done++;
482 }
483 return s;
484 }

507 step undo_top() noexcept
508 {
509 if(count == 0)
510 return {};
511 self().undoing();
512 // The pass merged into the newest one comes out of it.
513 if(unmerge.valid && unmerge.at == count - 1)
514 {
515 forget_aside();
516 const int at = unmerge.at;
517 unmerge.valid = false;
518 self().unmerged(at);
519 layers[at] = unmerge.a;
520 layers[at + 1] = unmerge.b;
521 layers[at + 1].loop_after = length;
522 length = unmerge.b.loop_before;
523 total = count + 1;
524 self().passes_changed();
525 return {.done = true};
526 }
527 drop_unmerge();
528 const int back = returns_on_undo();
529 auto& l = layers[count - 1];
530 if(count > 1)
531 {
532 l.loop_after = length;
533 length = l.loop_before;
534 }
535 count--;
536 if(back >= 0)
537 put_back();
538 self().passes_changed();
539 return {.done = true, .inserted = back};
540 }

659 void drop_oldest(int n) noexcept
660 {
661 n = std::clamp(n, 0, recorded);
662 for(int i = 0; i < n; i++)
663 if(history[i].kind == action::erase && history[i].kept >= 0)
664 {
665 const int j = history[i].kept;
666 kept_used &= ~(uint32_t(1) << j);
667 self().kept_forgotten(j);
668 }
669 for(int i = n; i < recorded; i++)
670 history[i - n] = history[i];
671 done = std::max(0, done - n);

677 int take_kept() noexcept
678 {
679 for(;;)
680 {
681 for(int j = 0; j < max_kept; j++)
682 if(!(kept_used & (uint32_t(1) << j)))
683 return j;
684 if(!forget_oldest_kept())
685 {
686 // Kept by actions redo would do again: those go.
687 recorded = done;
688 for(int j = 0; j < max_kept; j++)
689 self().kept_forgotten(j);
690 kept_used = 0;
691 }
692 }

722 }

780template <int MaxPasses, int MaxKept>
781struct tagged_material
782{
783 static constexpr int aside_pass = MaxPasses;
784 static constexpr int lost_pass = MaxPasses + 1;
785 static constexpr int kept_pass = MaxPasses + 2;
786 static_assert(kept_pass + MaxKept <= 64);
787 static constexpr uint64_t bit(int k) noexcept { return uint64_t(1) << k; }
789 static constexpr uint64_t passes_mask = bit(aside_pass) - 1;
790
791 static constexpr uint64_t without(uint64_t m, int k) noexcept
792 {
793 return (without_pass(m & passes_mask, k) & passes_mask) | (m & ~passes_mask);
794 }
795 static constexpr uint64_t with_room(uint64_t m, int k) noexcept
796 {
797 return (with_room_at(m & passes_mask, k) & passes_mask) | (m & ~passes_mask);
798 }

802 template <typename T>
803 static void set_aside(T& e, int k, int to = aside_pass) noexcept
804 {
805 const bool by_k = e.replaced_by & bit(k);
806 e.replaced_by = without(e.replaced_by, k);
807 if(by_k)
808 e.replaced_by |= bit(to);
809 if(e.pass == k)
810 e.pass = uint8_t(to);
811 else if(e.pass > k && e.pass < aside_pass)
812 e.pass--;
813 }
814
816 template <typename T>
817 static void put_back(T& e, int k, int from = aside_pass) noexcept
818 {
819 const bool by_it = e.replaced_by & bit(from);
820 e.replaced_by = with_room(e.replaced_by & ~bit(from), k);
821 if(by_it)
822 e.replaced_by |= bit(k);
823 if(e.pass == from)
824 e.pass = uint8_t(k);
825 else if(e.pass >= k && e.pass < aside_pass)
826 e.pass++;
827 }
828
833 template <typename T>
834 static void merge(T& e, int a, bool lost_a, bool lost_b) noexcept
835 {
836 const int b = a + 1;
837 const bool had_a = e.replaced_by & bit(a);
838 const bool had_b = e.replaced_by & bit(b);
839 e.merged = uint8_t(
840 (e.pass == b ? from_upper : 0) | (had_a ? replaced_by_lower : 0)
841 | (had_b ? replaced_by_upper : 0));
842 uint64_t r = e.replaced_by & ~(bit(a) | bit(b));
843 if((had_a && !lost_a) || (had_b && !lost_b))
844 r |= bit(a);
845 e.replaced_by = without(r, b);
846 if((e.pass == a && lost_a) || (e.pass == b && lost_b))
847 e.pass = uint8_t(lost_pass);
848 else if(e.pass >= b && e.pass < aside_pass)
849 e.pass--;
850 }
851
853 template <typename T>
854 static void unmerge(T& e, int a) noexcept
855 {
856 const int b = a + 1;
857 uint64_t r = with_room(e.replaced_by, b) & ~bit(a);
858 if(e.merged & replaced_by_lower)
859 r |= bit(a);
860 if(e.merged & replaced_by_upper)
861 r |= bit(b);
862 e.replaced_by = r;
863 if(e.pass == lost_pass || (e.merged & from_upper))
864 e.pass = uint8_t((e.merged & from_upper) ? b : a);
865 e.merged = 0;
866 }
867};

975 bool in_pass(int64_t k) const noexcept
