59template <typename Iterator>
60static constexpr void constexpr_rotate(Iterator first, Iterator middle, Iterator last)
61{
62 using namespace std;
63 Iterator next = middle;
64 while(first != next)
65 {
66 constexpr_swap(*first++, *next++);
67 if(next == last)
68 next = middle;
69 else if(first == middle)
70 middle = next;
71 }
72}

76static constexpr scales_array make_scale(std::initializer_list<bool> notes)
77{
78 std::array<scale_array, 12> r{};
79 for(std::size_t octave = 0; octave < 11; octave++)
80 {
81 std::size_t pos = 0;
82 for(bool note : notes)
83 {
84 if(octave * 12 + pos < 128)
85 {
86 r[0][octave * 12 + pos] = note;
87 pos++;
88 }
89 }
90 }
91
92 for(std::size_t octave = 1; octave < 12; octave++)
93 {
94 r[octave] = r[0];
95 constexpr_rotate(r[octave].rbegin(), r[octave].rbegin() + octave, r[octave].rend());
96 }
97 return r;
98}

171static std::optional<std::size_t> find_closest_index(const scale_array& arr, std::size_t i)
172{
173 const int n = int(arr.size());
174 const int note = int(i);
175 for(int r = 0; r < n; r++)
176 {
177 if(note + r < n && arr[note + r])
178 return note + r;
179 if(note - r >= 0 && arr[note - r])
180 return note - r;
181 }
182 return std::nullopt;
183}

250 void exec(const scale_array& scale, int transp)
251 {
252 for(const auto& msg : inputs.midi)
253 {
254 const auto type = msg.get_message_type();
255 const bool is_note
256 = type == libremidi::message_type::NOTE_ON
257 || type == libremidi::message_type::NOTE_OFF;
258 if(!is_note || msg.size() < 3)
259 {
260 if(type == libremidi::message_type::CONTROL_CHANGE && msg.size() >= 3
261 && (msg.bytes[1] == 120 || msg.bytes[1] == 123))
262 release_channel(msg.get_channel());
263 outputs.midi.push_back(msg);
264 continue;
265 }
266
267 const int chan = msg.get_channel();
268 const int pitch = msg.bytes[1] & 0x7F;
269 const int vel = msg.bytes[2] & 0x7F;
270 const auto ts = msg.timestamp;
271 const auto k = key(chan, pitch);
272
273 // A held key stops its note on re-press or release (note on with
274 // velocity 0 is a release).
275 if(auto it = map.find(k); it != map.end())
276 {
277 stop(it->second.chan, it->second.pitch, vel, ts);
278 map.erase(it);
279 }
280
281 if(type == libremidi::message_type::NOTE_ON && vel > 0)
282 {
283 if(auto out = map_note(scale, pitch, transp))
284 {
285 start(chan, *out, vel, ts);
286 map.insert({k, Note{*out, (uint8_t)vel, (uint8_t)chan}});
287 }
288 }
289 }

293 void update(const scale_array& scale, int transp)
294 {
295 for(auto it = map.begin(); it != map.end();)
296 {
297 Note& note = it->second;
298 const auto out = map_note(scale, it->first % 128, transp);
299 if(out && *out == note.pitch)
300 {
301 ++it;
302 continue;
303 }
304
305 stop(note.chan, note.pitch, 0, 0);
306 if(out)
307 {
308 start(note.chan, *out, note.vel, 0);
309 note.pitch = *out;
310 ++it;
311 }
312 else
313 {
314 it = map.erase(it);
315 }
316 }
317 }
