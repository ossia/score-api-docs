61template <typename Iterator>
62static constexpr void constexpr_rotate(Iterator first, Iterator middle, Iterator last)
63{
64 using namespace std;
65 Iterator next = middle;
66 while(first != next)
67 {
68 constexpr_swap(*first++, *next++);
69 if(next == last)
70 next = middle;
71 else if(first == middle)
72 middle = next;
73 }
74}

78static constexpr scales_array make_scale(std::initializer_list<bool> notes)
79{
80 std::array<scale_array, 12> r{};
81 for(std::size_t octave = 0; octave < 11; octave++)
82 {
83 std::size_t pos = 0;
84 for(bool note : notes)
85 {
86 if(octave * 12 + pos < 128)
87 {
88 r[0][octave * 12 + pos] = note;
89 pos++;
90 }
91 }
92 }
93
94 for(std::size_t octave = 1; octave < 12; octave++)
95 {
96 r[octave] = r[0];
97 constexpr_rotate(r[octave].rbegin(), r[octave].rbegin() + octave, r[octave].rend());
98 }
99 return r;
100}

173static std::optional<std::size_t> find_closest_index(const scale_array& arr, std::size_t i)
174{
175 const int n = int(arr.size());
176 const int note = int(i);
177 for(int r = 0; r < n; r++)
178 {
179 if(note + r < n && arr[note + r])
180 return note + r;
181 if(note - r >= 0 && arr[note - r])
182 return note - r;
183 }
184 return std::nullopt;
185}

204static std::optional<std::size_t>
205find_index(const scale_array& arr, std::size_t i, scale_direction dir)
206{
207 if(dir == scale_direction::nearest)
208 return find_closest_index(arr, i);
209 const int n = int(arr.size());
210 const int step = dir == scale_direction::up ? 1 : -1;
211 for(int k = int(i); k >= 0 && k < n; k += step)
212 if(arr[k])
213 return k;
214 for(int k = int(i) - step; k >= 0 && k < n; k -= step)
215 if(arr[k])
216 return k;
217 return std::nullopt;
218}

375 std::optional<uint8_t>
376 map_note(const scale_array& scale, int pitch, int transp) const noexcept
377 {
378 if(pitch < lowest || pitch > lowest + range)
379 return (uint8_t)pitch;
380 if(mode == scale_mode::remap)
381 {
382 const int degree = ((pitch - base) % 12 + 12) % 12;
383 const int target = degrees[degree];
384 if(target < 0 || target > 12)
385 return std::nullopt;
386 int offset = target - degree;
387 if(fold && offset > 6)
388 offset -= 12;
389 else if(fold && offset < -6)
390 offset += 12;
391 return (uint8_t)ossia::clamp(pitch + offset + transp, 0, 127);
392 }
393 if(auto index = find_index(scale, pitch, direction))
394 return (uint8_t)ossia::clamp(int(*index) + transp, 0, 127);
395 return std::nullopt;
396 }

413 void exec(const scale_array& scale, int transp)
414 {
415 for(const auto& msg : inputs.midi)
416 {
417 const auto type = msg.get_message_type();
418 const bool is_note
419 = type == libremidi::message_type::NOTE_ON
420 || type == libremidi::message_type::NOTE_OFF;
421 if(!is_note || msg.size() < 3)
422 {
423 if(type == libremidi::message_type::CONTROL_CHANGE && msg.size() >= 3
424 && (msg.bytes[1] == 120 || msg.bytes[1] == 123))
425 release_channel(msg.get_channel());
426 outputs.midi.push_back(msg);
427 continue;
428 }
429
430 const int chan = msg.get_channel();
431 const int pitch = msg.bytes[1] & 0x7F;
432 const int vel = msg.bytes[2] & 0x7F;
433 const auto ts = msg.timestamp;
434 const auto k = key(chan, pitch);
435
436 // A held key stops its note on re-press or release (note on with
437 // velocity 0 is a release).
438 if(auto it = map.find(k); it != map.end())
439 {
440 stop(it->second.chan, it->second.pitch, vel, ts);
441 map.erase(it);
442 }
443
444 if(type == libremidi::message_type::NOTE_ON && vel > 0)
445 {
446 if(auto out = map_note(scale, pitch, transp))
447 {
448 start(chan, *out, vel, ts);
449 map.insert({k, Note{*out, (uint8_t)vel, (uint8_t)chan}});
450 }
451 }
452 }
453 }

456 void update(const scale_array& scale, int transp)
457 {
458 for(auto it = map.begin(); it != map.end();)
459 {
460 Note& note = it->second;
461 const auto out = map_note(scale, it->first % 128, transp);
462 if(out && *out == note.pitch)
463 {
464 ++it;
465 continue;
466 }
467
468 stop(note.chan, note.pitch, 0, 0);
469 if(out)
470 {
471 start(note.chan, *out, note.vel, 0);
472 note.pitch = *out;
473 ++it;
474 }
475 else
476 {
477 it = map.erase(it);
478 }
479 }

499 static void
500 resolve_scale(std::string_view text, int base, scale_array& out) noexcept
501 {
502 const auto idx = get_scale(text);
503 if(idx >= 0 && idx < scale_type::custom)
504 {
505 out = scales[idx][base];
506 return;
507 }
508
509 const auto degrees = ossia::min(std::ssize(text), std::ptrdiff_t(12));
510 bool is_mask = degrees > 0;
511 for(std::ptrdiff_t d = 0; d < degrees; d++)
512 is_mask &= text[d] == '1' || text[d] == '0';
513 for(std::size_t note = 0; note < out.size(); note++)
514 {
515 const auto degree = std::ptrdiff_t((note + 12 - base) % 12);
516 out[note] = !is_mask || (degree < degrees && text[degree] == '1');
517 }
518 }

524 // A base of 12 is the same scale one octave up.
