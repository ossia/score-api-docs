65 void createOutput(score::gfx::OutputConfiguration conf) override
66 {
67 m_onResize = conf.onResize;
68 m_onReleaseRenderList = conf.onReleaseRenderList;
69 // Cache the requested graphics API so setSwapchainFormat can rebuild
70 // through createOutput when the format changes (live HDR↔SDR toggle):
71 // setting m_swapchainFormat alone leaves the QRhiTexture in its original
72 // format.
73 m_lastGraphicsApi = conf.graphicsApi;
74
75 QSize newSz = m_renderSize;
76 if(newSz.width() <= 0 || newSz.height() <= 0)
77 newSz = m_size;
78 if(newSz.width() <= 0 || newSz.height() <= 0)
79 newSz = QSize{1024, 1024};
80
81 m_renderState = score::gfx::createRenderState(
82 conf.graphicsApi, newSz, nullptr, m_deviceMode);
83 if(!m_renderState || !m_renderState->rhi)
84 {
85 qWarning() << "BackgroundNode: failed to create QRhi";
86 m_renderState.reset();
87 return;
88 }
89 m_renderState->outputSize = m_renderState->renderSize;
90 m_renderState->renderFormat
91 = (m_swapchainFormat != Gfx::SwapchainFormat::SDR)
92 ? QRhiTexture::RGBA32F
93 : QRhiTexture::RGBA8;
94
95 auto rhi = m_renderState->rhi;
96 m_texture = rhi->newTexture(
97 m_renderState->renderFormat, m_renderState->renderSize, 1,
98 QRhiTexture::RenderTarget | QRhiTexture::UsedAsTransferSource);
99 m_texture->create();
100
101 // Reverse-Z project rule: depth attachment is D32F (float). Fixed-point
102 // D24 combined with reverse-Z gives strictly worse precision than
103 // standard-Z, so we must allocate a float texture here. RenderTarget
104 // flag is required for attaching as a depth target.
105 m_depthTexture = rhi->newTexture(
106 QRhiTexture::D32F, m_renderState->renderSize, 1,
107 QRhiTexture::RenderTarget);
108 m_depthTexture->setName("BackgroundNode::m_depthTexture");
109 m_depthTexture->create();
110
111 QRhiTextureRenderTargetDescription desc;
112 desc.setColorAttachments({QRhiColorAttachment(m_texture)});
113
114 desc.setDepthTexture(m_depthTexture);
115 m_renderTarget = rhi->newTextureRenderTarget(desc);
116 m_renderState->renderPassDescriptor
117 = m_renderTarget->newCompatibleRenderPassDescriptor();
118 m_renderTarget->setRenderPassDescriptor(m_renderState->renderPassDescriptor);
119 m_renderTarget->create();
120
121 conf.onReady();
122 }

124 void destroyOutput() override
125 {
126 if(m_renderState)
127 {
128 // Drain the GPU before tearing resources down. Same rationale as
129 // ScreenNode::destroyOutput: when setSwapchainFormat invokes
130 // destroyOutput synchronously, an unfinished cbWrapper from a prior
131 // offscreen frame can still be referenced by ScenePreprocessor's
132 // per-frame copyBuffer. Recording into that CB after we've
133 // freed the rhi triggers VUID-vkCmdCopyBuffer-commandBuffer-
134 // recording and a device loss.
135 if(m_renderState->rhi)
136 {
137 SCORE_ASSERT(!m_renderState->rhi->isRecordingFrame());
138 m_renderState->rhi->finish();
139 }
140
141 // The registry survives render list teardown, so its QRhi resources
142 // must be released here, before RenderState::destroy() tears down the
143 // QRhi: destroyOwned() `delete`s the wrappers while the device is alive.
144 releaseRegistry();
145
146 delete m_renderTarget;
147 m_renderTarget = nullptr;
148
149 delete m_depthTexture;
150 m_depthTexture = nullptr;
151
152 delete m_texture;
153 m_texture = nullptr;
154
155 delete m_renderState->renderPassDescriptor;
156 m_renderState->renderPassDescriptor = nullptr;
157
158 m_renderState->destroy();
159 m_renderState.reset();
160 }

170 void setSwapchainFormat(Gfx::SwapchainFormat format)
171 {
172 if(m_swapchainFormat == format)
173 return;
174 m_swapchainFormat = format;
175
176 // Live format change while rendering: m_texture was allocated by
177 // createOutput with the prior format, and setFormat alone does not
178 // re-allocate its GPU memory backing. Route through destroyOutput +
179 // createOutput so the render target, RPD, depth and colour textures all
180 // come back in the matching format. With no output yet (m_renderState
181 // null) createOutput picks the format up from m_swapchainFormat.
182 if(m_renderState)
183 {
184 score::gfx::OutputConfiguration conf;
185 conf.graphicsApi = m_lastGraphicsApi;
186 conf.onResize = m_onResize;
187 conf.onReleaseRenderList = m_onReleaseRenderList;
188 // The Graph-owned RenderList holds QRhiResources belonging to the QRhi
189 // destroyOutput() is about to `delete`; release it first.
190 releaseOwnedRenderList();
191 destroyOutput();
192 createOutput(std::move(conf));
193 if(m_onResize)
194 m_onResize();
195 }

215 void resize()
216 {
217 QSize newSz = m_renderSize;
218 if(newSz.width() <= 0 || newSz.height() <= 0)
219 newSz = m_size;
220 if(newSz.width() <= 0 || newSz.height() <= 0)
221 newSz = QSize{1024, 1024};
222
223 if(m_renderState)
224 {
225 m_renderState->renderSize = newSz;
226 m_renderState->outputSize = newSz;
227
228 auto rhi = m_renderState->rhi;
229
230 // Drain the GPU before destroying m_renderTarget / m_texture /
231 // m_depthTexture, as destroyOutput does: the current frame's offscreen
232 // CB, or a queued one, may still reference these resources, and the
233 // setPixelSize + create below does not drain internally. Without this,
234 // validation fires on the next vkCmd* recording (-recording /
235 // -commandBuffer-recording / -in-use) and can lose the device.
236 rhi->finish();
237
238 m_renderTarget->destroy();
239 m_texture->destroy();
240 m_texture->setPixelSize(newSz);
241 m_texture->create();
242
243 if(m_depthTexture)
244 m_depthTexture->destroy();
245 else
246 m_depthTexture = rhi->newTexture(
247 QRhiTexture::D32F, newSz, 1, QRhiTexture::RenderTarget);
248 m_depthTexture->setPixelSize(newSz);
249 m_depthTexture->create();
250
251 m_renderTarget->deleteLater();
252 if(auto* rpd = m_renderState->renderPassDescriptor)
253 rpd->deleteLater();
254 m_renderState->renderPassDescriptor = nullptr;
255 m_renderTarget = nullptr;
256
257 QRhiTextureRenderTargetDescription desc;
258 desc.setColorAttachments({QRhiColorAttachment(m_texture)});
259 desc.setDepthTexture(m_depthTexture);
260 m_renderTarget = rhi->newTextureRenderTarget(desc);
261 m_renderState->renderPassDescriptor
262 = m_renderTarget->newCompatibleRenderPassDescriptor();
263 m_renderTarget->setRenderPassDescriptor(m_renderState->renderPassDescriptor);
264 m_renderTarget->create();
265 }
266
267 if(m_onResize)
268 m_onResize();
269 }
