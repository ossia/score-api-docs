166template <typename F>
167ossia::net::node_base* createScriptableNode(
168 const ScriptableRoots& roots, ossia::net::node_base& parent, const QString& name,
169 F&& writeBack)
170{
171 ossia::net::node_base* node{};
172 if(roots.revive)
173 node = roots.revive(parent, name.toStdString());
174 if(!node)
175 node = parent.create_child(name.toStdString());
176 if(!node)
177 return nullptr;
178 if(auto real = QString::fromStdString(node->get_name()); real != name)
179 writeBack(real);
180 return node;
181}

189 node.set_name(wanted);
190 if(auto real = QString::fromStdString(node.get_name()); real != name)
191 writeBack(real);
192}
