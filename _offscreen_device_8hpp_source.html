74 ~offscreen_device()
75 {
76 // Remove our output from the render graph synchronously, while both the
77 // BackgroundNode and its QRhi are still alive. This output is owned by the
78 // device (not the graph), so on shutdown it is torn down here before
79 // Graph::~Graph runs — and the graph's async REMOVE_NODE queue is never
80 // drained again. Without this, the graph is left with a dangling output
81 // pointer + a RenderList over a freed QRhi (SIGSEGV in ~Graph teardown).
82 if(auto* proto = static_cast<gfx_protocol_base*>(m_protocol.get());
83 proto && proto->context && proto->context->ui)
84 proto->context->ui->destroyOutput(m_node);
85
86 m_protocol->stop();
87 m_root.clear_children();
88 m_protocol.reset();
89 }
