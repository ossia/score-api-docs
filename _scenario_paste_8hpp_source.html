28template <typename CopiedObjects, typename ParentObject>
29ossia::flat_map<Id<Process::Cable>, Process::CableData> mapCopiedCables(
30 const score::DocumentContext& ctx, std::vector<Process::CableData>& cables,
31 std::vector<CopiedObjects*>& intervals,
32 const std::vector<Id<CopiedObjects>>& interval_ids, const ParentObject& scenario)
33{
34 ossia::flat_map<Id<Process::Cable>, Process::CableData> cable_map;
35
36 {
37 ossia::hash_map<Id<CopiedObjects>, Id<CopiedObjects>> id_map;
38 {
39 int i = 0;
40 for(CopiedObjects* interval : intervals)
41 {
42 id_map[interval->id()] = interval_ids[i];
43 i++;
44 }
45 }
46
47 auto& doc = score::IDocument::modelDelegate<ScenarioDocumentModel>(ctx.document);
48 auto cable_ids = getStrongIdRange<Process::Cable>(cables.size(), doc.cables);
49
50 int i = 0;
51 Path<ParentObject> p{scenario};
52 for(Process::CableData& cd : cables)
53 {
54 auto& source_vec = cd.source.unsafePath().vec();
55 auto& sink_vec = cd.sink.unsafePath().vec();
56 SCORE_ASSERT(!source_vec.empty());
57 SCORE_ASSERT(!sink_vec.empty());
58 int32_t source_itv_id = source_vec.front().id();
59 int32_t sink_itv_id = sink_vec.front().id();
60
61 for(CopiedObjects* interval : intervals)
62 {
63 auto id = interval->id().val();
64 if(id == source_itv_id)
65 source_itv_id = id_map.at(interval->id()).val();
66 if(id == sink_itv_id)
67 sink_itv_id = id_map.at(interval->id()).val();
68 }
69 source_vec.front()
70 = ObjectIdentifier{source_vec.front().objectName(), source_itv_id};
71 sink_vec.front() = ObjectIdentifier{sink_vec.front().objectName(), sink_itv_id};
72
73 source_vec.insert(
74 source_vec.begin(), p.unsafePath().vec().begin(), p.unsafePath().vec().end());
75 sink_vec.insert(
76 sink_vec.begin(), p.unsafePath().vec().begin(), p.unsafePath().vec().end());
77
78 cable_map.insert({cable_ids[i], std::move(cd)});
79 i++;
80 }
81
82 for(CopiedObjects* interval : intervals)
83 {
84 const auto ports = interval->template findChildren<Process::Port*>();
85 for(Process::Port* port : ports)
86 {
87 while(!port->cables().empty())
88 {
89 port->removeCable(port->cables().back());
90 }
91 }
92 }
93 }
94
95 return cable_map;
96}
