101 void run(const ossia::token_request& tk, ossia::exec_state_facade st) noexcept override
102 {
103 // Before any early return: the inlets are cleared once the node has run.
104 read_controls();
105
106 if(tk.paused() || sequences.empty())
107 return;
108 current_sequence = std::clamp(current_sequence, 0, int(sequences.size()) - 1);
109 if(pending_sequence >= int(sequences.size()))
110 pending_sequence = -1;
111
112 const double ratio = st.modelToSamples();
113 const int64_t tick_start = tk.physical_start(ratio);
114 const bool rewinding = tk.backward();
115

117 int64_t switch_at = never;

133 if(switch_at != never)

143 void play_steps(
144 const ossia::token_request& tk, double ratio, int64_t tick_start, bool rewinding,
145 int64_t from, int64_t to) noexcept
146 {
147 if(!(step_duration > 0.) || !std::isfinite(step_duration))
148 return;
149
150 auto play = [&](int64_t sample) {
151 if(sample >= from && sample < to)
152 play_step(sample, rewinding);
153 };
154
155 if(synced)
156 {
157 // The grid of the tempo and of the bar lines, in tick order both ways.
158 for(const auto& q : tk.get_quantification_dates(1. / step_duration))
159 play(tick_start + tk.physical_position(q.position, ratio));
160 }
161 else
162 {
163 // k * d from the start of the interval.
164 const double d = step_duration * ossia::flicks_per_second<double>;
165 const int64_t lo = std::min(tk.prev_date.impl, tk.date.impl);
166 const int64_t hi = std::max(tk.prev_date.impl, tk.date.impl);
167 // [prev; date[ going forward, ]date; prev] rewinding: the same steps
168 // are crossed on the way out and on the way back.
169 int64_t k0 = rewinding ? int64_t(std::floor(lo / d)) + 1 : int64_t(std::ceil(lo / d));
170 int64_t k1 = rewinding ? int64_t(std::floor(hi / d)) : int64_t(std::ceil(hi / d)) - 1;
171 // A tick never holds more steps than it has samples.
172 k1 = std::min(k1, k0 + 4096);
173 if(!rewinding)
174 {
175 for(int64_t k = k0; k <= k1; k++)
176 play(tk.to_physical_time_in_tick(ossia::time_value{int64_t(k * d)}, ratio));
177 }
178 else
179 {
180 for(int64_t k = k1; k >= k0; k--)
181 play(tk.to_physical_time_in_tick(ossia::time_value{int64_t(k * d)}, ratio));
182 }
183 }
184 }

186 void play_step(int64_t sample, bool rewinding) noexcept
187 {
188 const auto& seq = sequences[current_sequence];
189 const int n = seq.size();
190 if(n == 0)
191 return;
192 // Rewinding mirrors going forward: step back first, so going out and back
193 // over the same ground crosses the same steps.
194 if(rewinding)
195 current = (current + n - 1) % n;
196 if(current < 0 || current >= n)
197 current = 0;
198 out->write_value(seq[current], sample);
199 last.store(current, std::memory_order_relaxed);
200 if(!rewinding)
201 current = (current + 1) % n;
202 }
