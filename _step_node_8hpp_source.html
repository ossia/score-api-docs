109 {

168 void follow_jump(const ossia::token_request& tk) noexcept
169 {
170 // Quarter notes since the start of the process, at this tick's tempo.
171 const double elapsed
172 = tk.prev_date.impl / ossia::flicks_per_second<double> * tk.tempo / 60.;
173 origin = tk.musical_start_position - elapsed;
174
175 const int n = sequences.empty() ? 0 : int(sequences[current_sequence].size());
176 if(n == 0 || !(step_duration > 0.) || !std::isfinite(step_duration))
177 return;
178
179 // Steps whose time has come before this tick, the one on its start excluded.
180 double before{};
181 if(synced)
182 {
183 const double q = step_quarters(tk);
184 if(q > 0.)
185 before = std::ceil((tk.musical_start_position - origin) / q - 1e-9);
186 }
187 else
188 {
189 const double d = step_duration * ossia::flicks_per_second<double>;
190 before = std::ceil(tk.prev_date.impl / d);
191 }
192 if(std::isfinite(before) && before >= 0.)
193 current = int(std::fmod(before, double(n)));
194 }

197 void play_steps(
198 const ossia::token_request& tk, double ratio, int64_t tick_start, bool rewinding,
199 int64_t from, int64_t to) noexcept
200 {
201 if(!(step_duration > 0.) || !std::isfinite(step_duration))
202 return;
203
204 auto play = [&](int64_t sample) {
205 if(sample >= from && sample < to)
206 play_step(sample, rewinding);
207 };
208
209 if(synced)
210 {
211 // k steps from the start, in tick order both ways: the musical positions
212 // of [start; end[ going forward, ]end; start] rewinding.
213 const double q = step_quarters(tk);
214 const double ms = tk.musical_start_position;
215 const double me = tk.musical_end_position;
216 if(!(q > 0.) || ms == me)
217 return;
218 constexpr double eps = 1e-9;
219 int64_t k = rewinding ? int64_t(std::floor((ms - origin) / q + eps))
220 : int64_t(std::ceil((ms - origin) / q - eps));
221 // A tick never holds more steps than it has samples.
222 for(int guard = 0; guard < 4096; guard++, k += rewinding ? -1 : 1)
223 {
224 const double m = origin + k * q;
225 if(rewinding ? (m <= me + eps) : (m >= me - eps))
226 break;
227 play(tick_start + tk.physical_position(m, ratio));
228 }
229 }
230 else
231 {
232 // k * d from the start of the interval.
233 const double d = step_duration * ossia::flicks_per_second<double>;
234 const int64_t lo = std::min(tk.prev_date.impl, tk.date.impl);
235 const int64_t hi = std::max(tk.prev_date.impl, tk.date.impl);
236 // [prev; date[ going forward, ]date; prev] rewinding: the same steps
237 // are crossed on the way out and on the way back.
238 int64_t k0 = rewinding ? int64_t(std::floor(lo / d)) + 1 : int64_t(std::ceil(lo / d));
239 int64_t k1 = rewinding ? int64_t(std::floor(hi / d)) : int64_t(std::ceil(hi / d)) - 1;
240 // A tick never holds more steps than it has samples.
241 k1 = std::min(k1, k0 + 4096);
242 if(!rewinding)
243 {
244 for(int64_t k = k0; k <= k1; k++)
245 play(tk.to_physical_time_in_tick(ossia::time_value{int64_t(k * d)}, ratio));
246 }
247 else
248 {
249 for(int64_t k = k1; k >= k0; k--)
250 play(tk.to_physical_time_in_tick(ossia::time_value{int64_t(k * d)}, ratio));
251 }
252 }
253 }

255 void play_step(int64_t sample, bool rewinding) noexcept
256 {
257 const auto& seq = sequences[current_sequence];
258 const int n = seq.size();
259 if(n == 0)
260 return;
261 // Rewinding mirrors going forward: step back first, so going out and back
262 // over the same ground crosses the same steps.
263 if(rewinding)
264 current = (current + n - 1) % n;
265 if(current < 0 || current >= n)
266 current = 0;
267 out->write_value(seq[current], sample);
268 last.store(current, std::memory_order_relaxed);
269 if(!rewinding)
270 current = (current + 1) % n;
271 }
