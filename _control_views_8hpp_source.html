72inline double normalizedValue(const Process::ControlInlet& port, const ossia::value& v)
73{
74 const auto& dom = port.domain().get();
75 const auto lo = ossia::get_min(dom);
76 const auto hi = ossia::get_max(dom);
77 if(!lo.valid() || !hi.valid())
78 return std::clamp(scalarValue(v), 0., 1.);
79 const double l = scalarValue(lo), h = scalarValue(hi);
80 // A time chooser's domain is {min, 0} .. {max, 1}: its x is the range
81 if(auto a = lo.target<ossia::vec2f>())
82 if(auto b = hi.target<ossia::vec2f>())
83 {
84 const double tl = (*a)[0], th = (*b)[0];
85 return th == tl ? 0. : std::clamp((scalarValue(v) - tl) / (th - tl), 0., 1.);
86 }
87 return h == l ? 0. : std::clamp((scalarValue(v) - l) / (h - l), 0., 1.);
88}

288 static std::pair<double, double> range(const Process::ControlInlet& port)
289 {
290 const auto& dom = port.domain().get();
291 const auto lo = ossia::get_min(dom);
292 const auto hi = ossia::get_max(dom);
293 if(!lo.valid() || !hi.valid())
294 return {0., 1.};
295 // A time chooser's domain is {min, 0} .. {max, 1}
296 if(auto a = lo.target<ossia::vec2f>())
297 if(auto b = hi.target<ossia::vec2f>())
298 return {(*a)[0], (*b)[0]};
299 return {ossia::convert<double>(lo), ossia::convert<double>(hi)};

307 static ossia::value denormalize(const Process::ControlInlet& port, double n)
308 {
309 const auto [lo, hi] = range(port);
310 const double v = lo + std::clamp(n, 0., 1.) * (hi - lo);
311 // Keep the control's own type
312 if(port.value().get_type() == ossia::val_type::INT)
313 return int(std::lround(v));
314 // A time chooser: dragged on a graph, a length in seconds (a synced
315 // stage becomes free there)
316 if(port.value().get_type() == ossia::val_type::VEC2F)
317 return ossia::vec2f{float(v), 0.f};
318 return float(v);
319 }
