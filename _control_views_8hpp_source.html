71inline double normalizedValue(const Process::ControlInlet& port, const ossia::value& v)
72{
73 const auto& dom = port.domain().get();
74 const auto lo = ossia::get_min(dom);
75 const auto hi = ossia::get_max(dom);
76 if(!lo.valid() || !hi.valid())
77 return std::clamp(scalarValue(v), 0., 1.);
78 const double l = scalarValue(lo), h = scalarValue(hi);
79 // A time chooser's domain is {min, 0} .. {max, 1}: its x is the range
80 if(auto a = lo.target<ossia::vec2f>())
81 if(auto b = hi.target<ossia::vec2f>())
82 {
83 const double tl = (*a)[0], th = (*b)[0];
84 return th == tl ? 0. : std::clamp((scalarValue(v) - tl) / (th - tl), 0., 1.);
85 }
86 return h == l ? 0. : std::clamp((scalarValue(v) - l) / (h - l), 0., 1.);
87}

234 static std::pair<double, double> range(const Process::ControlInlet& port)
235 {
236 const auto& dom = port.domain().get();
237 const auto lo = ossia::get_min(dom);
238 const auto hi = ossia::get_max(dom);
239 if(!lo.valid() || !hi.valid())
240 return {0., 1.};

242 if(auto a = lo.target<ossia::vec2f>())

253 static ossia::value denormalize(const Process::ControlInlet& port, double n)
254 {
255 const auto [lo, hi] = range(port);
256 const double v = lo + std::clamp(n, 0., 1.) * (hi - lo);
257 // Keep the control's own type
258 if(port.value().get_type() == ossia::val_type::INT)
259 return int(std::lround(v));
260 // A time chooser: dragged on a graph, a length in seconds (a synced
261 // stage becomes free there)
262 if(port.value().get_type() == ossia::val_type::VEC2F)
263 return ossia::vec2f{float(v), 0.f};
264 return float(v);
265 }
