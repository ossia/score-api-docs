71inline bool loaderIsCurrent(QQmlEngine& e, const QUrl& url, const QByteArray& str)
72{
73 using Record = std::pair<QPointer<QQmlEngine>, QHash<QUrl, QByteArray>>;
74 thread_local std::vector<Record> known;
75
76 auto it = std::find_if(
77 known.begin(), known.end(), [&](const Record& r) { return r.first == &e; });
78 if(it == known.end())
79 {
80 std::erase_if(known, [](const Record& r) { return r.first.isNull(); });
81 it = known.emplace(known.end(), &e, QHash<QUrl, QByteArray>{});
82 }
83
84 const auto digest = QCryptographicHash::hash(str, QCryptographicHash::Sha1);
85 auto& digests = it->second;
86 const auto known_digest = digests.constFind(url);
87 if(known_digest == digests.cend())
88 {
89 digests.insert(url, digest);
90 return true;
91 }
92 return *known_digest == digest;
93}

174class EditedScriptRedirect final : public QQmlAbstractUrlInterceptor
175{
176public:
177 QUrl intercept(const QUrl& url, DataType) override
178 {
179 const QString& root = editStagingRoot();
180 if(root.isEmpty() || !url.isLocalFile())
181 return url;
182
183 // Staged folders are the children of one root, so what is asked for is
184 // <root>/<staged folder>/<what the script wants>, however deep that goes.
185 const QString path = url.toLocalFile();
186 if(!path.startsWith(root) || path.size() <= root.size() + 1
187 || path[root.size()] != QLatin1Char('/'))
188 return url;
189
190 const QString rest = path.mid(root.size() + 1);
191 const int slash = rest.indexOf(QLatin1Char('/'));
192 if(slash <= 0)
193 return url;
194
195 const QString wanted = rest.mid(slash + 1);
196
197 auto& s = detail::stagedScripts();
198 QReadLocker _{&s.lock};
199 const auto it = s.byDir.constFind(root + QLatin1Char('/') + rest.left(slash));
200 if(it == s.byDir.cend() || wanted == it->fileName)
201 return url;
202 return QUrl::fromLocalFile(it->originalDir + QLatin1Char('/') + wanted);
203 }
204};

280inline QString stageEditedScript(const QString& originalPath, const QByteArray& str)
281{
282 const QString root = editStagingRoot();
283 if(root.isEmpty())
284 return {};
285
286 const QFileInfo fi{originalPath};
287 if(fi.fileName().isEmpty())
288 return {};
289
290 // Not the name it had: a folder lends its .qml files to whatever sits in it
291 // as types, and a script staged under its own name would shadow the very
292 // type it is declaring itself to be. A leading dot is not an identifier, so
293 // nothing is lent.
294 const QString name = QStringLiteral(".score-edit")
295 + (originalPath.endsWith(QStringLiteral(".ui.qml"))
296 ? QStringLiteral(".ui.qml")
297 : QStringLiteral(".qml"));
298
299 const QString prefix = stagingFolderPrefix(originalPath);
300 const QString folder
301 = prefix
302 + QString::fromLatin1(
303 QCryptographicHash::hash(str, QCryptographicHash::Sha1).toHex().left(16));
304 const QString dir = root + QLatin1Char('/') + folder;
305 const QString file = dir + QLatin1Char('/') + name;
306
307 if(!QFileInfo::exists(file))
308 {
309 if(!QDir{}.mkpath(dir))
310 return {};
311 QFile f{file};
312 if(!f.open(QIODevice::WriteOnly))
313 return {};
314 const bool written = f.write(str) == str.size();
315 f.close();
316 if(!written)
317 {
318 QDir{dir}.removeRecursively();
319 return {};
320 }
321 }
322
323 {
324 auto& s = detail::stagedScripts();
325 QWriteLocker _{&s.lock};
326 s.byDir.insert(dir, {name, fi.absolutePath()});
327 }
328
329 sweepStagedFolders(root, prefix, folder);
330 return file;
331}

333inline void loadJSObjectFromString(
334 const QString& rootPath, const QByteArray& str, QQmlComponent& comp, bool is_ui)
335{
336 const QString path = is_ui ? uiPathOf(rootPath) : rootPath;
337
338 if(scriptIsItsFile(path, str))
339 {
340 loadJSObjectFromUrl(QUrl::fromLocalFile(path), str, comp);
341 return;
342 }
343
344 // An edit is in no file, and only something with a file can be compiled by
345 // url -- which is what lets qml hand back an already compiled form, and what
346 // will let it compile off this thread. So it is given one.
347 if(auto* engine = comp.engine())
348 {
349 if(const QString staged = stageEditedScript(path, str); !staged.isEmpty())
350 {
351 installEditRedirect(*engine);
352 loadJSObjectFromUrl(QUrl::fromLocalFile(staged), str, comp);
353 return;
354 }
355 }
356
357 comp.setData(str, QUrl::fromLocalFile(path));
358}

425 m_execFuncs, &ossia::qt::qml_engine_functions::compute, m_execFuncs,
426 [self, context, m_execFuncs, js_panel](const QString& code, const QString& cbname) {
427 // Exec thread
428
429 // Callback ran in UI thread
430 auto cb = [self
431 , context=QPointer{context}
432 , cur = QPointer{self->m_object}
433 , m_execFuncs
434 , cbname] (const QVariant& v) {
435 if(!self)
436 return;
437
438 // Go back to exec thread, we have to go through the normal engine exec ctx
439 ossia::qt::run_async(m_execFuncs, [self, context, cur, v, cbname] {
440 if(!context || !cur)
441 return;
442 if(self->m_object != cur)
443 return;
444
445 auto mo = self->m_object->metaObject();
446 for(int i = 0; i < mo->methodCount(); i++)
447 {
448 if(mo->method(i).name() == cbname)
449 {
450 mo->method(i).invoke(
451 self->m_object, Qt::DirectConnection, QGenericReturnArgument(),
452 QArgument<QVariant>{"v", v});
453 }
454 }
455 });
456 };
457
458 // Go to ui thread
459 ossia::qt::run_async(js_panel, [js_panel, code, cb]() {
460 js_panel->compute(code, cb); // This invokes cb
461 });
462 }, Qt::DirectConnection);
463 }
