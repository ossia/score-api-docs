87inline bool euclidHit(int i, int steps, int pulses, int rotation) noexcept
88{
89 if(steps <= 0 || pulses <= 0)

193inline bool tieStrikes(const Pattern& p, const Lane& l, int step) noexcept
194{
195 const int len = laneLength(p, l);
196 if(len <= 0)
197 return true;
198 const int prev = ((step - 1) % len + len) % len;
199 if(writtenNote(p, l, prev) == Note::Rest)
200 return true;
201 const StepParams& sp = stepParams(l, step);
202 const StepParams& pp = stepParams(l, prev);
203 const bool spread
204 = !(l.randomPerHit & Lane::PitchPerHit) && (sp.pitchRandom > 0 || pp.pitchRandom > 0);
205 return sp.ratchet > 1 || sp.pitch != pp.pitch || spread;
206}

210inline int struckStep(const Pattern& p, const Lane& l, int step) noexcept
211{
212 const int len = laneLength(p, l);
213 for(int k = 0; k < len; k++)
214 {
215 if(writtenNote(p, l, step) != Note::Legato || tieStrikes(p, l, step))
216 return step;
217 step = ((step - 1) % len + len) % len;
218 }
219 return step;
220}
