138 static double skewed_position(double x, double rise) noexcept
139 {
140 const double r = std::clamp(rise, 0.02, 0.98);
141 const double u = frac(x + 0.25); // 0 at the trough
142 return u < r ? 0.5 * u / r : 0.5 + 0.5 * (u - r) / (1. - r);
143 }
144 static double sine(double x, double rise) noexcept
145 {
146 return -std::cos(ossia::two_pi * skewed_position(x, rise));
147 }
148 static double triangle(double x, double rise) noexcept
149 {
150 const double w = skewed_position(x, rise); // 0 trough, 0.5 peak
151 return w < 0.5 ? -1. + 4. * w : 3. - 4. * w;
152 }
154 static double saw(double x, double shape) noexcept
155 {
156 const double e = std::exp2((0.5 - std::clamp(shape, 0., 1.)) * 4.); // 4 .. 1/4
157 return 2. * std::pow(frac(x + 0.5), e) - 1.;
158 }
160 static int step(double x, double duty) noexcept
161 {
162 return frac(x) < std::clamp(duty, 0., 1.) ? 1 : -1;
163 }

166 static float lattice(int64_t i, uint32_t seed) noexcept
167 {
168 uint64_t h = uint64_t(i) * 0x9E3779B97F4A7C15ull ^ (uint64_t(seed) << 17);
169 h ^= h >> 31;
170 h *= 0xBF58476D1CE4E5B9ull;
171 h ^= h >> 29;
172 return float(h >> 40) / float(1 << 23) - 1.f;
173 }

176 static float drift(double cycles, double roughness, uint32_t seed) noexcept
177 {
178 auto octave = [&](double t, uint32_t sd) {
179 const double f = std::floor(t);
180 const double u = t - f;
181 const double k = u * u * (3. - 2. * u); // smoothstep
182 const float a = lattice(int64_t(f), sd), b = lattice(int64_t(f) + 1, sd);
183 return a + (b - a) * float(k);
184 };
185 const double r = std::clamp(roughness, 0., 1.);
186 const float v = octave(cycles, seed) + float(r) * 0.5f * octave(cycles * 2.3, seed ^ 0x5bd1e995u)
187 + float(r * r) * 0.25f * octave(cycles * 5.1, seed ^ 0x27d4eb2du);
188 return std::clamp(v / float(1. + 0.5 * r + 0.25 * r * r), -1.f, 1.f);
189 }

191 using tick = halp::tick_flicks;
192 void operator()(const halp::tick_flicks& tk)
193 {
194 const auto ampl = inputs.ampl.value;
195 const auto offset = inputs.offset.value;
196 const auto jitter = inputs.jitter.value;
197 const auto type = inputs.waveform.value;
198 const double shape = inputs.shape.value;
199 const double period = std::max(1e-4, (double)inputs.period.value);
200 const bool sync = inputs.period.sync;
201 const bool locked = sync && inputs.lock_to_bars.value;
202
203 if(inputs.retrigger)
204 {
205 phase = 0.;
206 cycles = 0;
207 sync_origin = tk.start_position_in_quarters;
208 held_valid = false;
209 last_square = 0;
210 seed = uint32_t(rd());
211 }
212 else if(sync && !locked && tk.unexpected_bar_change())
213 {
214 phase = 0.;
215 cycles = 0;
216 }
217
218 // Cycles elapsed at the start and at the end of the tick. Locked to the
219 // bars, they are read off the musical position: the cycles stay on the
220 // bars through tempo changes, jumps and loops, with no drift from
221 // accumulating. Otherwise they accumulate from the start, synced in
222 // quarters. The binding hands a synced period over in seconds at the
223 // current tempo: back to quarter notes.
224 const double quarters_per_cycle = period * tk.tempo / 60.;
225 double c0{}, c1{};
226 if(locked)
227 {
228 if(quarters_per_cycle > 0.)
229 {
230 c0 = (tk.start_position_in_quarters - sync_origin) / quarters_per_cycle;
231 c1 = (tk.end_position_in_quarters - sync_origin) / quarters_per_cycle;
232 }
233 }
234 else
235 {
236 c0 = double(cycles) + phase;
237 if(!sync)
238 c1 = c0
239 + double(tk.model_read_duration())
240 / (period * ossia::flicks_per_second<double>);
241 else if(quarters_per_cycle > 0.)
242 c1 = c0
243 + (tk.end_position_in_quarters - tk.start_position_in_quarters)
244 / quarters_per_cycle;
245 else
246 c1 = c0;
247 }
248
249 double shift = inputs.phase.value;
250 if(jitter > 0)
251 shift += std::uniform_real_distribution<float>(-jitter, jitter)(this->rd);
252 shift /= 360.;
253 c0 += shift;
254 c1 += shift;
255 const double x = frac(c0);
256
257 // Every waveform, the noises included, spans [-1; 1]: Ampl. and Offset
258 // place them all the same way.
259 const auto add_val
260 = [&](auto new_val) { outputs.out.value = ampl * new_val + offset; };
261
262 // Sample and hold: a new value at each of the two steps of a cycle.
263 const auto crossed = [&] {
264 return std::floor(c0) != std::floor(c1) || step(c0, shape) != step(c1, shape);
265 };
266
267 switch(type)
268 {
269 case Sin:
270 add_val(sine(x, shape));
271 break;
272 case Triangle:
273 add_val(triangle(x, shape));
274 break;
275 case Saw:
276 add_val(saw(x, shape));
277 break;
278 case RampDown:
279 add_val(-saw(x, shape));
280 break;
281 case Square:
282 add_val(float(step(x, shape)));
283 break;
284 case SquareOnChange: {
285 const int s = step(x, shape);
286 if(s != last_square)
287 add_val(float(s));
288 last_square = s;
289 break;
290 }
291 case SampleAndHold:
292 if(!held_valid || crossed())
293 {
294 held = std::uniform_real_distribution<float>(-1.f, 1.f)(this->rd);
295 held_valid = true;
296 add_val(held);
297 }
298 break;
299 case SampleAndHoldEveryTick:
300 if(!held_valid || crossed())
301 {
302 held = std::uniform_real_distribution<float>(-1.f, 1.f)(this->rd);
303 held_valid = true;
304 }
305 add_val(held);
306 break;
307 case Noise1:
308 add_val(std::uniform_real_distribution<float>(-1.f, 1.f)(this->rd));
309 break;
310 case Noise2:
311 add_val(std::clamp(std::normal_distribution<float>(0.f, 0.4f)(this->rd), -1.f, 1.f));
312 break;
313 case Noise3:
314 add_val(std::clamp(std::cauchy_distribution<float>(0.f, 0.1f)(this->rd), -1.f, 1.f));
315 break;
316 case Drift:
317 add_val(drift(c0, shape, seed));
318 break;
319 }
320
321 // Whole cycles move to the counter, so the phase keeps its precision
322 // however long it runs, and in either direction. Locked, the count follows
323 // the position so that unlocking carries on from there; locking snaps the
324 // cycle to the bars.
325 const double end = locked ? c1 - shift : phase + (c1 - c0);
326 const double whole = std::floor(end);
327 phase = end - whole;
328 cycles = locked ? int64_t(whole) : cycles + int64_t(whole);
329 }
