62 if(frames <= 0)
63 return;
64 channel_count = std::clamp(channel_count, 0, max_channels);
65 m_channels.store(channel_count, std::memory_order_relaxed);
66
67 const uint64_t w = m_written.load(std::memory_order_relaxed);
68 const auto now = clock::now();
69 const double rate = m_rate.load(std::memory_order_relaxed);
70 const auto block = std::chrono::duration<double>(frames / std::max(rate, 1.));
71 if(w > 0 && now - m_last_write > std::max(4 * block, std::chrono::duration<double>(0.05)))
72 m_from.store(w, std::memory_order_relaxed);
73 m_last_write = now;
74
75 // Announce the block before touching the data
76 m_writing.store(w + frames, std::memory_order_relaxed);
77 std::atomic_thread_fence(std::memory_order_release);
78
79 for(int c = 0; c < channel_count; c++)
80 {

82 if(!buf)
83 continue;
84 // First block in a newly provisioned buffer, or after the channel was
85 // skipped: what is before it is not this channel's signal
86 if(m_since[c].load(std::memory_order_relaxed) == unwritten)
87 m_since[c].store(w, std::memory_order_relaxed);
88 const auto* src = channel(c);
89 for(int i = 0; i < frames; i++)

91 float(src[i]), std::memory_order_relaxed);
92 }

94 m_since[c].store(unwritten, std::memory_order_relaxed);
