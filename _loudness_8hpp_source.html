124 static constexpr int slots = 2;
125
126 Process::LatestValue<Readings> readings;
127 std::atomic_bool reset{};
128 std::array<std::atomic<ebur128_state*>, slots> spare{};
129 std::array<std::atomic<ebur128_state*>, slots> retired{};
131 std::atomic<double> rate{48000.};
133 std::atomic_bool recycling{};
134
137 void recycle()
138 {
139 for(auto& slot : retired)
140 if(auto* old = slot.exchange(nullptr))
141 ebur128_destroy(&old);
142 for(auto& slot : spare)
143 {
144 if(slot.load())
145 continue;
146 auto* fresh = makeState(rate.load());
147 ebur128_state* expected = nullptr;
148 if(!slot.compare_exchange_strong(expected, fresh))
149 ebur128_destroy(&fresh);
150 }
151 recycling = false;
152 }
153
156 int freeRetiredSlot() const noexcept
157 {
158 for(int i = 0; i < slots; i++)
159 if(!retired[i].load())
160 return i;
161 return -1;
162 }
163
169 ebur128_state* takeSpare(unsigned long samplerate) noexcept
170 {
171 for(auto& slot : spare)
172 {
173 ebur128_state* st = slot.exchange(nullptr);
174 if(!st)
175 continue;
176 if(st->samplerate == samplerate)
177 return st;
178 if(const int r = freeRetiredSlot(); r >= 0)
179 retired[r].store(st);
180 else
181 slot.store(st);
182 }
183 return nullptr;
184 }
185
186 ~LoudnessMonitor() override
187 {
188 for(auto* states : {&spare, &retired})
189 for(auto& slot : *states)
190 if(auto* st = slot.exchange(nullptr))
191 ebur128_destroy(&st);
192 }
193};

402 void operator()(int frames)
403 {
404 if(!state || frames <= 0)
405 return;
406 const int nch = std::min(inputs.audio.channels, max_channels);
407 if(nch == 0)
408 return;
409
410 if(restart || inputs.reset || monitor->reset.exchange(false, std::memory_order_relaxed))
411 {
412 restart = false;
413 reset_pending = true;
414 }
415 if(reset_pending)
416 {
417 const bool done = clear();
418 request_recycle();
419 if(!done)
420 return;
421 reset_pending = false;
422 }
423 map_channels(nch);
424
425 const bool running = inputs.running.value;
426 readings.running = running;
427 if(running)
428 {
429 for(int done = 0; done < frames;)
430 {
431 const int n = std::min(frames - done, block_frames);
432 std::fill_n(interleaved.begin(), std::size_t(n) * max_channels, 0.);
433 for(int c = 0; c < nch; c++)
434 {
435 const double* in = inputs.audio[c] + done;
436 for(int i = 0; i < n; i++)
437 interleaved[std::size_t(i) * max_channels + c] = in[i];
438 }
439 ebur128_add_frames_double(state.get(), interleaved.data(), n);
440 done += n;
441 }
442 for(int c = 0; c < nch; c++)
443 {
444 tp[c].process(inputs.audio[c], frames);
445 tp_max[c] = std::max(tp_max[c], tp[c].block_peak);
446 }
447 readings.duration += frames / rate;
448 }
449
450 since_readings += frames;
451 if(since_readings < rate / 10.)
452 return;
453 since_readings = 0;
454
455 auto* st = state.get();
456 double v{};
457 if(ebur128_loudness_momentary(st, &v) == EBUR128_SUCCESS)
458 readings.momentary = v;
459 if(ebur128_loudness_shortterm(st, &v) == EBUR128_SUCCESS)
460 readings.short_term = v;
461 if(ebur128_loudness_global(st, &v) == EBUR128_SUCCESS)
462 readings.integrated = v;
463 if(ebur128_loudness_range(st, &v) == EBUR128_SUCCESS)
464 readings.range = v;
465 double peak = 0;
466 for(int c = 0; c < nch; c++)
467 peak = std::max(peak, tp_max[c]);
468 readings.true_peak = ossia::detail::LinearGainToDecibels(std::max(peak, 1e-10));
469 if(running && std::isfinite(readings.short_term) && readings.short_term > histogram_low)
470 {
471 const int bin = int((readings.short_term - histogram_low) / histogram_step);
472 if(bin >= 0 && bin < histogram_bins)
473 readings.histogram[bin]++;
474 }
475 readings.sequence++;
476 if(monitor->watched())
477 monitor->readings.produce(readings);
478
479 outputs.momentary.value = float(lufsToDb(readings.momentary));
480 outputs.short_term.value = float(lufsToDb(readings.short_term));
481 outputs.integrated.value = float(lufsToDb(readings.integrated));
482 outputs.range.value = float(readings.range);
483 outputs.true_peak.value = float(readings.true_peak);
484 }
