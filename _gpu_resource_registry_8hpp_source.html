46 // Well-known arenas. Size tables live in GpuResourceRegistry.cpp and match the
47 // packed GPU layouts in SceneGPUState.hpp and CameraMath.hpp. Every entry
48 // implies a QRhiBuffer allocation at init time.
49 //
50 // The Raw* arenas are written by source halp nodes at their own operator()()
51 // time: view-independent, aspect-ratio-agnostic, pre-composition. The cooked
52 // arenas are populated by ScenePreprocessor's transform passes, which combine
53 // Raw inputs with the render target's aspect ratio and the scene-graph parent
54 // chain, and are what consumer shaders bind. Material and Env have no
55 // composition dependency, so source nodes write their cooked slot directly.
56 enum class Arena : uint8_t
57 {
58 // ── Shared / source-authored ──────────────────────────────────
59 // These arenas hold view- and filter-independent bytes: every
60 // preprocessor reads the same data regardless of its camera /
61 // render target / upstream scene filtering. The producer owns the
62 // slot; multiple preprocessors consume via gpu_slot_ref + isLive().
63 RawCamera, // RawCameraData — 64 B per slot, UBO
64 RawLight, // RawLightData — 64 B per slot, SSBO
65 RawTransform, // RawLocalTransform — 64 B per slot, SSBO
66 Material, // MaterialGPU — 64 B per slot, SSBO
67 Env, // EnvParamsUBO — 64 B per slot, UBO
68
69 // Cooked outputs (camera UBOs, composed world matrices, per-draw structs,
70 // LightGPU with world direction, MaterialGPU with resolved textureRefs) are
71 // preprocessor-private: they are view- and filter-dependent, so a shared arena
72 // would be wrong when two preprocessors see different filtered views.
73
74 Count_
75 };
76
77 // Fixed-stride slot: the arena buffer is a packed array of stride-byte slots,
78 // slot i at byte offset i * stride. The slot index is the arena-level identity
79 // consumer shaders use, e.g. scene_materials.entries[slot_index]. Allocation is
80 // O(1) through a free-list stack. Trades OffsetAllocator's variable-size
81 // flexibility for a shader-indexable layout and a 1:1 mapping between
82 // internal_index and byte offset, so arena reads need no translation table.
83 struct Slot
84 {
85 static constexpr uint32_t kInvalidIndex = 0xFFFFFFFFu;
86
87 Arena arena{Arena::RawCamera};
88 uint32_t slot_index{kInvalidIndex};
89 uint32_t size{0}; // requested payload size (≤ arena stride)
90 uint32_t generation{}; // stamped on allocate; bumps on free
91
92 bool valid() const noexcept { return slot_index != kInvalidIndex; }
93 };

295 // ─── Material texture arrays ──────────────────────────────────────
296 //
297 // Per-channel static texture arrays shared across every preprocessor in this
298 // RenderList. Static textures dedup by texture_source pointer, so every producer
299 // referencing an asset gets the same layer; dynamic handles (video textures,
300 // runtime GPU outputs) get per-slot bindings in `dynamicTextures`, named
301 // <channel>Dyn<slot> in consumer shaders.
302 //
303 // Sharing is safe because a texture belongs to an asset or a wired GPU handle,
304 // independent of which preprocessor is looking.
305
306 enum class TextureChannel : uint8_t
307 {
308 BaseColor = 0,
309 MetalRough = 1,
310 Normal = 2,
311 Emissive = 3,
312 Occlusion = 4, // Separate glTF occlusionTexture (when distinct from MR).
313 Count_ = 5
314 };

316 // Default layer size + max dynamic slots. Matched across channels so
317 // samplers are interchangeable and consumer shaders can declare a
318 // fixed sampler count.
319 static constexpr int kTextureLayerSize = 1024;
320 // One pool shared by every channel: a handle routed to two channels
321 // resolves to one slot.
322 static constexpr int kMaxDynamicSlots = 8;
323
324 // Static buckets in ONE pool shared by every channel, each holding textures
325 // of one (format, pixelSize, colourspace) tuple. baseColor and emissive are
326 // sRGB, metal-rough, normal and occlusion linear, so a model's channels land
327 // in a handful of shared buckets. Past the cap a texture is skipped and the
328 // shader reads tex_ref_none.
329 //
330 // The tex_ref_static encoding reserves a 7-bit bucket field, so the cap can
331 // grow to 128 without changing the packed layout or the shader decode masks,
332 // but the shader's ladder must be enlarged to match and the sampler budget
333 // re-checked.
334 static constexpr int kMaxBuckets = 8;

436 // Access or lazily create bucket 0 with an owned (format, size).
437 // Kept for init-time fallback allocation only — production code
438 // goes through findOrCreateBucket() which selects the right bucket
439 // for the texture's actual (format, size).
440 Bucket& ensurePrimary(QRhiTexture::Format fmt, QSize sz,
441 QRhiTexture::Flags flags = {})
442 {
443 if(buckets.empty())
444 buckets.emplace_back();
445 auto& b = buckets[0];
446 b.format = fmt;
447 b.pixelSize = sz;
448 b.flags = flags;
449 return b;
450 }
451
452 // Find a bucket matching (fmt, sz, flags), creating one if none matches and
453 // kMaxBuckets is not reached. Returns {bucket_index, pointer}, or
454 // {-1, nullptr} on overflow. Bucket identity is the exact tuple, no
455 // rounding.
456 std::pair<int, Bucket*>
457 findOrCreateBucket(QRhiTexture::Format fmt, QSize sz, QRhiTexture::Flags flags)
458 {
459 for(std::size_t i = 0; i < buckets.size(); ++i)
460 {
461 if(buckets[i].format == fmt && buckets[i].pixelSize == sz
462 && buckets[i].flags == flags)
463 return {(int)i, &buckets[i]};
464 }
465 if((int)buckets.size() >= kMaxBuckets)
466 return {-1, nullptr};
467 buckets.emplace_back();
468 auto& b = buckets.back();
469 b.format = fmt;
470 b.pixelSize = sz;
471 b.flags = flags;
472 return {(int)buckets.size() - 1, &b};
473 }

554 // ─── Mesh arena manager ───────────────────────────────────────────
555 //
556 // Per-mesh slab allocator over the attribute streams of the MDI concatenated
557 // geometry -- positions, normals, texcoords, tangents, colors, texcoords1,
558 // indices -- each a single growth-capped QRhiBuffer.
559 //
560 // Indirect-draw correctness invariant: one baseVertex is applied to ALL vertex
561 // bindings (VkDrawIndexedIndirectCommand::vertexOffset), so per-mesh byte
562 // offsets must satisfy pos/nrm/tan = baseVertex * 16 and uv = baseVertex * 8.
563 // Independent per-stream allocators cannot guarantee that once alloc/free
564 // traffic fragments the streams: they pick free blocks from different size bins
565 // and the offsets diverge, so the vertex shader reads attributes from the wrong
566 // slab.
567 //
568 // Hence TWO shared allocators: m_vertexAllocator in vertex units and
569 // m_indexAllocator in index units, both capped at 8M slots. Each slab carries
570 // one vertex_slot and one index_slot, and per-stream byte offsets derive as
571 // vertex_slot.offset * stride and index_slot.offset * 4, so lockstep is
572 // structural. Indirect draw takes baseVertex = vertex_slot.offset and
573 // firstIndex = index_slot.offset.
574 //
575 // Cache: a stable_id hit reuses the slab and skips the upload; a miss allocates.
576 // The sweep frees slabs unseen for `grace` frames.
577 //
578 // Backing buffers are pointer-stable for the registry's lifetime, so downstream
579 // bindings resolve once: 128 MB each for positions, normals, tangents and
580 // colors, 64 MB per texcoord stream, 32 MB indices.
581
582 enum class MeshStream : uint8_t
583 {
584 Positions = 0,
585 Normals = 1,
586 Texcoords = 2, // TEXCOORD_0 (primary UV).
587 Tangents = 3,
588 Colors = 4, // glTF COLOR_0, vec4 (vec3 sources padded with alpha=1).
589 Texcoords1 = 5, // glTF TEXCOORD_1 (lightmap / secondary UV).
590 Indices = 6,
591 Count_ = 7
592 };
593
594 // Bytes per element per stream, matching the MDI output layout the rasterizer
595 // presets consume: positions and normals are vec3 padded to vec4 for std430,
596 // tangents and colors vec4 (vec3 sources padded with alpha=1), texcoords vec2,
597 // indices uint32.
598 static constexpr uint32_t kMeshStride[(std::size_t)MeshStream::Count_]
599 = {16, 16, 8, 16, 16, 8, 4};
600
601 // Capacity reserved per stream at init: 128 MB for positions, normals, tangents
602 // and colors (16 B stride, 8M verts), 64 MB for each texcoord stream (8 B),
603 // 32 MB for indices (4 B). A scene exceeding these gets a sentinel allocation
604 // from allocate() and the caller skips the mesh.
605 static constexpr uint32_t kMeshCapBytes[(std::size_t)MeshStream::Count_]
606 = {
607 128u * 1024u * 1024u,
608 128u * 1024u * 1024u,
609 64u * 1024u * 1024u,
610 128u * 1024u * 1024u,
611 128u * 1024u * 1024u, // colors
612 64u * 1024u * 1024u, // texcoords1
613 32u * 1024u * 1024u,
614 };
