86 static PixelFormatInfo fromAVPixelFormat(AVPixelFormat fmt)
87 {
88 PixelFormatInfo info;
89 const AVPixFmtDescriptor* desc = av_pix_fmt_desc_get(fmt);
90 if(desc)
91 {
92 info.log2ChromaW = desc->log2_chroma_w;
93 info.log2ChromaH = desc->log2_chroma_h;
94 info.bitDepth = desc->comp[0].depth;
95 info.numPlanes = av_pix_fmt_count_planes(fmt);
96 info.hasAlpha = (desc->flags & AV_PIX_FMT_FLAG_ALPHA) != 0;
97 }
98 return info;
99 }
100
103 static PixelFormatInfo fromCodecParameters(
104 AVPixelFormat swFormat, AVPixelFormat codecparFormat,
105 int bitsPerRawSample)
106 {
107 PixelFormatInfo info;
108 // Prefer sw_format (from hw_frames_ctx) when available
109 AVPixelFormat fmt = (swFormat != AV_PIX_FMT_NONE) ? swFormat : codecparFormat;
110 if(fmt != AV_PIX_FMT_NONE)
111 info = fromAVPixelFormat(fmt);
112 // bits_per_raw_sample overrides depth when set (some containers report it)
113 if(bitsPerRawSample > info.bitDepth)
114 info.bitDepth = bitsPerRawSample;
115 return info;
116 }
117};
