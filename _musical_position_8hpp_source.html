74MusicalPosition musicalPosition(const Signatures& signatures, TimeVal date) noexcept
75{
76 constexpr int64_t whole = 4 * ossia::quarter_duration<int64_t>;
77 auto bar_duration = [](ossia::time_signature sig) {
78 return std::max<int64_t>(
79 1, whole * std::max<int64_t>(1, sig.upper) / std::max<int64_t>(1, sig.lower));
80 };
81
82 const int64_t t = std::max<int64_t>(0, date.impl);
83
84 int64_t bars = 0;
85 int64_t origin = 0;
86 ossia::time_signature sig{4, 4};
87
88 for(auto it = std::begin(signatures); it != std::end(signatures); ++it)
89 {
90 const int64_t change = it->first.impl;
91 if(change > t)
92 break;
93
94 if(change > origin)
95 {
96 const int64_t bar = bar_duration(sig);
97 bars += (change - origin + bar - 1) / bar;
98 origin = change;
99 }
100 sig = it->second;
101 }
102
103 sig.upper = std::max<uint16_t>(1, sig.upper);
104 sig.lower = std::max<uint16_t>(1, sig.lower);
105
106 const int64_t bar = bar_duration(sig);
107 const int64_t since = t - origin;
108 const int64_t in_bar = since % bar;
109
110 // Computed against the bar rather than from a rounded pulse duration, so
111 // that the last pulse of a bar never overflows into a pulse that does not
112 // exist.
113 const int64_t quarter_pulses = in_bar * sig.upper * 4 / bar;
114
115 MusicalPosition res;
116 res.bar = bars + since / bar;
117 res.pulse = int(std::min<int64_t>(quarter_pulses / 4, sig.upper - 1));
118 res.subdivision = int(quarter_pulses % 4);
119 res.signature = sig;
120 return res;
121}
