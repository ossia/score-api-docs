119 // Storage-image access for compute shaders, in the author's coordinate system.
120 //
121 // imageStore() addresses a raw texel index, where row 0 is first in memory on
122 // every backend, while the render target the result is copied into is
123 // bottom-up on OpenGL and top-down everywhere else -- what
124 // QRhi::isYUpInFramebuffer() reports. The two spaces are therefore mirrored
125 // against each other on OpenGL, and a compute shader that only stores comes
126 // out upside down there.
127 //
128 // The gate is the inverse of the fragment macros above: a fragment shader
129 // needs its correction on Vulkan, a compute shader needs it everywhere else.
130 // Read and store are corrected TOGETHER, so a shader that samples an input and
131 // stores it at the matching index keeps agreeing with itself.
132 //
133 // A cube face is addressed by direction, so no Y convention applies to it.
134 static constexpr auto computeImageMacros =
135 R"_(
136// Everything except Vulkan, measured rather than derived. The framebuffer
137// origin does not predict this: Direct3D and Metal put it at the top like
138// Vulkan, yet a compute storage image reaches the delivered picture mirrored on
139// them exactly as it does on OpenGL. Gating this on OpenGL alone leaves D3D11,
140// D3D12 and Metal upside down -- csf_orient_macros reports green=255 where 0
141// is expected at row 0 on all three.
142#if defined(QSHADER_SPIRV)
143#define ISF_STORE_COORD(img, coord) ivec2(coord)
144#define ISF_STORE_COORD_LAYER(img, coord) ivec3(coord)
145#define ISF_FIXUP_COMPUTE_TEXCOORD(coord) (coord)
146#define IMG_TEXEL_PREMULTIPLIED(tex, coord) texelFetch(tex, ivec2(coord), 0)
147#else
148#define ISF_STORE_COORD(img, coord) ivec2((coord).x, imageSize(img).y - 1 - (coord).y)
149#define ISF_STORE_COORD_LAYER(img, coord) ivec3((coord).x, imageSize(img).y - 1 - (coord).y, (coord).z)
150#define ISF_FIXUP_COMPUTE_TEXCOORD(coord) vec2((coord).x, 1. - (coord).y)
151#define IMG_TEXEL_PREMULTIPLIED(tex, coord) texelFetch(tex, ivec2(ivec2(coord).x, textureSize(tex, 0).y - 1 - ivec2(coord).y), 0)
152#endif
153
154#define IMG_STORE(img, coord, val) imageStore(img, ISF_STORE_COORD(img, coord), val)
155#define IMG_STORE_LAYER(img, coord, val) imageStore(img, ISF_STORE_COORD_LAYER(img, coord), val)
156#define IMG_LOAD(img, coord) imageLoad(img, ISF_STORE_COORD(img, coord))
157#define IMG_STORE_CUBE(img, coord, val) imageStore(img, ivec3(coord), val)
158#define IMG_LOAD_CUBE(img, coord) imageLoad(img, ivec3(coord))
159
160// Face size of a writable cubemap, as ivec2, on every backend.
161//
162// A cube storage image is declared `image2DArray` when the target is HLSL --
163// Direct3D has no writable cube texture type, so `imageCube` cannot be
164// translated at all (isf_emit_cube_image_decl). imageStore / imageLoad take the
165// same ivec3 (x, y, face) coordinate for both spellings, so nothing else about
166// the author's source changes; imageSize() is the single exception, returning
167// ivec2 for a cube and ivec3 for a 2D array. Take the size through this macro
168// (or write `imageSize(img).xy`, which is what it expands to) and the shader is
169// identical on Direct3D and everywhere else.
170#define IMG_SIZE_CUBE(img) (imageSize(img).xy)
171
172vec4 isf_unpremultiply(vec4 c) { return c.a > 0.0 ? vec4(c.rgb / c.a, c.a) : vec4(0.0); }
173#define IMG_NORM_PIXEL_PREMULTIPLIED(tex, coord) texture(tex, ISF_FIXUP_COMPUTE_TEXCOORD(coord))
174#define IMG_PIXEL_PREMULTIPLIED(tex, coord) texture(tex, ISF_FIXUP_COMPUTE_TEXCOORD((coord) / vec2(textureSize(tex, 0))))
175#define IMG_NORM_PIXEL(tex, coord) IMG_NORM_PIXEL_PREMULTIPLIED(tex, coord)
176#define IMG_PIXEL(tex, coord) IMG_PIXEL_PREMULTIPLIED(tex, coord)
177#define ISF_STRAIGHT_NORM_PIXEL(tex, coord) isf_unpremultiply(IMG_NORM_PIXEL_PREMULTIPLIED(tex, coord))
178#define ISF_STRAIGHT_PIXEL(tex, coord) isf_unpremultiply(IMG_PIXEL_PREMULTIPLIED(tex, coord))
179#define IMG_TEXEL(tex, coord) IMG_TEXEL_PREMULTIPLIED(tex, coord)
180#define ISF_STRAIGHT_TEXEL(tex, coord) isf_unpremultiply(IMG_TEXEL_PREMULTIPLIED(tex, coord))
181#define IMG_CUBE(tex, dir) texture(tex, dir)
182)_";
