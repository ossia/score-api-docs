15// Scene-graph filter. Takes a scene in, emits a scene out whose node
16// tree has been culled/pruned by a predicate selected via `mode`. The
17// dropped nodes and their descendants are excluded from flattening
18// downstream.
19//
20// Predicates run against each scene_node during the walk. Subtrees
21// whose nodes all match are returned by shared_ptr identity (no
22// cloning) so downstream caches stay warm on untouched branches.
23//
24// Path syntax: slash-joined scene_node::name chain from roots, glob
25// wildcards (`*` matches anything except `/`, `**` matches across
26// slashes). Example: `/*/Wheels/**` includes everything under any
27// root whose first-level child is named "Wheels".
28class SceneGraphFilter
29{

40 enum Mode
41 {
42 PassThrough,
43 VisibleOnly,
44 ByPath,
45 ByName,
46 ByComponent,
47 ByMaterialTag,
48 // SetVisibility mode: matching nodes have their `visible` flag
49 // flipped to !Invert (Invert=false → hidden, Invert=true → shown).
50 // Non-matching nodes kept untouched. Unlike the filter modes above
51 // this DOESN'T drop nodes — they stay in the tree so downstream
52 // material / transform / light data is preserved, just
53 // render-invisible.
54 SetVisibility,
55
56 // Schema-field predicates. Operate on well-known
57 // material_component / scene_node fields — no string hashing,
58 // no glob. Each mode reads one field and compares against the
59 // inline control.
60 ByAlphaMode, // material.alpha == (selected enum)
61 ByShadowCaster, // material.shadow_caster == (selected bool)
62 ByReflectionCaster, // material.reflection_caster == (selected bool)
63 ByPurpose, // scene_node.purpose == (selected enum)
64
65 // Property-dict predicates. Read scene_node::properties or
66 // material_component::properties by key and compare against a
67 // literal. Value type is inferred from the control (string/float/
68 // int). Useful for user-authored metadata — USD extra attributes,
69 // glTF material.extras JSON, custom layer tags.
70 ByNodeProperty, // scene_node.properties[key] matches value
71 ByMaterialProperty // material.properties[key] matches value
72 };
