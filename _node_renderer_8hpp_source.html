244struct Pass
245{
246 // User-declared ctors (including the implicit ones spelled out here)
247 // suppress -Wmissing-field-initializers on the many call sites that
248 // brace-init this struct with three arguments: the fallback plan is
249 // then default-constructed into an empty list, which is what
250 // non-fallback pipelines need. This costs aggregate initialization --
251 // call sites that want fallback_bindings assign it afterwards.
252 Pass() = default;
253 Pass(TextureRenderTarget rt, Pipeline pi, QRhiBuffer* ubo)
254 : renderTarget{std::move(rt)}, p{pi}, processUBO{ubo} {}
255 // Compat for addons that build a pass from a bare Pipeline; such passes
256 // fetch their render target through renderer.renderTargetForOutput(edge)
257 // at draw time.
258 Pass(Pipeline pi)
259 : p{pi} {}
260
261 TextureRenderTarget renderTarget;
262 Pipeline p;
263 QRhiBuffer* processUBO{};
264 // Bindings for "REQUIRED: false" VERTEX_INPUTS that had no matching
265 // upstream attribute when this pass's pipeline was built. Empty for
266 // pipelines where the shader is strict-matched (the common case).
267 // Consumed by the draw path: each slot's buffer is bound at its
268 // `binding_index` in the vertex-input array before the draw call.
269 // The buffers themselves are owned by VertexFallbackPool — the plan
270 // holds non-owning pointers.
271 FallbackBindingPlan fallback_bindings;
272
273 void release()
274 {
275 p.release();
276 if(processUBO)
277 {
278 processUBO->deleteLater();
279 processUBO = nullptr;
280 }
281 fallback_bindings.clear();
282 // renderTarget NOT released here — owned by RenderList
283 }
284};
