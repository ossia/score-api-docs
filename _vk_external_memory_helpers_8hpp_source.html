49struct ExternalHandle
50{
51#if defined(_WIN32)
52 void* handle{nullptr}; // HANDLE
53#else
54 int fd{-1};
55#endif
56 VkExternalMemoryHandleTypeFlagBits type{};
57
58 bool isValid() const noexcept
59 {
60#if defined(_WIN32)
61 return handle != nullptr;
62#else
63 return fd >= 0;
64#endif
65 }
66
67 // OS handle encoded as a void* for the CUDA interop layer / DVP, which expect
68 // the Win32 HANDLE directly or the fd cast to pointer width (see
69 // CudaInterop.cpp's CUDA_EXTERNAL_MEMORY_HANDLE_DESC fill-in).
70 void* osHandle() const noexcept
71 {
72#if defined(_WIN32)
73 return handle;
74#else
75 return reinterpret_cast<void*>(static_cast<intptr_t>(fd));
76#endif
77 }
78};

239// =============================================================================
240// Layout transition — one-time image layout change via a transient command
241// buffer (create pool + buffer, record a VkImageMemoryBarrier, submit, wait
242// idle). Used by the RDMA capture/output paths to move a freshly-created
243// exportable VkImage UNDEFINED -> GENERAL once, so the QRhi-adopted texture
244// (told GENERAL via createFrom) and CUDA (which reads/writes the memory) agree.
245//
246// Broad, conservative access/stage masks cover both the capture (shader-read)
247// and output (transfer-write) uses; the following vkQueueWaitIdle makes the
248// choice of masks immaterial to correctness for a one-time init transition.
249// @p queue / @p queueFamily are the graphics queue + family the barrier submits
250// on (from QRhiVulkanNativeHandles). Returns false on any Vulkan failure.
251// =============================================================================
252SCORE_PLUGIN_GFX_EXPORT bool transitionImageLayout(
253 const VulkanCtx& v, VkQueue queue, std::uint32_t queueFamily, VkImage image,
254 VkImageLayout oldLayout, VkImageLayout newLayout);
