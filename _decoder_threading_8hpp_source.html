124inline int frameThreads(const AVCodecParameters* par) noexcept
125{
126 int n = std::min(cpuCount(), max_frame_threads);
127 if(const auto bytes = frameBytes(par); bytes > 0)
128 n = std::min<int64_t>(n, std::max<int64_t>(2, frame_memory_budget / bytes));
129 return n;
130}

132inline int sliceThreads(const AVCodecParameters* par) noexcept
133{
134 int n = std::min(cpuCount(), max_slice_threads);
135 // No codec slices finer than a 16-line macroblock row.
136 if(par && par->height > 0)
137 n = std::min(n, std::max(1, (par->height + 15) / 16));
138 return n;

174inline DecoderThreading chooseDecoderThreading(
175 const AVCodec& codec, const AVCodecParameters* par, DecodeUseCase use,
176 int userThreads = 0, bool hardware = false) noexcept
177{
178 using namespace threading;
179#if defined(__EMSCRIPTEN__)
180 // A multithreaded context crashes on teardown on the emscripten pthread
181 // runtime, even when no frame was decoded.
182 return {1, 0, false, "single-threaded on wasm"};
183#endif
184 if(hardware || (codec.capabilities & AV_CODEC_CAP_HARDWARE))
185 return {1, 0, false, "hardware decoder"};
186
187 if(const int n = envThreadCount(); n > 0)
188 userThreads = n;
189 bool forced{};
190 const int forcedType = envThreadType(forced);
191
192 const int caps = codec.capabilities;
193 const bool canFrame = caps & AV_CODEC_CAP_FRAME_THREADS;
194 const bool canSlice = caps & AV_CODEC_CAP_SLICE_THREADS;
195 const bool delayOk = use == DecodeUseCase::Playback;
196
197 if(caps & AV_CODEC_CAP_OTHER_THREADS)
198 {
199 // Some wrappers (older libdav1d) map FF_THREAD_FRAME onto their frame
200 // pipelining, so the mask is what keeps it on or off.
201 const int count = userThreads > 0 ? userThreads : 0;
202 if(delayOk)
203 return {
204 count, FF_THREAD_FRAME | FF_THREAD_SLICE, false, "codec's own thread pool"};
205 if(ownPoolHasNoFrameDelay(codec))
206 return {count, FF_THREAD_SLICE, false, "codec's own thread pool, no frame delay"};
207 if(ownPoolHasDelayOption(codec))
208 return {
209 count, FF_THREAD_SLICE, true, "codec's own thread pool, frame delay disabled"};
210 return {1, 0, false, "codec's own thread pool may delay frames"};
211 }
212
213 if(forced)
214 {
215 if(forcedType == FF_THREAD_FRAME && canFrame)
216 return {
217 userThreads > 0 ? userThreads : frameThreads(par), FF_THREAD_FRAME,
218 false, "forced by SCORE_VIDEO_THREADING"};
219 if(forcedType == FF_THREAD_SLICE && canSlice)
220 return {
221 userThreads > 0 ? userThreads : sliceThreads(par), FF_THREAD_SLICE,
222 false, "forced by SCORE_VIDEO_THREADING"};
223 if(forcedType == 0)
224 return {1, 0, false, "forced by SCORE_VIDEO_THREADING"};
225 }
226
227 if(delayOk && canFrame && !(canSlice && sliceRich(codec)))
228 return {
229 userThreads > 0 ? userThreads : frameThreads(par), FF_THREAD_FRAME, false,
230 canSlice ? "frame threads: the stream may hold a single slice"
231 : "frame threads: the only model the codec has"};
232
233 if(canSlice)
234 return {
235 userThreads > 0 ? userThreads : sliceThreads(par), FF_THREAD_SLICE, false,
236 !delayOk ? "slice threads: no frame delay allowed"
237 : canFrame ? "slice threads: the format mandates many slices"
238 : "slice threads: the only model the codec has"};
239
240 if(canFrame)
241 return {1, 0, false, "frame threading would delay output"};
242 return {1, 0, false, "codec has no threading"};
243}

257inline std::string describeDecoderThreading(
258 const AVCodecContext& ctx, const DecoderThreading& t)
259{
260 std::string s = ctx.codec && ctx.codec->name ? ctx.codec->name : "?";
261 switch(ctx.active_thread_type)
262 {
263 case FF_THREAD_FRAME:
264 s += ": frame threads x";
265 break;
266 case FF_THREAD_SLICE:
267 s += ": slice threads x";
268 break;
269 default:
270 if(ctx.codec && (ctx.codec->capabilities & AV_CODEC_CAP_OTHER_THREADS))
271 s += ctx.thread_count > 0 ? ": own thread pool x" : ": own thread pool, auto";
272 else
273 s += ": single thread";
274 break;
275 }
276 if(ctx.active_thread_type
277 || (ctx.thread_count > 0 && ctx.codec
278 && (ctx.codec->capabilities & AV_CODEC_CAP_OTHER_THREADS)))
279 s += std::to_string(ctx.thread_count);
280 s += " (";
281 s += t.reason;
282 s += ")";
283 return s;
284}
