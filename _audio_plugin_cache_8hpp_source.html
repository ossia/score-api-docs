3// Versioned persistence for the audio plug-in scan caches (VST2 / VST3 /
4// CLAP / LV2), replacing the raw QVariant metatype blobs in QSettings.
5//
6// The old scheme stored `QVariant::fromValue(std::vector<Info>)`: any change
7// to the Info datastream layout made old blobs decode as garbage that
8// `QVariant::canConvert` happily accepted (the "vst_invalid_format" global
9// hack in the VST2 plug-in was a workaround for exactly this). The new blob
10// is explicit: magic, format version, element count, elements — and
11// deserialization fails cleanly instead of producing garbage when anything
12// does not line up.
13//
14// deduplicated() / dropShadowedInvalidEntries() heal caches that already
15// accumulated duplicates (each plug-in could end up multiplied by hundreds:
16// scan replies from *other* score instances used to be appended to whichever
17// instance owned the fixed notification port, and nothing ever pruned them).

51template <typename T>
52QByteArray serializePluginCache(quint32 version, const std::vector<T>& vec)
53{
54 QByteArray res;
55 {
56 QDataStream str{&res, QIODevice::WriteOnly};
57 str << pluginCacheMagic << version << (quint32)vec.size();
58 for(const auto& elt : vec)
59 {
60 QByteArray blob;
61 {
62 QDataStream es{&blob, QIODevice::WriteOnly};
63 es << elt;
64 }
65 str << blob;
66 }
67 }
68 return res;
69}

73template <typename T>
74std::optional<std::vector<T>>
75deserializePluginCache(quint32 version, const QByteArray& data)
76{
77 if(data.isEmpty())
78 return std::nullopt;
79
80 QDataStream str{data};
81 quint32 magic{}, ver{}, count{};
82 str >> magic >> ver >> count;
83 if(str.status() != QDataStream::Ok || magic != pluginCacheMagic || ver != version)
84 return std::nullopt;
85
86 // Each element blob costs at least its 4-byte length prefix; a larger
87 // count is a corrupt header, not a plausible plug-in collection.
88 if(count > (quint32)data.size() / 4)
89 return std::nullopt;
90
91 std::vector<T> res;
92 res.reserve(count);
93 for(quint32 i = 0; i < count; i++)
94 {
95 QByteArray blob;
96 str >> blob;
97 if(str.status() != QDataStream::Ok)
98 return std::nullopt;
99
100 QDataStream es{blob};
101 T elt;
102 es >> elt;
103 // An element that does not consume its blob exactly decoded with a
104 // different layout than it was written with
105 if(es.status() != QDataStream::Ok || !es.atEnd())
106 return std::nullopt;
107 res.push_back(std::move(elt));
108 }
109
110 if(!str.atEnd())
111 return std::nullopt;
112
113 return res;
114}

119template <typename T>
120std::vector<T> loadPluginCache(
121 quint32 version, const QString& key, const QString& legacyKey)
122{
123 QSettings set;
124
125 std::optional<std::vector<T>> versioned;
126 if(const auto val = set.value(key); val.canConvert<QByteArray>())
127 versioned = deserializePluginCache<T>(version, val.toByteArray());
128
129 std::vector<T> legacy;
130 if(!legacyKey.isEmpty())
131 {
132 if(!versioned)
133 {
134 if(const auto val = set.value(legacyKey); val.canConvert<std::vector<T>>())
135 legacy = val.value<std::vector<T>>();
136 }
137 // Removed even when the versioned cache won: an older score run may have
138 // rewritten it since the migration, and it must not resurrect stale
139 // entries after a future format-version bump.
140 set.remove(legacyKey);
141 }
142
143 return versioned ? *std::move(versioned) : legacy;
144}

154template <typename T, typename KeyFn>
155void deduplicate(std::vector<T>& vec, KeyFn&& key_of)
156{
157 QSet<QString> seen;
158 ossia::remove_erase_if(vec, [&](const T& elt) {
159 const QString k = key_of(elt);
160 if(seen.contains(k))
161 return true;
162 seen.insert(k);
163 return false;
164 });
165}

170template <typename T, typename PathFn, typename ValidFn>
171void dropShadowedInvalidEntries(std::vector<T>& vec, PathFn&& path_of, ValidFn&& is_valid)
172{
173 QSet<QString> valid_paths;
174 for(const auto& elt : vec)
175 if(is_valid(elt))
176 valid_paths.insert(path_of(elt));
177
178 ossia::remove_erase_if(vec, [&](const T& elt) {
179 return !is_valid(elt) && valid_paths.contains(path_of(elt));
180 });
181}

185template <typename T, typename KeyFn, typename PathFn, typename ValidFn>
186void sanitizePluginCache(
187 std::vector<T>& vec, KeyFn&& key_of, PathFn&& path_of, ValidFn&& is_valid)
188{
189 ossia::remove_erase_if(
190 vec, [&](const T& elt) { return path_of(elt).isEmpty(); });
191 dropShadowedInvalidEntries(vec, path_of, is_valid);
192 deduplicate(vec, key_of);
193}
