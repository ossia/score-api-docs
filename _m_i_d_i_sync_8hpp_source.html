74// https://blat-blatnik.github.io/computerBear/making-accurate-sleep-function/
75struct sleep_accurate
76{
77 double estimate = 5e-3;
78 double mean = 5e-3;
79 double m2 = 0;
80 int64_t count = 1;
81
82 void operator()(double seconds)
83 {
84 using namespace std;
85 using namespace std::chrono;
86
87 while(seconds > estimate)
88 {
89 auto start = high_resolution_clock::now();
90 this_thread::sleep_for(milliseconds(1));
91 auto end = high_resolution_clock::now();
92
93 double observed = (end - start).count() / 1e9;
94 seconds -= observed;
95
96 ++count;
97 double delta = observed - mean;
98 mean += delta / count;
99 m2 += delta * (observed - mean);
100 double stddev = std::sqrt(m2 / (count - 1));
101 estimate = mean + stddev;
102
103 // FIXME that's missing a rolling behaviour to be more
104 // precise for semi-large timescales, e.g.
105 // unplugging a laptop and powersave changing frequency
106 }
107
108 // spin lock
109 auto start = high_resolution_clock::now();
110 while((high_resolution_clock::now() - start).count() / 1e9 < seconds)
111 ;
112 }
113};

407 void clock_thread_main()
408 {
409 using namespace std::chrono;
410 ossia::set_thread_name("ossia midi clock");
411 ossia::set_thread_pinned(ossia::thread_type::Midi, 0);
412
413 sleep_accurate precise_sleep;
414 midi_clock::free_clock free_clock;
415 bool was_free = false;
416 device_message pending;
417 bool has_pending = false;
418 ossia::net::midi::midi_stream* device{};
419 auto send = [this, &device](const midi_clock::message& m) {
420 send_free_clock_message(device, m);
421 };
422
423 while(clock_thread_running.load(std::memory_order_acquire))
424 {
425 const auto p = load_free_params();
426 const int64_t now = steady_now_ns();
427 int64_t deadline = midi_clock::free_clock::never;
428 if(free_mode.load(std::memory_order_acquire))
429 {
430 was_free = true;
431 if(auto c = free_command.exchange(midi_clock::transport_command::None);
432 c != midi_clock::transport_command::None)
433 free_clock.command(c, now, p, send);
434 if(auto t = pending_tap.exchange(0); t != 0)
435 free_clock.tap(t, p);
436
437 free_clock.process(now, p, send);
438 free_tempo.store(free_clock.tempo(p), std::memory_order_relaxed);
439 deadline = free_clock.next_deadline(p);
440 }
441 else
442 {
443 if(was_free)
444 {
445 was_free = false;
446 free_clock.command(midi_clock::transport_command::Stop, now, p, send);
447 }
448 free_command.exchange(midi_clock::transport_command::None);
449 pending_tap.exchange(0);
450 }
451
452 for(;;)
453 {
454 if(!has_pending)
455 has_pending = to_device.try_dequeue(pending);
456 if(!has_pending)
457 break;
458 if(pending.rebind)
459 {
460 device = pending.device;
461 has_pending = false;
462 continue;
463 }
464 if(pending.msg.at > steady_now_ns())
465 {
466 deadline = std::min(deadline, pending.msg.at);
467 break;
468 }
469 send_midi(device, std::span<const uint8_t>(pending.msg.bytes.data(), pending.msg.size));
470 has_pending = false;
471 }
472
473 // Sleep 1 ms at most, to see new commands and messages.
474 const int64_t wait = deadline - steady_now_ns();
475 if(wait > 3'000'000)
476 std::this_thread::sleep_for(milliseconds(1));
477 else if(wait > 0)
478 precise_sleep(wait / 1e9);
479 }
480 }

499 [[nodiscard]]
500 mtc_time make_mtc_time(double total_seconds) const noexcept
501 {
502 mtc_time t;
503 t.frame_rate
504 = static_cast<std::underlying_type_t<MidiTimeCodeFrameRate>>(inputs.rate.value)
505 & 0b11;
506
507 total_seconds += this->inputs.offset;
508 if(total_seconds < 0.)
509 total_seconds = 0.;
510
511 auto h = std::div((long long)total_seconds, (long long)3600).quot;
512 total_seconds -= h * 3600;
513 auto m = std::div((long long)total_seconds, (long long)60).quot;
514 total_seconds -= m * 60;
515 double s;
516 auto frames = std::modf(total_seconds, &s);
517
518 t.h = h % 24;
519 t.m = m % 60;
520 t.s = uint8_t(s);
521 t.f = std::min(uint8_t(from_mtc_framerate(t.frame_rate) * frames), uint8_t(29));
522 return t;
523 }

527
528 void operator()(halp::tick_flicks tk)
529 {
530 if(setup.rate <= 0)
531 return;
532
533 bind_output();
534 publish_clock_controls();
535 begin_tick(tk.frames);
536 const bool is_free = inputs.source.value == MidiClockSource::Free;
537
538 if(start_requested.exchange(false, std::memory_order_acq_rel))
539 main_state.running = false;
540
541 // The executor ticks once more, with no frame, after stop().
542 const bool stopping = stop_requested.exchange(false, std::memory_order_acq_rel);
543 const bool pausing = pause_requested.exchange(false, std::memory_order_acq_rel);
544 if(stopping || pausing)
545 {
546 // No tick may come to send them later: on the port, what was delayed
547 // goes out now, then Stop.
548 flush_delayed();
549 if(main_state.running && !is_free
550 && inputs.clock_startstop == MidiStartStopMode::Enabled)
551 {
552 if(main_state.to_device)
553 emit_transport({0xFC});
554 else
555 emit(0, {0xFC});
556 }
557 main_state.running = false;
558 }
559 else
560 {
561 release_delayed(tk.frames);
562 }
563
564 if(tk.frames > 0)
565 process_timeline(tk, is_free);
566
567 if(tk.frames > 0)
568 main_state.out_frame += tk.frames;
569
570 outputs.tempo.value = free_mode.load(std::memory_order_relaxed)
571 ? free_tempo.load(std::memory_order_relaxed)
572 : float(tk.tempo);
573
574 // Free clock messages arrive when they are due: they go out at the start
575 // of the tick, one tick of latency at most. Older ones were sent while
576 // the process was not ticked, and are dropped.
577 const int64_t stale_before
578 = steady_now_ns() - 2 * main_state.latency_ns - 2'000'000;
579 midi_clock::message m;
580 while(to_output_port.try_dequeue(m))
581 {
582 if(m.at < stale_before)
583 continue;
584 halp::midi_msg msg;
585 msg.bytes.assign(m.bytes.begin(), m.bytes.begin() + m.size);
586 msg.timestamp = 0;
587 outputs.midi.push_back(std::move(msg));
588 }
589 sort_output();
590
591 // No tick may come before the device is removed: the clock thread lets go
592 // of it once it has sent what this tick dated for it.
593 if(stopping || pausing)
594 set_device(nullptr);
595 }

623 void publish_clock_controls() noexcept
624 {
625 const bool is_free = inputs.source.value == MidiClockSource::Free;
626 free_send_clock.store(inputs.clock == MidiClockMode::Enabled, std::memory_order_relaxed);
627 free_send_transport.store(
628 inputs.clock_startstop == MidiStartStopMode::Enabled, std::memory_order_relaxed);
629 free_bpm.store(inputs.bpm.value, std::memory_order_relaxed);
630 clock_rate_factor.store(
631 midi_clock::clock_rate_factor(int(inputs.clock_rate.value)),
632 std::memory_order_relaxed);
633 nudge.store(
634 int(inputs.nudge_up.value) - int(inputs.nudge_down.value),
635 std::memory_order_relaxed);
636 clock_offset_ms.store(inputs.clock_offset.value, std::memory_order_relaxed);
637 free_mode.store(is_free, std::memory_order_release);
638
639 if(!is_free)
640 return;
641
642 using enum midi_clock::transport_command;
643 if(inputs.stop_button)
644 free_command.store(Stop, std::memory_order_release);
645 else if(inputs.start_button)
646 free_command.store(Start, std::memory_order_release);
647 else if(inputs.continue_button)
648 free_command.store(Continue, std::memory_order_release);
649
650 if(inputs.tap)
651 {
652 const int64_t t = steady_now_ns();
653 pending_tap.store(t != 0 ? t : 1, std::memory_order_release);
654 }
655 }

664 void begin_tick(int frames) noexcept
665 {
666 const int64_t now = steady_now_ns();
667 const double ns_per_frame = 1e9 / setup.rate;
668 const int64_t buffer_ns = int64_t(std::max(setup.frames, frames) * ns_per_frame);
669 const int64_t drift = now - main_state.next_tick_ns;
670 if(main_state.next_tick_ns == 0 || std::abs(drift) > buffer_ns + 2'000'000)
671 main_state.tick_start_ns = now;
672 else
673 main_state.tick_start_ns = main_state.next_tick_ns + drift / 16;
674 main_state.ns_per_frame = ns_per_frame;
675 main_state.latency_ns = buffer_ns;
676 main_state.next_tick_ns
677 = main_state.tick_start_ns + int64_t(std::max(frames, 0) * ns_per_frame);
678 main_state.to_device = midi_out != nullptr;
679 main_state.tick_frames = std::max(frames, 0);
680 }

683 void emit(int64_t frame, std::initializer_list<uint8_t> bytes) noexcept
684 {
685 if(main_state.to_device)
686 {
687 device_message d;
688 auto& m = d.msg;
689 m.at = main_state.tick_start_ns + main_state.latency_ns
690 + int64_t(frame * main_state.ns_per_frame);
691 m.size = int(std::min(bytes.size(), m.bytes.size()));
692 std::copy_n(bytes.begin(), m.size, m.bytes.begin());
693 to_device.try_enqueue(d);
694 }
695 else
696 {
697 halp::midi_msg msg;
698 msg.bytes.assign(bytes.begin(), bytes.end());
699 msg.timestamp = frame;
700 outputs.midi.push_back(std::move(msg));
701 }
702 }
703
710 void emit_transport(std::initializer_list<uint8_t> bytes) noexcept
711 {
712 const double offset = inputs.clock_offset.value / 1000.;
713 const int64_t delay = offset > 0. ? int64_t(std::llround(offset * setup.rate)) : 0;
714 if(main_state.to_device || delay < main_state.tick_frames)
715 {
716 emit(delay, bytes);
717 return;
718 }
719
720 auto& d = main_state.delayed;
721 if(main_state.delayed_count == int(d.size()))
722 {
723 // Out of room: late rather than lost.
724 emit(std::max<int64_t>(main_state.tick_frames - 1, 0), bytes);
725 return;
726 }
727 auto& e = d[main_state.delayed_count++];
728 e.at = main_state.out_frame + delay;
729 e.size = int(std::min(bytes.size(), e.bytes.size()));
730 std::copy_n(bytes.begin(), e.size, e.bytes.begin());
731 }

742 void release_delayed(int frames) noexcept
743 {
744 const int64_t end = main_state.out_frame + std::max(frames, 0);
745 int kept = 0;
746 for(int i = 0; i < main_state.delayed_count; i++)
747 {
748 auto& e = main_state.delayed[i];
749 if(e.at < end)
750 emit_message(std::max<int64_t>(e.at - main_state.out_frame, 0), e);

753 }

773 void process_timeline(const halp::tick_flicks& tk, bool is_free)
774 {
775 const double current_time = tk.start_in_flicks / 705'600'000.0;
776 const double frame_duration
777 = std::abs(tk.end_in_flicks - tk.start_in_flicks) / 705'600'000.0;
778 const auto time = make_mtc_time(current_time);
779 const double factor
780 = midi_clock::nudge_factor(
781 int(inputs.nudge_up.value) - int(inputs.nudge_down.value))
782 * midi_clock::clock_rate_factor(int(inputs.clock_rate.value));
783
784 // A tick that does not start where the previous one ended is a jump of
785 // the timeline: a seek, or a loop starting over.
786 const bool jumped
787 = main_state.running && tk.start_in_flicks != main_state.next_start_in_flicks;
788 if(!main_state.running || jumped)
789 locate(tk, time, current_time, factor, jumped, is_free);
790 main_state.running = true;
791 main_state.next_start_in_flicks = tk.end_in_flicks;
792
793 if(!is_free && inputs.clock == MidiClockMode::Enabled)
794 {
795 main_state.clock.process(
796 current_time, frame_duration, tk.frames, tk.tempo, factor,
797 inputs.clock_offset.value / 1000.,
798 [this](int64_t sample_offset) { emit(sample_offset, {0xF8}); });
799 }
800
801 if(inputs.mtc == MidiTimeCodeMode::Enabled)
802 send_quarter_frames(tk, time, current_time, frame_duration);
803 }
804
811 void locate(
812 const halp::tick_flicks& tk, const mtc_time& time, double current_time,
813 double factor, bool jumped, bool is_free)
814 {
815 const bool transport
816 = !is_free && inputs.clock_startstop == MidiStartStopMode::Enabled;
817 const bool clock = !is_free && inputs.clock == MidiClockMode::Enabled;
818 const double quarters = tk.start_position_in_quarters;
819 const bool at_start = std::abs(quarters) < 1e-9;
820
821 if(jumped && transport)
822 emit_transport({0xFC});
823
824 // The clock resumes on the receiver's 16th note that the Song Position
825 // names. Nudge is a passing change of tempo, not of the receiver's beat.
826 const int64_t beats = main_state.clock.locate(
827 current_time, quarters, tk.tempo,
828 midi_clock::clock_rate_factor(int(inputs.clock_rate.value)));
829
830 if(clock || transport)
831 {
832 // A MIDI beat is a 16th note, 14 bits: past 1024 bars there is no
833 // position to send.
834 if(!(transport && at_start) && beats >= 0 && beats < 16384)
835 emit_transport({0xF2, uint8_t(beats & 0x7F), uint8_t((beats >> 7) & 0x7F)});
836 if(transport)
837 emit_transport({uint8_t(at_start ? 0xFA : 0xFB)});
838 }
839
840 if(inputs.mtc == MidiTimeCodeMode::Enabled)
841 {
842 emit(
843 0, {0xF0, 0x7F, 0x7F, 0x01, 0x01, uint8_t(time.h | (time.frame_rate << 5)),
844 time.m, time.s, time.f, 0xF7});
845 }
846 const double fps = from_mtc_framerate(time.frame_rate);
847 main_state.mtc_quarter_frame_index = 0;
848 main_state.last_mtc_quarter_frame_time
849 = next_even_frame(current_time, fps) - 1.0 / (fps * 4.0);
850 }
851
857 double next_even_frame(double t, double fps) const noexcept
858 {
859 const double offset = this->inputs.offset;
860 const double u = t + offset;
861 const double second = std::floor(u);
862 // As make_mtc_time numbers them: from 0 in each second
863 const int per_second = int(std::ceil(fps - 1e-9));
864 int k = int(std::ceil((u - second) * fps - 1e-6));
865 if(k % 2 != 0)
866 k++;
867 const double frame = k < per_second ? second + k / fps : second + 1.;
868 return frame - offset;
869 }

876 void send_quarter_frames(
877 const halp::tick_flicks& tk, const mtc_time& time, double current_time,
878 double frame_duration)
879 {
880 const double fps = from_mtc_framerate(time.frame_rate);
881 const double seconds_per_quarter_frame = 1.0 / (fps * 4.0);
882
883 double next_qf_time
884 = main_state.last_mtc_quarter_frame_time + seconds_per_quarter_frame;
885
886 while(next_qf_time < current_time + frame_duration)
887 {
888 // A piece due at the very start of the tick is not lost to rounding
889 if(next_qf_time >= current_time - 1e-9)
890 {
891 int64_t sample_offset
892 = ((next_qf_time - current_time) / frame_duration) * tk.frames;
893 sample_offset = std::clamp<int64_t>(sample_offset, 0, tk.frames - 1);
894
895 // Half a piece in, to stay clear of the frame boundary
896 if(main_state.mtc_quarter_frame_index == 0)
897 main_state.mtc_cycle
898 = make_mtc_time(next_qf_time + seconds_per_quarter_frame / 2.);
899 const mtc_time& cycle = main_state.mtc_cycle;
900
901 uint8_t qf_data = 0;
902 switch(main_state.mtc_quarter_frame_index)
903 {
904 case 0:
905 qf_data = 0x00 | (cycle.f & 0x0F);
906 break;
907 case 1:
908 qf_data = 0x10 | ((cycle.f >> 4) & 0x01);
909 break;
910 case 2:
911 qf_data = 0x20 | (cycle.s & 0x0F);
912 break;
913 case 3:
914 qf_data = 0x30 | ((cycle.s >> 4) & 0x03);
915 break;
916 case 4:
917 qf_data = 0x40 | (cycle.m & 0x0F);
918 break;
919 case 5:
920 qf_data = 0x50 | ((cycle.m >> 4) & 0x03);
921 break;
922 case 6:
923 qf_data = 0x60 | (cycle.h & 0x0F);
924 break;
925 case 7:
926 qf_data = 0x70 | (cycle.frame_rate << 1) | ((cycle.h >> 4) & 0x01);
927 break;
928 }
929
930 emit(sample_offset, {0xF1, qf_data});
931
932 main_state.mtc_quarter_frame_index
933 = (main_state.mtc_quarter_frame_index + 1) % 8;
934 }
935
936 main_state.last_mtc_quarter_frame_time = next_qf_time;
937 next_qf_time += seconds_per_quarter_frame;
938 }
939 }
