133void read_loop(Field& f, transport& t, loop_input& l, bool empty)
134{
135 const bool arrived = std::exchange(l.arrived, false) || f.value != l.left;
136 if(arrived)

138 const LoopMode v = f.value;
139 const bool repeated = l.streamed && v == l.last_in;

141 t.ask(v, empty);
142 l.last_in = v;
143 }
144 l.streamed = arrived;
145}

504struct solo_memory
505{
506 static constexpr int max_passes = 256;
507 std::array<bool, max_passes> muted{};
509 int count{-1};
510 int pass{-1};
511
512 void forget() noexcept { count = -1; }
513
515 template <typename Engine>
516 bool still_soloed(const Engine& e) const noexcept
517 {
518 if(count < 0 || e.looper_passes() != count)
519 return false;
520 for(int i = 0; i < count; i++)
521 if(e.looper_muted(i) != (i != pass))
522 return false;
523 return true;
524 }
525
527 template <typename Engine>
528 void toggle(Engine& e, int selected)
529 {
530 const bool soloed = still_soloed(e);
531 if(soloed && selected == pass)
532 {
533 for(int i = 0; i < count; i++)
534 e.looper_set_muted(i, muted[i]);
535 forget();
536 return;
537 }
538 const int n = std::min(e.looper_passes(), max_passes);
539 if(selected < 0 || selected >= n)
540 return;
541 // From one solo to another, the mutes to give back are the ones from
542 // before the first.
543 if(!soloed)
544 for(int i = 0; i < n; i++)
545 muted[i] = e.looper_muted(i);
546 for(int i = 0; i < n; i++)
547 e.looper_set_muted(i, i != selected);
548 count = n;
549 pass = selected;
550 }
551};

592struct switches
593{
594 static constexpr int max_actions = 16;
595 std::array<layer_action, max_actions> actions{};
596 int action_count{};
597 bool reverse{};
598 std::optional<float> speed;
599 solo_memory soloed;
601 int seen_passes{};
604 std::array<layer_action, max_actions> withdrawn{};
605 int withdrawn_count{};
606 undo_mark withdrawn_at{};
607
608 void reset() noexcept
609 {
610 action_count = 0;
611 withdrawn_count = 0;
612 reverse = false;
613 speed.reset();
614 }
615
617 void add(layer_action a, transport& t) noexcept
618 {
619 withdrawn_count = 0;
620 if(action_count < max_actions)
621 actions[action_count++] = a;
622 t.request_point();
623 }
624
627 bool withdraw_last(undo_mark at) noexcept
628 {
629 if(action_count == 0)
630 return false;
631 if(withdrawn_count > 0 && !(withdrawn_at == at))
632 withdrawn_count = 0;
633 withdrawn_at = at;
634 const auto a = actions[std::size_t(--action_count)];
635 if(withdrawn_count < max_actions)
636 withdrawn[std::size_t(withdrawn_count++)] = a;
637 return true;
638 }
639
642 bool may_press_again(undo_mark now) const noexcept
643 {
644 return withdrawn_count > 0 && withdrawn_at == now;
645 }
646 void press_again(transport& t) noexcept
647 {
648 const auto a = withdrawn[std::size_t(--withdrawn_count)];
649 if(action_count < max_actions)
650 actions[std::size_t(action_count++)] = a;
651 t.request_point();
652 }

656 template <typename Inputs, typename Engine>
657 void read(Inputs& in, transport& t, Engine& e, bool empty)
658 {
659 const auto press = [&](TransportAction a) {
660 t.press(a, empty, in.after_record.value, in.after_overdub.value);
661 };
662 const auto set_loop = [&](LoopMode m) {
663 if(t.wanted != m || t.queued())
664 t.ask(m, empty);
665 };
666 if(in.rec_play_dub.value)
667 press(TransportAction::RecPlayDub);
668 if(in.play_stop.value)
669 press(TransportAction::PlayStop);
670 if(in.play.value)
671 set_loop(LoopMode::Play);
672 if(in.stop.value)
673 set_loop(LoopMode::Stop);
674 if(in.record.value)
675 set_loop(LoopMode::Record);
676 if(in.overdub.value)
677 set_loop(LoopMode::Overdub);
678 if(in.new_layer.value)
679 press(TransportAction::NewLayer);
680 if(in.retrigger.value)
681 press(TransportAction::Retrigger);
682
683 if(const int n = e.looper_passes();
684 n > 0 && (in.next_layer.value || in.previous_layer.value))
685 {
686 const int step = (in.next_layer.value ? 1 : 0) - (in.previous_layer.value ? 1 : 0);
687 show_control(in.layer, std::clamp(std::clamp(in.layer.value, 1, n) + step, 1, n));
688 }
689
690 if(in.mute_layer.value)
691 add({.kind = layer_action::mute, .pass = -1, .layers = -1}, t);
692 if(in.erase_layer.value)
693 add({.kind = layer_action::erase, .pass = -1, .layers = -1}, t);
694 if(in.solo_layer.value)
695 add({.kind = layer_action::solo, .pass = -1, .layers = -1}, t);
696 if(in.stop_layer.value)
697 add({.kind = layer_action::stop, .pass = -1, .layers = -1}, t);
698
699 bool asked = false;
700 if(in.reverse_toggle.value)
701 reverse = !reverse, asked = true;
702 // From the speed a change still waiting for the grid leaves.
703 const auto toggle_speed = [&](float to) {
704 const float from = speed ? *speed : in.speed.value;
705 speed = from == to ? 1.f : to;
706 asked = true;
707 };
708 if(in.half_speed.value)
709 toggle_speed(0.5f);
710 if(in.double_speed.value)
711 toggle_speed(2.f);
712 if(asked)
713 t.request_point();
714 }

717 template <typename Inputs, typename Engine>
718 void apply(Inputs& in, Engine& e)
719 {
720 for(int i = 0; i < action_count; i++)
721 {
722 const auto& a = actions[i];
723 int pass = -1;
724 if(a.id != 0)
725 {
726 // The pass is gone, e.g. merged with another: nothing is left of it
727 // to act on.
728 pass = e.looper_pass_of(a.id);
729 if(pass < 0)
730 continue;
731 }
732 else
733 {
734 if(a.layers >= 0 && a.layers != e.looper_passes())
735 continue;
736 // A Layer past the layers there are names none.
737 pass = a.pass >= 0 ? a.pass : in.layer.value - 1;
738 }
739 switch(a.kind)
740 {
741 case layer_action::mute:
742 e.looper_mute(pass, a.muted);
743 break;
744 case layer_action::erase:
745 e.looper_erase(pass);
746 break;
747 case layer_action::solo:
748 e.looper_begin_actions();
749 soloed.toggle(e, pass);
750 e.looper_end_actions();
751 break;
752 case layer_action::stop:
753 e.looper_stop(pass, a.stopped);
754 break;
755 }
756 follow(in, e.looper_passes());
757 }
758 action_count = 0;
759 if(std::exchange(reverse, false))
760 show_control(in.reverse, !in.reverse.value);
761 if(speed)
762 show_control(in.speed, *std::exchange(speed, std::nullopt));
763 }

767 template <typename Inputs>
768 void follow(Inputs& in, int passes)
769 {
770 const int was = std::exchange(seen_passes, passes);
771 if(passes > was)
772 show_control(in.layer, passes);
773 else if(passes > 0 && in.layer.value > passes)
774 show_control(in.layer, passes);
775 }
776
778 template <typename Inputs>
779 void removed(Inputs& in, int pass)
780 {
781 seen_passes = std::max(0, seen_passes - 1);
782 if(pass < in.layer.value - 1 && in.layer.value > 1)
783 show_control(in.layer, in.layer.value - 1);
784 }
785
789 template <typename Inputs>
790 void merged(Inputs& in, int at)
791 {
792 seen_passes = std::max(0, seen_passes - 1);
793 if(in.layer.value - 1 > at && in.layer.value > 1)
794 show_control(in.layer, in.layer.value - 1);
795 }
796
799 template <typename Inputs>
800 void inserted(Inputs& in, int pass, int passes)
801 {
802 seen_passes = passes;
803 if(pass <= in.layer.value - 1 && in.layer.value < passes)
804 show_control(in.layer, in.layer.value + 1);
805 }

810 template <typename Inputs, typename Log>
811 void follow_merges(Inputs& in, Log& log)
812 {
813 if(log.count > 0)
814 soloed.forget();
815 for(int i = 0; i < log.count; i++)
816 merged(in, log.at[i]);
817 log.count = 0;
