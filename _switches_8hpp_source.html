128template <typename Field>
129void read_loop(Field& f, transport& t, loop_input& l, bool empty)
130{
131 const bool arrived = std::exchange(l.arrived, false) || f.value != l.left;
132 if(arrived)
133 {
134 const LoopMode v = f.value;
135 const bool repeated = l.streamed && v == l.last_in;
136 if(!repeated && v != t.wanted)
137 t.ask(v, empty);
138 l.last_in = v;
139 }
140 l.streamed = arrived;
141}

460struct solo_memory
461{
462 static constexpr int max_passes = 256;
463 std::array<bool, max_passes> muted{};
465 int count{-1};
466 int pass{-1};
467
468 void forget() noexcept { count = -1; }
469
471 template <typename Engine>
472 bool still_soloed(const Engine& e) const noexcept
473 {
474 if(count < 0 || e.looper_passes() != count)
475 return false;
476 for(int i = 0; i < count; i++)
477 if(e.looper_muted(i) != (i != pass))
478 return false;
479 return true;
480 }
481
483 template <typename Engine>
484 void toggle(Engine& e, int selected)
485 {
486 const bool soloed = still_soloed(e);
487 if(soloed && selected == pass)
488 {
489 for(int i = 0; i < count; i++)
490 e.looper_set_muted(i, muted[i]);
491 forget();
492 return;
493 }
494 const int n = std::min(e.looper_passes(), max_passes);
495 if(selected < 0 || selected >= n)
496 return;
497 // From one solo to another, the mutes to give back are the ones from
498 // before the first.
499 if(!soloed)
500 for(int i = 0; i < n; i++)
501 muted[i] = e.looper_muted(i);
502 for(int i = 0; i < n; i++)
503 e.looper_set_muted(i, i != selected);
504 count = n;
505 pass = selected;
506 }
507};

555 template <typename Inputs, typename Engine>
556 void read(Inputs& in, transport& t, Engine& e, bool empty)
557 {
558 const auto press = [&](TransportAction a) {
559 t.press(a, empty, in.after_record.value, in.after_overdub.value);
560 };
561 const auto set_loop = [&](LoopMode m) {
562 if(t.wanted != m || t.queued())
563 t.ask(m, empty);
564 };
565 if(in.rec_play_dub.value)
566 press(TransportAction::RecPlayDub);
567 if(in.play_stop.value)
568 press(TransportAction::PlayStop);
569 if(in.play.value)
570 set_loop(LoopMode::Play);
571 if(in.stop.value)
572 set_loop(LoopMode::Stop);
573 if(in.record.value)
574 set_loop(LoopMode::Record);
575 if(in.overdub.value)
576 set_loop(LoopMode::Overdub);
577 if(in.new_layer.value)
578 press(TransportAction::NewLayer);
579 if(in.retrigger.value)
580 press(TransportAction::Retrigger);
581
582 if(const int n = e.looper_passes();
583 n > 0 && (in.next_layer.value || in.previous_layer.value))
584 {
585 const int step = (in.next_layer.value ? 1 : 0) - (in.previous_layer.value ? 1 : 0);
586 show_control(in.layer, std::clamp(std::clamp(in.layer.value, 1, n) + step, 1, n));
587 }
588
589 if(in.mute_layer.value)
590 add({layer_action::mute, -1, -1}, t);
591 if(in.erase_layer.value)
592 add({layer_action::erase, -1, -1}, t);
593 if(in.solo_layer.value)
594 add({layer_action::solo, -1, -1}, t);
595
596 bool asked = false;
597 if(in.reverse_toggle.value)
598 reverse = !reverse, asked = true;
599 // From the speed a change still waiting for the grid leaves.
600 const auto toggle_speed = [&](float to) {
601 const float from = speed ? *speed : in.speed.value;
602 speed = from == to ? 1.f : to;
603 asked = true;
604 };
605 if(in.half_speed.value)
606 toggle_speed(0.5f);
607 if(in.double_speed.value)
608 toggle_speed(2.f);
609 if(asked)
610 t.request_point();
611 }

614 template <typename Inputs, typename Engine>
615 void apply(Inputs& in, Engine& e)
616 {
617 for(int i = 0; i < action_count; i++)
618 {
619 const auto& a = actions[i];
620 const int n = e.looper_passes();
621 if(a.layers >= 0 && a.layers != n)
622 continue;
623 // A Layer past the layers there are names none.
624 const int pass = a.pass >= 0 ? a.pass : in.layer.value - 1;
625 switch(a.kind)
626 {
627 case layer_action::mute:
628 e.looper_mute(pass);
629 break;
630 case layer_action::erase:
631 e.looper_erase(pass);
632 break;
633 case layer_action::solo:
634 soloed.toggle(e, pass);
635 break;
636 }
637 follow(in, e.looper_passes());
638 }
639 action_count = 0;
640 if(std::exchange(reverse, false))
641 show_control(in.reverse, !in.reverse.value);
642 if(speed)
643 show_control(in.speed, *std::exchange(speed, std::nullopt));
644 }

648 template <typename Inputs>
649 void follow(Inputs& in, int passes)
650 {
651 const int was = std::exchange(seen_passes, passes);
652 if(passes > was)
653 show_control(in.layer, passes);
654 else if(passes > 0 && in.layer.value > passes)
655 show_control(in.layer, passes);
656 }
657
659 template <typename Inputs>
660 void removed(Inputs& in, int pass)
661 {
662 seen_passes = std::max(0, seen_passes - 1);
663 if(pass < in.layer.value - 1 && in.layer.value > 1)
664 show_control(in.layer, in.layer.value - 1);
665 }
666
670 template <typename Inputs>
671 void merged(Inputs& in, int at)
672 {
673 seen_passes = std::max(0, seen_passes - 1);
674 if(in.layer.value - 1 > at && in.layer.value > 1)
675 show_control(in.layer, in.layer.value - 1);
676 }
677
680 template <typename Inputs>
681 void inserted(Inputs& in, int pass, int passes)
682 {
683 if(pass <= in.layer.value - 1 && in.layer.value < passes)
684 show_control(in.layer, in.layer.value + 1);
685 }
686
690 template <typename Inputs, typename Log>
691 void follow_merges(Inputs& in, Log& log)
692 {
693 if(log.count > 0)
694 soloed.forget();
695 for(int i = 0; i < log.count; i++)
696 merged(in, log.at[i]);
697 log.count = 0;
698 }
