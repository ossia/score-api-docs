98 const std::string& next_path(int slot, int& number, int layer)
99 {
100 // Loopers of a document write from several workers: the choice of a
101 // number and the creation of its first file go together.
102 static std::mutex choosing;
103 std::lock_guard _{choosing};
104
105 QDir dir{folder};
106 QDir{}.mkpath(folder);
107 if(number <= 0)
108 {
109 // A number is free when no file of the folder has it, whichever of a
110 // take's files they are: the files of some passes alone have neither
111 // the mix nor the first layer.
112 const QString head = prefix + QStringLiteral("-take");
113 std::vector<int> used;
114 for(const QString& name :
115 dir.entryList({head + QStringLiteral("*")}, QDir::Files | QDir::Hidden))
116 {
117 int end = int(head.size());
118 while(end < name.size() && name[end].isDigit())
119 end++;
120 bool ok = false;
121 const int n = name.mid(head.size(), end - head.size()).toInt(&ok);
122 if(ok && end < name.size() && (name[end] == '.' || name[end] == '-'))
123 used.push_back(n);
124 }
125 std::sort(used.begin(), used.end());
126 number = 1;
127 for(int n : used)
128 if(n == number)
129 number++;
130 else if(n > number)
131 break;
132 }
133 const QString path = file(number, layer);
134 if(QFile f{path}; f.open(QIODevice::WriteOnly))
135 f.close();
136 paths[slot].assign(path.toStdString());
137 bases[slot].assign(base(number).toStdString());
138 return paths[slot];
139 }
140
142 const std::string& take_base(int slot) const noexcept { return bases[slot]; }
143 const std::string& path(int slot) const noexcept { return paths[slot]; }
144};

252 {

278 struct release
279 {
280 audio_stop_take& s;
281 ~release()

283 s.passes.pool.reset();
284 s.in_use = false;

287 std::array<const float*, 16> data{};
288 channels = std::min<int>({channels, s.passes.channels, int(data.size())});
289 bool ok = writer && s.passes.pool && channels > 0 && frames > 0;
290 if(ok)
291 {
292 std::vector<float> buffer(std::size_t(channels) * std::size_t(frames));
293 const auto channel = [&](int c) { return buffer.data() + std::size_t(c) * frames; };
294 for(int c = 0; c < channels; c++)
295 data[c] = channel(c);
296 number = 0;
297 for(layer = -1; ok && layer < s.passes.count; layer++)
298 {
299 s.passes.flatten(channels, channel, layer);
300 ok = write(data.data());
301 }
302 }
