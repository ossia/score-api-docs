77 // The camera looks at the centre from yaw and pitch
78 constexpr double deg = std::numbers::pi / 180.;
79 const double yaw = value(m_yaw) * deg, pitch = value(m_pitch) * deg;
80 const QVector3D eye(std::cos(pitch) * std::sin(yaw), std::sin(pitch), std::cos(pitch) * std::cos(yaw));
81 const QVector3D fwd = -eye;
82 // As the viewer: up turns over with the camera past the poles
83 const QVector3D up(
84 -std::sin(pitch) * std::sin(yaw), std::cos(pitch), -std::sin(pitch) * std::cos(yaw));
85 const QVector3D right = QVector3D::crossProduct(fwd, up).normalized();
86
87 struct Axis
88 {
89 QVector3D dir;
90 QColor color;
91 const char* name;
92 };
93 Axis axes[]{
94 {{1, 0, 0}, QColor(235, 70, 60), "X"},
95 {{0, 1, 0}, QColor(90, 200, 90), "Y"},
96 {{0, 0, 1}, QColor(80, 140, 240), "Z"}};
97 // The farthest first, so that the nearest stays on top
98 std::sort(std::begin(axes), std::end(axes), [&](const Axis& a, const Axis& b) {
99 return QVector3D::dotProduct(a.dir, fwd) > QVector3D::dotProduct(b.dir, fwd);
100 });
101 const QPointF c = r.center();
102 const double len = size * 0.34;
103 QFont f = Process::ModeSwitch::font();
104 f.setBold(true);
105 p->setFont(f);
106 for(const auto& a : axes)
107 {
108 const QPointF tip = c + QPointF{QVector3D::dotProduct(a.dir, right) * len, -QVector3D::dotProduct(a.dir, up) * len};
109 // Darker the further it points away, so that it fades as it turns
110 const double away = std::clamp(QVector3D::dotProduct(a.dir, fwd), 0.f, 1.f);
111 p->setPen(QPen(mix(a.color, a.color.darker(200), away), 1.5));
112 p->drawLine(c, tip);
113 p->drawText(QRectF(tip.x() - 6, tip.y() - 6, 12, 12), Qt::AlignCenter, a.name);
114 }
115 }
