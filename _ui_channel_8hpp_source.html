58 bool try_write(
59 uint32_t index, uint32_t protocol, const void* data, uint32_t size) noexcept
60 {
61 const std::size_t n = packets_for(size);
62 // The producer never sees fewer packets than the consumer left: the room
63 // it computes is at most the actual one, and every try_enqueue succeeds.
64 const std::size_t used = m_queue.size_approx();
65 if(used > m_capacity || n > m_capacity - used)
66 return false;
67
68 packet p;
69 const header h{index, protocol, size};
70 std::memcpy(p.bytes, &h, sizeof(h));
71 std::size_t at = sizeof(h);
72 auto* src = static_cast<const uint8_t*>(data);
73 std::size_t left = size;
74 for(;;)
75 {
76 const std::size_t k = std::min(left, packet_size - at);
77 if(k > 0)
78 std::memcpy(p.bytes + at, src, k);
79 src += k;
80 left -= k;
81 [[maybe_unused]] const bool ok = m_queue.try_enqueue(p);
82 assert(ok);
83 if(left == 0)
84 return true;
85 at = 0;
86 }
87 }

92 template <typename F>
93 bool try_read(F&& dest) noexcept
94 {
95 const packet* p = m_queue.peek();
96 if(!p)
97 return false;
98 header h;
99 std::memcpy(&h, p->bytes, sizeof(h));
100 const std::size_t n = packets_for(h.size);
101 // The consumer never sees more packets than the producer wrote
102 if(m_queue.size_approx() < n)
103 return false;
104
105 const std::optional<uint8_t*> where = dest(std::as_const(h));
106 if(!where)
107 return false;
108 uint8_t* out = *where;
109
110 std::size_t at = sizeof(h);
111 std::size_t left = h.size;
112 for(std::size_t i = 0; i < n; i++)
113 {
114 const std::size_t k = std::min(left, packet_size - at);
115 if(out && k > 0)
116 {
117 std::memcpy(out, p->bytes + at, k);
118 out += k;
119 }
120 left -= k;
121 m_queue.pop();
122 if(i + 1 < n)
123 p = m_queue.peek();
124 at = 0;
125 }
126 return true;
127 }
128
129private:
130 struct alignas(8) packet
131 {
132 uint8_t bytes[packet_size];
133 };
134 // ossia::spsc_queue is a ConcurrentQueue in TSAN builds, which has no peek.
135 // With a block size no capacity reaches, the queue is one block, whose
136 // size_approx() and max_capacity() hold with no slot set apart.
137 moodycamel::ReaderWriterQueue<packet, (std::size_t(1) << 30)> m_queue;
138 std::size_t m_capacity{};
