125 static void from_value(score::gfx::PointerState& state, const ossia::value& v)
126 {
127 auto* l = v.target<std::vector<ossia::value>>();
128 if(!l)
129 return;
130 const int n = std::min<int>(int(l->size()), score::gfx::PointerState::slots);
131 for(int i = 0; i < n; i++)
132 {
133 const auto& e = (*l)[i];
134 if(auto* p = e.target<ossia::vec3f>())
135 {
136 state.move(i, (*p)[0], (*p)[1]);
137 state.press(i, (*p)[2]);
138 }
139 else if(auto* q = e.target<ossia::vec2f>())
140 {
141 state.move(i, (*q)[0], (*q)[1]);
142 }
143 else if(auto* sub = e.target<std::vector<ossia::value>>(); sub && sub->size() >= 2)
144 {
145 state.move(i, ossia::convert<float>((*sub)[0]), ossia::convert<float>((*sub)[1]));
146 if(sub->size() >= 3)
147 state.press(i, ossia::convert<float>((*sub)[2]));
148 }
149 }
