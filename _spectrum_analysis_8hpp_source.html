45 void prepare(int n, double sample_rate, Window w)
46 {
47 n = std::max(n, 64);
48 rate = sample_rate;
49 if(n == size && w == kind && !window.empty())
50 return;
51 if(n != size)
52 {
53 size = n;
54 fft.reset(n);
55 power.assign(n / 2 + 1, 0.);
56 }
57 kind = w;
58 window.resize(n);
59 constexpr double pi = std::numbers::pi;
60 for(int i = 0; i < n; i++)
61 {
62 const double t = double(i) / n;
63 switch(w)
64 {
65 case Window::Hann:
66 window[i] = 0.5 - 0.5 * std::cos(2 * pi * t);
67 break;
68 case Window::BlackmanHarris:
69 window[i] = 0.35875 - 0.48829 * std::cos(2 * pi * t)
70 + 0.14128 * std::cos(4 * pi * t) - 0.01168 * std::cos(6 * pi * t);
71 break;
72 case Window::FlatTop:
73 window[i] = 0.21557895 - 0.41663158 * std::cos(2 * pi * t)
74 + 0.277263158 * std::cos(4 * pi * t)
75 - 0.083578947 * std::cos(6 * pi * t)
76 + 0.006947368 * std::cos(8 * pi * t);
77 break;
78 }
79 }
80 double s = 0, s2 = 0;
81 for(double v : window)
82 {
83 s += v;
84 s2 += v * v;
85 }
86 coherent_gain = s;
87 sum_sq = s2;
88 }

92 void analyse(const T* x) noexcept
93 {
94 auto* in = fft.input();
95 for(int i = 0; i < size; i++)
96 in[i] = x[i] * window[i];
97 const auto* out = fft.execute();
98 // |X| * 2 / sum(w) is the amplitude of a sine on that bin
99 const double k = 4. / (coherent_gain * coherent_gain);
100 const int bins = size / 2 + 1;
101 for(int b = 0; b < bins; b++)
102 power[b] = (out[b][0] * out[b][0] + out[b][1] * out[b][1]) * k;
103 }

115 double bandPower(std::span<const double> p, double f0, double f1) const noexcept
116 {
117 if(p.empty())
118 return 0.;
119 const int last = int(p.size()) - 1;
120 const int b0 = std::clamp(int(std::ceil(frequencyBin(f0))), 0, last);
121 const int b1 = std::clamp(int(std::floor(frequencyBin(f1))), 0, last);
122 if(b1 < b0)
123 {
124 // Band narrower than a bin: the share of the bin it falls in that its
125 // width covers, so that the level per Hz is the same as for wide bands
126 const int b = std::clamp(int(std::lround(frequencyBin((f0 + f1) / 2))), 0, last);
127 return p[b] * (frequencyBin(f1) - frequencyBin(f0)) / enbw();
128 }
129 double sum = 0;
130 for(int b = b0; b <= b1; b++)
131 sum += p[b];
132 return sum / enbw();
133 }
