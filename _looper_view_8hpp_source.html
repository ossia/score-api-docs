288 void receive(const view_message& m) noexcept
289 {
290 if(m.looper == m_stopped_looper && m.stops < m_stopped_stops)
291 return;
292 if(m.stopped)
293 {
294 // The looper does not run any more: nothing asked of it will come.
295 m_stopped_looper = m.looper;
296 m_stopped_stops = m.stops;
297 m_last_message = {};
298 m_waiting = 0;
299 m_send_waiting = false;
300 m_part_waiting = 0;
301 state = m;
302 has_state = true;
303 if(update)
304 update();
305 return;
306 }
307 // After a silence the looper may be another one, made when the score
308 // played again, which knows nothing of what was asked before.
309 if(!live())
310 {
311 m_waiting = 0;
312 m_send_waiting = false;
313 m_part_waiting = 0;
314 }
315 m_last_message = clock::now();
316 state = m;
317 has_state = true;
318 if(m_waiting > 0 && m.answered >= m_waiting)
319 {
320 m_waiting = 0;
321 if(std::exchange(m_send_waiting, false))
322 send_now();
323 }
324 if(m_part_waiting > 0 && m.part_answered >= m_part_waiting)
325 {
326 // Answered with the files, or with nothing when the loop changed.
327 if(m.part_answer == m_part_waiting)
328 {
329 m_part_files = m.part_files();
330 m_part_placeable = m.part_placeable;
331 m_part_stretch = m.part_stretch;
332 }
333 else
334 m_part_files.clear();
335 m_part_waiting = 0;
336 }
337 if(update)
338 update();
339 }

561 void paint(auto ctx)
562 {
563 const double w = width(), h = height();
564 ctx.set_fill_color({24, 24, 26, 255});
565 ctx.begin_path();
566 ctx.draw_rect(0., 0., w, h);
567 ctx.fill();
568
569 const auto& s = state;
570 const auto col = mode_color(has_state ? s.mode : int(LoopMode::Stop), !has_state || s.empty);
571
572 // Header: progress ring, mode, what it is waiting for, length and passes.
573 {
574 const double cx = 13., cy = header / 2., r = 8.;
575 ctx.set_stroke_color({60, 60, 64, 255});
576 ctx.set_stroke_width(3.);
577 ctx.begin_path();
578 ctx.draw_circle(cx, cy, r);
579 ctx.stroke();
580 if(has_state && !s.empty)
581 {
582 ctx.set_stroke_color({col.r, col.g, col.b, col.a});
583 ctx.begin_path();
584 ctx.move_to(cx, cy - r);
585 ctx.arc_to(cx - r, cy - r, 2. * r, 2. * r, 90., -360. * s.position);
586 ctx.stroke();
587 }
588 ctx.set_stroke_width(1.);
589
590 char text[160];
591 if(!has_state)
592 std::snprintf(text, sizeof(text), "Play to start looping");
593 else if(s.stopped)
594 std::snprintf(text, sizeof(text), "Stopped with the score");
595 else if(s.wanted != s.requested)
596 std::snprintf(
597 text, sizeof(text), "%s, then %s on the grid", mode_name(s.mode),
598 mode_name(s.wanted));
599 else if(s.new_pass_pending || s.restart_pending)
600 std::snprintf(
601 text, sizeof(text), "%s, then %s on the grid", mode_name(s.mode),
602 s.new_pass_pending ? (s.restart_pending ? "new layer from the start" : "new layer")
603 : "from the start");
604 else
605 std::snprintf(
606 text, sizeof(text), "%s%s%s%s", mode_name(s.mode),
607 s.reverse ? ", reverse" : "", speed_name()[0] ? " " : "", speed_name());
608
609 ctx.set_fill_color({col.r, col.g, col.b, col.a});
610 ctx.begin_path();
611 ctx.draw_text(28., cy + 4., text);
612 ctx.fill();
613
614 if(has_state)
615 {
616 // Against the Send button, whatever its length.
617 char info[160];
618 std::snprintf(
619 info, sizeof(info), "%.1f s%s%s", s.length_seconds, s.full ? ", full" : "",
620 s.redo > 0 ? " (redo)" : "");
621 double info_w = 6.5 * std::strlen(info);
622 if constexpr(requires { ctx.text_width(std::string_view{}); })
623 info_w = ctx.text_width(info);
624 ctx.set_fill_color({170, 170, 176, 255});
625 ctx.begin_path();
626 ctx.draw_text(w - send_w - 10. - info_w, cy + 4., info);
627 ctx.fill();
628 }
629
630 // Send to timeline. A take the timeline cannot play is a file only.
631 const bool file_only = has_take() && !state.take_placeable && !taking_in();
632 const bool enabled = (placeable() || taking_in()) && bool(send_to_timeline);
633 const rgba button = enabled ? rgba{60, 110, 160, 255} : rgba{44, 44, 48, 255};
634 ctx.set_fill_color({button.r, button.g, button.b, button.a});
635 ctx.begin_path();
636 ctx.draw_rounded_rect(w - send_w - 4., 4., send_w, header - 8., 3.);
637 ctx.fill();
638 const rgba label = enabled ? rgba{235, 240, 245, 255} : rgba{110, 110, 116, 255};
639 ctx.set_fill_color({label.r, label.g, label.b, label.a});
640 // A grip: the button is also where the whole take is dragged from.
641 ctx.begin_path();
642 for(int row = 0; row < 3; row++)
643 for(int col = 0; col < 2; col++)
644 ctx.draw_rect(w - send_w + 1. + col * 3., cy - 4. + row * 3.5, 1.5, 1.5);
645 ctx.fill();
646 ctx.begin_path();
647 ctx.draw_text(
648 w - send_w + 9., cy + 4., saving() ? "Saving" : (file_only ? "File" : "Send"));
649 ctx.fill();
650 }
651
652 const double top = header, lanes_h = h - header - 4.;
653 if(!has_state || s.empty)
654 {
655 if(has_take())
656 {
657 ctx.set_fill_color({120, 120, 126, 255});
658 ctx.begin_path();
659 ctx.draw_text(10., top + lanes_h / 2., "The last take can still be sent.");
660 ctx.fill();
661 }
662 return;
663 }
664
665 if(s.midi)
666 paint_notes(ctx, top, lanes_h);
667 else if(s.values)
668 paint_values(ctx, top, lanes_h);
669 else
670 paint_lanes(ctx, top, lanes_h);
671
672 // Playhead, with an arrow the way it goes.
673 const double x = 4. + s.position * (w - 8.);
674 ctx.set_stroke_color({col.r, col.g, col.b, col.a});
675 ctx.set_stroke_width(2.);
676 ctx.begin_path();
677 ctx.move_to(x, top);
678 ctx.line_to(x, top + lane_count() * lane_height());
679 ctx.stroke();
680 ctx.set_stroke_width(1.);
681 {
682 const double dir = s.reverse ? -1. : 1.;
683 ctx.set_fill_color({col.r, col.g, col.b, col.a});
684 ctx.begin_path();
685 ctx.move_to(x, top);
686 ctx.line_to(x + dir * 7., top + 4.);
687 ctx.line_to(x, top + 8.);
688 ctx.close_path();
689 ctx.fill();
690 }
691
692 paint_erase_boxes(ctx);
693 paint_preparing(ctx);
694 }

914 void paint_notes(auto& ctx, double top, double)
915 {
916 const auto& s = state;
917 const int lanes = std::max(1, s.lanes);
918 const double lh = lane_height();
919 const double x0 = 4., lw = width() - 8.;
920
921 std::array<int, view_message::max_lanes> lo, hi;
922 lo.fill(127);
923 hi.fill(0);
924 for(int i = 0; i < s.notes; i++)
925 {
926 const auto& n = s.note[i];
927 const int l = std::min<int>(n.lane, view_message::max_lanes - 1);
928 lo[l] = std::min<int>(lo[l], n.pitch);
929 hi[l] = std::max<int>(hi[l], n.pitch);
930 }
931
932 for(int l = 0; l < lanes; l++)
933 {
934 const double y = top + l * lh;
935 const auto lbg = lane_background(l);
936 ctx.set_fill_color({lbg.r, lbg.g, lbg.b, lbg.a});
937 ctx.begin_path();
938 ctx.draw_rect(x0, y + 1., lw, lh - 2.);
939 ctx.fill();
940 if(lo[l] > hi[l])
941 continue;
942 lo[l] = std::max(0, lo[l] - 1);
943 hi[l] = std::min(127, hi[l] + 1);
944 }
945
946 // The notes not heard first, so that the ones heard are drawn over them.
947 for(int heard = 0; heard < 2; heard++)
948 {
949 for(int i = 0; i < s.notes; i++)
950 {
951 const auto& n = s.note[i];
952 if(bool(n.heard) != bool(heard))
953 continue;
954 const int l = std::min<int>(n.lane, lanes - 1);
955 const double rh = (lh - 3.) / (hi[l] - lo[l] + 1);
956 const double y = top + l * lh + 1.5 + (hi[l] - n.pitch) * rh;
957 const double nh = std::clamp(rh - 0.5, 2., 6.);
958 auto c = lane_color(l, lanes);
959 if(!heard)
960 c = dimmed(c);
961 ctx.set_fill_color({c.r, c.g, c.b, c.a});
962 ctx.begin_path();
963 const double a = x0 + n.start * lw;
964 const double yc = std::round(y + (rh - nh) / 2.);
965 if(n.end >= n.start)
966 ctx.draw_rect(a, yc, std::max(2., (n.end - n.start) * lw), nh);
967 else
968 {
969 ctx.draw_rect(a, yc, std::max(2., (1. - n.start) * lw), nh);
970 ctx.draw_rect(x0, yc, std::max(2., n.end * lw), nh);
971 }
972 ctx.fill();
973 }
974 }
975 outline_selected(ctx, top, lh);
976 }

982 void paint_values(auto& ctx, double top, double)
983 {
984 const auto& s = state;
985 const int lanes = std::max(1, s.lanes);
986 const double lh = lane_height();
987 const double x0 = 4., lw = width() - 8.;
988 const double bw = lw / view_message::bins;
989 for(int l = 0; l < lanes; l++)
990 {
991 const double y = top + l * lh;
992 const auto bg = lane_background(l);
993 ctx.set_fill_color({bg.r, bg.g, bg.b, bg.a});
994 ctx.begin_path();
995 ctx.draw_rect(x0, y + 1., lw, lh - 2.);
996 ctx.fill();
997
998 const float* p = s.peaks.data() + l * view_message::bins;
999 const uint8_t* f = s.value_flags.data() + l * view_message::bins;
1000 const auto base = lane_color(l, lanes);
1001 const auto quiet = dimmed(base);
1002 const double inner = std::max(2., lh - 8.);
1003 auto height = [&](int b) {
1004 if(f[b] & view_message::flag_number)
1005 return y + 4. + (1. - std::clamp(double(p[b]), 0., 1.)) * inner;
1006 return y + lh / 2.;
1007 };
1008 // The heard parts over the others.
1009 for(int heard = 0; heard < 2; heard++)
1010 {
1011 auto c = heard ? base : quiet;
1012 if(lane_muted(l))
1013 c = dimmed(c);
1014 ctx.set_stroke_color({c.r, c.g, c.b, c.a});
1015 ctx.set_stroke_width(heard ? 1.5 : 1.);
1016 ctx.begin_path();
1017 for(int b = 0; b < view_message::bins; b++)
1018 {
1019 const bool held = f[b] & (view_message::flag_number | view_message::flag_other);
1020 if(!held || bool(f[b] & view_message::flag_heard) != bool(heard))
1021 continue;
1022 const double xa = x0 + b * bw, ya = height(b);
1023 ctx.move_to(xa, ya);
1024 ctx.line_to(xa + bw, ya);
1025 // The step to the next bin's value.
1026 if(b + 1 < view_message::bins
1027 && (f[b + 1] & (view_message::flag_number | view_message::flag_other)))
1028 {
1029 const double yb = height(b + 1);
1030 if(std::abs(yb - ya) > 0.5)
1031 {
1032 ctx.move_to(xa + bw, ya);
1033 ctx.line_to(xa + bw, yb);
1034 }
1035 }
1036 }
1037 ctx.stroke();
1038 }
1039 ctx.set_stroke_width(1.);
1040
1041 // Impulses and values that are not numbers: ticks.
1042 for(int b = 0; b < view_message::bins; b++)
1043 {
1044 if(!(f[b] & (view_message::flag_impulse | view_message::flag_mark)))
1045 continue;
1046 const bool heard = (f[b] & view_message::flag_heard_mark)
1047 || ((f[b] & view_message::flag_mark) && !lane_muted(l));
1048 auto c = heard ? base : quiet;
1049 ctx.set_fill_color({c.r, c.g, c.b, c.a});
1050 ctx.begin_path();
1051 const double xa = x0 + b * bw;
1052 if(f[b] & view_message::flag_impulse)
1053 ctx.draw_rect(xa, y + 2., std::max(1.5, bw), lh - 4.);
1054 else
1055 ctx.draw_rect(xa, y + lh / 2. - 3., std::max(2., bw), 6.);
1056 ctx.fill();
1057 }
1058 }
1059 outline_selected(ctx, top, lh);
1060 }
